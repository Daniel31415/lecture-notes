\section{Binäre Klassifikation}

\begin{expl}\leavevmode
    \begin{itemize}
        \item SPAM oder noSPAM für E-Mails
        \item Hund oder kein Hund für Fotos
    \end{itemize}
\end{expl}
Wir haben also immer die \textbf{Labels}\index{Label} \emph{positiv} oder \emph{negativ} für solche \textbf{Instanzen}\index{Instanz}.

\begin{defi}{Merkmalsraum}{propertySpace}\index{Merkmalsraum}
    Ein \emph{Merkmalsraum} $\cX$ ist eine Menge, die eindeutig alle zu untersuchenden Instanzen eines Problems darstellt.
\end{defi}

\begin{expl}\leavevmode
    \begin{itemize}
        \item Für die SPAM-Klassifikation könnten wir beispielsweise folgendermaßen klassifizieren:
        \begin{enumerate}
            \item $\cX$ ist die Menge aller Strings in UTF-8 Unicode-Codierung
            \item $\cX$ ist eine Menge von Vektoren, also $\cX \subseteq \IZ_{\geq 0}^d$, wobei ein Vektor aus $\cX$ angibt, ob und wie häufig bestimmte Wörter in der E-Mail vorkommen. Dabei sind beispielsweise Wörter wie \emph{price}, \emph{winner} oder ähnliches Teile des \emph{bag of words}\index{bag of words}, und ihnen ist ein Index $\in \set{1,\ldots,d}$ zugeordnet. Der entsprechende Vektor zählt also diese Häufigkeiten
        \end{enumerate}
        \item Für die Image-Klassifikation könnten wir beispielsweise folgendermaßen klassifizieren:
        \begin{enumerate}
            \item $\cX$ ist die Menge aller Bilder in RGB-Pixeldarstellung mit fester Auflösung, fester Größe und fester Farbtiefe (beispielsweise 1 Megapixel, 24 bit Farbtiefe führen zu $\abs{\cX} = \left(2^{24}\right)^{10^6}$, das ist tatsächlich sehr groß)
            \item $\cX$ ist eine Menge von Vektoren, also $\cX \subseteq \IZ_{\geq 0}^d$, wobei ein Vektor aus $\cX$ angibt, ob und wie häufig bestimmte \emph{SIFT descriptors} vorkommen
        \end{enumerate}
    \end{itemize}
\end{expl}

Letztendlich suchen wir eine \emph{Hypothese}\index{Hypothese} $h \colon \cX \to \set{1, -1}$, sodass
\begin{equation*}
    h(x) = \begin{cases}
        1 & \text{$x$ ist SPAM/Hund/\dots} \\
        -1 & \text{sonst}.
    \end{cases}
\end{equation*}

\begin{defi}{Hypothesenklasse}{hypothesisClass}
    Eine \emph{Hypothesenklasse}\index{Hypothesenklasse} $\cH$ ist eine Menge von Funktionen $\cX \to \set{1,-1}$.
\end{defi}

\begin{expl}
    Wir betrachten einige Beispiele für $\cH$.
    \begin{itemize}
        \item Es könnte $\cH = \set{f \colon \cX \to \set{1,-1}}$ sein, also einfach \emph{alle} solche Funktionen beinhalten. 
        
        \emph{Nachteil:} Wir haben zu wenig Struktur.
        \item Sei $\cX = \IR$ und $\cH = \set{h_a \mid a \in \IR}$, wobei 
        \begin{equation*}
            h_a(x) = \begin{cases}
            1 & x \geq a \\
            -1 & x < a.
        \end{cases}
        \end{equation*}
        \item Sei $\cX$ die Menge aller Bilder und $\cH$ die Menge aller Hypothesen $h_w$, die einem neuronalen Netz mit fester Architektur und einem Gewichtsvektor $w$ entsprichen.
    \end{itemize}
\end{expl}

Wie wählen wir nun also unsere Hypothesen aus? Was macht eine gute Hypothese aus?

\begin{defi}{Grundwahrheit}{groundTruth}
    Die \emph{Grundwahrheit}\index{Grundwahrheit} $f$ ist eine Funktion $f \colon \cX \to \set{1, -1}$. Wenn dieses $f$ existiert, weist $f$ jedem $x \in \cX$ das korrekte Label zu.
\end{defi}
Wir nehmen an, dass so ein $f$ existiert. 

\emph{Ziel:} Finde eine Hypothese $h$, die $f$ möglichst nahe kommt.

Wir nehmen an, dass $m$ Datenpunkte (\emph{Instanzen}) aus einer Wahrscheinlichkeitsverteilung $\cD$ über $\cX$ gezogen werden. Diese nennen wir \emph{Sample}\index{Sample}.


\begin{defi}{Diskrete Wahrscheinlichkeitsverteilung}{discreteDistribution}
    Eine \emph{diskrete Wahrscheinlichkeitsverteilung}\index{Wahrscheinlichkeitsverteilung}\index{Wahrscheinlichkeitsverteilung!diskrete -} $\cD$ über $\cX$ ist definiet über ein \emph{Wahrscheinlichkeitsmaß}\index{Wahrscheinlichkeitsmaß} $\IP\colon \cX \to \edgys{0,1}$, wobei $\IP(x) \neq 0$ nur für abzählbar viele Punkte gelten soll und 
    \begin{equation*}
        \sum_{x \in \cX} \IP(x) = 1
    \end{equation*}
    gelten soll.
\end{defi}
\begin{expl}
    Sei $\cX = \IR$ und $\IP(x) = \begin{cases}
        \frac{1}{100} & x \in \set{1,\ldots,100} \\ 0 & \text{sonst}
    \end{cases} = \1_{\set{1,\ldots,100}}(x)$.
\end{expl}

\begin{defi}{Tatsächlicher Fehler}{actualError}
    Der \emph{tatsächliche Fehler}\index{tatsächlicher Fehler}\index{Fehler!tatsächlicher -} $\err_{\cD, f}(h)$ einer Hypothese $h$ hinsichtlich Verteilung $\cD$ über Merkmalsraum $\cX$ und Grundwahrheit $f$ ist 
    \begin{equation*}
        \err_{\cD, f}(h) \coloneqq \IP_{x \sim \cD}[h(x) \neq f(x)].
    \end{equation*}
\end{defi}

\begin{expl}
    Es sei $f(x) = h_{30}$ und $h(x) = h_{50}$. Dann ist
    \begin{equation*}
        \err_{\cD, f}(h) = \frac{\abs{\set{30,\ldots,49}}}{\abs{\set{1,\ldots,100}}} = \frac{20}{100} = \frac{1}{5}.
    \end{equation*}
\end{expl}

\begin{prob}{Lernproblem}{learningProb}
    \prblmObj{$m$ Datenpunkte $x_1, \ldots, x_m$, die \emph{\ac{iid}} aus $\cD$ gezogen sind, und $y_1 = f(x_1), \ldots, y_m = f(x_m)$ (die \emph{korrekten} Labels). Sei $S = \set{(x_1, y_1), \ldots, (x_m, y_m)}$ die Menge der Samples\index{Sample} also gegeben}{Hypothese $h_S \in \cH$}{$\err_{\cD,f}(h_S)$ möglichst klein}
\end{prob}
\begin{defi}{Trainingsfehler}{trainingError}\index{Trainingsfehler}\index{Fehler!Trainings-}
    Der \emph{Trainingsfehler} $\err_S(h)$ einer Hypothese $h \in \cH$ hinsichtlich einer Trainingsmenge $S$ ist 
    \begin{equation*}
        \err_S(h) \coloneqq \frac{1}{m} \abs*{\set*{h(x_i) \neq y_i}}.
    \end{equation*}
\end{defi}

\section{Lernen von Schwellenwertfunktionen}

Sei im Folgenden $\cX = \IR$ unser Merkmalsraum.

In diesem Beispiel und auch im Folgenden vereinfachen wir uns das Leben durch die Annahme, dass die Grundwahrheit \emph{realisierbar}\index{realisierbar} ist. Das heißt, dass $f \in \cH$. Wir werden unsere Theorie später verallgemeinern, dass sie auch ohne diese Annahme auskommt. Wir nehmen also an, dass es ein $a \in \IR$ gibt, sodass die Grundwahrheit $f \colon \cX \to \set{-1, 1}$ gegeben ist durch
\begin{equation*}
    f(x) = \begin{cases}
        1 & x \geq a \\
        -1 & \text{sonst}.
    \end{cases}
\end{equation*}

Wir nehmen an, dass uns $m$ Datenpunkte mit korrekten Labels gegeben sind: Seien $x_1, \dots , x_m$ Datenpunkte, die \ac{iid} aus $\cD$ gezogen sind. Außerdem seien $y_i = f(x_i)$ die zugehörigen, korrekten Labels. Die Menge aller Samples bezeichnen wir mit $S = \set{(x_1, y_1), \dots ,(x_m, y_m)}$.

Ein einfacher Lernalgorithmus wählt nun das größte $a'$, sodass $h_{a'}$ die Menge $S$ korrekt klassifiziert. Das heißt, wir setzen $a'$ auf den Wert des kleinsten $x_i$ mit $y_i = 1$, wenn es ein solches gibt. Anderenfalls $a' = \infty$.

\begin{thm}{Abschätzung tatsächlicher Fehler der Schwellenwertfunktionen}{realErrorBoundTargetValueFunciton}
    Sei $h_{a'}$ die vom einfachen Lernalgorithmus berechnete Hypothese, der als Eingabe eine Menge von Samples von $m$ Datenpunkten mit korrekten Labels gemäß $f$ erhält, die \ac{iid} aus $\cD$ gezogen werden. Dann gilt für alle $\varepsilon > 0$, dass
    \begin{equation*}
        \IP[\err_{\cD, f}(h_{a'}) > \varepsilon] \leq e^{-\varepsilon m}.
    \end{equation*}
\end{thm}
\begin{proof}
    Weil unser Lernalgorithmus unter allen Hypothesen, die $S$ korrekt klassifizieren, die Hypothese $h_{a'}$ mit dem größten $a'$ wählt, wird auf jeden Fall gelten, dass $a' \geq a$. Falsch klassifiziert werden alle Punkte im Bereich $\edgyround{a, a'}$. Somit gilt nun für jede Verteilung $\cD$
    \begin{equation*}
        \err_{\cD, f}(h_{a'}) = \IP_{x \sim \cD}[x \in \roundedgy{a, a'}].
    \end{equation*}
    Sei nun $a'' \geq a$ die kleinste Zahl, sodass $\IP_{x \sim D}[x \in \rounds{a, a''}] \geq \varepsilon$. Es könnte sein, dass $a''$ nicht existiert, weil $\IP_{x \sim \cD}[x \in \edgyround{a, \infty}] < \varepsilon$. Dann ist aber die Wahrscheinlichkeit, dass der Fehler $\err_{\cD,f} (h_{a'})$ den Wert $\varepsilon$ überschreitet, gleich null, und die Aussage des Satzes gilt trivialerweise.

    Es bleibt zu zeigen, dass die Aussage des Satzes auch dann gilt, wenn $\cD$ dergestalt ist, dass $\IP_{x \sim D} [x \in \edgyround{a,\infty}] \geq \varepsilon$ und somit $a''$ wohldefiniert ist. Damit muss auch gelten $\IP_{x\sim D} [x \in \roundedgy{a, a''}] \leq \varepsilon$. Der Fehler von $h_{a'}$ wird also höchstens $\varepsilon$ sein, falls $a' \leq a''$. Dies geschieht, wenn es mindestens ein $i$ gibt, sodass $x_i \in \edgys{a, a''}$. Sei $\cE_i$ das Ereignis, dass $x_i \in \edgys{a, a''}$. Es gilt $\IP[\cE_i] = \IP_{x_i \sim \cD} [x_i \in \edgys{a, a''}] \geq \varepsilon$.
    Damit \emph{nicht} $a' \leq a''$ gilt, darf keines der Ereignisse $\cE_i$ eintreten. Wir interessieren uns also für $\bigcap_i \overline{\cE_i}$. Weil $x_1, \dots , x_m$ unabhängige Züge aus $\cD$ sind, gilt nun auch
    \begin{equation*}
        \IP\left[ \bigcap_i \overline{\cE_i} \right] = \prod_i \IP[\overline{\cE_i}] \leq (1 - \varepsilon)^m.
    \end{equation*}
    Mit $1 - x \leq e^{-x}$ für alle $x \in \IR$ folgt die Behauptung.
\end{proof}

\section{\acs{pac}-Lernbarkeit}
Satz~\ref{thm:realErrorBoundTargetValueFunciton} sagt uns nun insbesondere, dass wenn wir $m = \frac{1}{\varepsilon} \ln\left( \frac{1}{\delta} \right)$ wählen, die Wahrscheinlichkeit, dass der tatsächliche Fehler unserer gefundenen Hypothese größer als $\varepsilon$ ist, kleiner als $\delta$ wird. Diese Aussage gilt für alle $\varepsilon > 0$ und alle $\delta > 0$. Wenn die Anzahl der Samples also nur groß genug ist, werden wir mit großer Wahrscheinlichkeit nur einen sehr kleinen Fehler haben. Die Hypothesenklassen, für die dies gilt, heißen \emph{\ac{pac}-lernbar}\index{PAC-Lernbarkeit}:

\begin{defi}{\ac{pac}-Lernbarkeit}{pac-learnable}
    Eine Hypothesenklasse $\cH$ heißt \emph{\ac{pac}-lernbar} (im realisierbaren Sinn), wenn es eine Funktion $m_\cH$ und einen Lernalgorithmus $\cA$ gibt, sodass $A$ für alle $\varepsilon, \delta > 0$ jede Verteilung $\cD$ und alle $f \in \cH$, gegeben ein Sample $S$ von Größe mindestens $m_\cH(\varepsilon, \delta)$ von Datenpunkten mit korrekten Labels, eine Hypothese $h_S \in \cH$ berechnet, sodass 
    \begin{equation*}
        \IP[\err_{\cD, f}(h_S) < \varepsilon] \geq 1 - \delta.
    \end{equation*}
\end{defi}

\ac{pac}-Lernbarkeit nach dieser Definition impliziert nicht unbedingt, dass das Lernproblem effizient lösbar ist. Man spricht von \emph{effizienter \ac{pac}-Lernbarkeit}\index{PAC-Lernbarkeit!effiziente -}, wenn die Laufzeit von $\cA$ polynomiell ist und außerdem gilt, dass $m_\cH(\varepsilon, \delta)$ polynomiell in $\frac{1}{\varepsilon}$ und $\frac{1}{\delta}$ beschränkt ist.

\acs{pac} steht für \emph{\acl{pac}}. Dabei bedeutet \emph{probably} in diesem Fall, dass die Wahrscheinlichkeit mindestens $1 - \delta$ ist, und \emph{approximately correct} bezieht sich darauf, dass $\err_{\cD, f}(h_S) < \varepsilon$.

Nicht jede Hypothesenklasse ist \ac{pac}-lernbar. Zum Beispiel ist die Klasse aller Hypothesen $\IN \to \set{-1, 1}$ nicht \ac{pac}-lernbar, was wir noch beweisen werden.

\section{Lernen von Intervallen}

Wir betrachten als weiteres Beispiel nun die Klasse der Intervallfunktionen. Weiterhin sei $\cX = \IR$, aber $\cH$ sei nun die Menge der Funktionen der Form $h_{a', b'} \colon \IR \to \set{-1,1}$ mit 
\begin{equation*}
    h_{a', b'} = \begin{cases}
        1 & x \in \edgys{a', b'} \\
        -1 & \text{sonst}.
    \end{cases}
\end{equation*}
Wir betrachten weiterhin den realisierbaren Fall. Das heißt, es gibt eine Grundwahrheit $f \in \cH$, die alle Datenpunkte richtig klassifiziert. Diese hat die Form

\begin{equation*}
    f(x) = \begin{cases}
        1 & x \in \edgys{a, b} \\
        -1 & \text{sonst}.
    \end{cases}
\end{equation*}

Gegeben ist wieder eine Menge $S = \set{(x_1, y_1), \dots ,(x_m, y_m)}$ von Datenpunkten mit korrekten Labels. Unser Lernalgorithmus wählt das kleinste Intervall $\edgys{a', b'}$, das $S$ korrekt klassifiziert. Das heißt, wir setzen $a'$ auf den Wert des kleinsten $x_i$ mit $y_i = 1$ und $b'$ auf den Wert des größten $x_i$ mit $y_i = 1$. 
Den Fall, dass es kein $i$ mit $y_i = 1$ gibt, schieben wir ab in die Übung und ignorieren ihn.

\begin{thm}{Abschätzung tatsächlicher Fehler Intervalllernen}{realErrorBoundInterval}
    Für alle $\varepsilon > 0$ gilt 
    \begin{equation*}
        \IP[\err_{\cD, f}(h_{a', b'}) > \varepsilon] \leq 2e^{-\frac{\varepsilon m}{2}}.
    \end{equation*}
\end{thm}

\begin{lem}{Union Bound}{unionBound}
    Es seien $\cE_1, \ldots, \cE_n$ (nicht notwendigerweise disjunkte) Ereignisse. Dann gilt 
    \begin{equation*}
        \IP\left[ \bigcup_{i = 1}^n \cE_i \right] \leq \sum_{i = 1}^n \IP[\cE_i].
    \end{equation*}
\end{lem}
\begin{proof}
    Induktives Anwenden von $\IP[A \cup B] = \IP[A] + \IP[B] - \IP[A \cap B] \leq \IP[A] + \IP[B]$.
\end{proof}

\begin{proof}[Beweis von Satz~\ref{thm:realErrorBoundInterval}]
    Weil unser Lernalgorithmus das kleinste Intervall wählt, welches $S$ korrekt klassifiziert, wird auf jeden Fall gelten, dass $a' \geq a$ und $b' \leq b$. Falsch klassifiziert werden alle Punkte im Bereich $\edgyround{a, a'} \cup \roundedgy{b', b}$. Somit gilt nun für jede Verteilung $\cD$
    \begin{equation*}
        \err_{\cD, f}(h_{a', b'}) = \IP_{x \sim \cD}[x \in \edgyround{a, a'} \cup \roundedgy{b', b}].
    \end{equation*}
    Analog zu oben betrachten wir den Fall, dass $\IP_{x \sim \cD}[x \in \edgys{a,b}] < \varepsilon$. In diesem Fall ist die Wahrscheinlichkeit, dass der Fehler $\err_{\cD,f} (h_{a', b'})$ den Wert $\varepsilon$ überschreitet, wieder gleich null, und die Aussage des Satzes gilt trivialerweise.

    Es bleibt zu zeigen, dass die Aussage des Satzes auch dann gilt, wenn $\cD$ dergestalt ist, dass $\IP_{x\sim D} [x \in \edgys{a,b}] \geq \varepsilon$. Sei nun, ähnlich wie oben, $a'' \in \edgys{a,b}$ die kleinste Zahl, sodass $\IP_{x \sim D} [x \in \edgys{a, a''}] \geq \frac{\varepsilon}{2}$. Analog sei $b'' \in \edgys{a,b}$ die größte Zahl, sodass $\IP_{x \sim \cD} [x \in \edgys{b'', b}] \geq \frac{\varepsilon}{2}$. Damit $\err_{\cD,f} (h_{a', b'}) \leq \varepsilon$, ist es nun hinreichend, dass $a' \leq a''$ und $b' \geq b''$. Dies geschieht, wenn es mindestens ein $i$ und mindestens ein $j$ gibt, sodass $x_i \in \edgys{a, a''}$ und $x_j \in \edgys{b'', b}$ wobei $i$ und $j$ nicht verschieden sein müssen. Für jedes $i$ gilt
    \begin{equation*}
        \IP[x_i \in \edgys{a, a''}] \geq \frac{\varepsilon}{2} \quad \text{und} \quad \IP[x_i \in \edgys{b'', b}] \geq \frac{\varepsilon}{2}.
    \end{equation*}
    Sei $\cE_a$ das Ereignis, dass $x_1, \ldots, x_m \notin \edgys{a, a''}$ und sei $\cE_b$ das Ereignis, dass $x_1, \ldots, x_m \notin \edgys{b'', b}$. Weil $x_1, \ldots, x_m$ unabhängige Züge aus $\cD$ sind, gilt nun auch 
    \begin{equation*}
        \IP[\cE_a] \leq \left( 1 - \frac{\varepsilon}{2} \right)^m \quad \text{und} \quad \IP[\cE_b] \leq \left( 1 - \frac{\varepsilon}{2} \right)^m
    \end{equation*}
    und mit~\Cref{lem:unionBound} folgt 
    \begin{equation*}
        \IP[\cE_a \cup \cE_b] \leq 2\left( 1 - \frac{\varepsilon}{2} \right)^m.
    \end{equation*}
    Die Behauptung folgt wieder mit $1 - x \leq e^{-x}$.
\end{proof}

Aus dem Satz können wir folgern, dass auch die Klasse der Intervalle \ac{pac}-lernbar ist. Setzen wir etwa $m = \frac{2}{\varepsilon'}\ln\left( \frac{2}{\delta} \right)$, dann folgt aus Satz~\ref{thm:realErrorBoundInterval}, dass $\IP[\err_{\cD, f}(h_S) \leq \varepsilon'] \geq 1 - \delta$, wobei wir $\varepsilon'$ und $\delta$ als Parameter wählen können. Es gilt für jedes $\varepsilon' < \varepsilon$, dass 
\begin{equation*}
    \IP[\err_{D, f}(h_S) < \varepsilon] \geq \IP[\err_{\cD, f}(h_S) \leq \varepsilon'].
\end{equation*}
Wählen wir also beispielsweise $m_\cH(\varepsilon, \delta) = \frac{4}{\varepsilon} \ln\left( \frac{2}{\delta} \right)$, dann folgt $\IP[\err_{\cD, f}(h_S) < \varepsilon] \geq 1 - \delta$. Die Klasse der Intervalle ist also \ac{pac}-lernbar.

\chapter{Endliche Hypothesenklassen, Wachstumsfunktion und \acs{vc}-Dimension}

\section{Endliche Hypothesenklassen, Wachstumsfunktion}

Wir betrachten den Fall, dass $\cH$ endlich ist. 
\begin{thm}{Tatsächlicher Fehler bei endlicher Hypothesenklasse}{finiteHypothesisClassRealError}
    Wenn $m \geq \frac{1}{\varepsilon} \ln\left( \frac{\abs{\cH}}{\delta} \right)$, dann gilt mit Wahrscheinlichkeit mindestens $1 - \delta$, dass alle $h \in \cH$ mit $\err_S(h) = 0$ auch $\err_{\cD, f}(h) < \varepsilon$ erfüllen.
\end{thm}

\begin{proof}
    Sei $h \in \cH$ fest mit $\err_{\cD, f}(h) \geq \varepsilon$. Dann gilt, da $S$ \ac{iid} mittels $\cD$ gezogen wird, und mit der Voraussetzung $\IP[h(x_i) \neq y_i] \geq \varepsilon$
    \begin{equation*}
        \IP\left[ \err_S(h) = 0 \right] = \IP[h(x_1) = y_1, \ldots, h(x_m) = y_m] = \prod_{i = 1}^m \IP[h(x_i) = y_i] \leq (1 - \varepsilon)^m \leq e^{-\varepsilon m}.
    \end{equation*}
    Für jede Hypothese definiere $\cE_h$ als das Ereignis, dass $\err_S(h) = 0$ ist. Dann ist 
    \begin{equation*}
        \begin{split}
            &\IP[\exists h \in \cH \colon \text{$\err_S(h) = 0$ und $\err_{\cD, f}(h) \geq \varepsilon$}] = \IP\left[ \bigcup_{\substack{h \in \cH \\ \err_{\cD, f}(h) \geq \varepsilon}}  \cE_h \right]  \\
            \leq~& \sum_{\substack{h \in \cH \\ \err_{\cD, f}(h) \geq \varepsilon}} \IP[\err_S(h) = 0] \leq \abs{\cH} e^{-\varepsilon m} \leq \delta,
        \end{split}
    \end{equation*}
    also exakt, was wir zeigen wollten.
\end{proof}

\begin{defi}{Wachstumsfunktion}{growthFunction}
    Gegeben $A \subseteq \cX$, sei $\cH|_A$ die Menge aller Hypothesen $h \in \cH$ mit Definitionsbereich eingeschränkt auf $A$, also
    \begin{equation*}
        \cH|_A = \set*{h|_A \mid h \in \cH}.
    \end{equation*}
    Die \emph{Wachstumsfunktion}\index{Wachstumsfunktion} von $\cH$ ist definiert als 
    \begin{equation*}
        \Pi_\cH(m) \coloneqq \max_{\substack{A \subseteq \cX \\ \abs{A} = m}} \abs*{\cH|_A}.
    \end{equation*}
\end{defi}
Es gilt immer $\Pi_\cH(m) \leq 2^m$, es ist aber oft wesentlich kleiner.
\begin{expl}
    Sei $\cX = \IR$ und $\cH$ die Klasse der Schwellenwertfunktionen $h_a(x) = \begin{cases}
        1 & x \geq a \\ -1 & \text{sonst}.
    \end{cases}$ Für $A = \set{2,3,4}$ ist 
    \begin{equation*}
        \cH|_A = \set*{x \mapsto -1 \forall x \in A x \mapsto 1 \forall x \in A, x \mapsto \begin{cases}
            1 & x = 4 \\ -1 & x \in \set{2,3}
        \end{cases}, x \mapsto \begin{cases}
            1 & x \in \set{3,4} \\ -1 & x = 2
        \end{cases}}
    \end{equation*}
    und damit ist $\abs*{\cH|_A} = 4 \ll 2^3$. Allgemein ist für $\abs{A} = m$ für Schwellenwertfunktionen immer $\abs*{\cH|_A} = m + 1$.
\end{expl}
\section{Subexponentielles Wachstum impliziert \acs{pac}-Lernbarkeit}

\begin{thm}{\ac{pac}-lernbar wenn Wachstumsfunktion subexponentiell}{pacLearnableForSubexponentialGrowthFct}
    Es seien $\varepsilon, \delta > 0$ fest, und 
    \begin{equation}
        m \geq \max\set*{\frac{8}{\varepsilon}, \frac{2}{\varepsilon}\log_2\left( \frac{2\Pi_\cH(2m)}{\delta} \right)}.\label{eq:pacLearnableForSubexponentialGrowthFct}
    \end{equation}
    Betrachte eine Sample-Menge $S$ von $m$ Datenpunkten mit korrekten Labels gemäß der Grundwahrheit $f$, die \ac{iid} aus $\cD$ gezogen werden. Dann gilt mit Wahrscheinlichkeit $\geq 1 - \delta$, dass alle $h \in \cH$ mit $\err_S(h) = 0$ auch $\err_{\cD, f}(h) < \varepsilon$ erfüllen.
\end{thm}

Der Beweis folgt später. Wir fragen uns erstmal, wann die Ungleichung
\begin{equation}
    m \geq \frac{2}{\varepsilon} \log_2\left( \frac{2\Pi_\cH(2m)}{\delta} \right) = \frac{2}{\varepsilon} \log_2(\Pi_\cH(2m)) + \frac{2}{\varepsilon}\log_2\left( \frac{2}{\delta} \right)\label{eq:pacLearnableSubexponentialBoundary}
\end{equation}
erfüllt ist. Es ist~\eqref{eq:pacLearnableSubexponentialBoundary} äquivalent zu 
\begin{equation*}
    \frac{m - \frac{2}{\varepsilon}\log_2\left( \frac{2}{\delta} \right)}{\log_2(\Pi_\cH(2m))} \geq \frac{2}{\varepsilon}.
\end{equation*}
Wir testen ein paar Schranken aus.
\begin{enumerate}[$(a)$]
    \item Wenn $\Pi_\cH(2m) = 2^{2m}$, dann ist $\log_2(\Pi_\cH(2m)) = 2m$ und die Ungleichung kann nicht für alle $\varepsilon, \delta > 0$ erfüllt werden.
    \item Gilt $\log_2(\Pi_\cH(2m)) \in o(m)$, so müssen wir $m$ nur hinreichend groß wählen, um diese Schranke zu erfüllen. 
\end{enumerate}
\begin{expl}
    Bei den Schwellenwertfunktionen ist $\Pi_\cH(2m) = 2m + 1$. Dann gilt für alle $\delta > 0$
    \begin{equation*}
        \frac{m - \frac{2}{\varepsilon}\log_2\left( \frac{2}{\delta} \right)}{\log_2(2m + 1)} \overset{m \to \infty}{\longrightarrow} \infty.
    \end{equation*}
\end{expl}
Damit erhalten wir (mit dem Beweis von Satz~\ref{thm:pacLearnableForSubexponentialGrowthFct} als Black Box) \ac{pac}-Lernbarkeit für Hypothesenklassen, die die angesprochenen Ungleichungen erfüllen.

\section{Schranke über die \acs{vc}-Dimension}
Über die \acf{vc}-Dimension werden wir die Wachstumsfunktion beschränken (in mehr Details in der nächsten Vorlesung):

\begin{lem}{Sneak Peak: Wachstumslemma}{growthLemmaSneakPeak}
    Wenn $\cH$ \ac{vc}-Dimension $d$ hat, so gilt 
    \begin{equation*}
        \Pi_\cH(m) \leq \left( \frac{em}{d} \right)^d
    \end{equation*}
\end{lem}
Damit motivieren wir folgenden Satz:
\begin{thm}{\ac{pac}-Lernbarkeit mit \ac{vc}-Dimension}{PACusingVC}
    Sei $\cH$ eine Hypothesenklasse mit \ac{vc}-Dimension $d < \infty$. Seien $1 \geq \varepsilon > 0$, $\delta > 0$ und sei 
    \begin{equation}
        m \geq \max\set*{\frac{4}{\varepsilon} \log_2\left( \frac{2}{\delta} \right), \frac{8d}{\varepsilon}\log_2\left( \frac{16}{\varepsilon} \right)}. \label{eq:inequalityForMUsingVC}
    \end{equation}
    Betrachte eine Sample-Menge $S$ von $m$ Datenpunkten mit korrekten Labels gemäß der Grundwahrheit $f$, die \ac{iid} aus $\cD$ gezogen werden. Dann gilt mit Wahrscheinlichkeit $ \geq 1 - \delta$, dass alle $h \in \cH$ mit $\err_S(h) = 0$ auch $\err_{\cD, f}(h) < \varepsilon$ erfüllen.
\end{thm}
\begin{proof}
    Es genügt zu zeigen, dass~\eqref{eq:inequalityForMUsingVC} bereits~\eqref{eq:pacLearnableSubexponentialBoundary} impliziert (wenn wir später Satz~\ref{thm:pacLearnableForSubexponentialGrowthFct} später gezeigt haben).

    Zunächst, da $d \geq 1$ und $\varepsilon \leq 1$, gilt 
    \begin{equation*}
        m \geq \frac{8d}{\varepsilon}\log_2\left( \frac{16}{\varepsilon} \right) \geq \frac{8}{\varepsilon}.
    \end{equation*}
    Weiterhin gilt
    \begin{equation*}
        \frac{m - \frac{2}{\varepsilon}\log_2\left( \frac{2}{\delta} \right)}{\log_2(\Pi_\cH(2m))} \geq \frac{m}{2d\log_2\left( \frac{2em}{d} \right)} \geq \frac{\frac{8d}{\varepsilon}\log_2\left( \frac{16}{\varepsilon} \right)}{2d\log_2\left( \frac{2e}{d} \cdot \frac{8d}{\varepsilon}\log_2\left( \frac{16}{\varepsilon} \right) \right)} \geq \frac{4}{\varepsilon}\frac{\log_2\left( \frac{16}{\varepsilon} \right)}{\log_2\left(\frac{16e}{\varepsilon} \cdot \log_2\left(\frac{16e}{\varepsilon}   \right)\right)} \geq \frac{2}{\varepsilon},
    \end{equation*}
    da der Nenner monoton fallend in $\varepsilon$ und im Intervall $\rounds{0,1}$ mindestens $\frac{1}{2}$ ist.
\end{proof}

\chapter{Mengensysteme und deren \acs{vc}-Dimension}

Heute fragen wir uns, welche Struktur einer Hypothesenklasse $\cH$ zu subexponentiellem Wachstum führt.

\begin{defi}{Mengensystem, Aufspalten, Abspalten}{setSystem}
    Sei $\cX$ eine beliebige Menge und $\cR \subseteq \cP(\cX)$. Dann nennen wir $\cR$ ein \emph{Mengensystem mit Grundmenge $\cX$}\index{Mengensystem}.
    
    Eine Menge $A' \subseteq \cX$ wird \emph{durch $\cR$} von einer Menge $A$ \emph{abgespalten}\index{Abspaltung (Menge)}\index{Mengensystem!Abspaltung}, wenn $r \in \cR$ existiert, sodass $A' = r \cap A$.

    Eine Menge $A \subseteq \cX$ wird \emph{durch $\cR$ aufgespalten} (engl. \emph{Shattering})\index{Aufspaltung (Menge)}\index{Mengensystem!Aufspaltung}, wenn jede Teilmenge $A' \subseteq A$ von $A$ durch $\cR$ abgespalten wird.
\end{defi}

\begin{defi}{\acs{vc}-Dimension}{vcDimension}
    Die \emph{\acf{vc}-Dimension}\index{VC-Dimension} von $\cR$ ist die Anzahl der Elemente der größten aufgespaltenen Menge. Wir schreiben dafür $\dim(\cR)$.

    Wir setzen $\dim(\cR) \coloneqq \infty$, wenn für jedes $n_0 \in \IN$ eine aufgespaltene Menge mit $n_0$ Elementen existiert, und $\dim(\cR) \coloneqq 0$, wenn keine solche Menge existiert.
\end{defi}

\begin{expl}\leavevmode
    \begin{enumerate}
        \item Für $\cX = \IR^2$ und $\cR$ die Menge aller achsenparalleler Rechtecke. Betrachte $A \subseteq \cX$. Es ist $\dim(\cR) \geq 4$. Wenn wir mindestens fünf Punkte haben, so ist einer dieser Punkte in keiner Koordinate \enquote{extrem} bzw. \enquote{maximal}, und wir können also nie ein Rechteck um alle Punkte außer diesen Mittelpunkt bauen. Also ist $\dim(\cR) = 4$.
        \item Für $\cX = \IR^2$ und $\cR$ die Menge aller konvexen Polygone. Dann ist für jedes $n_0$ die Menge der $n_0$ Punkte, die ein regelmäßiges $n_0$-Eck bilden, aufspaltbar. Also ist $\dim(\cR) = \infty$.
    \end{enumerate}
\end{expl}

\section{Wachstum von endlichen Mengensystemen}
\begin{expl}\label{expl:setSystemSize}
    Sei $\cX = \set{1,\ldots,m}$ und sei $\cR$ das Mengensystem, das alle Teilmengen von maximaler Größe $k$ enthält. Dann ist $\dim(\cR) = k$ und 
    \begin{equation*}
        \abs{\cR} = \sum_{i = 0}^k \binom{m}{i} = \sum_{i = 0}^k \left( \frac{m}{k} \right)^i \frac{k^i}{i!} \leq \left(\frac{m}{k}\right)^k \sum_{i = 0}^k \frac{k^i}{i!} \leq \left( \frac{em}{k} \right)^k.
    \end{equation*}
    mit 
    \begin{equation*}
        \binom{m}{i} = \frac{m!}{(m-i)! i!} \leq \frac{m^i k^i}{i! k^i} \leq \left(\frac{m}{k}\right)^i\frac{k^i}{i!}
    \end{equation*}
    und der Reihendarstellung von $\exp$
    \begin{equation*}
        e^x = \sum_{i = 0}^\infty \frac{x^i}{i!}.
    \end{equation*}
    Damit wächst $\abs{\cR}$ subexponentiell in $m$ für konstante $k$.
\end{expl}

\begin{lem}{Größe von Mengensystemen}{sizeOfSetSystems}
    Es gilt für jedes Mengensystem $\cR$ mit $m$-elementiger Grundmenge $\cX$ und \ac{vc}-Dimension $d$, dass 
    \begin{equation*}
        \abs{\cR} \leq \sum_{i = 0}^d \binom{m}{i}.
    \end{equation*}
\end{lem}
\begin{proof}
    Wir machen Induktion über $m$.
    \begin{description}
        \item[IA: $m = 0$.] Dann ist $\abs{\cR} \leq 1$ und $d = 0$ und es gilt 
        \begin{equation*}
            \sum_{i = 0}^d \binom{0}{i} = \binom{0}{0} = 1.
        \end{equation*}
        \item[IS: $m > 0$.] Sei $\cR$ ein Mengensystem mit Grundmenge $\cX$ mit $\abs{\cX} = m$ und \ac{vc}-Dimension $d$. Angenommen, $d = 0$. Dann ist $\abs{\cR} \leq 1$ und die Aussage ist erfüllt, da $\binom{m}{i} \geq \binom{1}{0} = 1$. 
        
        Sei also auch $d > 0$. Sei $x \in \cX$ fest. Sei $\cR_1 \coloneqq \set{r \setminus \set{x} \mid r \in \cR}$ und sei $\dim(\cR_1) = d_1 \leq d$. Nach Induktionsvoraussetzung folgt 
        \begin{equation*}
            \abs{\cR_1} \leq \sum_{i = 0}^{d_1} \binom{m-1}{i} \leq \sum_{i = 0}^d \binom{m-1}{i}.
        \end{equation*}
        Definiere $\cR_2 \coloneqq \set{r \setminus \set{x} \mid r \cup \set{x} \in \cR \text{ und } r \setminus \set{x} \in \cR}$. Es gilt also $\abs{\cR} = \abs{\cR_1} + \abs{\cR_2}$. Dieses $\cR_2$ dient dazu, es auszugleichen, dass in $\cR_1$ durch Beschränkung von $\cR$ auf $\cX \setminus \set{x}$ zwei verschiedene Mengen identisch werden können. Diese Mengen werden in $\cR_2$ quasi doppelt gezählt und somit ausgeglichen.
        \begin{example}
            Für $\cX = \set{1,2,3}, x = 1$ und $\cR= \set{\set{1,2}, \set{2}, \set{1,3}}$, dann ist $\cR_1 = \set{\set{2}, \set{3}}$ und $\cR_2 = \set{\set{2}}.$
        \end{example}
        Sei $d_2 = \dim(\cR_2)$. Wir behaupten, dass $d_2 \leq d - 1$: Angenommen, $d_2 \geq d$. Dann gibt es $A \subseteq \cX \setminus \set{x}$ mit $\abs{A} = d$, sodass $A$ durch $\cR_2$ aufgespalten wird. Dann wird aber $A \cup \set{x}$ nach Definition von $\cR_2$ durch $\cR_1$ aufgespalten, und die \ac{vc}-Dimension von $\cR$ wäre somit mindestens $d_2 + 1 > d$. Widerspruch!
        
        Somit gilt nach Induktionsvoraussetzung
        \begin{equation*}
            \abs{\cR_2} \leq \sum_{i = 0}^{d_2} \binom{m-1}{i} \leq \sum_{i = 0}^{d-1} \binom{m-1}{i}.
        \end{equation*}
        Nun gilt mit $\binom{a}{b} = \binom{a-1}{b} + \binom{a-1}{b-1}$
        \begin{equation*}
            \begin{split}
                \abs{\cR} = \abs{\cR_1} + \abs{\cR_2} &\leq \sum_{i = 0}^d \binom{m-1}{i} + \sum_{i = 0}^{d-1} \binom{m-1}{i} = \sum_{i = 0}^d \binom{m-1}{i}+ \sum_{i = 1}^d \binom{m-1}{i-1} \\
                &= \binom{m-1}{0} + \sum_{i = 1}^d \binom{m-1}{i} + \binom{m-1}{i-1} = \binom{m}{0} + \sum_{i = 1}^d \binom{m}{i} = \sum_{i = 0}^d \binom{m}{i},
            \end{split}
        \end{equation*}
    \end{description}
    was wir zeigen wollten.
\end{proof}

\section{Unendliche Mengensysteme}
\begin{defi}{Untersystem}{subsystem}
    Sei $\cR$ ein Mengensystem mit Grundmenge $\cX$. Jede Menge $A \subseteq \cX$ bestimmt ein \emph{Untersystem}\index{Untersystem}\index{Mengensystem!Untersystem} von $\cR$ wie folgt:
    \begin{equation*}
        \cR|_A = \set{r \cap A \mid r \in \cR}
    \end{equation*}
\end{defi}
Es gilt $\dim(\cR|_A) \leq \dim(\cR)$.


Wir betrachten nun das Wachstumslemma (Sneak Peak war~\Cref{lem:growthLemmaSneakPeak}).

\begin{lem}{Wachstumslemma}{growthLemma}
    Sei $\cR$ ein Mengensystem mit Grundmenge $\cX$ und \ac{vc}-Dimension $d$. Für jedes $n \in \IN$ gilt 
    \begin{equation*}
        \Pi_\cR(m) \coloneqq \max_{\substack{A \subseteq \cX \\ \abs{A} = m}} \abs{\cR|_A} \leq \left( \frac{em}{d} \right)^d.
    \end{equation*}
    Wir nennen $\Pi$ die \emph{Wachstumsfunktion} von $\cR$.\index{Wachstumfsunktion}
\end{lem}

\begin{proof}
    Für jedes $A \subseteq \cX$ mit $\abs{A} = m$ gilt 
    \begin{equation*}
        \abs{\cR|_A} \leq \sum_{i = 0}^d \binom{m}{i} \leq \left( \frac{em}{d} \right)^d
    \end{equation*}
    nach~\Cref{lem:sizeOfSetSystems} und der Rechnung aus Beispiel~\ref{expl:setSystemSize}. Es solft somit direkt $\Pi_\cR(m) \leq \left( \frac{em}{d} \right)^d$. 
\end{proof}

Nun folgt mit $\cR \coloneqq \set{r_h \mid h \in \cH}$, wobei $r_h \coloneqq \set{x \in \cX \mid h(x) = 1}$, auch direkt $\Pi_\cH(m) = \Pi_\cR(m)$ und damit~\Cref{lem:growthLemmaSneakPeak}.

\chapter{Lineare Klassifikation und ihre \acs{vc}-Dimension}

Sei die Hypothesenklasse $\cH$ die Menge der Funktionen der Form 
\begin{equation*}
    \begin{split}
        h_{a, b} \colon \IR^d \to& \set{-1,1} \\
            x \mapsto& \begin{cases}
                1 & \inner{a, x} \geq b \\

            \end{cases}
    \end{split}
\end{equation*}
Äquivalent als Mengensystem $\cR$ mit den Mengen 
\begin{equation*}
    r_{a, b} = \set{x \in \IR^d \mid \inner{a, x} \geq b}.
\end{equation*}
Dabei definiert $r_{a, b}$ einen Halbraum, beschränkt durch eine Hyperebene 
\begin{equation*}
    \ell = \set{x \in \IR^d \mid \inner{a,x} = b}.
\end{equation*}

\section{\acs{vc}-Dimension von Halbräumen}

\begin{defi}{Affinkombination, affine Abhängigkeit}{affineCombination}
    Eine Linearkombination 
    \begin{equation*}
        \sum_{i = 1}^n \alpha_i p^i \quad \text{für $p_1, \ldots, p_n \in \IR^d, \alpha_1, \dots, \alpha_n \in \IR$}
    \end{equation*}
    ist eine \emph{Affinkombination}\index{Affinkombination}, wenn 
    \begin{equation*}
        \sum_{i = 1}^n \alpha_i = 1
    \end{equation*}
    gilt.

    Die Menge $\set{p_1, \ldots, p_n} \subseteq \IR^d$ ist \emph{affin abhängig}\index{affin abhängig}, wenn es ein $i \in \set{1,\ldots,n}$ gibt, sodass 
    \begin{equation*}
        p_i = \sum_{\substack{j = 1 \\ j \neq i}}^n \beta_i p_j \quad \text{mit} \quad \sum_{\substack{j = 1 \\ j \neq i}}^n \beta_i = 1.
    \end{equation*}
\end{defi}

\begin{lem}{$d+2$ Punkte sind affin abhängig}{dPlus2PtsAreAffineDependent}
    Jede Menge von $d + 2$ Punkten in $\IR^d$ sind affin abhängig.
\end{lem}

\begin{proof}
    Sei $A = \set{p_1, \ldots, p_{d + 2}} \subseteq \IR^d$. Definiere $q_i \coloneqq p_i - p_{d + 2}$ für alle $i$.

    Da jede Menge von $d + 1$ Punkten im $\IR^d$ linear abhängig ist, gibt es ein $r \in \set{1, \ldots, d + 1}$, sodass
    \begin{equation*}
        q_r = \sum_{\substack{i = 1 \\ i \neq r}}^{d + 1} \beta_i q_i
    \end{equation*}
    gilt. OBdA sei $r = d + 1$. Dann ist also 
    \begin{equation*}
        q_r = \sum_{i = 1}^d \beta_i q_i,
    \end{equation*}
    und somit gilt 
    \begin{equation*}
        p_{d + 1} = q_{d + 1} + p_{d + 2} = \left( \sum_{j = 1}^d \beta_i q_i \right) + p_{d + 2} = \left( \sum_{j = 1}^d \beta_i p_i \right) + \left( - \sum_{j = 1}^d \beta_i  + 1\right) p_{d + 2}.
    \end{equation*}
    Nun definieren wir $\beta_{d + 2} \coloneqq - \sum_{j = 1}^d \beta_i + 1$, somit ist $\beta_{d + 2} + \sum_{j = 1}^d \beta_i = 1$ und $p_{d + 1}$ ist die Affinkombination der anderen $d + 1$ Punkte mit diesen $\beta_i$.
\end{proof}

\begin{defi}{Konvexkombination}{convexCombination}
    Eine Linearkombination 
    \begin{equation*}
        \sum_{i = 1}^n \alpha_i p^i \quad \text{für $p_1, \ldots, p_n \in \IR^d, \alpha_1, \dots, \alpha_n \in \IR$}
    \end{equation*}
    ist eine \emph{Konvexkombination}\index{Konvexkombination}, wenn 
    \begin{equation*}
        \sum_{i = 1}^n \alpha_i = 1 \quad \text{und} \quad \alpha_i \geq 0 \text{ für alle $i \in \set{1,\ldots, n}$}
    \end{equation*}
    gilt.

    Die Menge aller Konvexkombinationen einer Menge $P \subseteq \IR^d$ ist ihre \emph{konvexe Hülle}\index{konvexe Hülle} $\conv(P)$.
\end{defi}

\begin{lem}{Radon}{radon}
    Für jede Menge $A$ von $d + 2$ Punkten im $\IR^d$ existieren $A_1 , A_2 \subseteq A$ mit $A_1 \cap A_2 = \emptyset$ und ein Punkt $q$, sodass $q \in \conv(A_1)$ und $q \in \conv(A_2)$.
\end{lem}
Wir nennen $q$ dann \emph{Radonpunkt}\index{Radonpunkt}.

\begin{proof}
    Sei $A = \set{p_1, \ldots, p_{d + 2}}$. Diese ist nach~\Cref{lem:dPlus2PtsAreAffineDependent} affin abhängig. Also existiert ein $i \in \set{1, \ldots, d + 2}$, sodass 
    \begin{equation*}
        p_i = \sum_{\substack{j = 1 \\ j \neq i}}^{d + 2} \alpha_j p_j \quad \text{mit} \quad \sum_{\substack{j = 1 \\ j \neq i}}^{d + 2} \alpha_j = 1.
    \end{equation*}
    Wir setzen $\alpha_i \coloneqq -1$. Dann ist 
    \begin{align}
        0 &= p_i + \alpha_i p_i = \sum_{j = 1}^{d + 2} \alpha_j p_j \quad \text{und} \label{eq:radonProof-1} \\
        0 &= \sum_{j = 1}^{d + 2} \alpha_j. \label{eq:radonProof-2}
    \end{align}
    Wir setzen $I_1 \coloneqq \set{j \mid \alpha_j > 0}$ und $I_2 \coloneqq \set{j \mid \alpha_j = 0}$. Dann folgt aus~\eqref{eq:radonProof-1} 
    \begin{equation*}
        0 = \sum_{j \in I_1} \alpha_j p_j + \sum_{j \in I_2} \alpha_j p_j,
    \end{equation*}
    und umgeformt 
    \begin{equation*}
        - \sum_{j \in I_2} \alpha_j p_j = \sum_{j \in I_1} \alpha_j p_j.
    \end{equation*}
    Analog folgt aus~\eqref{eq:radonProof-2} 
    \begin{equation*}
        -\sum_{j \in I_2} \alpha_j = \sum_{j \in I_1} \alpha_j.
    \end{equation*}
    Setze nun $\gamma \coloneqq \sum_{j \in I_1} \alpha_j$ und $\beta_j \coloneqq \frac{\alpha_j}{\gamma}$ für $j \in I_1$, sowie $\beta_j \coloneqq -\frac{\alpha_j}{\gamma}$ für $j \in I_2$. Dann ist $\sum_{j \in I_1} \beta_j = 1 = \sum_{j \in I_2} \beta_j$ und 
    \begin{equation*}
        q \coloneqq \sum_{j \in I_2} -\frac{\alpha_j}{\gamma} p_j = \sum_{j \in I_1} \frac{\alpha_j}{\gamma} p_j.
    \end{equation*}
    Beide Seiten sind also disjunkte Konvexkombinationen für denselben Punkt $q$.
\end{proof}

\begin{thm}{\acs{vc}-Dimension von Halbräumen: obere Schranke}{VCDimOfHalfSpacesUpperBound}
    Die \ac{vc}-Dimension von Halbräumen im $\IR^d$ ist höchstens $d + 1$.
\end{thm}
\begin{proof}
    Angenommen, die \ac{vc}-Dimension wäre mindestens $d + 2$. Wähle $A = \set{p_1, \ldots, p_{d + 2}} \subseteq \IR^d$, sodass $A$ vollständig durch $\cR$ aufgespalten werden kann. Nach~\Cref{lem:radon} existieren Indexmengen $I_1, I_2 \subseteq \set{1,\ldots,d + 2}$ mit $I_1 \cap I_2 = \emptyset$, sodass Konvexkombinationen und ein Punkt $q$ existieren mit 
    \begin{equation*}
        \sum_{i \in I_1} \alpha_i p_i = q = \sum_{i \in I_2} \beta_i p_i
    \end{equation*}
    und $\sum_{i \in I_1} \alpha_i = 1 = \sum_{i \in I_2} \beta_i$ mit $\alpha_i, \beta_i \geq 0$.

    Sei $r_{a, b}$ der Halbraum, der $A_1 = \set{p_i \mid i \in I_1}$ von $A$ abspaltet. Dann ist 
    \begin{equation*}
        A_1 = r_{a, b} \cap A \quad \text{und} \quad A_2 \cap r_{a, b} = \emptyset
    \end{equation*}
    für $A_2 = \set{p_i \mid i \in I_2}$. Also ist $\inner{a, p_i} \geq b$ für $i \in I_1$ und $\inner{a, p_i} < b$ für $i \in I_2$.

    Nun gilt mit Bilinearität des Skalarproduktes sowie $\sum_{i \in I_1}\alpha_i = 1 = \sum_{i \in I_2} \beta_i$ und $\alpha_i, \beta_i \geq 0$ 
    \begin{equation*}
        \inner{a, q} = \inner*{a, \sum_{i \in I_1} \alpha_i p_i} =  \sum_{i \in I_1} \alpha_i \inner{a, p_i} \geq \sum_{i \in I_1} \alpha_i \cdot b = b \cdot \sum_{i \in I_1} \alpha_i = b,
    \end{equation*}
    andererseits aber auch 
    \begin{equation*}
        \inner{a, q} = \inner*{a, \sum_{i \in I_2} \beta_i p_i} = \sum_{i \in I_1} \beta_i \inner{a, p_i} < \sum_{i \in I_1} \beta_i \cdot b = b \cdot \sum_{i \in I_1} \beta_i = b,
    \end{equation*}
    Widerspruch! $\lightning$
\end{proof}

\begin{thm}{\acs{vc}-Dimension von Halbräumen: untere Schranke}{VCDimOfHalfSpacesLowerBound}
    Die \ac{vc}-Dimension von Halbräumen im $\IR^d$ ist mindestens $d + 1$.
\end{thm}
\begin{proof}
    Wir konstruieren $A \subset \IR^d$ mit $\abs{A} = d + 1$, sodass $A$ aufgespalten wird. Sei $e_i$ der $i$te Einheitsvektor und $e_0$ der Nullvektor. Dabei ist $A = \set{e_i \mid i \in \set{0, \ldots, d}}$ aufspaltbar:

    Sei $A' \subseteq A$ beliebig. Dann wähle $a = (a_1, \ldots, a_d)$ und $b \in \IR$ wie folgt, sodass $r_{a, b} \cap A = A'$ und $r_{a, b} \cap (A \setminus A') = \emptyset$:
    \begin{equation*}
        a_i = \begin{cases}
            1 & e_i \in A' \\
            -1 & e_i \notin A' ,
        \end{cases}
        \quad \quad b = \begin{cases}
            0 & e_0 \in A' \\
            1 & e_0 \notin A'.
        \end{cases}
    \end{equation*}

    Dann ist $\inner{a, e_0} = 0$, passt also immer. Für $1 \leq i \leq d$ gilt
    \begin{equation*}
        \inner{a, e_i} = a_i = \begin{cases}
            1 \geq b & e_i \in A' \\
            -1 < b & e_i \notin A'.
        \end{cases}
    \end{equation*}
    Somit ist $\inner{a, e_i} \geq b \iff e_i \in A'$, und $A'$ kann immer durch einen Halbraum aufgespalten werden.
\end{proof}

\chapter{\acs{pac}-Lernbarkeit und $\varepsilon$-Netze}

Heute wollen wir endlich Satz~\ref{thm:pacLearnableForSubexponentialGrowthFct} beweisen.

Wir betrachten dafür das \emph{Gegenereignis}: Es existiert eine Hypothese $h \in \cH$ mit $\err_S(h) = 0$, aber $\err_{\cD, f}(h) \geq \varepsilon$.

Sei $\cR$ das Mengensystem, das $\cH$ entspricht, also sei $r_h = \set{x \in \cX \mid h(x) = 1}$ für $h \in \cH$. Sei $g = r_f$. Sei die \emph{symmetrische Differenz}\index{symmetrische Differenz} von $g$ und $r$ definiert als $g \Delta r_h \coloneqq (g \setminus r_h) \cup (r_h \setminus g)$.

\begin{expl}
    Sei $\cX = \IR$ und $\cR$ das Mengensystem der Intervalle. Sei $g \in \cR$ das Intervall $\edgys{a, b}$, das der Grundwahrheit entspricht. Sei $r_h \in \cR$ das Intervall $\edgys{a', b'}$ welches der Hypothese $h$ entspricht. Angenommen, $a \leq a' \leq b \leq b'$, dann ist $g \Delta r_h = \edgyround{a, a'} \cup \roundedgy{b, b'}$.
\end{expl}
Es gilt damit naheliegenderweise
\begin{equation*}
    \err_S(h) = \frac{1}{m} \abs{\set{h(x_i) \neq y_i}} = \frac{\abs{S \cap (g \Delta r_h)}}{\abs{S}}
\end{equation*}
und 
\begin{equation*}
    \err_{\cD, f}(h) = \IP_{x \sim \cD} [h(x) \neq f(x)] = \IP_{x \sim \cD} [x \in (g \Delta r_h)].
\end{equation*}

Damit das Gegenereignis bezüglich $h \in \cH$ eintritt, muss gelten 
\begin{enumerate}[$(i)$]
    \item $(g \Delta r_h) \cap S = \emptyset$, und
    \item $\IP_{x \sim \cD} [x \in (g \Delta r_h)] \geq \varepsilon$.
\end{enumerate}

Für den Beweis von Satz~\ref{thm:pacLearnableForSubexponentialGrowthFct} müssen wir über \emph{alle} Hypothesen argumentieren. Dafür definieren wir das Mengensystem aller möglichen so bildbaren symmetrischen Differenzen: Sei 
\begin{equation*}
    \cF \coloneqq \set{g \Delta r \mid r \in \cR}.
\end{equation*}

\begin{defi}{$\varepsilon$-Netz}{epsilonNet}\index{epsilon@$\varepsilon$-Netz}
    Sei $\cF$ ein Mengensystem mit Grundmenge $\cX$ und $\cD$ eine Wahrscheinlichkeitsverteilung über $\cX$. Sei $\varepsilon \in \edgys{0,1}$. Eine Menge $A \subseteq \cX$ ist ein \emph{$\varepsilon$-Netz für $\cF$}, wenn für jedes $r \in \cF$ mit $\IP_{x \sim \cD}[x \in r] \geq \varepsilon$ gilt, dass $A \cap r \neq \emptyset$.
\end{defi}

Damit können wir Satz~\ref{thm:pacLearnableForSubexponentialGrowthFct} wie folgt äquivalent formulieren:
\begin{addmargin}{.05\textwidth}
    \emph{Wenn Bedingung~$\eqref{eq:pacLearnableForSubexponentialGrowthFct}$ erfüllt ist, dann ist die Sample-Menge $S$, die \ac{iid} aus $\cD$ gezogen ist, mit Wahrscheinlichkeit mindestens $1 - \delta$ ein $\varepsilon$-Netz für $\cF$.}
\end{addmargin}

Beweisen wollen wir den Satz mithilfe der Wachstumsfunktion, dafür schauen wir zunächst auf folgendes Lemma:

\begin{lem}{Wachstumsfunktion des $\varepsilon$-Netzes}{GrowthFunctionOfEpsilonNet}
    Es gilt $\Pi_\cF(m) \leq \Pi_\cH(m)$ für alle $m \in \IN$.
\end{lem}
\begin{proof}
    Jede Menge in $r \in \cR$ erzeugt eine eindeutige Menge in $\cF$. Daraus folgt $\abs{\cF} \leq \abs{\cR}$. Das gleiche Argument gilt in der Beschränkung auf beliebiges $S \subseteq \cX$, also
    \begin{equation*}
        \abs{\cF|_S} = \abs{\set{(g \Delta r) \cap r \mid r \in \cR}} = \abs{\set{(g \cap S) \Delta r \mid r \in \cR|_S}} \leq \abs{\cR|_S}
    \end{equation*}
    und somit 
    \begin{equation*}
        \Pi_\cF(m) = \max_{\substack{S \subseteq \cX \\ \abs{S} = m}} \abs{\cF|_S} \leq \max_{\substack{S \subseteq \cX \\ \abs{S} = m}} \abs{\cR|_S} = \Pi_\cR(m) = \Pi_\cH(m),
    \end{equation*}
    was wir zeigen wollten.
\end{proof}

\begin{proof}[Beweis von Satz~\ref{thm:pacLearnableForSubexponentialGrowthFct}]
    Sei $\cF$ das Mengensystem der symmetrischen Differenzen wie oben definiert, beschrieben durch $\cH$ und $f$. Betrachte Samplemenge $S$ von $m$ Datenpunkten, \ac{iid} verteilt aus $\cD$, wobei $m$ Bedingung~\eqref{eq:pacLearnableForSubexponentialGrowthFct} erfüllt. Sei $A$ das folgende Ereignis:
    \begin{equation*}
        \exists r \in \cF\colon \IP_{x \sim \cD}[x \in r] \geq \varepsilon \quad \text{und} \quad \abs{r \cap S} = 0.
    \end{equation*}
    Falls $A$ eintritt, so ist $S$ kein $\varepsilon$-Netz für $\cF$. Für die Abschätzung definieren wir uns ein Hilfsereignis $B$. Dafür sei $S'$ eine andere Menge von $m$ Datenpunkten, die auch \ac{iid} aus $\cD$ gezogen werden. Sei $B$ das folgende Ereignis:
    \begin{equation*}
        \exists r \in \cF \colon \abs{r \cap S'} \geq \frac{\varepsilon m}{2} \quad \text{und} \quad \abs{r \cap S} = 0.
    \end{equation*}
    Wir verwenden die folgenden zwei Behauptungen, die im Anschluss gezeigt werden:
    \begin{clm}\label{clm:pacLearnableForSubexponentialGrowthFct-1}
        Es gilt $\IP[B \mid A] \geq \frac{1}{2}$. 
    \end{clm}
    \begin{clm}\label{clm:pacLearnableForSubexponentialGrowthFct-2}
        Es gilt $\IP[B] \leq \frac{\delta}{2}$
    \end{clm}
    Es folgt mit beiden Behauptungen 
    \begin{equation*}
        \IP[A] = \frac{\IP[A \cap B]}{\IP[B \mid A]} \leq \frac{\IP[B]}{\IP[B \mid A]} \leq \frac{\frac{\delta}{2}}{\frac{1}{2}} \leq \delta,
    \end{equation*}
    und damit ist der Satz gezegigt.
\end{proof}

Wir zeigen nun beide Behauptungen, brauchen dafür aber noch folgendes Lemma:
\begin{lem}{Summe von Bernoulli-Variablen}{sumOfBernoulliVariables}
    Seien $Z_1, \ldots, Z_m$ \ac{iid} Bernoulli-verteilt mit $\IP[Z_i = 1] = p$ für alle $i$. Es gilt für die Summe $Z = \sum_{i = 1}^m Z_i$
    \begin{equation*}
        \IP\left[ Z \leq \frac{pm}{2} \right] \leq \frac{4(1-p)}{pm}.
    \end{equation*}
\end{lem}
\begin{proof}
    Es gelten $\IE[Z] = pm$ und $\Var[Z] = p(1-p)m$. Die \v{C}eby\v{s}ev-Ungleichung besagt, dass für alle $a \in \IR$ 
    \begin{equation*}
        \IP[\abs{Z - \IE[Z]} \geq a] \leq \frac{\Var[Z]}{a^2}.
    \end{equation*}
    Wir setzen $a = \frac{pm}{2}$ und erhalten 
    \begin{equation*}
        \frac{\Var[Z]}{a^2} \leq \frac{p(1-p)m}{\left( \frac{pm}{2} \right)^2} = \frac{4(1-p)}{m}.
    \end{equation*}
    Das liefert uns 
    \begin{equation*}
        \IP\left[ Z \leq \frac{pm}{2} \right] = \IP\left[ \abs{Z - pm} \geq \frac{pm}{2} \right] = \IP\left[ \abs{Z - \IE[Z]} \geq \frac{pm}{2} \right] \leq \frac{4(1 - p)}{pm}. \qedhere
    \end{equation*}
\end{proof}

\begin{proof}[Beweis von Behauptung~\ref{clm:pacLearnableForSubexponentialGrowthFct-1}]
    Unter der Annahme, dass $A$ eingetreten ist, existiert $r \in \cF$, sodass $\abs{r \cap S} = 0$ und $\IP_{x \sim \cD} [x \in r] \geq \varepsilon$.

    Sei $Z_i = 1$ genau dann, wenn der $i$te Datenpunkt von $S'$ in $r$ fällt. Nach~\Cref{lem:sumOfBernoulliVariables} mit $p \geq \varepsilon$ und $Z = \abs{r \cap S'}$ erhalten wir 
    \begin{equation*}
        \IP\left[ \abs{r \cap S'} \leq \frac{\varepsilon m}{2} \right] \leq \IP\left[ \abs{r \cap S'} \leq \frac{pm}{2} \right] \leq \frac{4(1-p)}{pm} \leq \frac{1}{2},
    \end{equation*}
    da nach~\eqref{eq:pacLearnableForSubexponentialGrowthFct} gilt $m \geq \frac{8}{\varepsilon}$ und damit $pm \geq \varepsilon m \geq 8$.
\end{proof}

\begin{proof}[Beweis von Behauptung~\ref{clm:pacLearnableForSubexponentialGrowthFct-2}]
    Äquivalenter Weg, um $S$ und $S'$ zu bestimmen: Ziehe $2m$ Datenpunkte \ac{iid} gemäß $\cD$, sei das Ergebnis $T$. Ziehe nun $m$ mal \emph{ohne} Zurücklegen aus $T$, nenne das Ergebnis $S$. Schließlich setze $S' = T \setminus S$.

    Für festes $T$ und festes $r \in \cF$ setze $k \coloneqq \abs{r \cap T}$. Es kann nur $\abs{r \cap S'} \geq \frac{\varepsilon m}{2}$ gelten, wenn $k \geq \frac{\varepsilon}{2} m$ gilt. Es ist 
    \begin{equation*}
        \begin{split}
            \IP[\abs{r \cap S} = 0\mid T] &= \frac{2m-k}{2m} \cdot \frac{2m-k-1}{2m - 1} \cdot \cdots \cdot \frac{m - k + 1}{m + 1} \\
            &= \frac{m(m-1)\cdots(m-k+1)}{(2m)(2m-1)\cdots(2m-k+1)} \leq 2^{-k},
        \end{split}
    \end{equation*}
    weil sich alle Faktoren bis auf die ersten $k$ im Nenner und die letzten $k$ im Zähler wegkürzen. Für festes $r$ und $T$ erhalte 
    \begin{equation*}
        \IP\left[ \abs{r \cap S'} \geq \frac{\varepsilon m}{2} \text{ und } \abs{r \cap S} = 0 ~\middle|~ T \right] \leq \begin{cases}
            0 & k < \frac{\varepsilon}{2}m \\
            2^{-k} & \text{sonst}
        \end{cases} \leq 2^{-\frac{\varepsilon}{2}m}.
    \end{equation*}
    Nun ist Ereignis $B$ die Vereinignug all dieser Ereignisse über alle $r \in \cF$
    \begin{equation*}
        \IP[B \mid T] = \Pi_\cF(2m) 2^{-\frac{\varepsilon}{2}m} \leq \Pi_\cH(2m)2^{-\frac{\varepsilon}{2}m} \leq \frac{\delta}{2},
    \end{equation*}
    wobei die Ungleichungen durch~\eqref{eq:pacLearnableForSubexponentialGrowthFct} und~\Cref{lem:unionBound} und~\Cref{lem:GrowthFunctionOfEpsilonNet} folgen, da es genügt, Mengen zu zählen, die sich auf $T$ unterscheiden. Die Wachstumsfunktion davon ist beschränkt, da $\abs{T} = 2m$.
\end{proof}

\section{Einführendes Beispiel}
\begin{expl}\label{expl:boosting}
    Sei $\cX = \IR$ und $\cH$ die Klasse der Schwellenwertfunktionen 
    \begin{equation*}
        h(x) \geq \begin{cases}
            1 & x \geq a \\
            -1 & x < a
        \end{cases} \quad \text{oder} \quad h(x) \geq \begin{cases}
            1 & x \leq a \\ 
            -1 & x > a
        \end{cases}
    \end{equation*}
    Für $a \in \IR$ und $u \in \set{-1, 1}$ sei $h_{a, u}(x) = u \cdot \sgn(x - a)$ mit der Konvention $\sgn(0) = 1$
\end{expl}

Angenommen, die Grundwahrheit klassifiziert $x \in \edgys{a, b}$ positiv und $x \notin \edgys{a, b}$ negativ. Im Allgemeinen können wir in diesem Fall keinen Trainingsfehler unter $\frac{1}{3}$ erreichen, da immer ein Punkt aus $S = \set{(-1, -1), (0, 1), (1, -1)}$ falsch klassifiziert wird. Deswegen definieren wir 
\begin{equation*}
    h^* = \sgn(h_{a,1}(x) + h_{b,-1}(x) + h_{z, -1}(x))
\end{equation*}
mit $z < \min_i x_i$. Nun ist für $x \in \edgys{a,b}$ jeweils $h_{a, 1}(x) = h_{b, -1}(x) = 1$ und $h_{z, -1}(x) = -1$, also $h^*(x) = 1$. Für $x \in \rounds{z, a} \cup \rounds{b, \infty}$ folgt $h^*(x) = -1$. Damit ist für alle $i$ garantiert, dass $h^*(x_i) = y_i$ und somit $\err_S(h^*) = 0$ gilt. Da die Klassifizierung aber unterhalb von $z$ falsch ist, ist $h^*$ trotzdem noch nicht die Grundwahrheit.

\section{Problemstellung}

Sei $S = \set{(x_i, y_i) \mid i = 1, \ldots, m}$ mit $x_i \in \cX$ und $y_i \in \set{-1, 1}$ gegeben. 
\begin{defi}{Schwacher Lernalgorithmus}{weakLearningAlgo}\index{Lernalgorithmus!schwacher -}
    Ein Lernalgorithmus heißt \emph{schwach}, wenn er neben der Samplemenge $S$ auch beliebige Gewichte $p_1, \ldots, p_m \geq 0$ mit $\sum_{i = 1}^m p_i = 1$ erhält. 
\end{defi}
Die Gewichte definieren also eine Wahrscheinlichkeitsverteilung auf $S$. Der schwache Lernalgorithmus berechnet daraufhin eine Hypothese $h_p$, sodass 
\begin{equation*}
    \err_p(h_p) \coloneqq \sum_{\substack{i = 1 \\ h_p(x_i) \neq y_i}}^m p_i \leq \frac{1}{2} - \gamma
\end{equation*}
für ein festes $\gamma > 0$. Für $p_i = \frac{1}{m}$ ist $\err_p(h_p)$ genau der Trainingsfehler. Das heißt, die Hypothese stellt sicher, dass eine gewichtete Verteilung des Trainingsfehlers klein ist.

\begin{expl}
    Im Beispiel~\ref{expl:boosting} erfüllt ein $h \in \cH$, welches $\err_S(h)$ minimiert, diese Garantie mit $\gamma = \frac{1}{6}$:

    Es gilt für alle $p$ und alle $a < b$
    \begin{equation*}
        \sum_{i : x_i < a} p_i \leq \frac{1}{3} \quad \text{oder} \quad \sum_{i: a \leq x_i \leq b} p_i \leq \frac{1}{3} \quad \text{oder} \quad \sum_{i: x_i > b} p_i \leq \frac{1}{3}.
    \end{equation*}
    Zwei der drei Datenpunkte können wir leicht durch eine Schwellenwertfunktion klassifizieren. Der kleinste Fehler wird also höchstens $\frac{1}{3} = \frac{1}{2} - \frac{1}{6}$ sein.
\end{expl}

\section{Idee für einen Algorithmus}

\begin{defi}{Starker Lernalgorithmus}{strongLearningAlgo}\index{Lernalgorithmus!starker -}
    Seien $\varepsilon > 0$, die Samplemenge $S$ und ein schwacher Lernalgorithmus gegeben. Ein Lernalgorithmus heißt \emph{stark}, wenn er Hypothesen $h_1, \ldots, h_T$ und Koeffizienten $\alpha_1, \ldots \alpha_T$ berechnet, sodass für
    \begin{equation*}
        h^*(x) \coloneqq \sgn\left( \sum_{i = 1}^T \alpha_i h_i(x)\right)
    \end{equation*}
    gilt, dass $\err_S(h^*) \leq \varepsilon$.
\end{defi}

Die Idee für solch einen starken Lernalgorithmus ist wie folgt: Wir starten mit $p_i^{(1)} = \frac{1}{m}$ für alle $i$. Dann nutze den schwachen Lernalgorithmus, um die erste Hypothese $h_1$ zu berechnen.

Nun aktualisieren wir die $p_i$. Dafür definiere $w_i^{(t)}$ als die Gewichte, die in Iteration $t$ neu berechnet werden. Setze dann 
\begin{equation*}
    p_i^{(t)} \coloneqq \frac{w_i^{(t)}}{W^t} \quad \text{wobei} \quad W^t = \sum_{i = 1}^m w_i^{(t)}.
\end{equation*}

Dabei nutzen wir folgende Update-Regel:
\begin{equation*}
    w_i^{(t+1)} = \begin{cases}
        e^{-\eta} w_i^{(t)} & h_t(x_i) = y_i \\
        e^{\eta_t} w_i^{(t)} & h_t(x_i) \neq y_i,
    \end{cases}
\end{equation*}
wobei $\eta^t > 0$ für alle $i$ gleich ist. Wegen 
\begin{equation*}
    h_t(x_i) y_i = \begin{cases}
        1 & h_t(x_i) = y_i \\
        -1 & h_t(x_i) \neq y_i
    \end{cases}
\end{equation*}
ist 
\begin{equation*}
    w_i^{(t+1)} = w_i^{(t)} e^{-\eta_t y_i h_t(x_i)}.
\end{equation*}

\section{\acs{AdaBoost}}
Der Algoritmus~\ref{alg:adaboost} (\textsc{\acf{AdaBoost}}) benutzt genau diese Ideen:
\begin{algorithm}[h]
    \caption{\textsc{\acs{AdaBoost}}}\label{alg:adaboost}
    \begin{algorithmic}[1]
        \State{Initialisiere $w_i^{(1)}$ für alle $i$}
        \For{$t = 1, \ldots, T$}
            \State{$W^{(t)} \gets \sum_{i = 1}^m w_i^{(t)}, \quad p_i^{(t)} \gets \frac{w_i^{(t)}}{W^t}$ für alle $i$}
            \State{$h_t \gets$ vom schwachen Lernalgorithmus auf $p^(t)$ berechnetes Ergenbnis}
            \State{$\varepsilon_t \gets \sum_{i: h_t(x_i) \neq y_i} p_i^{(t)}$} \Comment{Fehler von $h_t$ auf $p^{(t)}$}
            \State{$\eta_t \gets \frac{1}{2} \ln\left( \frac{1}{\varepsilon_t} - 1 \right)$}
            \State{$w_i^{(t+1)} \gets w_i^{(t)} e^{-\eta_t h_t(x_i) y_i}$ für alle $i$}
        \EndFor{}
        \State{$h^*(x) \coloneqq \sgn\left(\sum_{t = 1}^T \eta_t h_t(x)\right)$}
        \State{\Return{$h^*$}}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{\acs{AdaBoost}}{adaBoost}
    \textsc{\ac{AdaBoost}} garantiert $\err_S(h^*) \leq e^{-2\gamma^2 T}$.
\end{thm}
\begin{proof}
    Wir werden zeigen, dass 
    \begin{equation}
        W^{t + 1} \leq e^{-2\gamma^2} W^t \label{eq:adaBoost}
    \end{equation}
    gilt.

    Mit~\eqref{eq:adaBoost} folgt 
    \begin{equation*}
        W^{(T + 1)} \leq e^{-2\gamma^2 T} W^{(1)} = e^{-2\gamma^2 T} \cdot m.
    \end{equation*}
    Betrachte $\err_S(h^*) = \frac{1}{m} \abs*{\set*{i \mid h^*(x_i) \neq y_i}}$. Wir zeigen als nächstes, dass die Anzahl falsch klassifizierter Datenpunkte durch $W^{(T + 1)}$ nach oben beschränkt ist. Dazu definiere 
    \begin{equation*}
        g_t(x) \coloneqq \sum_{t' = 1}^t \eta_{t'} h_{t'}(x).
    \end{equation*}
    Per Definition ist $h^* = \sgn(g_T)$ und $w_i^{(t)} = e^{-y_i g_{t-1}(x_i)}$.

    Es gilt für alle $i$ mit $h^*(x_i) \neq y_i$ wieder $y_i g_T(x_i) \leq 0$ und damit erhalten wir 
    \begin{equation*}
        w_i^{(T + 1)} = e^{-y_i g_T(x_i)} \geq 1.
    \end{equation*}
    Außerdem ist $w_i^{(T + 1)} \geq 0$ für alle $i$, und daher gilt 
    \begin{equation*}
        \abs*{\set{i \mid h^*(x_i) \neq y_i}} \leq \sum_{i = 1}^m w_i^{(t + 1)} = W^{(T + 1)}.
    \end{equation*}
    Damit folgt der Satz aus~\eqref{eq:adaBoost} und es bleibt lediglich~\eqref{eq:adaBoost} zu zeigen:
    \begin{equation*}
        W^{(t + 1)} = \sum_{i = 1}^m w_i^{(t + 1)} = \sum_{i = 1}^m w_i^{(t)} e^{-y_i \eta_t h_t(x_i)},
    \end{equation*}
    und für den Quotienten erhalten wir 
    \begin{equation*}
        \frac{W^{(t + 1)}}{W^{(t)}} = \sum_{i = 1}^m \frac{w_i^{(t)}}{W^{(t)}} e^{-y_i \eta_t h_t(x_i)} = \sum_{i = 1}^m p_i^{(t)} e^{-y_i \eta_t h_t(x_i)} = \sum_{i: h_t(x_i) = y_i} p_i^{(t)} e^{-\eta_t} + \sum_{i: h_t(x_i) \neq y_i} p_i^{(t)} e^{\eta_t}.
    \end{equation*}
    Nun gilt nach Definition $\sum_{i: h_t(x_i) \neq y_i} p_i^{(t)} = \varepsilon_t$ und $\eta_t$ so, dass $e^{\eta_t} = \sqrt{\frac{1}{\varepsilon_t} - 1}$, damit folgt
    \begin{equation*}
        \begin{split}
            \frac{W^{(t + 1)}}{W^{(t)}} &= (1 - \varepsilon_t) e^{-\eta_t} + \varepsilon_t e^{\eta_t} = (1 - \varepsilon_t) \frac{1}{\sqrt{\frac{1}{\varepsilon_t} - 1}} + \varepsilon_t \sqrt{\frac{1}{\varepsilon_t} - 1} \\
            &= (1 - \varepsilon_t) \sqrt{\frac{\varepsilon_t}{1 - \varepsilon_t}} + \varepsilon_t \sqrt{\frac{1 - \varepsilon_t}{\varepsilon_t}} = 2\sqrt{\varepsilon_t(1-\varepsilon_t)}.
        \end{split}
    \end{equation*}
    Setze nun $\gamma_t = \frac{1}{2}-\gamma_t$. Der schwache Lernalgorithmus garantiert $\gamma_t \geq \gamma$ und somit 
    \begin{equation*}
        \frac{W^{(t + 1)}}{W^{(t)}} = 2\sqrt{\varepsilon_t(1-\varepsilon_t)} = 2\sqrt{\left( \frac{1}{2}  + \gamma_t \right)\left(\frac{1}{2} - \gamma_t\right)} = \sqrt{1 - 4\gamma_t^2} \leq \sqrt{e^{-4\gamma_t^2}} = e^{-2\gamma_t^2} \leq e^{-2\gamma^2}
    \end{equation*}
    mit $1 - x \leq e^{-x}$. Damit ist~\eqref{eq:adaBoost} gezeigt und Satz~\ref{thm:adaBoost} vollständig bewiesen.
\end{proof} 

Man muss aber aufpassen, möglicherweise führt ein sehr großes $T$ einfach zu Overfitting, und nicht unbedingt zu kleinerem Fehler.
