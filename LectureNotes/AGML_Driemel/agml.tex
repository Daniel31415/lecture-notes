\section{Binäre Klassifikation}

\begin{expl}\leavevmode
    \begin{itemize}
        \item SPAM oder noSPAM für E-Mails
        \item Hund oder kein Hund für Fotos
    \end{itemize}
\end{expl}
Wir haben also immer die \textbf{Labels}\index{Label} \emph{positiv} oder \emph{negativ} für solche \textbf{Instanzen}\index{Instanz}.

\begin{defi}{Merkmalsraum}{propertySpace}\index{Merkmalsraum}
    Ein \emph{Merkmalsraum} $X$ ist eine Menge, die eindeutig alle zu untersuchenden Instanzen eines Problems darstellt.
\end{defi}

\begin{expl}\leavevmode
    \begin{itemize}
        \item Für die SPAM-Klassifikation könnten wir beispielsweise folgendermaßen klassifizieren:
        \begin{enumerate}
            \item $X$ ist die Menge aller Strings in UTF-8 Unicode-Codierung
            \item $X$ ist eine Menge von Vektoren, also $X \subseteq \IZ_{\geq 0}^d$, wobei ein Vektor aus $X$ angibt, ob und wie häufig bestimmte Wörter in der E-Mail vorkommen. Dabei sind beispielsweise Wörter wie \emph{price}, \emph{winner} oder ähnliches Teile des \emph{bag of words}\index{bag of words}, und ihnen ist ein Index $\in \set{1,\ldots,d}$ zugeordnet. Der entsprechende Vektor zählt also diese Häufigkeiten
        \end{enumerate}
        \item Für die Image-Klassifikation könnten wir beispielsweise folgendermaßen klassifizieren:
        \begin{enumerate}
            \item $X$ ist die Menge aller Bilder in RGB-Pixeldarstellung mit fester Auflösung, fester Größe und fester Farbtiefe (beispielsweise 1 Megapixel, 24 bit Farbtiefe führen zu $\abs{X} = \left(2^{24}\right)^{10^6}$, das ist tatsächlich sehr groß)
            \item $X$ ist eine Menge von Vektoren, also $X \subseteq \IZ_{\geq 0}^d$, wobei ein Vektor aus $X$ angibt, ob und wie häufig bestimmte \emph{SIFT descriptors} vorkommen
        \end{enumerate}
    \end{itemize}
\end{expl}

Letztendlich suchen wir eine \emph{Hypothese}\index{Hypothese} $h: X \to \set{1, -1}$, sodass
\begin{equation*}
    h(x) = \begin{cases}
        1 & \text{$x$ ist SPAM/Hund/\dots} \\
        -1 & \text{sonst}.
    \end{cases}
\end{equation*}

\begin{defi}{Hypothesenklasse}{hypothesisClass}
    Eine \emph{Hypothesenklasse}\index{Hypothesenklasse} $\cH$ ist eine Menge von Funktionen $X \to \set{1,-1}$.
\end{defi}

\begin{expl}
    Wir betrachten einige Beispiele für $\cH$.
    \begin{itemize}
        \item Es könnte $\cH = \set{f: X \to \set{1,-1}}$ sein, also einfach \emph{alle} solche Funktionen beinhalten. 
        
        \emph{Nachteil:} Wir haben zu wenig Struktur.
        \item Sei $X = \IR$ und $\cH = \set{h_a \mid a \in \IR}$, wobei 
        \begin{equation*}
            h_a(x) = \begin{cases}
            1 & x \geq a \\
            -1 & x < a.
        \end{cases}
        \end{equation*}
        \item Sei $X$ die Menge aller Bilder und $\cH$ die Menge aller Hypothesen $h_w$, die einem neuronalen Netz mit fester Architektur und einem Gewichtsvektor $w$ entsprichen.
    \end{itemize}
\end{expl}

Wie wählen wir nun also unsere Hypothesen aus? Was macht eine gute Hypothese aus?

\begin{defi}{Grundwahrheit}{groundTruth}
    Die \emph{Grundwahrheit}\index{Grundwahrheit} $f$ ist eine Funktion $f: X \to \set{1, -1}$. Wenn dieses $f$ existiert, weist $f$ jedem $x \in X$ das korrekte Label zu.
\end{defi}
Wir nehmen an, dass so ein $f$ existiert. 

\emph{Ziel:} Finde eine Hypothese $h$, die $f$ möglichst nahe kommt.

Wir nehmen an, dass $m$ Datenpunkte (\emph{Instanzen}) aus einer Wahrscheinlichkeitsverteilung $\cD$ über $X$ gezogen werden. Diese nennen wir \emph{Sample}\index{Sample}.


\begin{defi}{Diskrete Wahrscheinlichkeitsverteilung}{discreteDistribution}
    Eine \emph{diskrete Wahrscheinlichkeitsverteilung}\index{Wahrscheinlichkeitsverteilung}\index{Wahrscheinlichkeitsverteilung!diskrete -} $\cD$ über $X$ ist definiet über ein \emph{Wahrscheinlichkeitsmaß}\index{Wahrscheinlichkeitsmaß} $\IP: X \to \edgys{0,1}$, wobei $\IP(x) \neq 0$ nur für abzählbar viele Punkte gelten soll und 
    \begin{equation*}
        \sum_{x \in X} \IP(x) = 1
    \end{equation*}
    gelten soll.
\end{defi}
\begin{expl}
    Sei $X = \IR$ und $\IP(x) = \begin{cases}
        \frac{1}{100} & x \in \set{1,\ldots,100} \\ 0 & \text{sonst}
    \end{cases} = \1_{\set{1,\ldots,100}}(x)$.
\end{expl}

\begin{defi}{Tatsächlicher Fehler}{actualError}
    Der \emph{tatsächliche Fehler}\index{tatsächlicher Fehler}\index{Fehler!tatsächlicher -} $\err_{\cD, f}(h)$ einer Hypothese $h$ hinsichtlich Verteilung $\cD$ über Merkmalsraum $X$ und Grundwahrheit $f$ ist 
    \begin{equation*}
        \err_{\cD, f}(h) \coloneqq \IP_{x \sim \cD}[h(x) \neq f(x)].
    \end{equation*}
\end{defi}

\begin{expl}
    Es sei $f(x) = h_{30}$ und $h(x) = h_{50}$. Dann ist
    \begin{equation*}
        \err_{\cD, f}(h) = \frac{\abs{\set{30,\ldots,49}}}{\abs{\set{1,\ldots,100}}} = \frac{20}{100} = \frac{1}{5}.
    \end{equation*}
\end{expl}

\begin{prob}{Lernproblem}{learningProb}
    \prblmObj{$m$ Datenpunkte $x_1, \ldots, x_m$, die \emph{\ac{iid}} aus $\cD$ gezogen sind, und $y_1 = f(x_1), \ldots, y_m = f(x_m)$ (die \emph{korrekten} Labels). Sei $S = \set{(x_1, y_1), \ldots, (x_m, y_m)}$ die Menge der Samples\index{Sample} also gegeben}{Hypothese $h_S \in \cH$}{$\err_{\cD,f}(h_S)$ möglichst klein}
\end{prob}
\begin{defi}{Trainingsfehler}{trainingError}\index{Trainingsfehler}\index{Fehler!Trainings-}
    Der \emph{Trainingsfehler} $\err_S(h)$ einer Hypothese $h \in \cH$ hinsichtlich einer Trainingsmenge $S$ ist 
    \begin{equation*}
        \err_S(h) \coloneqq \frac{1}{m} \abs*{\set*{h(x_i) \neq y_i}}.
    \end{equation*}
\end{defi}

\section{Lernen von Schwellenwertfunktionen}

Sei im Folgenden $X = \IR$ unser Merkmalsraum.

In diesem Beispiel und auch im Folgenden vereinfachen wir uns das Leben durch die Annahme, dass die Grundwahrheit \emph{realisierbar}\index{realisierbar} ist. Das heißt, dass $f \in \cH$. Wir werden unsere Theorie später verallgemeinern, dass sie auch ohne diese Annahme auskommt. Wir nehmen also an, dass es ein $a \in \IR$ gibt, sodass die Grundwahrheit $f : X \to \set{-1, 1}$ gegeben ist durch
\begin{equation*}
    f(x) = \begin{cases}
        1 & x \geq a \\
        -1 & \text{sonst}.
    \end{cases}
\end{equation*}

Wir nehmen an, dass uns $m$ Datenpunkte mit korrekten Labels gegeben sind: Seien $x_1, \dots , x_m$ Datenpunkte, die \ac{iid} aus $\cD$ gezogen sind. Außerdem seien $y_i = f(x_i)$ die zugehörigen, korrekten Labels. Die Menge aller Samples bezeichnen wir mit $S = \set{(x_1, y_1), \dots ,(x_m, y_m)}$.

Ein einfacher Lernalgorithmus wählt nun das größte $a'$, sodass $h_{a'}$ die Menge $S$ korrekt klassifiziert. Das heißt, wir setzen $a'$ auf den Wert des kleinsten $x_i$ mit $y_i = 1$, wenn es ein solches gibt. Anderenfalls $a' = \infty$.

\begin{thm}{Abschätzung tatsächlicher Fehler der Schwellenwertfunktionen}{realErrorBoundTargetValueFunciton}
    Sei $h_{a'}$ die vom einfachen Lernalgorithmus berechnete Hypothese, der als Eingabe eine Menge von Samples von $m$ Datenpunkten mit korrekten Labels gemäß $f$ erhält, die \ac{iid} aus $\cD$ gezogen werden. Dann gilt für alle $\varepsilon > 0$, dass
    \begin{equation*}
        \IP[\err_{\cD, f}(h_{a'}) > \varepsilon] \leq e^{-\varepsilon m}.
    \end{equation*}
\end{thm}
\begin{proof}
    Weil unser Lernalgorithmus unter allen Hypothesen, die $S$ korrekt klassifizieren, die Hypothese $h_{a'}$ mit dem größten $a'$ wählt, wird auf jeden Fall gelten, dass $a' \geq a$. Falsch klassifiziert werden alle Punkte im Bereich $\edgyround{a, a'}$. Somit gilt nun für jede Verteilung $\cD$
    \begin{equation*}
        \err_{\cD, f}(h_{a'}) = \IP_{x \sim \cD}[x \in \roundedgy{a, a'}].
    \end{equation*}
    Sei nun $a'' \geq a$ die kleinste Zahl, sodass $\IP_{x \sim D}[x \in \rounds{a, a''}] \geq \varepsilon$. Es könnte sein, dass $a''$ nicht existiert, weil $\IP_{x \sim \cD}[x \in \edgyround{a, \infty}] < \varepsilon$. Dann ist aber die Wahrscheinlichkeit, dass der Fehler $\err_{\cD,f} (h_{a'})$ den Wert $\varepsilon$ überschreitet, gleich null, und die Aussage des Satzes gilt trivialerweise.

    Es bleibt zu zeigen, dass die Aussage des Satzes auch dann gilt, wenn $\cD$ dergestalt ist, dass $\IP_{x \sim D} [x \in \edgyround{a,\infty}] \geq \varepsilon$ und somit $a''$ wohldefiniert ist. Damit muss auch gelten $\IP_{x\sim D} [x \in \roundedgy{a, a''}] \leq \varepsilon$. Der Fehler von $h_{a'}$ wird also höchstens $\varepsilon$ sein, falls $a' \leq a''$. Dies geschieht, wenn es mindestens ein $i$ gibt, sodass $x_i \in \edgys{a, a''}$. Sei $\cE_i$ das Ereignis, dass $x_i \in \edgys{a, a''}$. Es gilt $\IP[\cE_i] = \IP_{x_i \sim \cD} [x_i \in \edgys{a, a''}] \geq \varepsilon$.
    Damit \emph{nicht} $a' \leq a''$ gilt, darf keines der Ereignisse $\cE_i$ eintreten. Wir interessieren uns also für $\bigcap_i \overline{\cE_i}$. Weil $x_1, \dots , x_m$ unabhängige Züge aus $\cD$ sind, gilt nun auch
    \begin{equation*}
        \IP\left[ \bigcap_i \overline{\cE_i} \right] = \prod_i \IP[\overline{\cE_i}] \leq (1 - \varepsilon)^m.
    \end{equation*}
    Mit $1 - x \leq e^{-x}$ für alle $x \in \IR$ folgt die Behauptung.
\end{proof}

\section{\acs{pac}-Lernbarkeit}
Satz~\ref{thm:realErrorBoundTargetValueFunciton} sagt uns nun insbesondere, dass wenn wir $m = \frac{1}{\varepsilon} \ln\left( \frac{1}{\delta} \right)$ wählen, die Wahrscheinlichkeit, dass der tatsächliche Fehler unserer gefundenen Hypothese größer als $\varepsilon$ ist, kleiner als $\delta$ wird. Diese Aussage gilt für alle $\varepsilon > 0$ und alle $\delta > 0$. Wenn die Anzahl der Samples also nur groß genug ist, werden wir mit großer Wahrscheinlichkeit nur einen sehr kleinen Fehler haben. Die Hypothesenklassen, für die dies gilt, heißen \emph{\ac{pac}-lernbar}\index{PAC-Lernbarkeit}:

\begin{defi}{\ac{pac}-Lernbarkeit}{pac-learnable}
    Eine Hypothesenklasse $\cH$ heißt \emph{\ac{pac}-lernbar} (im realisierbaren Sinn), wenn es eine Funktion $m_\cH$ und einen Lernalgorithmus $\cA$ gibt, sodass $A$ für alle $\varepsilon, \delta > 0$ jede Verteilung $\cD$ und alle $f \in \cH$, gegeben ein Sample $S$ von Größe mindestens $m_\cH(\varepsilon, \delta)$ von Datenpunkten mit korrekten Labels, eine Hypothese $h_S \in \cH$ berechnet, sodass 
    \begin{equation*}
        \IP[\err_{\cD, f}(h_S) < \varepsilon] \geq 1 - \delta.
    \end{equation*}
\end{defi}

\ac{pac}-Lernbarkeit nach dieser Definition impliziert nicht unbedingt, dass das Lernproblem effizient lösbar ist. Man spricht von \emph{effizienter \ac{pac}-Lernbarkeit}\index{PAC-Lernbarkeit!effiziente -}, wenn die Laufzeit von $\cA$ polynomiell ist und außerdem gilt, dass $m_\cH(\varepsilon, \delta)$ polynomiell in $\frac{1}{\varepsilon}$ und $\frac{1}{\delta}$ beschränkt ist.

\acs{pac} steht für \emph{\acl{pac}}. Dabei bedeutet \emph{probably} in diesem Fall, dass die Wahrscheinlichkeit mindestens $1 - \delta$ ist, und \emph{approximately correct} bezieht sich darauf, dass $\err_{\cD, f}(h_S) < \varepsilon$.

Nicht jede Hypothesenklasse ist \ac{pac}-lernbar. Zum Beispiel ist die Klasse aller Hypothesen $\IN \to \set{-1, 1}$ nicht \ac{pac}-lernbar, was wir noch beweisen werden.

\section{Lernen von Intervallen}

Wir betrachten als weiteres Beispiel nun die Klasse der Intervallfunktionen. Weiterhin sei $X = \IR$, aber $\cH$ sei nun die Menge der Funktionen der Form $h_{a', b'} : \IR \to \set{-1,1}$ mit 
\begin{equation*}
    h_{a', b'} = \begin{cases}
        1 & x \in \edgys{a', b'} \\
        -1 & \text{sonst}.
    \end{cases}
\end{equation*}
Wir betrachten weiterhin den realisierbaren Fall. Das heißt, es gibt eine Grundwahrheit $f \in \cH$, die alle Datenpunkte richtig klassifiziert. Diese hat die Form

\begin{equation*}
    f(x) = \begin{cases}
        1 & x \in \edgys{a, b} \\
        -1 & \text{sonst}.
    \end{cases}
\end{equation*}

Gegeben ist wieder eine Menge $S = \set{(x_1, y_1), \dots ,(x_m, y_m)}$ von Datenpunkten mit korrekten Labels. Unser Lernalgorithmus wählt das kleinste Intervall $\edgys{a', b'}$, das $S$ korrekt klassifiziert. Das heißt, wir setzen $a'$ auf den Wert des kleinsten $x_i$ mit $y_i = 1$ und $b'$ auf den Wert des größten $x_i$ mit $y_i = 1$. 
Den Fall, dass es kein $i$ mit $y_i = 1$ gibt, schieben wir ab in die Übung und ignorieren ihn.

\begin{thm}{Abschätzung tatsächlicher Fehler Intervalllernen}{realErrorBoundInterval}
    Für alle $\varepsilon > 0$ gilt 
    \begin{equation*}
        \IP[\err_{\cD, f}(h_{a', b'}) > \varepsilon] \leq 2e^{-\frac{\varepsilon m}{2}}.
    \end{equation*}
\end{thm}

\begin{lem}{Union Bound}{unionBound}
    Es seien $\cE_1, \ldots, \cE_n$ (nicht notwendigerweise disjunkte) Ereignisse. Dann gilt 
    \begin{equation*}
        \IP\left[ \bigcup_{i = 1}^n \cE_i \right] \leq \sum_{i = 1}^n \IP[\cE_i].
    \end{equation*}
\end{lem}
\begin{proof}
    Induktives Anwenden von $\IP[A \cup B] = \IP[A] + \IP[B] - \IP[A \cap B] \leq \IP[A] + \IP[B]$.
\end{proof}

\begin{proof}[Beweis von Satz~\ref{thm:realErrorBoundInterval}]
    Weil unser Lernalgorithmus das kleinste Intervall wählt, welches $S$ korrekt klassifiziert, wird auf jeden Fall gelten, dass $a' \geq a$ und $b' \leq b$. Falsch klassifiziert werden alle Punkte im Bereich $\edgyround{a, a'} \cup \roundedgy{b', b}$. Somit gilt nun für jede Verteilung $\cD$
    \begin{equation*}
        \err_{\cD, f}(h_{a', b'}) = \IP_{x \sim \cD}[x \in \edgyround{a, a'} \cup \roundedgy{b', b}].
    \end{equation*}
    Analog zu oben betrachten wir den Fall, dass $\IP_{x \sim \cD}[x \in \edgys{a,b}] < \varepsilon$. In diesem Fall ist die Wahrscheinlichkeit, dass der Fehler $\err_{\cD,f} (h_{a', b'})$ den Wert $\varepsilon$ überschreitet, wieder gleich null, und die Aussage des Satzes gilt trivialerweise.

    Es bleibt zu zeigen, dass die Aussage des Satzes auch dann gilt, wenn $\cD$ dergestalt ist, dass $\IP_{x\sim D} [x \in \edgys{a,b}] \geq \varepsilon$. Sei nun, ähnlich wie oben, $a'' \in \edgys{a,b}$ die kleinste Zahl, sodass $\IP_{x \sim D} [x \in \edgys{a, a''}] \geq \frac{\varepsilon}{2}$. Analog sei $b'' \in \edgys{a,b}$ die größte Zahl, sodass $\IP_{x \sim \cD} [x \in \edgys{b'', b}] \geq \frac{\varepsilon}{2}$. Damit $\err_{\cD,f} (h_{a', b'}) \leq \varepsilon$, ist es nun hinreichend, dass $a' \leq a''$ und $b' \geq b''$. Dies geschieht, wenn es mindestens ein $i$ und mindestens ein $j$ gibt, sodass $x_i \in \edgys{a, a''}$ und $x_j \in \edgys{b'', b}$ wobei $i$ und $j$ nicht verschieden sein müssen. Für jedes $i$ gilt
    \begin{equation*}
        \IP[x_i \in \edgys{a, a''}] \geq \frac{\varepsilon}{2} \quad \text{und} \quad \IP[x_i \in \edgys{b'', b}] \geq \frac{\varepsilon}{2}.
    \end{equation*}
    Sei $\cE_a$ das Ereignis, dass $x_1, \ldots, x_m \notin \edgys{a, a''}$ und sei $\cE_b$ das Ereignis, dass $x_1, \ldots, x_m \notin \edgys{b'', b}$. Weil $x_1, \ldots, x_m$ unabhängige Züge aus $\cD$ sind, gilt nun auch 
    \begin{equation*}
        \IP[\cE_a] \leq \left( 1 - \frac{\varepsilon}{2} \right)^m \quad \text{und} \quad \IP[\cE_b] \leq \left( 1 - \frac{\varepsilon}{2} \right)^m
    \end{equation*}
    und mit~\Cref{lem:unionBound} folgt 
    \begin{equation*}
        \IP[\cE_a \cup \cE_b] \leq 2\left( 1 - \frac{\varepsilon}{2} \right)^m.
    \end{equation*}
    Die Behauptung folgt wieder mit $1 - x \leq e^{-x}$.
\end{proof}

Aus dem Satz können wir folgern, dass auch die Klasse der Intervalle \ac{pac}-lernbar ist. Setzen wir etwa $m = \frac{2}{\varepsilon'}\ln\left( \frac{2}{\delta} \right)$, dann folgt aus Satz~\ref{thm:realErrorBoundInterval}, dass $\IP[\err_{\cD, f}(h_S) \leq \varepsilon'] \geq 1 - \delta$, wobei wir $\varepsilon'$ und $\delta$ als Parameter wählen können. Es gilt für jedes $\varepsilon' < \varepsilon$, dass 
\begin{equation*}
    \IP[\err_{D, f}(h_S) < \varepsilon] \geq \IP[\err_{\cD, f}(h_S) \leq \varepsilon'].
\end{equation*}
Wählen wir also beispielsweise $m_\cH(\varepsilon, \delta) = \frac{4}{\varepsilon} \ln\left( \frac{2}{\delta} \right)$, dann folgt $\IP[\err_{\cD, f}(h_S) < \varepsilon] \geq 1 - \delta$. Die Klasse der Intervalle ist also \ac{pac}-lernbar.

\chapter{Endliche Hypothesenklassen, Wachstumsfunktion und \acs{vc}-Dimension}

\section{Endliche Hypothesenklassen, Wachstumsfunktion}

Wir betrachten den Fall, dass $\cH$ endlich ist. 
\begin{thm}{Tatsächlicher Fehler bei endlicher Hypothesenklasse}{finiteHypothesisClassRealError}
    Wenn $m \geq \frac{1}{\varepsilon} \ln\left( \frac{\abs{\cH}}{\delta} \right)$, dann gilt mit Wahrscheinlichkeit mindestens $1 - \delta$, dass alle $h \in \cH$ mit $\err_S(h) = 0$ auch $\err_{\cD, f}(h) < \varepsilon$ erfüllen.
\end{thm}

\begin{proof}
    Sei $h \in \cH$ fest mit $\err_{\cD, f}(h) \geq \varepsilon$. Dann gilt, da $S$ \ac{iid} mittels $\cD$ gezogen wird, und mit der Voraussetzung $\IP[h(x_i) \neq y_i] \geq \varepsilon$
    \begin{equation*}
        \IP\left[ \err_S(h) = 0 \right] = \IP[h(x_1) = y_1, \ldots, h(x_m) = y_m] = \prod_{i = 1}^m \IP[h(x_i) = y_i] \leq (1 - \varepsilon)^m \leq e^{-\varepsilon m}.
    \end{equation*}
    Für jede Hypothese definiere $\cE_h$ als das Ereignis, dass $\err_S(h) = 0$ ist. Dann ist 
    \begin{equation*}
        \begin{split}
            &\IP[\exists h \in \cH: \text{$\err_S(h) = 0$ und $\err_{\cD, f}(h) \geq \varepsilon$}] = \IP\left[ \bigcup_{\substack{h \in \cH \\ \err_{\cD, f}(h) \geq \varepsilon}}  \cE_h \right]  \\
            \leq~& \sum_{\substack{h \in \cH \\ \err_{\cD, f}(h) \geq \varepsilon}} \IP[\err_S(h) = 0] \leq \abs{\cH} e^{-\varepsilon m} \leq \delta,
        \end{split}
    \end{equation*}
    also exakt, was wir zeigen wollten.
\end{proof}

\begin{defi}{Wachstumsfunktion}{growthFunction}
    Gegeben $A \subseteq X$, sei $\cH|_A$ die Menge aller Hypothesen $h \in \cH$ mit Definitionsbereich eingeschränkt auf $A$, also
    \begin{equation*}
        \cH|_A = \set*{h|_A \mid h \in \cH}.
    \end{equation*}
    Die \emph{Wachstumsfunktion}\index{Wachstumsfunktion} von $\cH$ ist definiert als 
    \begin{equation*}
        \Pi_\cH(m) \coloneqq \max_{\substack{A \subseteq X \\ \abs{A} = m}} \abs*{\cH|_A}.
    \end{equation*}
\end{defi}
Es gilt immer $\Pi_\cH(m) \leq 2^m$, es ist aber oft wesentlich kleiner.
\begin{expl}
    Sei $X = \IR$ und $\cH$ die Klasse der Schwellenwertfunktionen $h_a(x) = \begin{cases}
        1 & x \geq a \\ -1 & \text{sonst}.
    \end{cases}$ Für $A = \set{2,3,4}$ ist 
    \begin{equation*}
        \cH|_A = \set*{x \mapsto -1 \forall x \in A x \mapsto 1 \forall x \in A, x \mapsto \begin{cases}
            1 & x = 4 \\ -1 & x \in \set{2,3}
        \end{cases}, x \mapsto \begin{cases}
            1 & x \in \set{3,4} \\ -1 & x = 2
        \end{cases}}
    \end{equation*}
    und damit ist $\abs*{\cH|_A} = 4 \ll 2^3$. Allgemein ist für $\abs{A} = m$ für Schwellenwertfunktionen immer $\abs*{\cH|_A} = m + 1$.
\end{expl}
\section{Subexponentielles Wachstum impliziert \acs{pac}-Lernbarkeit}

\begin{thm}{\ac{pac}-lernbar wenn Wachstumsfunktion subexponentiell}{pacLearnableForSubexponentialGrowthFct}
    Es seien $\varepsilon, \delta > 0$ fest, und 
    \begin{equation*}
        m \geq \max\set*{\frac{8}{\varepsilon}, \frac{2}{\varepsilon}\log_2\left( \frac{2\Pi_\cH(2m)}{\delta} \right)}.
    \end{equation*}
    Betrachte eine Sample-Menge $S$ von $m$ Datenpunkten mit korrekten Labels gemäß der Grundwahrheit $f$, die \ac{iid} aus $\cD$ gezogen werden. Dann gilt mit Wahrscheinlichkeit $\geq 1 - \delta$, dass alle $h \in \cH$ mit $\err_S(h) = 0$ auch $\err_{\cD, f}(h) < \varepsilon$ erfüllen.
\end{thm}

Der Beweis folgt später. Wir fragen uns erstmal, wann die Ungleichung
\begin{equation}
    m \geq \frac{2}{\varepsilon} \log_2\left( \frac{2\Pi_\cH(2m)}{\delta} \right) = \frac{2}{\varepsilon} \log_2(\Pi_\cH(2m)) + \frac{2}{\varepsilon}\log_2\left( \frac{2}{\delta} \right)\label{eq:pacLearnableSubexponentialBoundary}
\end{equation}
erfüllt ist. Es ist~\eqref{eq:pacLearnableSubexponentialBoundary} äquivalent zu 
\begin{equation*}
    \frac{m - \frac{2}{\varepsilon}\log_2\left( \frac{2}{\delta} \right)}{\log_2(\Pi_\cH(2m))} \geq \frac{2}{\varepsilon}.
\end{equation*}
Wir testen ein paar Schranken aus.
\begin{enumerate}[$(a)$]
    \item Wenn $\Pi_\cH(2m) = 2^{2m}$, dann ist $\log_2(\Pi_\cH(2m)) = 2m$ und die Ungleichung kann nicht für alle $\varepsilon, \delta > 0$ erfüllt werden.
    \item Gilt $\log_2(\Pi_\cH(2m)) \in o(m)$, so müssen wir $m$ nur hinreichend groß wählen, um diese Schranke zu erfüllen. 
\end{enumerate}
\begin{expl}
    Bei den Schwellenwertfunktionen ist $\Pi_\cH(2m) = 2m + 1$. Dann gilt für alle $\delta > 0$
    \begin{equation*}
        \frac{m - \frac{2}{\varepsilon}\log_2\left( \frac{2}{\delta} \right)}{\log_2(2m + 1)} \overset{m \to \infty}{\longrightarrow} \infty.
    \end{equation*}
\end{expl}
Damit erhalten wir (mit dem Beweis von Satz~\ref{thm:pacLearnableForSubexponentialGrowthFct} als Black Box) \ac{pac}-Lernbarkeit für Hypothesenklassen, die die angesprochenen Ungleichungen erfüllen.

\section{Schranke über die \acs{vc}-Dimension}
Über die \acf{vc}-Dimension werden wir die Wachstumsfunktion beschränken (in mehr Details in der nächsten Vorlesung):

\begin{lem}{Sneak Peak: Wachstumslemma}{growthLemmaSneakPeak}
    Wenn $\cH$ \ac{vc}-Dimension $d$ hat, so gilt 
    \begin{equation*}
        \Pi_\cH(m) \leq \left( \frac{em}{d} \right)^d
    \end{equation*}
\end{lem}
Damit motivieren wir folgenden Satz:
\begin{thm}{\ac{pac}-Lernbarkeit mit \ac{vc}-Dimension}{PACusingVC}
    Sei $\cH$ eine Hypothesenklasse mit \ac{vc}-Dimension $d < \infty$. Seien $1 \geq \varepsilon > 0$, $\delta > 0$ und sei 
    \begin{equation}
        m \geq \max\set*{\frac{4}{\varepsilon} \log_2\left( \frac{2}{\delta} \right), \frac{8d}{\varepsilon}\log_2\left( \frac{16}{\varepsilon} \right)}. \label{eq:inequalityForMUsingVC}
    \end{equation}
    Betrachte eine Sample-Menge $S$ von $m$ Datenpunkten mit korrekten Labels gemäß der Grundwahrheit $f$, die \ac{iid} aus $\cD$ gezogen werden. Dann gilt mit Wahrscheinlichkeit $ \geq 1 - \delta$, dass alle $h \in \cH$ mit $\err_S(h) = 0$ auch $\err_{\cD, f}(h) < \varepsilon$ erfüllen.
\end{thm}
\begin{proof}
    Es genügt zu zeigen, dass~\eqref{eq:inequalityForMUsingVC} bereits~\eqref{eq:pacLearnableSubexponentialBoundary} impliziert (wenn wir später Satz~\ref{thm:pacLearnableForSubexponentialGrowthFct} später gezeigt haben).

    Zunächst, da $d \geq 1$ und $\varepsilon \leq 1$, gilt 
    \begin{equation*}
        m \geq \frac{8d}{\varepsilon}\log_2\left( \frac{16}{\varepsilon} \right) \geq \frac{8}{\varepsilon}.
    \end{equation*}
    Weiterhin gilt
    \begin{equation*}
        \frac{m - \frac{2}{\varepsilon}\log_2\left( \frac{2}{\delta} \right)}{\log_2(\Pi_\cH(2m))} \geq \frac{m}{2d\log_2\left( \frac{2em}{d} \right)} \geq \frac{\frac{8d}{\varepsilon}\log_2\left( \frac{16}{\varepsilon} \right)}{2d\log_2\left( \frac{2e}{d} \cdot \frac{8d}{\varepsilon}\log_2\left( \frac{16}{\varepsilon} \right) \right)} \geq \frac{4}{\varepsilon}\frac{\log_2\left( \frac{16}{\varepsilon} \right)}{\log_2\left(\frac{16e}{\varepsilon} \cdot \log_2\left(\frac{16e}{\varepsilon}   \right)\right)} \geq \frac{2}{\varepsilon},
    \end{equation*}
    da der Nenner monoton fallend in $\varepsilon$ und im Intervall $\rounds{0,1}$ mindestens $\frac{1}{2}$ ist.
\end{proof}

\subsection{Mengensysteme und deren \acs{vc}-Dimension}

\begin{defi}{Mengensystem, Aufspalten, Abspalten}{setSystem}
    Sei $X$ eine beliebige Menge und $\cR \subseteq \cP(X)$. Dann nennen wir $\cR$ ein \emph{Mengensystem mit Grundmenge $X$}\index{Mengensystem}.
    
    Eine Menge $A' \subseteq X$ wird \emph{durch $\cR$} von einer Menge $A$ \emph{abgespalten}\index{Abspaltung (Menge)}\index{Mengensystem!Abspaltung}, wenn $r \in \cR$ existiert, sodass $A' = r \cap A$.

    Eine Menge $A \subseteq X$ wird \emph{durch $\cR$ aufgespalten} (engl. \emph{Chattering})\index{Aufspaltung (Menge)}\index{Mengensystem!Aufspaltung}, wenn jede Teilmenge $A' \subseteq A$ von $A$ durch $\cR$ abgespalten wird.
\end{defi}

\begin{defi}{\acs{vc}-Dimension}{vcDimension}
    Die \emph{\acf{vc}-Dimension}\index{VC-Dimension} von $\cR$ ist die Anzahl der Elemente der größten aufgespaltenen Menge. Wir schreiben dafür $\dim(\cR)$.

    Wir setzen $\dim(\cR) \coloneqq \infty$, wenn für jedes $n_0 \in \IN$ eine aufgespaltene Menge mit $n_0$ Elementen existiert, und $\dim(\cR) \coloneqq 0$, wenn keine solche Menge existiert.
\end{defi}

\begin{expl}\leavevmode
    \begin{enumerate}
        \item Für $X = \IR^2$ und $\cR$ die Menge aller achsenparalleler Rechtecke. Betrachte $A \subseteq X$. Es ist $\dim(\cR) \geq 4$. Wenn wir mindestens fünf Punkte haben, so ist einer dieser Punkte in keiner Koordinate \enquote{extrem} bzw. \enquote{maximal}, und wir können also nie ein Rechteck um alle Punkte außer diesen Mittelpunkt bauen. Also ist $\dim(\cR) = 4$.
        \item Für $X = \IR^2$ und $\cR$ die Menge aller konvexen Polygone. Dann ist für jedes $n_0$ die Menge der $n_0$ Punkte, die ein regelmäßiges $n_0$-Eck bilden, aufspaltbar. Also ist $\dim(\cR) = \infty$.
    \end{enumerate}
\end{expl}

\subsection{Wachstum von endlichen Mengensystemen}
\begin{expl}\label{expl:setSystemSize}
    Sei $X = \set{1,\ldots,m}$ und sei $\cR$ das Mengensystem, das alle Teilmengen von maximaler Größe $k$ enthält. Dann ist $\dim(\cR) = k$ und 
    \begin{equation*}
        \abs{\cR} = \sum_{i = 0}^k \binom{m}{i} = \sum_{i = 0}^k \left( \frac{m}{k} \right)^i \frac{k^i}{i!} \leq \left(\frac{m}{k}\right)^k \sum_{i = 0}^k \frac{k^i}{i!} \leq \left( \frac{em}{k} \right)^k.
    \end{equation*}
    mit 
    \begin{equation*}
        \binom{m}{i} = \frac{m!}{(m-i)! i!} \leq \frac{m^i k^i}{i! k^i} \leq \left(\frac{m}{k}\right)^i\frac{k^i}{i!}
    \end{equation*}
    und der Reihendarstellung von $\exp$
    \begin{equation*}
        e^x = \sum_{i = 0}^\infty \frac{x^i}{i!}.
    \end{equation*}
    Damit wächst $\abs{\cR}$ subexponentiell in $m$ für konstante $k$.
\end{expl}

\begin{lem}{Größe von Mengensystemen}{sizeOfSetSystems}
    Es gilt für jedes Mengensystem $\cR$ mit $m$-elementiger Grundmenge $X$ und \ac{vc}-Dimension $d$, dass 
    \begin{equation*}
        \abs{\cR} \leq \sum_{i = 0}^d \binom{m}{i}.
    \end{equation*}
\end{lem}
\begin{proof}
    Wir machen Induktion über $m$.
    \begin{description}
        \item[IA: $m = 0$.] Dann ist $\abs{\cR} \leq 1$ und $d = 0$ und es gilt 
        \begin{equation*}
            \sum_{i = 0}^d \binom{0}{i} = \binom{0}{0} = 1.
        \end{equation*}
        \item[IS: $m > 0$.] Sei $\cR$ ein Mengensystem mit Grundmenge $X$ mit $\abs{X} = m$ und \ac{vc}-Dimension $d$. Angenommen, $d = 0$. Dann ist $\abs{\cR} \leq 1$ und die Aussage ist erfüllt, da $\binom{m}{i} \geq \binom{1}{0} = 1$. 
        
        Sei also auch $d > 0$. Sei $x \in X$ fest. Sei $\cR_1 \coloneqq \set{r \setminus \set{x} \mid r \in \cR}$ und sei $\dim(\cR_1) = d_1 \leq d$. Nach Induktionsvoraussetzung folgt 
        \begin{equation*}
            \abs{\cR_1} \leq \sum_{i = 0}^{d_1} \binom{m-1}{i} \leq \sum_{i = 0}^d \binom{m-1}{i}.
        \end{equation*}
        Definiere $\cR_2 \coloneqq \set{r \setminus \set{x} \mid r \cup \set{x} \in \cR \text{ und } r \setminus \set{x} \in \cR}$. Es gilt also $\abs{\cR} = \abs{\cR_1} + \abs{\cR_2}$. Dieses $\cR_2$ dient dazu, es auszugleichen, dass in $\cR_1$ durch Beschränkung von $\cR$ auf $X \setminus \set{x}$ zwei verschiedene Mengen identisch werden können. Diese Mengen werden in $\cR_2$ quasi doppelt gezählt und somit ausgeglichen.
        \begin{example}
            Für $X = \set{1,2,3}, x = 1$ und $\cR= \set{\set{1,2}, \set{2}, \set{1,3}}$, dann ist $\cR_1 = \set{\set{2}, \set{3}}$ und $\cR_2 = \set{\set{2}}.$
        \end{example}
        Sei $d_2 = \dim(\cR_2)$. Wir behaupten, dass $d_2 \leq d - 1$: Angenommen, $d_2 \geq d$. Dann gibt es $A \subseteq X \setminus \set{x}$ mit $\abs{A} = d$, sodass $A$ durch $\cR_2$ aufgespalten wird. Dann wird aber $A \cup \set{x}$ nach Definition von $\cR_2$ durch $\cR_1$ aufgespalten, und die \ac{vc}-Dimension von $\cR$ wäre somit mindestens $d_2 + 1 > d$. Widerspruch!
        
        Somit gilt nach Induktionsvoraussetzung
        \begin{equation*}
            \abs{\cR_2} \leq \sum_{i = 0}^{d_2} \binom{m-1}{i} \leq \sum_{i = 0}^{d-1} \binom{m-1}{i}.
        \end{equation*}
        Nun gilt mit $\binom{a}{b} = \binom{a-1}{b} + \binom{a-1}{b-1}$
        \begin{equation*}
            \begin{split}
                \abs{\cR} = \abs{\cR_1} + \abs{\cR_2} &\leq \sum_{i = 0}^d \binom{m-1}{i} + \sum_{i = 0}^{d-1} \binom{m-1}{i} = \sum_{i = 0}^d \binom{m-1}{i}+ \sum_{i = 1}^d \binom{m-1}{i-1} \\
                &= \binom{m-1}{0} + \sum_{i = 1}^d \binom{m-1}{i} + \binom{m-1}{i-1} = \binom{m}{0} + \sum_{i = 1}^d \binom{m}{i} = \sum_{i = 0}^d \binom{m}{i},
            \end{split}
        \end{equation*}
    \end{description}
    was wir zeigen wollten.
\end{proof}

\subsection{Unendliche Mengensysteme}
\begin{defi}{Untersystem}{subsystem}
    Sei $\cR$ ein Mengensystem mit Grundmenge $X$. Jede Menge $A \subseteq X$ bestimmt ein \emph{Untersystem}\index{Untersystem}\index{Mengensystem!Untersystem} von $\cR$ wie folgt:
    \begin{equation*}
        \cR|_A = \set{r \cap A \mid r \in \cR}
    \end{equation*}
\end{defi}
Es gilt $\dim(\cR|_A) \leq \dim(\cR)$.


Wir betrachten nun das Wachstumslemma (Sneak Peak war~\Cref{lem:growthLemmaSneakPeak}).

\begin{lem}{Wachstumslemma}{growthLemma}
    Sei $\cR$ ein Mengensystem mit Grundmenge $X$ und \ac{vc}-Dimension $d$. Für jedes $n \in \IN$ gilt 
    \begin{equation*}
        \Pi_\cR(m) \coloneqq \max_{\substack{A \subseteq X \\ \abs{A} = m}} \abs{\cR|_A} \leq \left( \frac{em}{d} \right)^d.
    \end{equation*}
    Wir nennen $\Pi$ die \emph{Wachstumsfunktion} von $\cR$.\index{Wachstumfsunktion}
\end{lem}

\begin{proof}
    Für jedes $A \subseteq X$ mit $\abs{A} = m$ gilt 
    \begin{equation*}
        \abs{\cR|_A} \leq \sum_{i = 0}^d \binom{m}{i} \leq \left( \frac{em}{d} \right)^d
    \end{equation*}
    nach~\Cref{lem:sizeOfSetSystems} und der Rechnung aus Beispiel~\ref{expl:setSystemSize}. Es solft somit direkt $\Pi_\cR(m) \leq \left( \frac{em}{d} \right)^d$. 
\end{proof}

Nun folgt mit $\cR \coloneqq \set{r_h \mid h \in \cH}$, wobei $r_h \coloneqq \set{x \in X \mid h(x) = 1}$, auch direkt $\Pi_\cH(m) = \Pi_\cR(m)$ und damit~\Cref{lem:growthLemmaSneakPeak}.
