\chapter{Das Sweep-Verfahren}
\section{Einführung}

\newLecture{14.04.2026}

Das \emph{Sweep}\index{Sweep}-Verfahren ist ähnlich vielseitig wie \emph{Divide and Conquer}, also ein hilfreiches Pardigma. Divide and Conquer ist beispielsweise sinnvoll, wenn die erwarteten Kosten des Ergebnisses klein sind. Beispielsweise erhalten wir $T(n) \leq 2 \cdot T(n/2) + c(\text{merge})$. 

\emph{Sweep} ist ein geometrisches Verfahren, bei dem wir lokal eine Dimensionsreduktion von einem $n$-dimensionalen Problem auf ein $(n-1)$-dimensionales Problem verwandeln, indem wir die Ebene mit einer \emph{Sweepline} ausfegen, indem wir gemäß einer vorher angefertigten Sortierung fegen. Dabei folgt die Korrektheit des Verfahrens über eine Invariante nahe der Sweepline.

\section{Sweep im Eindimensionalen}
\subsection{Das Maximum einer Menge von Objekten}

\begin{prob}{Find maximum}{findMax}
    \prblm{$Q = [q_1, \ldots, q_n] \in \IR^n$}{$q_i \in Q$ mit $q_i \geq q_j \forall j = 1,\ldots,n$}
\end{prob}

Hierfür haben wir folgenden optimalen Algorithmus mit Laufzeit $\Theta(n)$. Dabei sweepen wir über die Indizes, erhalten also lokal ein $0$-dimensionales Problem

\begin{algorithm}[h]
    \caption{\textsc{FindMax}}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:findMax}}
        \Output{Wie in~\Cref{prob:findMax}}
        \State{$\MaxSF \gets q[1]$}
        \For{$j = 2$ \textbf{to} $n$ \textbf{do}}
            \If{$\MaxSF < q[j]$}
                \State{$\MaxSF \gets q[j]$}
            \EndIf{}
        \EndFor{}
        \State{\Return{$\MaxSF$}}
    \end{algorithmic}
\end{algorithm}

\subsection{Das dichteste Paar in einer Menge von Zahlen}

 
\begin{prob}{Closest pair in einer Dimension}{closestPair1D}
    \prblm{$x_1, \ldots, x_n \in \IR$}{$i \neq j \in \set{1,\ldots,n}$ sodass $\displaystyle\abs{x_i - x_j} = \min_{\ell \neq k} \abs{x_\ell - x_k}$ für $\ell, k \in \set{1,\ldots,n}$}
\end{prob}

Wir sortieren die Zahlen aufsteigend, sodass wir ein neues Array $x_1' \leq  \cdots \leq x_n' \in \IR$ bekommen (im Mehrdimensionalen sortieren wir sie nach einer bestimmten Koordinate). Das braucht aber natürlich bereits $\cO(n \log(n))$ Zeit.

\begin{algorithm}[h]
    \caption{\textsc{FindClosestPair-1D}}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:closestPair1D}, aber als aufsteigend sortiertes Array $x'$}
        \Output{Wie in~\Cref{prob:closestPair1D}}
        \State{$\MinDistSF \gets x'[2] - x'[1]$}
        \State{$\ClosPos \gets 2$}
        \For{$j = 3$ \textbf{to} $n$ \textbf{do}}
            \If{$\MinDistSF > x'[j] - x'[j-1]$}
                \State{$\MinDistSF \gets x'[j] - x'[j-1]$}
                \State{$\ClosPos \gets j$}
            \EndIf{}
        \EndFor{}
        \State{\Return{$x'[\ClosPos-1], x'[\ClosPos]$}}
    \end{algorithmic}
\end{algorithm}

\begin{cor}
    Die Berechnung des \textsc{Closest Pair}s von $n$ reellen Zahlen benötigt Laufzeit $\cO(n \log(n))$.
\end{cor}

\subsection{Die maximale Teilsumme}

\begin{prob}{Maximum subvector}{maximumSubvector}
    \prblm{Array $V \in \IR^n$}{Finde $i \leq j$, sodass $\displaystyle V[i] + V[i+1] + \cdots + V[j] = \max_{\ell \leq k} V[\ell] + \cdots + V[k]$, mit $i, j, \ell, k \in \set{1,\ldots,n}$}
\end{prob}

\begin{algorithm}[h]
    \caption{\textsc{FindMaximumSubvector}}
    \begin{algorithmic}[1]
        \State{$\MaxSF \gets 0$}
        \State{$\MaxEndHere \gets 0$}
        \For{$j = 1$ \textbf{to} $n$ \textbf{do}}
            \State{$\MaxEndHere \gets \max(0, \MaxEndHere + V[j])$}
            \State{$\MaxSF \gets \max(\MaxSF, \MaxEndHere)$}
        \EndFor{}
        \State{\Return{$\MaxSF$}}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{FindMaximumSubvector}{findMaxSubvec}
    In einer Folge von $n$ Zahlen die maximale Teilsumme konsekutiver Zahlen zu bestimmen, hat die Zeitkomplexität $\Theta(n)$.
\end{thm}



\section{Sweep in der Ebene}

Im Allgemeinen ist $\abs{x} \coloneqq \norm{x}_2$ die Euklidische Norm des Vektor $x$.

\subsection{Das dichteste Punktepaar in der Ebene}
 
\begin{prob}{Closest pair in zwei Dimensionen}{closestPair2D}
    \prblm{$x_1, \ldots, x_n \in \IR^2$}{$i \neq j \in \set{1,\ldots,n}$ sodass $\displaystyle\abs{x_i - x_j} = \min_{\ell \neq k} \abs{x_\ell - x_k}$ für $\ell, k \in \set{1,\ldots,n}$}
\end{prob}

Wir führen zum Verwalten ein Verzeichnis ein, in dem zu jedem Zeitpunkt den Status der \emph{Sweep Line} tracked. Dieses heißt \emph{\ac{sss}}. Übergeordnet führen wir eine \emph{\ac{es}} ein, in dieser sind einfach alle Punkte nach ihrer $x$-Koordinate sortiert. Die \ac{sss} teilt diese in Streifen der Breite $\MinSF$ ein. Erreicht ein neuer Punkt solch einen Streifen, testen wir unter allen Punkten im Streifen, welche davon den geringsten Abstand haben. Das aktualisiert $\MinSF$ und der Streifen der \ac{sss} wird entsprechend schmaler.

Wir erhalten den folgenden Algorithmus:

\begin{algorithm}[h]
    \caption{\textsc{FindClosestPair-2D}}
    \begin{algorithmic}[1]
        \Input{Array von $n$ Punkten aus $\IR^2$}
        \Output{Zwei Elemente des Arrays mit dem geringsten euklidischen Abstand}
        \State{Sortiere die $n$ Punkte nach aufsteigenden $x$-Koordinaten und füge sie ins Array $P$ ein. Das bildet die \ac{es}}
        \State{Füge $P[1], P[2]$ in \ac{sss} ein}
        \State{$\MinSF \gets \abs{P[1] - P[2]}$}
        \State{$\links \gets 1$}
        \State{$\rechts \gets 3$}
        \While{$\rechts \leq n$}
            \If{$P[\links].x + \MinSF \leq P[\rechts].x$}
                \State{Entferne $P[\links]$ aus \ac{sss}}
                \State{$\links \gets \links + 1$}
            \Else{}
                \State{$\MinSF \gets \MinDist(\text{\ac{sss}}, P[\rechts], \MinSF)$}
                \State{Füge $P[\rechts]$ in \ac{sss} ein}
                \State{$\rechts \gets \rechts + 1$}
            \EndIf{}
        \EndWhile{}
        \State{\Return{$\MinSF$}}
    \end{algorithmic}
\end{algorithm}


\newLecture{16.04.2026}

Zu spezifizieren ist noch \MinDist. Wir müssen uns dafür Gedanken darüber machen, welche Datenstruktur wir am besten verwenden, um die \ac{sss} zu verwalten. Eine Liste hätte lineare Laufzeit, um die Nachbarn zu prüfen, das kann im schlimmsten Fall $\cO(n^2)$ pro Ausführung von \MinDist benötigen kann.

Sei $r \coloneqq P[\rechts]$ und $M \coloneqq \MinSF$. Dann genügt es, für $\MinDist(\text{\ac{sss}}, r, M)$ nur die Punkte aus \ac{sss} anzuschauen, deren $x$-Koordinate in $\edgys{r.x - M, r.x + M}$ liegt. Folgende Operationen muss unsere \ac{sss} unterstützen:
\begin{itemize}
    \item $\MinDist(\text{\ac{sss}}, r, M)$
    \item $\Insert(p)$
    \item $\Delete(p)$
\end{itemize}
Wir verwenden als Datenstruktur einen balancierten Binärbaum über die $y$-Koordinaten als Schlüssel. Damit benötigen $\Insert$ und $\Delete$ jeweils $\cO(\log(n))$, und Ausführung der Bereichsanfrage nach allen $p.y$ im Invervall $\edgys{r.x - M, r.x + M}$ benötigt Laufzeit $\cO(\log(n) + k)$, wobei $k$ die Größe der Antwort bedeutet.

Der betrachtete Streifen des Rechtecks um $r$ hat einen Flächeninhalt von $2M^2 = 2M \cdot M$. Ein Punkt aus dem Rechteck deckt mit seinem Kreis mit Radius $\frac{M}{2}$ mindestens einen Viertelkreis als Flächeninhalt des Rechtecks ab, also mindestens $\pi\frac{M^2}{16}$. Wir erhalten mit
\begin{equation*}
    k \leq \frac{2M^2}{\pi\frac{M^2}{16}} \leq \frac{32}{\pi} < 11,
\end{equation*}
also ist $k \leq 10\in \cO(1)$ und die Bereichsanfrage benötigt auch nur $\cO(\log(n))$ Zeit. Die Datenstruktur für \ac{sss} in diesem Fall ist also ein \emph{\ac{avl}-Baum}. Wir erhalten folgendes Lemma:

\begin{lem}{Höchstens $10$ Punkte in Rechteck}{max10ptsInRect}
    Sei $M > 0$ und $P$ eine Menge von Pnukten im $\IR^2$, von denen je zwei den Abstand $\geq M$ haben. Ein Rechteck mit Kantenlängen $M$ und $2M$ enthält höchstens $10$ Punkte.
\end{lem}

\begin{thm}{Laufzeit FindClosestPair}{findClosestPairRuntime}
    Der minimale Abstand aller Paare einer $n$-elementigen Punktmenge in der Ebene lässt sich in Zeit $\cO(n \log(n))$ bestimmen.
\end{thm}

\subsection{Schnittpunkte von Strecken}

\begin{prob}{Line Segment Intersection in zwei Dimensionen}{lineSegmentIntersection2D}
    \prblm{$n$ Liniensegmente $s_i = l_i r_i$ in der Ebene für $i \in \set{1,\ldots,n}$}{Berechne die \emph{echten Schnittpunkte}, also Punkte, die zum relativen Inneren beider Strecken bestehen}
\end{prob}
Wir nehmen an, dass keine Mehrfachschnittpunkte vorliegen und nie gleiche $x$-Koordinaten an beiden Segmentenden vorkommen. Außerdem sollen die geraden in \emph{allgemeiner Lage}\index{allgemeine Lage} liegen, sie weisen also keinerlei geometrische Besonderheiten wie Parallelität oder Orthogonalität auf.

Im \emph{Existenzproblem}\index{Existenzproblem} fragen wir uns, ob es unter den $n$ Strecken mindestens zwei gibt, die einen echten Schittpunkt haben, bein \emph{Aufzählungsproblem}\index{Aufzählungsproblem} wollen wir \emph{alle} echten Schnittpunkte bestimmen.

Wir fragen uns wieder nach einer sinnvollen Wahl von \acl{es} und \acl{sss}: Die \ac{es} bestehe aus den linken und rechten Endpunkten $l_i$ und $r_i$, insbesondere deren $x$-Koordinaten. Als \ac{sss} wählen wir die Segmente gemäß einer noch festzulegenden Ordnung. Außerdem gibt es drei Typen von Ereignissen, die auftreten können:
\begin{enumerate}
    \item Neuer linker Endpunkt eines Segments gefunden (\LeftEnd),
    \item Neuer rechter Endpunkt eines Segments gefunden (\RightEnd),
    \item Neuer Schnittpunkt gefunden (\Intersect).
\end{enumerate}

\begin{lem}{Echte Schnittpunkte sind Nachbarn}{intersectAreNeighbours}
    Haben zwei Liniensegmente einen echten Schnittpunkt zum Zeitpunkt $t$, dann sind die Segmente unmittelbar vor $t$ Nachbarin in der \ac{sss} geworden.
\end{lem}

wir fragen uns jetzt, welche Operationen wir für die Lösung des \emph{Existenzproblems} unterstützen müssen:
\begin{itemize}
    \item $\Insert(\text{\ac{sss}}, seg, x)$ fügt Segment $seg$ entsprechend der Ordnung zur Zeit $x$ in die \ac{sss} ein.
    \item $\Delete(\text{\ac{sss}}, seg, x)$ entfernt Segment $seg$ zur Zeit $x$ aus der \ac{sss}.
    \item $\Prec(\text{\ac{sss}}, seg, x)$ bestimmt den Vorgänger von $seg$ in der \ac{sss} zum Zeitpunkt $x$.
    \item $\Succ(\text{\ac{sss}}, seg, x)$ bestimmt den Nachfolger von $seg$ in der \ac{sss} zum Zeitpunkt $x$.
    \item $\IntersectTest(seg_1, seg_2)$ testet auf Schnitt zwischen $seg_1$ und $seg_2$.
\end{itemize}

Für die Implementierun der \ac{sss} verwenden wir wieder einen \ac{avl}-Baum als balancierten Suchbaum, also ist jede der Operationen in $\cO(\log(n))$ ausführbar, weil höchstens $n$ Segmente in der \ac{sss} gespeichert sind und weil der Suchbaum balanciert ist. Da die Bearbeitung von jedem der höchstens $2n$ vielen Ereignisse nur drei solche Operationen erfordert, erhalten wir folgende Aussage:

\begin{thm}{Existenzproblem Line Segment Intersection}{existanceProblemLineSegmentIntersection}
    Man kann in optimaler Zeit $\cO(n \log(n))$ und mit Speicherplatz $\cO(n)$ herausfinden, ob von $n$ Strecken in der Ebene mindestens zwei einen echten Schnittpunkt haben.
\end{thm}

Dabei haben \Insert{} und \Delete{} wieder Laufzeit $\cO(\log(n))$, während $\IntersectTest$ nur $\cO(1)$ benötigt.

Das Lösen des \emph{Aufzählungsproblem} ist etwas anspruchsvoller. Nach einem Schnittpunkt wechselt die Ordnung zwischen zwei Segmenten, das vorher obere ist dann das neue untere Segment. Die konkreten Instanzen der drei oben beschriebenen Typen von Ereignissen, die während des Sweep zu behandeln sind, sind nicht alle schon von Anfang bekannt, deswegen brauchen wir eine \emph{dynamische} \ac{es}. Außer Initialisierung und dem Test, ob \ac{es} leer ist, sind folgende Operationen möglich:
\begin{itemize}
    \item $\Insert(\text{\ac{es}},\event)$ fügt das Ereignis \event{} entsprechend seiner Eintrittszeit in die \ac{es} ein.
    \item $\NextEvent$ liefert das zeitlich erste in \ac{es} gespeicherte Ereignis und entfernt es aus der \ac{es}.
\end{itemize}

Dafür werden wir eine \emph{Priority Queue} verwenden.

Zusammen haben wir folgenden Algorithmus:

\begin{algorithm}[h]
    \caption{\textsc{FindAllIntersections-2D}}\label{alg:findAllIntersections2D}
    \begin{algorithmic}[1]
        \State{Initialisiere \ac{sss} und \ac{es}}
        \State{Sortiere die $2n$ Endpunkte nach aufsteigenden $x$-Koordinaten}
        \State{Erzeuge daraus \event{}s und füge diese in die \ac{es} ein}
        \While{$\text{\ac{es}} \neq \emptyset$}
            \State{$\event \gets \NextEvent(\text{\ac{es}})$}
            \If{\event{} \textbf{is} \LeftEnd}
                \State{$\Insert(\text{\ac{sss}}, seg, t)$}
                \State{$prevSeg \gets \Prec(\text{\ac{sss}}, seg, t)$}
                \State{$\IntersectTestCreateEvent(prevSeg,seg)$}
                \State{$nextSeg \gets \Succ(\text{\ac{sss}}, seg, t)$}
                \State{$\IntersectTestCreateEvent(seg,nextSeg)$}
            \Else{} 
                \If{\event{} \textbf{is} \RightEnd}
                    \State{$prevSeg \gets \Prec(\text{\ac{sss}}, seg, t)$}
                    \State{$nextSeg \gets \Succ(\text{\ac{sss}}, seg, t)$}
                    \State{$\Delete(\text{\ac{sss}}, seg, t)$}
                    \State{$\IntersectTestCreateEvent(prevSeg,nextSeg)$}
                \Else{}
                    \State{$(uSeg, lSeg) \gets (\text{vorher oberes Segment}, \text{vorher unteres Segment})$}
                    \State{Nehme $(uSeg, lSeg)$ als Paar mit Schnitt auf}
                    \State{$prevSeg \gets \Prec(\text{\ac{sss}}, uSeg, t)$}
                    \State{$\IntersectTestCreateEvent(prevSeg,uSeg)$}
                    \State{$nextSeg \gets \Succ(\text{\ac{sss}}, uSeg, t)$}
                    \State{$\IntersectTestCreateEvent(lSeg,nextSeg)$}
                \EndIf{}
            \EndIf{}
        \EndWhile{}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{FindAllIntersections-2D}{findAllIntersections2D}
    Das Sweep-Verfahren berechnet die $k$ echten Schnittpunkte von $n$ Liniensegmenten in Zeit $\cO((n + k) \log(n))$ mit Platz $\cO(k + n)$.
\end{thm}

\begin{proof}
    Insgesamt gibt es $2n+k$ Ereignisse. Weil keines von ihnen mehrfach in der Ereignis-Struktur vorkommt (da jeder Versuch, ein schon vorhandenes Element erneut einzufügen, von der Datenstruktur abgewiesen wird), sind niemals mehr als $\cO(n^2)$ Ereignisse in der \ac{es} gespeichert. Jeder Zugriff auf die \ac{es} ist also in Zeit $\cO(\log(n^2)) = \cO(\log(n))$ ausführbar.
    
    Die \textbf{while}-Schleife wird $(2n + k)$-mal durchlaufen; bei jedem Durchlauf werden höchstens sieben Operationen auf der \ac{sss} oder der \ac{es} ausgeführt. Daraus folgt die Behauptung.
\end{proof}

\begin{thm}{FindAllIntersections-2D weniger Platzbedarf}{findAllIntersections2DLessSpace}
    Das Sweep-Verfahren berechnet die $k$ echten Schnittpunkte von $n$ Liniensegmenten in Zeit $\cO((n + k) \log(n))$ mit Platz $\cO(k + n)$.
\end{thm}

Wir können also auch degenerierte\index{degeneriert} Fälle behandeln.

\subsection{Einschub -- Untere Schranken}

Wir bekommen für Closest Pair in 2D einen $\cO(n \log(n))$ Sweep-Algorithmus. Für den Schnitt von Liniensegmenten bekommen wir Sweep-Algorithmen mit $\cO(n \log(n))$ für das Existenzproblem und $\cO((n + k) \log(n))$ für das Aufzählungsproblem. Für die Frage, ob es bessere Algorithmen gibt, wollen wir Lower Bounds erarbeiten, dafür folgt jetzt etwas Technik.

Außerdem wollen wir den Begriff der \emph{allgemeinen Lage}\index{allgemeine Lage} besser definieren, um \emph{degenerierte Fälle}\index{degeneriert} behandeln, beispielsweise Mehrfachschnittpunkte. Dabei können wir entweder \emph{intrinsisch}\index{intrinsisch}, also \emph{im} Algorithmus stattfinden (beispielsweise durch \emph{lexikographische Ordnung}\index{lexikographische Ordnung}, also Sortierung nach $x$-Koordinaten, und bei Gleichheit weiter nach $y$-Koordinaten); oder wir arbeiten \emph{extrinsisch}\index{extrinsisch}, das heißt, wir manipulieren die Eingabe.

Wir erinnern uns an die Landau-Symbole. Es ist $g \in \Omega(f)$ genau dann, wenn ein $n_0 \geq 0$ und $c > 0$ existieren mit $g(n) \geq c \cdot f(n)$ für alle $n \geq n_0$.

Für einen Algorithmus $A$, ein Problem $\Pi$ und $P \in \Pi$ definieren wir $T_A(n) \coloneqq \max\set{T_A(P) \mid \abs{P} = n}$ als die Anzahl wesentlicher Opertionen.

Für die Sortierung von $n$ paarweise verschiedenen Objekten $q_1, \ldots, q_n$ (\enquote{in allgemeiner Lage}), die eine \emph{totale Ordnung} besitzen. Wir sagen
\begin{equation*}
    K(i, j) = \begin{cases}
        1 & q_i < q_j \\ 
        -1 & q_j < q_i,
    \end{cases}
\end{equation*}
diese Operation (\emph{Schlüsselvergleich}) benötigt $\cO(1)$ Operationen. Wir bestimmen eine Permutation $\pi: \set{1,\ldots,n} \bij \set{1,\ldots,n}$, sodass $q_{\pi(1)} < q_{\pi(2)} < \cdots < q_{\pi(n)}$ gilt.

% \setcounter{algorithm}{3}
\begin{thm}{Sortieren durch Schlüsselvergleiche}{keyCompareSorting}
    Sortieren durch Schlüsselvergleiche hat Komplexität $\Omega(n \log(n))$.
\end{thm}

\begin{proof}
    Sei $A$ ein deterministischer Algorithmus zur Sortierung durch Schlüsselvergleiche. Wir definieren $W(i, j) \coloneqq \set{(q_1, \ldots, q_n) \in Q^n \mid q_i < q_j}$. Alle Eingaben, die dieser Menge angehören, erfüllen $K(i, j) = 1$. Wird also $K(i, j)$ als erster Aufruf der Vergleichsfunktion berechnet, folgt darauf die Folge der weiteren Vergleiche.  Dann erhalten wir einen \emph{Entscheidungsbaum} von $A$ als binären Baum, dessen Knoten mit Vergleichen \enquote{$q_i < q_j$?} beschriftet sind. An den Kanten steht entweder $1$ oder $0$, je nach Wahrheitsgehalt. Weil $A$ das Sortierproblem löst, gibt es zu jedem Blatt des Entscheidungsbaums eine Permutation $\pi$, sodass nur Inputfolgen mit 
    \begin{equation*}
        q_{\pi(1)} < q_{\pi(2)} < \cdots < q_{\pi(n)}
    \end{equation*}
    zu diesem Blatt führen. Für jede Permutation muss ein Blatt vorhanden sein, also hat der Entscheidungsbaum mindestens $n!$ Blätter und somit mindestens die Höhe 
    \begin{equation*}
        \log_2(n!) \geq \log_2\left( \left( \frac{n}{2} \right)^{\frac{n}{2}} \right) = \frac{n}{2} \log_2\left( \frac{n}{2} \right) \geq \frac{1}{3}n \log_2(n) \quad \forall n \geq 8.
    \end{equation*}
    Es gibt also eine Permutation, für die das Verfahren $A$ mindestens $\frac{1}{3}n \log_2(n) \in \Omega(n \log(n))$ viele Vergleiche ausführt.
\end{proof}
Da Heapsort oder Mergesort in $\cO(n \log(n))$ sortieren, hat das Sortierproblem Komplexität $\Theta(n \log(n))$.

\begin{prob}{Lineares Modell / Elementtest}{elementTest}
    \prblm{$c_1, \ldots, c_n, d \in \IR$, Punkt $(x_1, \ldots, x_n) \in \IR^n$}{Ist $c_1 x_1 + \cdots + c_n x_n + d < 0$?}
\end{prob}
Wir werten also eine lineare Funktion $h$ aus ($\cO(1)$) Beispielsweise für $c_i = 1, c_j = -1$ ist $h(x_1, x_2) = x_1 - x_2$. Diese Funktion beschreibt eine (Hyper-)Ebene im $\IR^n$. Wir fragen uns nach einer unteren Schranke für den \emph{Elementtest} bzw. sein Entscheidungsproblem: Für $W \subseteq \IR^n$, liegt $x \in W$?

\begin{thm}{Elementtest auf Zusammenhangskomponenten}{elementTestConnectedComponents}
    Sei $W \subseteq \IR^n$ eine Menge mit $m$ Zusammenhangskomponenten. Dann benötigt jeder Algorithmus für den Elementtest von $W$ mindestens $\log_2(m)$ viele Schritte, das Problem liegt also in $\Omega(\log(m))$.
\end{thm}
Ein Beispielproblem wäre beispielsweise binäre Suche für $m$ disjunkte Teilintervalle von $\IR$ zur Bestimmung, ob $x$ in einem dieser Punkte liegt.

\begin{proof}
    Sei $A$ ein Algorithmus für den Elementtest für $W$. Wie im Beweis von Satz~\ref{thm:keyCompareSorting} betrachten wir den linearen Entscheidungsbaum $E$ von $A$, dessen Konten die Tests 
    \begin{equation*}
        h(x_1, \ldots, x_n) < 0?
    \end{equation*}
    in allen möglichen Rechenabläufen von $A$ darstellen. Es genügt zu zeigen, dass $E$ mindestens so viele Blätter hat wie die Menge $W$ Zusammenhangskomponenten. Für jedes Blatt $b$ von $E$ sei 
    \begin{equation*}
        A_b = \set{x \in \IR^n \mid \text{$A$ terminiert bei Eingabe von $x$ im Blatt $b$}}.
    \end{equation*}
    Dieses $A_b$ ist -- nach Definition des linearen Modells -- Durchschnitt von offenen und abgeschlossenen affinen Teilräumen 
    \begin{equation*}
        \begin{split}
            T_h \coloneqq& \set{(x_1, \ldots, x_n) \in \IR^n \mid h(x_1, \ldots, x_n) < 0} \\
            \overline{T_h} \coloneqq& \set{(x_1, \ldots, x_n) \in \IR^n \mid h(x_1, \ldots, x_n) \geq 0}.
        \end{split}
    \end{equation*}
    Wir merken, dass $T_h$ und $\overline{T_h}$ konvex sind, insbesondere also zusammenhängend. Für alle $x \in A_b$ trifft $A$ dieselbe Entscheidung; folglich ist $A_b \subseteq W$ oder $A_b \cap W = \emptyset$. Da der Algirithmus für jede Eingabe $x$ eine Entscheidung treffen muss, folgt 
    \begin{equation*}
        \IR^n = \bigcup_{\text{$b$ Blatt}} A_b.
    \end{equation*}
    Damit erhalten wir 
    \begin{equation*}
        \begin{split}
            W &= \IR^n \cap W \\
            &= \bigcup_{\text{$b$ Blatt}} A_b \cap W \\
            &= \bigcup_{b \in B} A_b
        \end{split}
    \end{equation*}
    für eine Teilmenge $B$ aller Blätter von $E$. Die Anzahl der Zusammenhangskomponenten von dieser Menge ist also höchstens so groß wie $\abs{B}$, der Anzahl der zusammenhängenden Teilmengnen $A_b$, weil jedes $A_b$ vollständig in einer Zusammenhangskomponente enthalten ist. Andererseits ist $\abs{B}$ höchstens so groß wie die Anzal \emph{aller} Blätter von $E$.
\end{proof}

\begin{prob}{$\varepsilon$-Closeness}{epsCloseness}
    \prblm{$x_1, \ldots, x_n \in \IR$, $\varepsilon > 0$}{Gibt es $(i, j)$ mit $\abs{x_i - x_j} < \varepsilon$?}
\end{prob}

\begin{cor}
    Das Problem \emph{$\varepsilon$-Closeness} hat im lineraen Modell eine Zeitkomplexität von $\Theta(n \log(n))$.\label{cor:epsCloseness}
\end{cor}
\begin{proof}
    Durch Sortieren ist auf jeden Fall eine Zeit in $\cO(n \log(n))$ möglich.

    Dass die Zeit in $\Omega(n \log(n))$ liegt, zeigen wir mittels Elementtest: Wir setzen
    \begin{equation*}
        W \coloneqq \set{(x_1, \ldots, x_n) \in \IR^n \mid \abs{x_i - x_j} \geq \varepsilon \text{ für alle } i \neq j}.
    \end{equation*}
    Dann ist $\varepsilon$-Closeness einfach nur der Eiementtest für $W$.

    Wir behaupten, dass 
    \begin{equation*}
        W_\pi \coloneqq \set{(x_1, \ldots, x_n) \in W \mid x_{\pi(1)} < x_{\pi(2)} < \cdots < x_{\pi(n)}}
    \end{equation*}
    die Zusammenhangskomponenten von $W$ sind, wobei $\pi \in \cS_n$. Dann wären es genau $n!$ viele Zusammenhangskomponenten und wieder analog zum Beweis von Satz~\ref{thm:keyCompareSorting} folgt dann, dass $\Omega(n \log(n))$ Zeit gebraucht wird. 

    \emph{Beweis der Behauptung.} Es gilt 
    \begin{equation*}
        \bigcup_{\pi \in \cS_n} W_\pi = W.
    \end{equation*}
    (Es ist $\bigcup_{\pi \in \cS_n} W_\pi \subseteq W$, da $W_\pi \subseteq W$ per Definition gilt. Es ist $\bigcup_{\pi \in \cS_n} W_\pi \supseteq W$, da für jedes $x \in W$ ein $\pi$ existiert mit $x \in W_\pi$, denn da $x \in W$, hat $x$ keine zwei gleichen Koordinaten.)

    Wir zeigen nun, dass $W_\pi$ konvex (und damit (weg)zusammenhängend) ist: Seien $p, q \in W_\pi$. Sei oBdA $p_i > p_j$ für beliebige $i \neq j$. Da $p, q \in W_\pi \subseteq W$, gilt auch $q_i > q_j$ und insbesondere $p_i \geq p_j + \varepsilon$ sowie $q_i \geq q_j + \varepsilon$. Für beliebiges $t \in \edgys{0,1}$ ist 
    \begin{equation*}
        p_i' \coloneqq (1 - t) p_i + t q_i \geq (1 - t) (p_j + \varepsilon) + t (q_j + \varepsilon) = (1 - t) p_j + t q_j + \varepsilon = p_j' + \varepsilon \in W_\pi.
    \end{equation*}
    Damit ist $W_\pi$ konvex.

    Es bleibt zu zeigen, dass für $\pi_1 \neq \pi_2$ auch $W_{\pi_1} \cap W_{\pi_2} = \emptyset$ gilt, also $W_{\pi_1} \cup W_{\pi_2}$ \emph{nicht} wegzusammenhängend ist: Seien $\pi_1 \neq \pi_2$ und $p \in W_{\pi_1}, q \in W_{\pi_2}$. Angenommen, es gäbe einen Weg zwischen $q$ und $q$, parametrisiert also 
    \begin{equation*}
        \varphi(t) \coloneqq \left( \varphi_1(t), \ldots, \varphi_n(t) \right)
    \end{equation*}
    für $t \in \edgys{0,1}$ und $\varphi_i \in C^0(\edgys{0,1}, \IR)$, wobei $f(0) = p$ und $f(1) = q$. Da $\pi_1 \neq \pi_2$, gibt es $i, j$ mit oBdA $p_i < p_j$, aber $q_i > q_j$. Dann ist $\varphi_i(0) - \varphi_j(0) = p_i - p_j < 0$, aber $\varphi_i(1) - \varphi_j(1) = q_i - q_j > 0$. Für $g(t) \coloneqq \varphi_i(t) - \varphi_j(t)$ gibt es nach Zwischenwertsatz also ein $t \in \rounds{0,1}$ mit $g(t) = 0$, also $\varphi_i(t) = \varphi_j(t)$. Dann ist aber $\varphi(t) \notin W$, da zwei Koordinaten gleich sind. Das bedeutet, $W_{\pi_1} \cup W_{\pi_2}$ sind nicht wegzusammenhängend (für das feste $\varepsilon$ aus der Eingabe).
\end{proof}

\begin{prob}{Element-Uniqueness}{elementUniqueness}
    \prblm{$x_1, \ldots, x_n \in \IR$}{Existieren $i, \neq j$ mit $x_i = x_j$?}
\end{prob}

\begin{cor}
    Das Problem \emph{Element-Uniqueness} hat im lineraen Modell eine Zeitkomplexität von $\Theta(n \log(n))$.
\end{cor}
\begin{proof}
    Analog zu Korollar~\ref{cor:epsCloseness} mit dem modifizierten 
    \begin{equation*}
        W = \set{(x_1, \ldots, x_n) \in \IR^n \mid \abs{x_i - x_j} > 0 \text{ für alle }i \neq j}.
    \end{equation*}
    Der Rest geht genau gleich. Jedes \enquote{$\geq \varepsilon$} wird durch ein \enquote{$> 0$} ausgetauscht.
\end{proof}

\begin{cor}
    Sortieren hat auch im linearen Modell eine Zeitkomplexität von $\Theta(n \log(n))$.
\end{cor}
\begin{proof}
    Wir \emph{reduzieren}\index{Reduktion} $\varepsilon$-Closeness bzw. Element Uniqueness in linearer Laufzt auf das Sortierproblem, also für $P \in \cO(n)$ zeigen wir 
    \begin{equation*}
        \text{$\varepsilon$-Closesness / Element-Uniqueness} \leq_P \text{Sortieren:}
    \end{equation*}

    Sei $(x_1, \ldots, x_n) \in \IR^n$ gegeben. Wir sortieren diese zu $x_1', \ldots, x_n'$. Der anschließende Test, ob diese $\varepsilon$-close (oder identisch) sind, benötigt nur $\cO(n)$ Zeit. Das heißt, da nach Korollar~\ref{cor:epsCloseness} das $\varepsilon$-Closeness-Problem nur $\Omega(n \log(n))$ möglich ist, muss das Sortieren bereits $\Omega(n \log(n))$ benötigen.
\end{proof}

\begin{prob}{All-Nearest-Neighbours}{allNearestNeighbours}
    \prblm{$S \subseteq \IR^d$ mit $\abs{S} = n$}{Für jedes $x \in S$ das $y \in S$ mit $\abs{x - y} \leq \abs{x - z}$ für $z \in S$}
\end{prob}

\begin{cor}
    Das Problem \emph{All-Nearest-Neighbours} hat Zeitkomplexität $\Omega(n \log(n))$.
\end{cor}

\begin{proof}
    Wir reduzieren wieder $\varepsilon\text{-Closeness} \leq_P \text{All-Nearest-Neighbours}$: Sei $x_1, \ldots, x_n \in \IR$ und $\varepsilon > 0$, so können wir Punkte 
    \begin{equation*}
        p_i \coloneqq (x_i, 0, \ldots, 0) \in \IR^d
    \end{equation*}
    bilden und für jeden von ihnen einen nächsten Nachbarn bestimmen. Unter diesen $n$ Punktepaaren befindet sich auch ein \emph{dichtestes Paar} $(p_i, p_j)$ mit
    \begin{equation*}
        \abs{p_i - p_j} = \min_{1 \leq k \neq \ell \leq n} \abs{p_k p_\ell},
    \end{equation*} 
    und wir können dieses dichteste Paar in $\cO(n)$ finden. Nun brauchen wir nur noch zu testen, on $\abs{x_i - x_j} < \varepsilon$ gilt. Da muss also $\Omega(n \log(n))$ gelten.
\end{proof}

\begin{cor}
    Ein dichtestes Paar von $n$ Punkten im $\IR^d$ zu finden, hat die Zeitkomplexität $\Omega(n \log(n))$. \hfill $\qed$
\end{cor}

\breakline{Ende Einschub}

\begin{lem}{Untere Schranke für Line Segment Intersection}{lowerBoundLineSegmentIntersection}
    Das Existenzproblem für den Schnitt von $n$ Liniensegmenten hat Zeitkomplexität $(\Omega(n \log(n)))$. Das Aufzählungsproblem für $k$ Schnittpunkte hat Zeitkomplexität $\Omega(n \log(n) + k)$.
\end{lem}

\begin{proof}
    Reduktion von $\varepsilon$-Closeness $\leq_P$ Line Segment Intersection: 
    
    Seien $x_1, \ldots, x_n \in \IR$ und $\varepsilon > 0$ gegeben. Wir bilden die Strecken $s_i$ von $(x_i, 0)$ nach $(x_i + \frac{\varepsilon}{2}, \frac{\varepsilon}{2})$ und $t_i$ von $(x_i, 0)$ nach $(x_i - \frac{\varepsilon}{2}, \frac{\varepsilon}{2})$. Parallele Strecken können keinen echten Schnittpunkt miteinaneder haben, selbst dann nicht, wenn sie übereinander liegen. Ein echter Schnittpunkt kann nur von zwei Segmenten $s_i$ und $t_j$ mit $i \neq j$ passieren, für die dann $0 < \abs{x_i - x_j} < \varepsilon$ gelten muss. Umgekehrt verursacht jedes solche Zahlenpaar einen echten Schnittpunkt von $s_i$ und $t_j$ oder von $s_j$ mit $t_i$.
\end{proof}
