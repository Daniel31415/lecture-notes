\section{Topological spaces}

Let $X$ a set and $2^X$ the power set.

\begin{defi}{Topological space, Topology}{topSpace}
    A \emph{topological space}\index{space!topological -}\index{topological space} $(X, \cT)$ is a pair of a set $X$ and subset $\cT$ of $2^X$ with the following properties:
    \begin{enumerate}[$(i)$]
        \item $\emptyset \in \cT, X \in \cT$
        \item finite intersections: $U, V \in \cT \Rightarrow U \cap V \in \cT$
        \item arbitrary unions: $U_\alpha \in \cT ~\forall \alpha \in A \Rightarrow \bigcup_{\alpha \in A} U_\alpha \in \cT$.
    \end{enumerate}
    The set $\cT$ is called a \emph{Topology on $X$}\index{Topology} and the elements in $\cT$ are often called \emph{open sets}\index{open set}\index{set!open -}. A set $C$ is called \emph{closed}\index{closed set}\index{set!closed -} if $X \setminus C$ is open.

    If $\cT_1$ and $\cT$ are topologies on $X$, then $\cT_1$ is called \emph{finer} than $\cT_2$ or $\cT_2$ is called \emph{coarser} than $\cT_1$ if $\cT_1 \supset \cT_2$.
\end{defi}

\begin{bsp}
    \begin{enumerate}[$(i)$]
        \item The finest topology is $\cT = 2^X$.
        \item The coarsest topology is $\cT = \set{\emptyset, X}$.
        \item Standard topology on $\IR$: $U \in \cT_{st} \iff \forall x \in U \exists \varepsilon > 0: \intervaloo{x-\varepsilon, x+\varepsilon} \subseteq U$. We will usually use this topology.
        \item Let $X = \IR$ and $\cT = \set{U \subseteq \IR \mid U = \emptyset \text{ or $\IR \setminus U$ is countable}}$. Then, the closed sets are countable\footnote{We call a set \emph{countable}\index{countable}\index{set!countable} if it is empty, finite or has an bijection to $\IN$.} or $\IR$. 
        \item Let $(X, \cT)$ be a topological space. Then, for $A \subseteq X$, the \emph{relative topology}\index{topology!relative -} $\cT_A$ on $A$ is defined by $\cT_A = \set{U \cap A \mid U \in \cT}$.
        \item Let $B$ be a set and $\cT_\beta$ a topology on $X$ for all $\beta \in B$. Then, $\cT := \bigcap_{\beta \in B} \beta$ defines a topology on $X$.
        
        Given any $\cS \subseteq 2^X$,
        \begin{equation}
            \bigcap_{\substack{\cT~\text{top.}\\\cT \supseteq \cS}} \cT
        \end{equation}
        is a topology and is called the \emph{coarsest topology containing $\cS$}.
        \item \emph{Product topology} of $(X, \cT_X), (Y, \cT_Y)$ as topological spaces is definied with 
        \begin{equation}
            \cS := \set{U \times Y \mid U \in \cT_X} \cup \set{X \times V \mid V \in \cT_Y}.
        \end{equation}
        The product topology is $\cT_\cS$, the coarsest topology containing $\cS$.
        
        The product topology is the coarsest topology so that the projections $\pi_1(x,y) = x$ and $\pi_2(x,y) = y$ are \enquote{continuous}.
    \end{enumerate}
\end{bsp}

\begin{defi}{Interior, Closure, Boundary}{interorClosureBoundary}
    Let $(X, \cT)$ be a topological space and $A \subseteq X$. We define the \emph{interior}\index{set!interior} $A^\circ$, the \emph{closure}\index{set!closure} $\overline{A}$ and the \emph{boundary}\index{set!boundary} $\partial A$ by 
    \begin{equation}
        A^\circ = \bigcup_{\substack{U \subseteq A \\ U \in \cT}} U, \quad \overline{A} = \bigcap_{\substack{C \supseteq A \\ X \setminus C \in \cT}} C, \quad \partial A = \overline{A} \setminus A^\circ.
    \end{equation}
\end{defi}

\begin{defi}{Dense subsets, separable topological space}{denseSubsets}
    $A \subseteq X$ is \emph{dense}\index{set!dense} if $\overline{A} = X$. The space $X$ is \emph{separable}\index{separable}\index{topological space!separable -} if there exists a countable, dense
    subset.
\end{defi}

\begin{defi}{Open neighbourhood}{openNeighbourhood}
    Let $x \in X$. Then, $U \subseteq X$ is an \emph{open neighbourhood of $x$}\index{set!open neighbourhood}\index{open neighbourhood} if $U \in \cT$ and $x \in U$.  
\end{defi}

\begin{defi}{Continuous maps, Homeomorphisms}{continuity}
    Let $(X, \cT_X), (Y, \cT_Y)$ topological spaces. Then, $f: X \to Y$ is called \emph{continuous}\index{continuous} if
    \begin{equation}
        \forall V \supseteq \set{x} \in \cT_Y: \quad f^{-1}(V) = \set{x \in X \mid f(x) \in V} \in \cT_X.
    \end{equation}
    We call $f$ a \emph{homeomorphism}\index{homeomorphism} if $f$ is bijective and both $f$ and $f^{-1}$ are continuous.
\end{defi}

If $\cT = 2^X$, and $f: X \to Y$ is continuous.

If $\cT = \set{\emptyset, \cT}$, then every constant map is continuous.




\newLecture{11.10.2024}

\begin{defi}{Convergence}{convergence}
    Let $(X, \cT)$ be a topological space and $x: \IN \to X$ a sequence. Let $x^* \in X$. We say that \emph{$x$ converges to $x^*$}\index{convergence} if for \textbf{every} neighbourhood $U$ of $x$, the set $\set{j \in \IN \mid x_j \notin U}$ is finite.
\end{defi}

\begin{beispiel}
    \begin{enumerate}[$(i)$]
        \item If $\cT = \set{\emptyset, X}$, every sequence converges to every point $x^* \in X$.
        \item If $\cT = 2^X$, take $U = \set{x^*}$, so only sequences which are constant with up to finitely many exceptions converge.
    \end{enumerate}
\end{beispiel}

\begin{defi}{Hausdorff space}{hausdorffSpace}
    A topological space is called a \emph{Hausdorff space}\index{Hausdorff}\index{Hausdorff!- space}\index{space!Hausdorff -} if 
    \begin{enumerate}[$(i)$]
        \item sets consisting of a single point are closed,
        \item for different points $x \neq y$, there exist disjoint open neighbourhoods: $\exists U_x, U_y \in \cT$ with $x \in U_x, y \in U_y$ and $U_x \cap U_y = \emptyset$.
    \end{enumerate}
\end{defi}

\begin{prop}
    If $(X, \cT)$ is Hausdorff, then every sequence has at most one limit point. \hfill $\qed$
\end{prop}

\begin{defi}{Compactness}{compactness}
    A subset $K$ of a topological space $(X, \cT)$ is \emph{compact}\index{compact}\index{set!compact -} if every open cover of $K$ contains a \textbf{finite} subcover: If $A$ is an index set and $U_\alpha \in \cT$ for every $\alpha \in A$ and 
    \begin{equation}
        \bigcup_{\alpha \in A} U_\alpha \supseteq K,
    \end{equation}
    then there exists a \textbf{finite} set $A' \subseteq A$ so that 
    \begin{equation}
        \bigcup_{\alpha \in A'} U_\alpha \supseteq K.
    \end{equation}
\end{defi}

\begin{lem}{Continuous maps compact to compact}{compactUnderContinuous}
    If $f: (X, \cT_X) \to (Y, \cT_Y)$ is continuous and $K \subseteq X$ is compact in $X$, then $f(X)$ is compact in $Y$.
\end{lem}

\begin{proof}
    Follows directly from the definitions.
\end{proof}

\begin{thm}{Continuous maps with compact domain attain max and min}{compactToMaxMin}
    If $f: (X, \cT_X) \to \IR$ is continuous and $K \subseteq X$ is compact, then $f$ attains its minimum and maximum on $K$, i.e. $\exists \underline{x}, \overline{x} \in K$ so that $f(\underline{x}) = \inf_{x \in K} f(x), f(\overline{x}) = \sup_{x \in K} f(x)$.
\end{thm}

\begin{proof}
    $f(K) \subseteq \IR$ is compact, the rest is easy.
\end{proof}

\begin{defi}{Connectedness}{connectedness}
    Let $(X, \cT)$ be a topological space. We call $X$ \emph{connected}\index{topological space!connected -}\index{connected}\index{space!connected -} if it cannot be written as a disjoint union of two open subsets. A subset $A \subseteq X$ is connected if $(A, \cT_A)$ is connected.
\end{defi}

\begin{aufgabe}
    Show that intervals in $\IR$ are connected.
\end{aufgabe}

\begin{thm}{Intermediate value theorem}{intermediateValue}
    If $f: (X, \cT_X) \to \IR$ is continuous and $(X, \cT_X)$ is connected, and there exist $a, b \in X$ so that $f(a) < 0, f(b) > 0$. Then, there exists an $x \in X$ with $f(x) = 0$.
\end{thm}

\begin{proof}
    Assume $f(x) \neq 0$ for all $x \in X$. We know that $U := f^{-1}(\intervaloo{0,\infty})$ is open, and $X \setminus U = f^{-1}(\intervaloc{-\infty,0}) = f^{-1}(\intervaloo{-\infty,0})$ is also open. Then, $X = U \cup (X \setminus U) \Rightarrow X$ not connected. $\lightning$
\end{proof}

\section{Metric spaces}
\subsection{Definition and examples}

Let $X$ be a set and $d: X \times X \to \intervalco{0,\infty}$.

\begin{defi}{Metric space, metric}{metricSpace}
    $(X,d)$ is called a \emph{metric space}\index{space!metric -}\index{metric space} and $d$ is called \emph{metric}\index{metric} if
    \begin{enumerate}[$(i)$]
        \item $d(x,y) = 0 \Leftrightarrow x = y$,
        \item $\forall x,y: d(x,y) = d(y,x)$,
        \item (triangle inequality)\index{triangle inequality} $d(x,y) \leq d(x,z) + d(z,y)$ for all $x,y,z \in X$.\footnote{\enquote{Where the triangle inequality is not true is on airline flights. Sometimes, it is cheaper to fly from Köln/Bonn to Mallorca first before flying to mainland Spain later.} -- Stefan Müller, 11.10.2024}
    \end{enumerate}
    If only $(ii)$ and $(iii)$ are fulfilled, we call $d$ a \emph{semimetric} or \emph{pseudometric}\index{semimetric}\index{pseudometric}\index{metric!semi-}\index{metric!pseudo-}, and with $x \sim y :\Leftrightarrow d(x,y) = 0$, we get a metric $\overline{d}$ on the equivalence classes.
\end{defi}

\begin{lem}{Composition of concave map and metric is metric}{compositionConcaveMetric}
    Assume $\psi: \intervalco{0,\infty} \to \intervalco{0,\infty}$ is continuous, non-decreasing, \textbf{concave} and $\psi(0) = 0, \psi(x) > 0$ for $x > 0$. If $d$ is a metric on $X$, so is $\psi \circ d$.
\end{lem}

\begin{proof}
    It is enough and easy to check that $\psi(a+b) \leq \psi(a) + \psi(b)$. 
\end{proof}

\begin{beispiel}
    Define $\psi(t) := \frac{t}{1+t} = 1 - \frac{1}{1+t}$. Then, $\psi \in C^0(\intervalco{0,\infty}) \cap C^2(\intervaloo{0,\infty})$. Non-decreasing and concave are equivalent to $\psi' \geq 0$ and $\psi'' \leq 0$.
\end{beispiel}

\begin{beispiel}
    \begin{enumerate}[$(i)$]
        \item Let $X = \IR^n$ and $d_\infty(x,y) = \max_{1 \leq i \leq n} \abs{x_i - y_i}$.
        \item $X = \IR^n, d_p(x,y) := \left(\sum_{i=1}^n \abs{x_i - y_i}^p\right)^{\frac{1}{p}}$ for $1 \leq p < \infty$,
        \item $\ell_p = \set{a: \IN \to \IR \mid \left(\sum_{i=1}^\infty \abs{x_i}^p\right)^{\frac{1}{p} < \infty}}$ and $d_p(a,b) := \left(\sum_{i=1}^\infty \abs{a_i - b_i}^p\right)^{\frac{1}{p}}$ for $1 \leq p < \infty$
        \item $\ell_\infty := \set{a: \IN \to \IR \mid \sup_{i \in \IN} \abs{a_i} < \infty}$ with $d_\infty(a,b) := \sup_{i \in \IN} \abs{a_i - b_i}$
        \item $X = \IR$ with $d(x,y) = \frac{\abs{x-y}}{1+\abs{x-y}} = \psi(\abs{x-y})$ for $\psi(t) = \frac{t}{1+t}$. By Lemma~\ref{lem:compositionConcaveMetric}, $d$ is a metric on $\IR$. Fun fact: $0 \leq d(x,y) < 1$.
        \item Metric on the space of all sequences:
        \begin{equation}
            d(a,b) = \sum_{i = 0}^\infty 2^{-i} \frac{\abs{a_i - b_i}}{1 + \abs{a_i - b_i}} \leq \sum_{i = 0}^\infty 2^{-i} \leq 2.
        \end{equation}
        \item Let $X$ be a set and $(Y, d_Y)$ a metric space and $f: X \to Y$ \textbf{injective}. Then, we can define the \emph{pullback metric}\index{metric!pullback -}\index{pullback metric} $d_X(x,y) := d_Y(f(x),f(y))$. $d_X$ is a metric on $X$.
        \item $\overline{\IR} := \IR \cup \set{-\infty,\infty}$. Then, 
        \begin{equation}
            f: \overline{\IR} \to \intervalcc{-1,1}, x \mapsto \begin{cases}
                -1 & x = -\infty \\ \frac{x}{1+\abs{x}} & x \in \IR \\ 1 & x = \infty
            \end{cases}
        \end{equation}
        is bijective. Then, $d_{\overline{\IR}}(x,y) = \abs{f(x)-f(y)}$ is a pullback metric.
        \item Shortest path on a graph
    \end{enumerate}
\end{beispiel}

\begin{bemerkung}
    Let $(X,d)$ be a metric space, let $A, B \subseteq X$, let $x \in X$ and let $r > 0$. Further distances:
    \begin{description}
        \item[Diameter of a subset] 
        \begin{equation}
            \sup\set{d(x,y) \mid x,y \in A}
        \end{equation} 
        (with the convention $\diam(A) = \infty$ if $\set{d(x,y) \mid x,y \in A}$ is unbounded)
        \item[Distance between two sets]
        \begin{equation}
            \dist(A,B) = \inf\set{d(x,y) \mid x \in A, y \in B}
        \end{equation}
        \item[Distance between a point and a set]
        \begin{equation}
            \dist(x,A) = \inf\set{d(x,y) \mid y \in B} = \dist(\set{x},A)
        \end{equation}
        \item[$r$-neighbourhood of a set]
        \begin{equation}
            B_r(A) := \set{x \in X \mid \dist(x,A) < r}
        \end{equation}  
        \item[$r$-ball of a point]
        \begin{equation}
            B_r(x) = B(x,r) = \set{y \in X \mid d(y,x) < r}
        \end{equation}
        \item[Hausdorff distance of two sets]\index{Hausdorff distance}\index{set!Hausdorff distance between -s}
        \begin{equation}
            d_H(A,B) := \inf\set{r > 0 \mid A \subseteq B_r(B) \text{ and } B \subseteq B_r(A)}
        \end{equation}
        and $d_H(\set{x},\set{y}) = d(x,y)$.   
    \end{description}
\end{bemerkung}

\subsection{Convergence and continuity in metric spaces}

\begin{prop}
    Let $(X,d)$ be a metric space and $x : \IN \to X$ a sequence, and $x^* \in X$. Then, the following two statements are equivalent:
    \begin{enumerate}[$(i)$]
        \item $x$ converges to $x^*$ in $(X,\cT_d)$,
        \item $\lim_{n \to \infty} d(x_n, x^*) = 0$.
    \end{enumerate}
\end{prop}

\textbf{Moral.} \enquote{In metric spaces, anything can be done by sequences}

\begin{prop}
    Let $(X,d)$ be a metric space and let $A \subseteq X$. Then, the following two statements are equivalent:
    \begin{enumerate}[$(i)$]
        \item $A$ is closed, so $X \setminus A \in \cT_d$,
        \item $A$ is \emph{sequentially closed}\index{set!sequentially closed -}\index{set!closed -!sequentially -}, so for every sequence $x: \IN \to A$ which converges to $x^* \in X$, one has $x^* \in A$.
    \end{enumerate}
    Moreover, the closure of any set $A$ is given by the \emph{sequential closure}\index{sequential closure}, this means 
    \begin{equation}
        \overline{A} = \set{x^* \in X \mid \exists x: \IN \to A, \text{$x$ converges to $x^*$}}.
    \end{equation}
\end{prop}

\begin{proof}
    Easy exercise.
\end{proof}

\begin{prop}
    Let $(X, d_X), (Y, d_Y)$ be two metric spaces and $f: X \to Y$. Then, the following three statements are equivalent:
    \begin{enumerate}[$(i)$]
        \item $f$ is continuous in as a map $f: (X, \cT_{d_X}) \to (Y, \cT_{d_Y})$,
        \item For every $x\in X, \varepsilon > 0$ there exists a $\delta > 0$ so that $d_X(z,x) < \delta \Rightarrow d_Y(f(z), f(x)) < \varepsilon$,
        \item For every sequence $x$ convergent to $x^*$, also $f(x)$ converges to $f(x^*)$.
    \end{enumerate}
\end{prop}

We know that every metric space is Hausdorff. \emph{Question:} If $(X, \cT)$ is a Hausdorff space, does there exist a metric $d: X \times X \to \intervalco{0,\infty}$ such that $\cT = \cT_d$? Or shorter, is $X$ metrizable?

If $X$ is second countable (so there exist countable neighbourhoods), $X$ is metrizable, and then, $X$ ist first countable.

\begin{defi}{Equivalent metrics, stronger metrics}{equivalentMetrices}
    Let $d_1$ and $d_2$ be metrics on $X$. 
    
    We say \emph{$d_1$ is stronger than $d_2$} if $\cT_{d_1} \supset \cT_{d_2}$.\index{metric!stronger -}

    We say \emph{$d_1$ is equivalent to $d_2$} if $\cT_{d_1} = \cT_{d_2}$.\index{metric!equivalent -}
\end{defi}

\begin{prop}
    The following statements are equivalent:
    \begin{enumerate}[$(i)$]
        \item $d_1$ is stronger than $d_2$,
        \item The identity map $x \mapsto x$ for $(X,d_1) \to (X,d_2)$ is continuous,
        \item Every sequence that converges in $d_1$ also converges in $d_2$,
        \item $\forall x \in X \forall \varepsilon > 0 \exists \delta > 0$ so that $d_1(x,y) < \delta \Rightarrow d_2(x,y) < \varepsilon$.
    \end{enumerate}
\end{prop}

\begin{beispiel}\leavevmode
    \begin{enumerate}[$(i)$]
        \item Let $X = \IR$ and $d_1(x,y) = \abs{x - y}, d_2 = \abs{\arctan(x) - \arctan(y)}$. Then, $d_1$ and $d_2$ are equivalent (Exercise 1.4).
        \item Let $\ell_1 := \set{x: \IN \to \IR \mid \sum_{k = 0}^\infty \abs{x_k} < \infty}$ with $d_1(x,y) := \sum_{k = 0}^\infty \abs{x_k - y_k}$ and $d_\infty(x,y) = \sup_{k \in \IN} \abs{x_k - y_k} \leq d_1(x,y)$. We see that $d_1$ is stronger than $d_\infty$, but not equivalent: See the map 
        \begin{equation}
            \IN \to (\IN \to \IR), \quad j \mapsto x^{(j)}_k := \begin{cases}
                0 & k < j-1 \text{ or } k > 2j \\
                \frac{1}{k} & j \leq k \leq 2j.
            \end{cases}
        \end{equation} 
        Then, $\abs{x_k^{(j)}} \leq \frac{1}{j}$ with $d_\infty(x^{(j)},0) \leq \frac{1}{j} \to 0$ as $j \to \infty$, but $d_1(x_k^{(j)},0) = \sum_{k = 0}^\infty x_k^{(j)} = \sum_{k = j}^{2j} \frac{1}{k} \geq j\frac{1}{2j} = \frac{1}{2}$.
    \end{enumerate}
\end{beispiel}

\begin{defi}{Separable metric space}{separable}

\subsection{Completeness}

\begin{defi}{Cauchy sequence, completeness}{cauchySeqCompleteness}\index{Cauchy sequence}\index{sequence!Cauchy -}\index{complete}\index{metric space!complete -}
    Let $(X,d)$ be a metric space.
    \begin{enumerate}[$(i)$]
        \item A sequence $x: \IN \to X$ is a \emph{Cauchy sequence} if $\forall \varepsilon > 0 \exists k_0$ so that for all $k, \ell \geq k_0: d(x_k, x_\ell) < \varepsilon$,
        \item $(X,d)$ is called \emph{complete} if every Cauchy sequence converges.
    \end{enumerate}
\end{defi}

\begin{beispiel}\leavevmode

        \item $(\ell_p, d_p)$ are complete for all $1 \leq p \leq \infty$.

\begin{prop}
    Let $(X,d)$ be a complete metric space and $A \subseteq X$ closed. Let $d_A := d|_{A \times A}$. Then, $(A, d_A)$ is complete.
\end{prop}
\begin{proof}
    If $x: \IN \to A$ is a Cauchy sequence in $X$, by completeness, $x_k \to x^*$ in $X$. As $A$ is (sequentially) closed, $x^* \in A$.
\end{proof}

\begin{defi}{Isometric immersion, isometry}{isometry}\index{isometry}\index{isometric immersion}
    Let $(X, d_X)$ and $(Y, d_Y)$ be metric spaces. A map $f: X \to Y$ is called an \emph{isometric immersion} if 
    \begin{equation}
        d_Y(f(x),f(y)) = d_X(x,y)
    \end{equation}
    for all $x, y \in X$. The map $f$ is called an \emph{isometry} if it is in addition bijective
\end{defi}

\begin{bemerkung}
    An isometric immersion is automatically bijective. Thus, if $f$ is an isometric immersion, than $f$ is an isometry from $X$ to $f(X)$. The spaces $X$ and $f(X)$ (equipped with the restriction of the metric $d_Y$) are indistinguishable as metric spaces. In particular, a sequence $f: \IN \to \IR$ is convergent in $X$ if and only if $f \circ x$ is convergent in $f(X)$, and $X$ is a Cauchy sequence in $X$ if and only if $f \circ x$ is a Cauchy sequence in $f(X)$.
\end{bemerkung}

\begin{thm}{Completion}{completion}
    Let $(X,d)$ be a metric space. Then, there exists a complete, metric space $(\tilde{X}, \tilde{d})$ and an isometric immersion $j: X \to \tilde{X}$ such that $j(X)$ is dense in $\tilde{X}$.
\end{thm}
The space $\tilde{X}$ is called the \emph{completion of $X$}\index{space!completion (of -)}\index{completion} (and is unique up to isometries).

\begin{proof}
    We will see a short proof later. Here we just sketch the standard proof which is based on considering the space of Cauchy sequences in $X$ modulo converging sequences. This proof is modelled on the construction of the real numbers $\IR$ from the rational numbers $\IQ$ (see e.g. Alt's book for details).

    Let
    \begin{equation}
        \hat{X} := \set{x: \IN \to X \mid \text{$x$ Cauchy sequence}}.
    \end{equation}
    Define 
    \begin{equation}
        x \sim y \quad :\iff \quad \lim_{k \to \infty} d(x_k, y_k) = 0.
    \end{equation}
    It is easy to see that $\sim$ is an equivalence relation. The equivalence class of $x \in \hat{X}$ is defined by 
    \begin{equation}
        [x] := \set{z \in \hat{X} \mid z \sim x},
    \end{equation}
    and the space of equivalences is defined by $\hat{X} := \nicefrac{X}{\sim}$. On $\hat{X}$, we define 
    \begin{equation}
        \tilde{d}([x],[y]) := \lim_{j \to \infty} d(x_j, y_j).
    \end{equation}
    It follows from the definition of the equivalence classes and the triangle inequality that $\tilde{d}$ is well-defined, i.e., the right hand side only depends on the equivalence classes of the Cauchy sequences $x$ and $y$. Since $x$ and $y$ are Cauchy sequences, it is also easy to see that $j \mapsto d(x_j, y_j)$ is a Cauchy sequence in $\IR$ and that therefore $\lim_{j \to \infty} d(x_j, y_j)$ exists. It is easy to see that $\tilde{d}$ is a metric on $\tilde{X}$. Then, one shows that
    \begin{itemize}
        \item $(\tilde{X},\tilde{d})$ is complete,
        \item the map $j: X \to \tilde{X}$ which maps $a$ to the (equivalence class of the) constant sequence with value $a$ is an isometry
        \item $j(X)$ is dense in $(\tilde{X},\tilde{d})$
    \end{itemize}
\end{proof}

\section{Normed spaces}
Let $X$ be a $\IK$-vector space. Recall our convention that $\IK = \IR$ or $\IK = \IC$.

\begin{defi}{Norm}{norm}
    A pair $(X, \norm{\cdot})$ is called \emph{normed space}\index{normed space}\index{norm}\index{space!normed -} if $X$ is a $\IK$-vector space and $\norm{\cdot}: X \to \intervalco{0,\infty}$ with the following properties:
    \begin{enumerate}[$(i)$]
        \item Definiteness: $\norm{x} = 0 \Rightarrow x = 0$ \index{definiteness}
        \item Homogeneity: $\norm{\lambda x} = \abs{\lambda} \cdot \norm{x}$ for all $\lambda \in \IK, x \in X$ \index{homogeneity}
        \item Triangle inequality: $\norm{x + y} \leq \norm{x} + \norm{y}$ for all $x, y \in X$. \index{triangle inequality}
    \end{enumerate}
\end{defi}
If only $(ii)$ and $(iii)$ hold, we call $\norm{\cdot}$ a \emph{seminorm}.\index{norm!semi-}\index{seminorm}

If $(i)-(iii)$ hold, the map $d$ defined by $d(x,y) = \norm{x-y}$ is a metric that is translation invariant.

It follows from the triangle inequality that 
\begin{equation}
    \abs{\norm{x} - \norm{y}} \leq \norm{x - y}
\end{equation}\index{triangle inequality!reversed -}
That implies, that the map $X \to \IR, x \mapsto \norm{x}$ is continuous (actually, Lipschits continuous with Lipschitz constant $1$).

\begin{defi}{Banach space}{banachSpace}
    A normed space which is complete is called \emph{Banach space}\index{Banach space}\index{space!Banach -}.
\end{defi}

\begin{beispiel}
    For $1 \leq p \leq \infty$, the sequence spaces $\ell_p$ are Banach spaces if equipped with norms $\norm{x}_p := \left(\sum_{k = 0}^\infty \abs{x_k}^p\right)^{\frac{1}{p}}$ for $p < \infty$ and $\norm{x}_\infty := \sup_{k \in \IN} \abs{x_k}$ respectively.
\end{beispiel}

\begin{defi}{Stronger norms, equivalent norms}{strongerEquivalentNorms}
    Let $\norm{\cdot}_1$ and $\norm{\cdot}_2$ be norms on a $\IK$-vector space $X$. We say that $\norm{\cdot}_1$ is \emph{stronger than} $\norm{\cdot}_2$\index{norm!stronger -} if the corresponding metric $d_1$ is stronger than $d_2$.

    We say that $\norm{\cdot}_1$ and $\norm{\cdot}_2$ \emph{are equivalent}\index{norm!equivalent -} if $d_1$ and $d_2$ are equivalent.
\end{defi}

\begin{prop}
    Let $\norm{\cdot}_1$ and $\norm{\cdot}_2$ be norms on a $\IK$-vector space $X$. Then, $\norm{\cdot}_1$ is stronger that $\norm{\cdot}_2$ iff there exists a $C > 0$ such that 
    \begin{equation}
        \forall x \in X: \quad \norm{x}_2 \leq C \cdot \norm{x}_1.\label{eq:strongerNorm}
    \end{equation}
    Moreover, $\norm{\cdot}_1$ and $\norm{\cdot}_2$ are equivalent iff three exist $c, C > 0$ such that 
    \begin{equation}
        \forall x \in X: \quad c\norm{x}_1 \leq \norm{x}_2 \leq C\norm{x}_1.
    \end{equation}
\end{prop}
\begin{proof}
    The second statement follows immediately from the first statement.

    Recall that $d_1$ is stronger than $d_2$ iff the identity map from $(X,d_1)$ to $(X,d_2)$ is continuous. If~\eqref{eq:strongerNorm} holds, the identity map is continuous.

    Now assume that $\id: (X,d_1) \to (X,d_2)$ is continuous. Apply the $\varepsilon$-$\delta$ definition of continuity at the point $x = 0$:
    \begin{equation}
        \exists \delta > 0: \quad \forall y: \quad \norm{y}_1 < \delta \Rightarrow \norm{y}_2 < 1.
    \end{equation}
    Let $C > 0$ be any number such that $C > \frac{1}{\delta}$. Let $X \in X \setminus \set{0}$. Define $y := \frac{x}{C\norm{x}_1} \Rightarrow \norm{y}_1 = \frac{1}{C} < \delta$. This implies that $\norm{y}_2 < 1$. Now, $x = C\norm{x}_1 y$ and thus $\norm{x}_2 ) C\norm{x}_1 \norm{y}_2 \leq C\norm{x}_1$. 
\end{proof}

\begin{beispiel}
    \begin{enumerate}[$(i)$]
        \item On $\IR^n$, all norms are equivalent. (Analysis II)
        \item See $\ell_1 := \set{x: \IN \to \IR \mid \sum_{k = 1}^\infty \abs{x_k} < \infty}$ with $\norm{x}_1 = \sum_{k = 1}^\infty, \abs{x_k}$ and $\norm{x}_\infty = \sup_{k \in \IN} \abs{x_k}$. We have $\norm{x}_\infty \leq \norm{x}$, so $\norm{\cdot}_1$ is stronger than $\norm{\cdot}_\infty$ but $\norm{\cdot}_\infty$ is not equivalent to $\norm{\cdot}_1$.
    \end{enumerate}
\end{beispiel}

\section{Hilbert spaces}
For $\IC \ni \alpha = a + bi$ with $a, b \in \IR$, we say $\overline{\alpha} := a - bi$ is the \emph{complex conjugate}. We get $\alpha \overline{\alpha} = a^2 + b^2$.

\begin{defi}{Sesquilinear form}{sesquilinearForm}
    Let $X$ be a $\IK$-vector space. A map $X \times X \to \IK, (x_1, x_2) \mapsto \Span{x_1,x_2}_X$ is called a \emph{sesquilinear form}\index{sesquilinear form} if for all $x, y, x_1, x_2, y_1, y_2 \in X$ and for alle $\alpha \in \IK$, we have 
    \begin{enumerate}[$(i)$]
        \item $\Span{\alpha x, y}_X = \alpha \Span{x,y}_X$ and $\Span{x, \alpha y}_X = \overline{\alpha}\Span{x,y}_X$,
        \item $\Span{x_1 + x_2,y}_X = \Span{x_1,y}_X + \Span{x_2,y}_X$, $\Span{x,y_1 + y_2}_X = \Span{x,y_1}_X + \Span{x,y_2}_X$.
    \end{enumerate}
    A sesquilinear form is \emph{symmetric}\index{sesquilinear form!symmetric -} if 
    \begin{equation}
        \Span{y,x}_X = \Span{x,y}_X \quad \quad \forall x,y \in X.
    \end{equation}
    A sesquilinear form is \emph{positive semidefinite}\index{sesquilinear form!positive semidefinite -} if 
    \begin{equation}
        \Span{x,x}_X \geq 0 \quad \quad \forall x \in X,
    \end{equation}
    and \emph{positive definite} if $\Span{x,x}_X \geq 0$ and $\Span{x,x}_ = 0 \Rightarrow X = 0$ for all $x \in X$.\index{sesquilinear form!positive definite -}
\end{defi}

\begin{beispiel}
    \begin{enumerate}[$(i)$]
        \item $X = \IC$ and $\Span{x,y} = x\overline{y}$,
        \item $X = \IC^n$ and $\Span{x,y} = \sum_{i = 1}^n x_i \overline{y_i}$,
        \item $X = (\ell_2,\IC)$ and $\Span{x,y} = \sum_{i \in \IN} x_i \overline{y_i}$.
    \end{enumerate}
\end{beispiel}

\begin{defi}{Scalar product}{scalarProduct}
    A positive definite, symmetric sesquilinear form on a $\IK$-vector space is a \emph{Scalar product}.\index{scalar product}. Then, the space $(X, \Scalar_X)$ is called a \emph{Pre-Hilbert space}\index{Hilbert space!Pre-}\index{space!Hilbert -!Pre-}.
\end{defi}
\emph{What are the key features of a scalar product?}
\begin{itemize}
    \item Geometrically: Orthogonal if $\Span{x,y}_X = 0$; orthogonal projections; Pythagoras
    \item Analytically: Cauchy-Schwarz inequality
\end{itemize}

\begin{lem}{Scalar product traits}{scalarProdTraits}
    Let $(X)$ be a $\IK$-vector space and $\Scalar_X$ a positive semidefinite, symmetric sesquilinear form and define 
    \begin{equation}
        \norm{x} := \sqrt{\Span{x,x}_X}.
    \end{equation}
    Then,
    \begin{enumerate}[$(i)$]
        \item Homogeneity: $\norm{\alpha x} = \abs{\alpha} \cdot \norm{x}$,
        \item Cauchy-Schwarz inequality\index{Cauchy-Schwarz inequality}: $\abs{\Span{x,y}} \leq \norm{x} \cdot \norm{y}$,
        \item Triangle inequality\index{triangle inequality}: $\norm{x + y}\leq \norm{x} + \norm{y}$,
        \item Parallelogram identity\index{parallelogram identity}: $\norm{x + y}^2 + \norm{x-y}^2 = 2(\norm{x}^2 + \norm{x}^2)$.
    \end{enumerate}
    In particular, $\norm{\cdot}$ is a seminorm. If, in addition, $\Scalar_X$ is positive definite, then $\norm{\cdot}$ is a norm.
\end{lem}
\begin{proof}
    \begin{enumerate}[$(i)$]
        \item clear by definition.
        \item Without loss of generality, assume $\Span{x,y} \neq 0$. 
        
        \emph{Special case:} $\Span{x,y} \in \IR_+$. Assume in addition that $\norm{x} \neq 0, \norm{y} \neq 0$. Set $\xi := \frac{x}{\norm{x}}$ and $\eta := \frac{y}{\norm{y}}$, so $\norm{\xi} = \norm{\eta} = 1$. We have
        \begin{equation}
            \begin{split}
                0 &\leq \Span{\xi - \eta, \xi - \eta} = \norm{\xi}^2 + \norm{\eta}^2 - \Span{\xi,\eta} - \Span{\eta,\xi} = \norm{\xi}^2 + \norm{\eta}^2 - 2\Rea\Span{\xi,\eta} \\
                \iff 1 &= \frac{1}{2}\left(\norm{\xi}^2 + \norm{\eta}^2\right) \geq \Rea\Span{\xi,\eta} = \frac{\Rea\Span{x,y}}{\norm{x}\norm{y}} \cdot \abs{\Span{x,y}} = \Rea\Span{x,y} \leq \norm{x}\norm{y}.
            \end{split}
        \end{equation}
        If $\norm{x} = 0$, apply the calculation with $\xi = kx$ and $\eta = \frac{1}{k}y$. Then, $\abs{\Span{x,y}} = \abs{\Span{\xi,\eta}}\leq \frac{\norm{y}^2}{k^2}$ for all $k > 0$ which implies $\Span{x,y} = 0$. A similar argument applies if $\norm{y} = 0$.

        \emph{From special case to general case:} Assume $\Span{x,y} \neq 0$. Define $\alpha := \frac{\Span{x,y}}{\norm{\Span{x,y}}}$, so $\norm{\alpha} = 1$, and define $\tilde{x} := \overline{\alpha} \cdot x$, so $\Span{\tilde{x},y} = \overline{\alpha} \cdot \Span{x,y} = \overline{\alpha} \alpha \abs{\Span{x,y}} = \abs{\Span{x,y}} > 0$.

        By the special case,
        \begin{equation}
            \abs{\Span{x,y}} = \abs{\Span{\tilde{x},y}} \leq \norm{\tilde{x}} \cdot \norm{y} = \norm{x} \cdot \norm{y}.
        \end{equation}
        \item Exercise.
        \item Exercise.
    \end{enumerate}
\end{proof}

\begin{prop}\label{prop:BxSeparableIffXfinite}
    The space $B(X) = B(X; \IR)$ is separable if and only if $X$ is finite.
\end{prop}

\begin{beispiel}
    If $X = \IN$, then $B(X) = \ell_\infty$, the space of bounded sequences, is not separable.
\end{beispiel}

\begin{proof}
    If $X$ is finite, wlog $X = \set{1,\ldots,n}$. Then,
    \begin{equation}
        B(X) = \set{f: \set{1,\ldots,n} \to \IR} = \set{(f_1,\ldots,f_n) \mid f_i \in \IR} \iso \IR^n
    \end{equation}
    As $\IR$ is separable because $\IQ$ is dense and countable, and finite products of separable spaces are separable, we have that $B(X)$ is separable.

    Assume $X$ is not finite. Idea: Find a set $A \subseteq B(X)$ which is not countable and such that
    \begin{equation}
        \norm{a - b} \geq \delta > 0 \text{ for all } a \neq b \in A. \label{eq:denseSeparableProp}
    \end{equation}
    If $B(X)$ was separable, there exists a $D \subseteq B(X)$ which is dense and countable.
    
    \emph{Claim:}~\eqref{eq:denseSeparableProp} implies that there exists an injective map $j: A \to D$.

    \emph{Proof of the claim:} Let $a \in A$. By density of $D$, there exists $a' \in D$ such that $\norm{a - a'} < \frac{\delta}{2}$. Similar, take $\norm{b - b'} < \frac{\delta}{2}$. If $a' = b'$, by the triangle inequality, we get $\norm{a - b} < \delta$, so we get $a = b$.

    Take $A = \set{f: X \to \set{0,1}} = \set{0,1}^X$; this set is uncountable by Cantor's second diagonalization argument (see Analysis I). If $f \neq g \in A$, then there exists an $A$ such that $f(x) - g(x) \in \set{-1,1}$, so we get that $\norm{f - g} \geq 1$.
\end{proof}

\begin{defi}{$C(X;Y), C_b(X;Y)$: Space of continuous (bounded) functions}{spaceOfContFct}
    Let $(X, \cT)$ be a topological space and $(Y, \norm{\cdot})$ a normed space. Then, we define\index{$C(X;Y)$}\index{$C_b(X;Y)$}\index{space!- of continuous functions}\index{space!- of continuous functions!- (bounded)}
    \begin{equation}
        \begin{split}
            C(X;Y) &:= \set{f: X \to Y \mid \text{$f$ is continuous}} \\
            C_b(X;Y) &:= \set{f: X \to Y \mid \text{$f$ is continuous and bounded}}.
        \end{split}
    \end{equation}
\end{defi}

In general, $C \neq C_b$, e.g. $X = \intervaloo{0,1}, Y = \IR, f = \frac{1}{x}.$

\begin{prop}
    Let $(Y, \norm{\cdot})$ be a Banach space and for $f \in C_b(X;Y)$ define 
    \begin{equation}
        \norm{f}_{C_{b}} := \sup_{x \in X} \norm{f(x)}.
    \end{equation}
    Then, $(C_b(X;Y), \norm{\cdot}_{C_b})$ is a Banach space.
\end{prop}

\begin{proof}
    We have $C_b(X; Y) \subseteq B(X; Y)$. Hence, it suffices to show that if $f_k \in C_b(X;Y)$ and if $\norm{f_k - f_*} \to 0$ for $f_* \in B(X;Y)$, then $f_* \in C_b(X;Y)$. The key for that is the triangle inequality:
    \begin{equation}
        \begin{split}
            \norm{f_*(z) - f_*(x)} \leq \underset{\leq \norm{f_k - f_*}}{\underbrace{\norm{f_*(z) - f_k(z)}}} + \underset{\substack{\text{use continuity} \\ \text{of $f_k$}}}{\underbrace{\norm{f_k(z) - f(k)_x}}} + \underset{\leq \norm{f_k - f_*}}{\underbrace{\norm{f_*(x) - f_k(x)}}}.
        \end{split}
    \end{equation}
    Let $\varepsilon > 0, x \in X$. There exists a $k \in \IN$ such that $\norm{f_k - f_*} < \frac{\varepsilon}{3}$.

    By the continuity of $f_k$ there exists an open neighbourhood $U$ of $x$ such that for all $z \in U$ we have $\norm{f_k(z) - f_k(x)} < \frac{\varepsilon}{3}$. Hence, if $z \in U$, we get 
    \begin{equation}
        \norm{f_*(z) - f_*(x)} < \frac{\varepsilon}{3} + \frac{\varepsilon}{3} + \frac{\varepsilon}{3} = \varepsilon.
    \end{equation}
\end{proof}

        \forall \varepsilon > 0 \exists \delta_{f,\varepsilon} > 0: \quad d(x,y) < \delta \Rightarrow \norm{f(x) - f(y)} < \varepsilon.
    \end{equation}
    Cover $X$ by finitely many $\delta$-balls $B(x_i,\delta)$ for $ = 1,\ldots,N$.
\end{proof}

\begin{thm}{$C(U;Y)$ for $U \subseteq \IR^n$}{CUYinRn}
    Let $U \subseteq \IR^n$ open, $(Y, \norm{\cdot})$ normed space. Let $K_i \subseteq U$ be compact with $K_i \subseteq K_{i+1}$ and $\bigcup_{i = 1}^\infty K_i = U$, and assume that for every $x \in U$, there exists an $r > 0$ and an $i$ such that $B(x,r) \subset K_i$. For $f \in C(U;Y)$ define 
    \begin{equation}
        [f]_i := \sup_{x \in K_i} \norm{f(x)}, \quad \quad d(f,g) := \sum_{i = 1}^\infty 2^{-i} \frac{[f - g]_i}{1 + [f - g]_i}.
    \end{equation}
    Then, $[\cdot]_i$ is a seminorm on $C(U; Y)$ and $d$ is a metric on $C(U; Y)$.

    If $Y$ is a Banach space, then $(C(U;Y), d)$ is complete. If $Y$ is separable, then $(C(U;Y), d)$ is separable.
\end{thm}
\begin{proof}
    $[\cdot]_i$ is a seminorm, see the triangle inequality above. A function could be $0$ on $K_i$ but not $0$ on the entire space.

    We get that 
    \begin{equation}
        d_i(f,g) = \frac{[f-g]_i}{1 + [f-g]_i}
    \end{equation}
    is a semimetric, so also 
    \begin{equation}
        d(f,g) = \sum_{i = 1}^\infty 2^{-i} d_i(f,g)
    \end{equation}
    is a semimetric. So we need to show that $d(f,g) = 0 \Rightarrow f = g$: If $d(f,g) = 0 \Rightarrow [f-g]_i = 0$, so $f - g = 0$ on $K_i$. As $U = \bigcup_{i = 1}^\infty K_i$, we get $f - g = 0$ on $U$.

    \emph{Completeness:} Let $j \mapsto f_j$ be a Cauchy sequence for $d$, so $f_j$ is Cauchy for each $[\cdot]_i$, so also $f_j|_{K_i}$ is a Cauchy sequence in $C(K_i; Y) = C_b(K_i; Y)$ as $K_i$ is compact. Hence, there exists a $f_{*,i}: K_i \to Y$ such that $f_j|_{K_i} \to f_{*,i}$. As $K_i \subseteq K_{i+1}$, we get that $f_{*,i+1}|_{K_i} = f_{*,i}$. With that, we can define $f_*: U \to Y$ such that $f_{*,i} = f_*|_{K_i}$. With that, we get that $\norm{f_j|_{K_i} - f_*|_{K_i}} \to 0$
    and with that, we get that $d(f_j, f_*) \to 0$.
\end{proof}

\begin{bemerkung}
    \begin{enumerate}[$(i)$]
        \item $\sum_{i = 1}^\infty 2^{-i} \frac{d_i}{1+d_i}$ is used in various examples e.g. $\IR^\IN$.
        \item There exists no norm on $C(U;Y)$ which induces the same topology as $d$ (similar to the homework for $\IR^\IN$).
        \item For each $U \subseteq \IR^n$ open, there exists a sequence $K_i$ of compact sets, e.g.
        \begin{equation}
            K_i := \set{x \in U \mid \dist(x, \IR^n \setminus U) \geq 2^{-i}, \abs{x} \leq 2^i}.
        \end{equation}
        \item Different choices of the sequence $K_i$ lead to the same topology on $C(U;Y)$ (or to equivalent metrics)
    \end{enumerate}
\end{bemerkung}

\begin{prop}
    Let $U \subseteq \IR^n$ open, $C(U;Y)$ as in Theorem~\ref{thm:CUYinRn}. Let $f: \IN \to C(U;Y)$ and $g \in C(U;Y)$. Then, the following statements are equivalent:
    \begin{enumerate}[$(i)$]
        \item $\lim_{j \to \infty} d(f_j,g) = 0$,
        \item $\lim_{j \to \infty} d_i(f_j, g) = 0$ for all $i$ (where $d_i(f,g) = [f - g]_i$),
        \item \textbf{for all} compact set $K \subseteq U$ we have 
        \begin{equation}
            \lim_{j \to \infty} \sup_{x \in K} \norm{f_j(x) - g(x)} = 0,
        \end{equation}
        we call this \emph{locally uniform convergence}\index{convergence!locally uniform -}\index{locally uniform convergence}.
    \end{enumerate}
\end{prop}

\begin{proof}
    \begin{description}
        \item[\enquote{$(iii) \Rightarrow (ii)$}:] obvious.
        \item[\enquote{$(ii) \Rightarrow (i)$}:] We have $\lim_{j \to \infty} d_i(f_j, g) = 0$, so also 
        \begin{equation}
            \lim_{j \to \infty} \frac{d_i(f_j, g)}{1 + d_i(f_j, g)} \to 0 \quad \Rightarrow \quad \sum_{i = 1}^N 2^{-i} \frac{d_i(f_j, g)}{1 + d_i(f_j, g)} \to 0
        \end{equation}
        and with that also 
        \begin{equation}
            \limsup_{j \to \infty} d(f_j, g) \leq \sum_{i = 1}^N \frac{d_i(f_j, g)}{1 + d_i(f_j, g)} + \sum_{i = N + 1}^\infty 2^{-i} \leq 2^{-N}
        \end{equation}
        and as $N$ was arbitary, this goes to $0$.
        \item[\enquote{$(i) \Rightarrow (iii)$}:] For each $x \in K$ there exist $i_x, r_x > 0$ such that $B(x, r_x) \subseteq K_{i_x}$. Then, we have $\bigcup_{x \in K} B(x, r_x) \supseteq K$. As $K$ is compact, there exist $x_1, \ldots, x_N, r_1, \ldots, r_n$ such that 
        \begin{equation}
            K \subseteq \bigcup_{i = 1}^N B(x_k, r_k) \quad \quad \text{where} \quad B(x_k, r_k) \subseteq K_{i_k}. 
        \end{equation}
    \end{description}
\end{proof}

\begin{defi}{Functions with compact support}{compactSupport}
    \begin{enumerate}[$(i)$]
        \item Let $(X,\cT)$ be a topological space, $(Y,\norm{\cdot})$ a normed space and $f: X \to Y$. Then, we define the \emph{support}\index{support} of $f$
        \begin{equation}
            \supp(f) := \overline{\set{x \in X \mid f(x) \neq 0}}.
        \end{equation}
        \item The space of continuous functions with compact support is defined as 
        \begin{equation}
            C_c(X; Y) := \set{f \in C(X; Y) \mid \supp(f) \text{ is compact}}.
        \end{equation}
        \item Let $U \subseteq \IR^n$ oben, $Y = \IR^m$. Then, we set 
        \begin{equation}
            C_c^k(U; \IR^m) := C^k(U; \IR^m) \cap C_c(U; \IR^m) \subseteq C^k(U; \IR^m)
        \end{equation}
        and
        \begin{equation}
            \cD(U; \IR^m) := C_c^\infty(U; \IR^m) = C^\infty(U; \IR^m) \cap C_c(U; \IR^m).
        \end{equation}
    \end{enumerate}
\end{defi}

\textbf{Notation.} Some people write $C_0(\ldots)$ instead of $C_c(\ldots)$. People wbo use $C_c(\ldots)$ sometimes define $C_0(U;\IR^m) := \overline{C_c(U;\IR^m)}$ in $C(\overline{U};\IR^m)$. Those are not the same as for $U = \intervaloo{-1,1}$ and $f(x) = 1 - \abs{x}$, we have $f \in C_0(U)$ but $f \notin C_c(U)$.

Interesting subspaces of the space of continuous functions arise when we consider functions with a given modulus of continuity $\rho : \intervalco{0,\infty} \to \intervalco{0,\infty}$, i.e., functions for which $\norm{f(x) - f(y)} \leq C_f \rho(d(x,y))$ where $\lim_{t \downarrow 0} \rho(t) = 0$. The most important examples are Hölder continuous functions, which correspond to $\rho(t) = t^\alpha$. For simplicity we focus on functions on $A \subseteq \IR^n$.

\begin{defi}{Hölder continuity}{hoelderCont}
    Let $\alpha \in \intervaloc{0,1}, A \subseteq \IR^n$ and $f: A \to \IR^n$. Then, we say that $f$ is \emph{Hölder continuous} if\index{Hölder!- continuous}\index{continuity!Hölder -} 
    \begin{equation}
        [f]_{\alpha,A} := \sup_{\substack{x,y \in A \\ x \neq y}} \frac{\abs{f(x) - f(y)}}{\abs{x - y}^\alpha}.
    \end{equation}
    Then, we define
    \begin{equation}
        C^{0,\alpha} := \set{f \in C_b(A; \IR^m) \mid [f]_{\alpha, A} < \infty}
    \end{equation}
    and
    \begin{equation}
        \norm{f}_{\alpha, A} := \sup_{x \in A} \abs{f(x)} + [f]_{\alpha,A}.
    \end{equation}
    If $U \subseteq \IR^n$ is open and bounded, we define 
    \begin{equation}
        C^{k,\alpha} := \set{f \in C^k(\overline{U}; \IR^m \mid \left[D^k f\right]_{0,\alpha,U} < \infty)}
    \end{equation}
    and 
    \begin{equation}
        \norm{f}_{k,\alpha,U} := \norm{f}_{C^k(U)} + \left[ D^k f \right]_{0,\alpha,U}.
    \end{equation}
\end{defi}

\begin{prop}
    $[\cdot]_{\alpha,A}$ is a seminorm on $C^{0,\alpha}(A; \IR^m)$ and the spaces $C^{0,\alpha}(A; \IR^m)$ and $C^{k,\alpha}(A; \IR^m)$ are Banach spaces with norms $\norm{\cdot}_{\alpha,U}$ and $\norm{\cdot}_{k,\alpha,U}$.
\end{prop}
