\chapter{Einleitung}
\begin{defi}{Optimierungsproblem, Approximationsalgorithmus}{optimizationProblemApproxAlgo}
    Ein \emph{Optimierungsproblem} $\Pi$ besteht aus
    \begin{itemize}
        \item einer Menge von Instanzen $\cI_\Pi$
        \item für jede Instanz $ I \in \cI_\Pi$
        \begin{itemize}
            \item einer Menge $\cS_I$ von Lösungen
            \item einer Zielfunktion $f_I: \cS_I \to \IR_{\geq 0}$.
        \end{itemize} 
    \end{itemize}
    Ein \emph{Approximationsalgorithmus}\index{Approximationsalgorithmus} $A$ nimmt $I$ als Eingabe und berechnet $A(I) \in \cS_I$ (in polynomieller Zeit).

    Sei $w_A(I) = f_I(A(I))$. Ein Approximationsalgorithmus erreicht Approximationsfaktor $r$, falls 
    \begin{equation*}
        w_A(I) \leq r \cdot \OPT(I) \quad \forall I \in \cI_\Pi \quad \quad \text{(für Minimierungsprobleme)},
    \end{equation*}
    bzw.
    \begin{equation*}
        w_A(I) \geq \frac{1}{r} \cdot \OPT(I) \quad \forall I \in \cI_\Pi \quad \quad \text{(für Maximierungsprobleme)}.
    \end{equation*}
\end{defi}

\section{Vertex Cover}
\newLecture{14.10.2025}


\begin{prob}{Vertex Cover}{VertexCover}
    \prblmObj{Ungerichteter Graph $G = (V, E)$}{Alle $V' \subseteq V$, sodass $\forall e = (u, v) \in E$ gilt $u \in V'$ oder $v \in V'$}{Minimiere $\abs{V'}$}\index{Vertex Cover}
\end{prob}




\begin{defi}{Matching}{Matching}
    Ein \emph{Matching}\index{Matching} $M \subseteq E$ eines Graphen $G = (V, E)$ ist eine Teilmenge der Kanten, sodass es keinen Knoten gibt, der zu mehr als einer Kante aus $M$ inzident ist.

    Es heißt \emph{inklusionsmaximal}\index{Matching!inklusionsmaximales -}, wenn für alle Kanten $e \in E \setminus M$ gilt: $M \cup \set{e}$ ist kein Matching.
\end{defi}
\begin{algorithm}[h]
    \caption{\textsc{Matching-VC}}\label{alg:matching-vc}\index{Matching-VC}
    \begin{algorithmic}[1] % Matching-VC
        \Input{ungerichteter Graph $G = (V, E)$}
        \Output{Vertex Cover $V' \subseteq V$ mit $\abs{V'} \leq 2 \cdot \OPT(G)$}
        \LComment{Berechne inklusionsmaximales Matching $M \subseteq E$:}
        \State{} $M := \emptyset$
        \ForAll{$e = \set{u, v} \in E$}
            \If{weder $u$ noch $v$ ist inzident zu einer Kante in $M$}
                \State{} $M := M \cup \set{e}$
            \EndIf{}
        \EndFor{}
        \State{} $V(M) := \set{v \in V \mid \text{$v$ ist inzident zu einer Kante in $M$}}$
        \State{} \Return{$V(M)$}
    \end{algorithmic}
\end{algorithm}


\begin{thm}{Matching-VC}{MatchingVC}
    Matching-VC hat Laufzeit $\cO(\abs{V} + \abs{E})$ für Graphen in Adjazenzlistendarstellung. Er berechnet ein Vertex Cover der Größe höchstens $2 \cdot \OPT(G)$, wobei $\OPT(G)$ die Größe des kleinsten Vertex Covers ist.
\end{thm}

\begin{proof}
    \begin{description}
        \item[Laufzeit:] In Schritt 1 wird jede Kante genau einmal betrachtet, $\cO(\abs{V} + \abs{E})$. 
        
        In Schritt 2 wird jeder Knoten höchstens einmal betrachtet, $\cO(\abs{V})$. Gemeinsam ergibt sich $\cO(\abs{V} + \abs{E})$.
        \item[Korrektheit:] Angenommen, $e = (u,v) \in E$ wurde nicht von $V(M)$ abgedeckt. Das heißt, es wären weder $u \in V(M)$ noch $v \in V(M)$. Dann wäre $M \cup \set{e}$ ein Matching und $e$ würde in Schritt 1 zu $M$ hinzugefügt werden. Widerspruch zur Inklusionsmaximalität von $M$. Also ist $V(M)$ ein Vertex Cover.
        \item[Approximation:] Sei $V^*$ ein minimales Vertex Cover. $V^*$ muss alle Kanten $e \in E$ abdecken, also auch alle $e \in M$. 
        
        Außerdem haben in $M$ keine zwei Kanten denselben Endpunkt. Deshalb ist $\abs{M} \leq \abs{V^*}$. 
        
        Für unsere Lösung gilt $\abs{V(M)} = 2 \cdot \abs{M} \leq 2 \cdot \OPT(G)$.
    \end{description}
\end{proof}

\section{Set Cover}\label{sec:SetCover}

\begin{prob}{Set Cover}{SetCover}
    \prblmObj{Menge $S$ mit $n$ Elementen, $m$ Teilmengen $S_1, \ldots, S_m \subseteq S$ mit $\bigcup_{i=1}^m S_i = S$, Kosten $c_1, \ldots, c_m \in \IN$.}{Alle $A \subseteq \set{1, \ldots, m}$ mit $\bigcup_{i \in A} S_i = S$}{Minimiere $\sum_{i \in A} c_i$}\index{Set Cover}
\end{prob}


\begin{algorithm}[h] % Greedy-SC
    \caption{\textsc{Greedy-SC}}\label{alg:greedySC}\index{Greedy-SC}
    \begin{algorithmic}[1]
        \Input{Grundmenge $S$ mit $n$ Elementen, $m$ Teilmengen $S_1, \ldots, S_m \subseteq S$ mit $\bigcup_{i=1}^m S_i = S$, Kosten $c_1, \ldots, c_m \in \IN$.}
        \Output{$A \subseteq \set{1, \ldots, m}$ mit $\bigcup_{i \in A} S_i = S$ und $\sum_{i \in A} c_i$ minimal}
        \State{} $A := \emptyset, C := \emptyset$
        \While{$C \neq S$}
            \State{} Wähle $S_i$ mit minimalen relativen Kosten $r_i(C) = \frac{c_i}{\abs{S_i \setminus C}}$
            \State{} Setze $\price(x) := r_i(C)$ für alle $x \in S_i \setminus C$
            \State{} $A := A \cup \set{i}$
            \State{} $C := C \cup S_i$
        \EndWhile{}
        \State{} \Return{$A$}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{Greedy-SC}{greedySC}
    \textsc{Greedy-SC} ist ein $H_n$-Approximationsalgorithmus für Set Cover, wobei
    \begin{equation*}
        H_n = \sum_{i = 1}^n \frac{1}{i}
    \end{equation*}
    die $n$-te harmonische Zahl ist.
\end{thm}

\emph{Untere Schranke für Greedy-SC:} Sei $\varepsilon > 0$ sehr klein. Betrachte $n$ Knoten und $m = n + 1$ Mengen, wobei $S_i = \set{i}$ mit Kosten $\frac{1}{n+1-i}$ ist (für $i \in S$) und $S_{n+1} = S$ mit Kosten $1 + \varepsilon$. Dann wählt \textsc{Greedy-SC} wählt alle Einzelnen, OPT die gesamte Menge.

\newLecture{15.10.2025}

\begin{proof}
    \emph{Beobachtung:} Es gilt 
    \begin{equation*}
        c(A) = \sum_{x \in S} \price(x).
    \end{equation*}
    Sei $x_1, \ldots, x_n$ die Reihenfolge der Elemente in $S$, wie sie vom \textsc{Greedy-SC} abgedeckt werden (Beim Abdecken von mehreren Elementen in gleicher Iteration wähle beliebige Reihenfolge).

    \begin{lem}{Preis in Greedy-SC}{GreedySCPrice}
        Für alle $k \in \set{1,\ldots,n}$ gilt
        \begin{equation*}
            \price(x_k) \leq \frac{\OPT}{n-k+1}.
        \end{equation*}
    \end{lem}
    \begin{proof}[Beweis: Lemma~$\ref{lem:GreedySCPrice}$]
        Betrachte ein beliebiges $k \in \set{1,\ldots,n}$. Sei $i \in \set{1,\ldots,m}$ der Index der Menge $S_i$, mit der $x_k$ erstmalig abgedeckt wird. Bezeichne $A$ die Auswahl der Mengen, \emph{bevor} $x_k$ abgedeckt wird.

        Sei $C := \bigcup_{i \in A} S_i$. Es gilt $\abs{C} \leq k - 1$. Eine optimale Auswahl $A^*$ von Mengen deckt ganz $S$ ab, insbesondere auch $S \setminus C$, wobei $\abs{S \setminus C} \geq n - k + 1$.

        Es gilt 
        \begin{equation*}
            \begin{split}
                \OPT &= \sum_{j \in A^*} c_j \geq \sum_{\substack{j \in A^* \\ S_j \setminus C \neq \emptyset}} c_j = \sum_{\substack{j \in A^* \\ S_j \setminus C \neq \emptyset}} \abs{S_j \setminus C} \underset{=r_j(C) \geq r_i(C)}{\underbrace{\frac{c_j}{\abs{S_j \setminus C}}}}  \\
                &\geq r_i(C) \underset{\geq \abs{S \setminus C}}{\underbrace{\sum_{\substack{j \in A^* \\ S_j \setminus C \neq \emptyset}} \abs{S_j \setminus C}}} \geq \underset{=\price(x_k)}{\underbrace{r_i(C)}} \cdot (n-k+1)
            \end{split}
        \end{equation*}
        wobei $r_j(C) \geq r_i(C)$ wegen der Greedy-Auswahl und 
        \begin{equation*}
            \sum_{\substack{j \in A^* \\ S_j \setminus C \neq \emptyset}} \abs{S_j \setminus C} \geq \abs{S \setminus C},
        \end{equation*}
        weil $A^*$ ganz $S \setminus C$ abdeckt.
    \end{proof}
    Nun gilt mit Lemma~\ref{lem:GreedySCPrice}
    \begin{equation*}
        \begin{split}
            c(A) &= \sum_{k = 1}^n \price(x_k) \leq \sum_{k = 1}^n \frac{\OPT}{n-k+1} = \OPT\sum_{i=1}^n \frac{1}{i} = \OPT \cdot H_n
        \end{split}
    \end{equation*}
    und Satz~\ref{thm:greedySC} ist gezeigt.
\end{proof}

\section{Scheduling auf identischen Maschinen}
\begin{prob}{Scheduling auf identischen Maschinen}{SchedulingIdenticalMachines}\index{Scheduling}\index{Scheduling!auf identischen Maschinen}
    \prblmObj{Menge $J = \set{1,\ldots,n}$ von Jobs, Menge $M = \set{1,\ldots,m}$ von Maschinen, Größen $p_j > 0$ für jeden Job $j \in J$}{Schedule $\pi: J \to M$, Last von Maschine $i$ in Schedule $\pi$: $L_i(\pi) = \displaystyle\sum_{\substack{j \in J \\ \pi(j) = i}} p_j$, Makespan $C(\pi) = \max_{i \in M} L_i(\pi)$}{Finde Schedule $\pi$ mit kleinstem Makespan}
\end{prob}

\begin{algorithm}[H]
    \caption{\textsc{LeastLoaded}}\label{alg:leastloaded}\index{LeastLoaded}
    \begin{algorithmic}[1]
        \Input{Siehe~\Cref{prob:SchedulingIdenticalMachines}}
        \Output{Siehe~\Cref{prob:SchedulingIdenticalMachines}}
        \State{Betrachte Jobs in Reihenfolge $1,\ldots,n$}
        \State{Weise jedem Job der Maschine mit bislang kleinster Last zu}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{LeastLoaded}{LeastLoaded}
    \textsc{LeastLoaded} ist ein $(2-\frac{1}{m})$-Approximationsalgorithmus für Scheduling auf identischen Maschinen.
\end{thm}

\begin{proof}
    Sei $\pi^*$ ein optimaler Schedule. Dann ist 
    \begin{equation*}
        C(\pi^*) \geq \frac{1}{m}\sum_{j \in J}  p_j,
    \end{equation*}
    denn
    \begin{equation*}
        C(\pi^*) = \max_{i \in M} \sum_{\substack{j \in J \\ \pi(j) = i}} p_j \geq \frac{1}{m}\sum_{i \in M}\sum_{\substack{j \in J \\ \pi(j) = i}} p_j = \frac{1}{m}\sum_{j \in J} p_j.
    \end{equation*}
    Außerdem gilt $C(\pi^*) \geq\displaystyle\max_{j \in J} p_j$.

    Sei $\pi$ der Schedule, der \textsc{LeastLoaded} berechnet. Sei $i$ die Maschine mit größter Last. Sei $j$ der Job, der zuletzt zu $i$ hinzugefügt wurde. Zu diesem Zeitpunkt war Maschine $i$ diejenige mit der geringsten Last. Ihre Last war höchstens $\frac{1}{m}\sum_{k = 1}^{j-1} p_k$. Es gilt 
    \begin{equation*}
        C(\pi) = L_i(\pi) \leq \frac{1}{m}\underset{\leq\sum_{k \in J \setminus \set{j}} p_k}{\underbrace{\sum_{k = 1}^{j-1} p_k}} + p_j \leq \underset{\leq C(\pi^*)}{\underbrace{\frac{1}{m} \sum_{k \in J} p_k}} + \left( 1 - \frac{1}{m} \right) \underset{\leq C(\pi^*)}{\underbrace{p_j}} \leq \left( 2 - \frac{1}{m} \right)C(\pi^*)
    \end{equation*}
    und die Abschätzung ist gezeigt.
\end{proof}

\begin{rmk}
    $2-\frac{1}{m}$ ist eine untere Schranke für \textsc{LeastLoaded}.
    \begin{proof}
        Sei $n = m(m-1) + 1$ und $p_1 = \dots = p_{m(m-1)} = 1$ und $p_{m(m-1)+1} = m$. 

        \textsc{LeastLoaded} führt auf jeder Maschine $m-1$ Jobs der Größe $1$ durch. Auf einer Maschine wird zusätzlich ein Job der Größe $m$ ausgeführt. Somit ist der Makespan $m-1+m = 2m-1$.

        Eine optimale Belegung führt auf $m-1$ Maschinen je $m$ Jobs der Größe $1$ aus und auf der letzten Maschine einen Job der Größe $m$, wir haben also einen Makespan von $m$, und der Faktor stimmt.
    \end{proof}
\end{rmk}

\begin{algorithm}
    \caption{\textsc{LongestProcessingTime} (LPT)}\index{LongestProcessingTime}\index{LPT}
    \begin{algorithmic}[1]
        \Input{Siehe~\Cref{prob:SchedulingIdenticalMachines}}
        \Output{Siehe~\Cref{prob:SchedulingIdenticalMachines}}
        \State{Sortiere Jobs so, dass $p_1 \geq p_2 \geq \dots \geq p_n$.}
        \State{Führe \textsc{LeastLoaded} auf den sortierten Jobs aus.}
    \end{algorithmic}
\end{algorithm}
\begin{thm}{LPT}{LPT}
    \textsc{LongestProcessingTime} ist ein $\frac{4}{3}$-Approximationsalgorithmus.
\end{thm}
\begin{proof}
    Angenommen, es gäbe eine Instanz, sodass LPT einen Makespan von mehr als $\frac{4}{3} \cdot \OPT$ erreicht. Betrachte unter diesen Instanzen eine mit geringster Anzahl Jobs.
    
    Sei $i$ eine Maschine mit höchster Last in LPT-Schedule $\pi$. Sei $j$ der letzte Job, der Maschine $i$ zugewiesen wird. Es ist $j = n$, da sonst $p_1, \ldots, p_j$ ein Gegenbeispiel mit weniger Jobs wäre. Es gilt 
    \begin{equation*}
        C(\pi) = L_i(\pi) \leq \frac{1}{m}\sum_{k=1}^{n-1} p_k + p_n \leq \OPT + p_n.
    \end{equation*}
    Aus $C(\pi) > \frac{4}{3}\OPT$ folgt $p_n > \frac{1}{3}\OPT$. Da $p_1 \geq \dots \geq p_n$ folgt $p_j > \frac{1}{3}\OPT$ für alle $j \in J$. Also erhält im optimalen Schedule jede Maschine höchstens zwei Jobs. 
    
    Sei $k$ die Anzahl Jobs mit Größe $\frac{2}{3}\OPT$. Im optimalen Schedule sind $k$ Jobs alleine auf einer Maschine und $n-k$ Jobs höchstens zu zweit. 

    \begin{description}
        \item[Fall 1: $k = n$.] Also ist $n \leq m$, Job $n$ wird also einer leeren Maschine zugewiesen. 
        \item[Fall 2: $k < n$.] Zum Zeitpunkt bevor Job $n$ hinzugefügt wird, hat LPT mindestens eine Maschine mit höchstens einem Job und dieser hat Größe höchstens $\frac{2}{3}\OPT$. Wegen $p_n \leq \frac{2}{3}\OPT$ ist die Last dieser Maschine im Anschluss höchstens $\frac{4}{3}\OPT$.
    \end{description}
    Beides ist ein Widerspruch zur Annahme. $\lightning$
\end{proof}

\begin{rmk}
    $\frac{4}{3}$ ist eine untere Schranke für \textsc{LongestProcessingTime}.
    \begin{proof}
        Übungsaufgabe.
    \end{proof}
    Wir werden später sehen, dass für alle $\varepsilon > 0$ ein $(1+\varepsilon)$-Approximationsalgorithmus existiert.
\end{rmk}

\section{Rucksackproblem}

\begin{prob}{Rucksackproblem (Knapsack Problem, KP)}{knapsackProblem}
    \prblmObj{Kapazität $t \in \IN$, Nutzen $p_1, \ldots, p_n \in \IN$, Gewichte $w_1, \ldots, w_n \in \set{1,\ldots,t}$}{Teilmenge $A \subseteq \set{1,\ldots,n}$ mit $w(A) := \displaystyle \sum_{i \in A} w_i \leq t$}{Maximiere $p(A) := \displaystyle \sum_{i \in A} p_i$}\index{Rucksackproblem}\index{Knapsack-Problem}\index{KP}
\end{prob}

\begin{algorithm}
    \caption{\textsc{Greedy-KP}}\index{Greedy-KP}\label{alg:greedyKP}
    \begin{algorithmic}[1]
        \Input{Siehe~\Cref{prob:knapsackProblem}}
        \Output{Siehe~\Cref{prob:knapsackProblem}}
        \State{Sortiere Objekte nach Effizienz, sodass $\frac{p_1}{w_1} \geq \dots \geq \frac{p_n}{w_n}$}
        \State{$B := \emptyset$, $i := 1$}
        \While{$i \leq n$ \textbf{and} $w(B) + w_i \leq t$}
            \State{$B := B \cup \set{i}$}
            \State{$i := i + 1$}
        \EndWhile{}
        \State{Sei $i^*$ ein Objekt mit dem größten Nutzen}
        \If{$p(B) < p_{i^*}$}
            \State{$A := \set{i^*}$}
            \Else{} $A := B$
        \EndIf{}
        \State{\Return{$A$}}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{Greedy-KP}{greedyKP}
    \textsc{Greedy-KP} ist ein $\frac{1}{2}$-Approximationsalgorithmus.
\end{thm}

\begin{proof}
    Sei $B := \set{1,\ldots,i}$ die im Algorithmus berechnete Menge. Falls $i = n$, gibt es nichts zu zeigen.

    Falls $i < n$, gilt $w(B) + w_{i+1} > t$. 

    \emph{Behauptung:} Es gilt $p(B) + p_{i+1} \geq \OPT$:

    Angenommen, $\OPT > p(B) + p_{i+1}$. Dann gibt es eine Menge $A' \subseteq \set{1,\ldots,n}$, für die gilt, dass $w(A') \leq t < w(B) + w_{i+1}$ und $p(A') = \OPT > p(B) + p_{i+1}$. 
    
    Betrachte $J := \set{1,\ldots,i+1} \setminus A'$ und $J' := A' \setminus \set{1,\ldots,i+1}$. Nun gilt 
    \begin{equation*}
        p(B) + p_{i+1} < p(A') = (p(B) + p_{i+1}) + p(J') - p(J) \implies p(J') > p(J)
    \end{equation*}
    und analog
    \begin{equation*}
        w(B) + w_{i+1} > w(A') = (w(B) + w_{i+1}) + w(J') - w(J) \implies w(J') < w(J).
    \end{equation*}
    Sei $r = \frac{p_{i+1}}{w_{i+1}}$, dann 
    \begin{equation*}
        p(J) \geq r \cdot w(J), \quad \quad p(J') \leq r \cdot w(J').
    \end{equation*}
    Zusammen ergibt sich 
    \begin{equation*}
        p(J) \geq r \cdot w(J) > r \cdot w(J') \geq p(J') > p(J). \quad \quad \lightning
    \end{equation*}
    Damit ist die Behauptung gezeigt und es gilt $p(B) + p_{i+1} \geq \OPT$.

    Somit gilt 
    \begin{equation*}
        p(A) = \max\set{p(B), p_{i^*}} \geq \frac{1}{2}(p(B) + p_{i^*}) \geq \frac{1}{2}(p(B) + p_{i+1}) \geq \frac{1}{2}\OPT
    \end{equation*}
    und die Aussage ist gezeigt.
\end{proof}


\chapter{Runden und dynamische Programmierung}

\begin{defi}{Approximationsschema, PTAS, FPATS}{ApproxScheme}
    Ein \emph{Approximationsschema}\index{Approximationsschema} $A$ für ein Optimierungsproblem $\Pi$ ist ein Algorithmus, der zu jeder Eingabe $(I, \varepsilon)$ mit $I \in \cI_\Pi, \varepsilon > 0$ eine Lösung $A(I, \varepsilon) \in \cS_I$ berechnet und es gilt 
    \begin{equation*}
        \begin{split}
            f(A(I,\varepsilon)) &\leq (1+\varepsilon)\OPT(I) \quad \quad \text{bei Minimierungsproblemen, bzw.} \\ 
            f(A(I,\varepsilon)) &\geq (1-\varepsilon)\OPT(I) \quad \quad \text{bei Maximierungsproblemen.}
        \end{split}
    \end{equation*}
    $A$ heißt \emph{polynomielles Approximationsschema}\index{Approximationsschema!polynomielles -} (PTAS)\index{PTAS}, wenn die Laufzeit für jedes $\varepsilon$ polynomiell in $\abs{I}$ beschränkt ist.

    $A$ heißt \emph{voll-polynomielles Approximationsschema}\index{Approximationsschema!voll-polynomielles -} (FPTAS)\index{FPATS}, wenn die Laufzeit polynomiell in $\frac{1}{\varepsilon}$ und $\abs{I}$ ist.
\end{defi}

\begin{expl}\leavevmode
    \begin{itemize}
        \item $\Theta(\abs{I}^{1/\varepsilon})$ ist PTAS, aber kein FPTAS
        \item $\Theta\left( \frac{\abs{I}^{100}}{\varepsilon^{10}} \right)$ ist ein FPTAS
    \end{itemize}
\end{expl}

\section{Rucksackproblem}

\subsection{Dynamische Programmierung}\index{Rucksackproblem}\index{KP}\index{Knapsack-Problem}

Sei $W(i,p)$ \emph{geringste} Gewicht unter allen Teilmengen von $\set{1,\ldots,i}$, die Nutzen mindestens $p$ erreichen, bzw. $\infty$, wenn das nicht möglich ist.

Falls $I \subseteq \set{1,\ldots,i}$ gilt: Falls $\sum_{i \in I} p_i \geq p$, dann ist $\sum_{i \in I} w_i \geq W(i, p)$.

Es gilt
\begin{equation*}
    \begin{split}
        W(0,p) &= \begin{cases}
            0 & p \leq 0 \\ 
            \infty & p > 0,
        \end{cases} \\
        W(1,p) &= \begin{cases}
            0 & p \leq 0 \\
            w_1 & 0 < p \leq p_1 \\
            \infty & p > p_1.
        \end{cases}
    \end{split}
\end{equation*}

Sei $P := \max_{\set{1,\ldots,n}} p_i$. Uns interessieren $W(i,p)$ für $i = 0,\ldots,n$ und $p = 0,\ldots,nP$.

\emph{Rekursionsformel für $W(i,p)$:}

Sei $I \subseteq \set{1,\ldots,i}$ diejenige Teilmenge mit minimalem Gewicht unter allen mit $\sum_{j \in I} p_j \geq p$.

\begin{description}
    \item[Fall 1: $i \notin I$.] Dann gilt $W(i,p) = W(i-1,p)$.
    \item[Fall 2: $i \in I$.] Dann betrachte $I \setminus \set{i}$. Es ist $W(I \setminus \set{i}) = W(i,p) - w(i)$ und $p(I \setminus \set{i}) \geq p - p_i$. Daraus folgt 
    \begin{equation*}
        W(i-1, p-p_i) \leq W(i,p) - w_i.
    \end{equation*}
    Umgekehrt sei $I' \subseteq \set{1,\ldots,i-1}$ diejenige Menge, die $W(i-1,p-p_i)$ liefert. Nun gilt
    \begin{equation*}
        W(I' \cup \set{i}) = W(i-1,p-p_i) + w_i, \quad \quad p(I' \cup \set{i}) = p(I') + p_i \geq p.
    \end{equation*}
    Daraus folgt, dass $W(i,p) \leq W(i-1,p-p_i) + w_i$.

    Umstellen liefert 
    \begin{equation*}
        W(i,p) = W(i-1, p-p_i) + w_i.
    \end{equation*}
\end{description}
Insgesamt ergibt sich die Rekursion 
\begin{equation*}
    W(i,p) = \min\set{W(i-1,p), W(i-1,p-p_i) + w_i}, \quad \quad W(0,p) = \begin{cases}
        \infty & p > 0 \\ 
        0 & p \leq 0.
    \end{cases}
\end{equation*}


\begin{algorithm}
    \caption{\textsc{DynKP}}\label{alg:DynKP}\index{DynKP}
    \begin{algorithmic}[1]
        \Input{Siehe~\Cref{prob:knapsackProblem}}
        \Output{Siehe~\Cref{prob:knapsackProblem}}
        \State{$P := \max_{i \in \set{1,\ldots,n}} p_i$}
        \For{$i = 1$ \textbf{to} $n$}
            \For{$p = 1$ \textbf{to} $nP$}
                \State{$W(i,p) = \min\set{W(i-1,p),W(i-1,p-p_i) + w_i}$}
            \EndFor{}
        \EndFor{}
        \State{\Return{maximales $p \in \set{1,\ldots,nP}$ mit $W(n,p) \leq t$.}}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{DynKP}{DynKP}
    \textsc{DynKP} liefert in Zeit $\Theta(n^2 P)$ den maximal erreichbaren Nutzen.
\end{thm}

\subsection{Approximationsschema}
\begin{algorithm}
    \caption{\textsc{FPTAS-KP}}\label{alg:FPTAS-KP}\index{FPTAS-KP}
    \begin{algorithmic}[1]
        \Input{Siehe~\Cref{prob:knapsackProblem}, $\varepsilon > 0$}
        \Output{Siehe~\Cref{prob:knapsackProblem}}
        \State{$P := \max_{i \in \set{1,\ldots,n}} p_i$}
        \State{$K := \frac{\varepsilon P}{n}$}
        \For{$i = 1$ \textbf{to} $n$}
            \State{$p_i' = \floor*{\frac{p_i}{K}}$}
        \EndFor{}
        \State{Benutze \textsc{DynKP} auf $p_1', \ldots, p_n', w_1, \ldots, w_n, t$}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{FPTAS-KP}{FPTAS-KP}
    \textsc{FPTAS-KP} hat eine Laufzeit von $\cO(\frac{n^3}{\varepsilon})$ und berechnet eine $1-\varepsilon$-Approximation.
\end{thm}
\begin{proof}\leavevmode
    \begin{description}
        \item[Laufzeit:] \textsc{DynKP} braucht Zeit $\Theta(n^2 P')$, wobei $P' = \max_{i \in \set{1,\ldots,n}} p_i'$. Es gilt für alle $i$:
        \begin{equation*}
            p_i' = \floor*{\frac{p_i}{K}} \leq \floor*{\frac{P}{K}} = \floor*{\frac{n}{\varepsilon}} \leq \frac{n}{\varepsilon},
        \end{equation*}
        also ist auch $P' \leq \frac{n}{\varepsilon}$.
        \item[Approximationsfaktor:] Sei $I'$ eine optimale Lösung auf $p_1', \ldots, p_n'$ und $I$ eine optimale Lösung auf $p_1, \ldots, p_n$.
        
        $I'$ ist auch eine optimale Lösung auf $\set{p_i^*} := \set{K \cdot p_i'}$.
        \begin{expl}
            Für $n = 4, P = 50, \varepsilon = \frac{4}{5}, K = \frac{\varepsilon P}{n} = 10$. Für $p_1 = 33$ erhalten wir $p_1' = 3$ bzw. $p_1^* = 30$.
        \end{expl}

        Es gilt $p_i^* = K \cdot \floor*{\frac{p_i}{K}} \geq K \cdot \left( \frac{p_i}{K} - 1 \right) = p_i - K$ und $p_i^* \leq K \cdot \frac{p_i}{K} = p_i$, also $p_i^* \in \edgys{p_i - K, p_i}$.

        Für alle $J \subseteq \set{1,\ldots,n}$ gilt 
        \begin{equation*}
            p^*(J) := \sum_{i \in J} p_i^* \leq \sum_{i \in J} p_i = p(J)
        \end{equation*}
        und 
        \begin{equation*}
            p^*(J) = \sum_{i \in J} p_i^* \geq \sum_{i \in J} (p_i - K) = \sum_{i \in J} p_i - \abs{J} \cdot K \geq p(J) - nK.
        \end{equation*}
        Damit ist $p(I') \geq p^*(I') \geq p^*(I) \geq p(I) - nK$, da $I'$ optimal hinsichtlich $p^*$ ist.

        Weiterhin ist $p(I) \geq P$, also gilt 
        \begin{equation*}
            \frac{p(I')}{\OPT} \geq \frac{p(I) - nK}{p(I)} \geq 1 - \frac{nK}{P} = 1 - \varepsilon
        \end{equation*}
        und wir erhalten $1-\varepsilon$ als Approximationsfaktor.
    \end{description}
\end{proof}

\section{Scheduling auf identischen Maschinen}\index{Scheduling}\index{Scheduling!auf identischen Maschinen}

\emph{Erster Ansatz für Approximationsschema:} Wir nennen einen Job $j \in J$ \emph{klein}, falls 
\begin{equation*}
    p_j \leq \varepsilon \cdot \frac{1}{m} \sum_{k = 1}^n p_k,
\end{equation*}
sonst heißt er \emph{groß}. Wir betrachten folgenden Algorithmus:

\begin{algorithm}
    \caption{\textsc{GroßKleinScheduling}}\label{alg:grossKlein}
    \begin{algorithmic}[1]
        \Input{Siehe~\Cref{prob:SchedulingIdenticalMachines}, $\varepsilon > 0$}
        \Output{Siehe~\Cref{prob:SchedulingIdenticalMachines}}
        \State{Berechne optimalen Schedule auf großen Jobs} \Comment{Bspw. durch Ausprobieren aller Schedules}
        \State{Ergänze kleine Jobs mit \textsc{LeastLoaded}}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{GroßKleinScheduling}{grossKlein}
    \textsc{GroßKleinScheduling} ist ein PTAS mit Laufzeit $m^{\frac{m}{\varepsilon}}$.
\end{thm}

\begin{proof}
    \begin{description}
        \item[Laufzeit:] Für Algorithmus~\ref{alg:grossKlein} gibt es höchstens $\frac{m}{\varepsilon}$ große Jobs, denn jeder ist mindestens $\varepsilon\frac{1}{m}\sum_{k=1}^n p_k$ groß. Damit gibt es maximal $m^{\frac{m}{\varepsilon}}$ mögliche Schedules. Damit hat Algorithmus~\ref{alg:grossKlein} polynomielle Laufzeit für jedes $\varepsilon > 0$, wenn $m$ konstant ist.
        \item[Approximationsfaktor:] Betrachte die Maschine mit der größten Last.
        \begin{description}
            \item[Fall 1:] Diese Maschine erhält nur große Jobs in unserem Schedule. Dann ist der Makespan durch die kleinen Jobs unverändert. Unser Schedule ist optimal auf den großen Jobs, und somit auch optimal auf allen Jobs.
            \item[Fall 2:] Diese Maschine erhält auch einen kleinen Job. Im Moment der Zuweisung des letzten solchen Jobs durch \textsc{LeastLoaded} ist die Last höchstens die $\frac{1}{m}\sum_{k=1}^n p_k$. Der letzte Job ist ein kleiner Job, somit haben wir am Ende eine Gesamtlast von höchstens 
            \begin{equation*}
                (1+\varepsilon)\frac{1}{m}\sum_{k=1}^n p_k \leq (1+\varepsilon)\OPT.
            \end{equation*}
        \end{description}
    \end{description}
\end{proof}

\subsection{Bin-Packing mit konstant vielen Objektgrößen}

\begin{prob}{Bin-Packing}{binPacking}\index{Bin-Packing}
    \prblmObj{Objektgrößen $s_1, \ldots, s_n \in \IN$, Kapazität $T \in \IN$}{Partitionen $C_1, \ldots, C_\ell$ der Menge $\set{1,\ldots,n}$ mit $\sum_{j \in C_i} s_j \leq T$ für alle $i$}{Minimiere $\ell$}
\end{prob}

\emph{Spezialfall:} Wir wählen als Eingabe $(s_1, \ldots, s_z) \in \IN^z, (n_1, \ldots, n_z) \in \IN^z, T \in \IN$. Dabei haben wir jeweils $n_i$ Objekte der Größe $s_i$.

Wir bezeichnen mit $\BINS(i_1, \ldots, i_z)$ den optimalen Zielfunktionswert, wenn es $i_j \leq n_j$ Objekte der Größe $s_j$ gibt. Uns interessiert $\BINS(n_1, \ldots, n_z)$. Berechne aller Werte $\BINS(i_1, \ldots, i_z)$ für $(i_1, \ldots, i_z) \in \IN_0^z$ mit $i_j \leq n_j$. Wenn $z$ konstant ist, gibt es höchstens $(n+1)^z \in \cO(n^z)$ viele solche Vektoren.

Definiere 
\begin{equation*}
    C := \set*{(i_1, \ldots, i_z) \in \IN_0^z ~\middle|~ \sum_{j = 1}^z i_j s_j \leq T} \setminus \set{(0,\ldots,0)}.
\end{equation*}

Wir bemerken:
\begin{itemize}
    \item $\BINS(0,\ldots,0) = 0$
    \item $\BINS(i_1, \ldots, i_z) = 1$ für $(i_1, \ldots, i_z) \in C$
\end{itemize}
Für die allgemeine Größe von $\BINS(i_1, \ldots, i_z)$ betrachte ein optimales Packing. Der erste Bin erhalte $(c_1, \ldots, c_z) \in C$ viele Objekte. So erhalten die übrigen Bins insgesamt $(i_1 - c_1, \ldots, i_z - c_z)$ Objekte und es gilt 
\begin{equation*}
    \BINS(i_1, \ldots, i_z) = 1 + \BINS(i_1 - c_1, \ldots, i_z - c_z).
\end{equation*}
Als allgemeine Rekursionsformel ergibt sich 
\begin{equation*}
    \BINS(i_1, \ldots, i_z) = 1 + \min_{(c_1, \ldots, c_z) \in C} \BINS(i_1 - c_1, \ldots, i_z - c_z) \quad \quad \text{für alle } (i_1, \ldots, i_z) \neq (0, \ldots, 0)
\end{equation*}
und $\BINS(0,\ldots,0) = 0$.

Wir berechnen Werte von $\BINS(i_1, \ldots, i_z)$ aufsteigend nach $\sum_{j=1}^z i_j$. Insgesamt werden $\cO(n^z)$ viele Vektoren mit jeweils Laufzeit $\cO(n^z)$ pro Vektor, damit ergibt sich eine Gesamtlaufzeit von $\cO(n^{2z})$, also folgender Satz:

\begin{thm}{Bin-Packing}{binPacking}
    Das Bin-Packing-Problem mit $z$ verschiedenen Objektgrößen kann für Instanzen mit $n$ Objekten in Zeit $\cO(n^{2z})$ gelöst werden (wobei die Konstante in der $\cO$-Notation von $z$ abhängt.) 
\end{thm}

\subsection{Approximationsschema}

\emph{Zusätzliche Eingabe:} $T$. Der Algorithmus liefert entweder einen Schedule mit Makespan höchstens $(1+\varepsilon)T$ oder er beweist, dass $\OPT > T$. 

Wir bezeichnen einen Job $j$ als \emph{groß}, falls $p_j > \varepsilon T$. Wir runden die Größen aller großen Jobs auf Vielfache von $\varepsilon^2 T$ ab, somit gibt es höchstens $\frac{1}{\varepsilon^2}$ viele Jobgrößen. Auf diesen bestimmen wir eine optimale Bin-Packing-Lösung. Wir erhalten als Antwort entweder eine Packung der großen Jobs auf $m$ Maschinen mit Makespan $\leq T$ (bzgl. der gerundeten Jobgrößen), oder dass es nicht möglich ist (dann brechen wir ab). Für die kleinen Jobs verwenden wir anschließend \textsc{LeastLoaded}.

Für den Makespan gilt insgesamt:
\begin{itemize}
    \item Erhält eine Maschine einen kleinen Job, betrachte den letzten derartigen Job. Dann gilt 
    \begin{equation*}
        L_i(\pi) \leq \underset{\leq T}{\underbrace{\left( \frac{1}{m}\sum_{k=1}^n p_k \right)}} + p_j \leq (1+\varepsilon)T.
    \end{equation*}
    \item Erhält eine Maschine keinen kleinen Job, dann ist die Last durch die gerundeten Jobgrößen höchstens $T$. Sei $p_j'$ die abgerundete Jobgröße und $p_j$ die tatsächliche Jobgröße. Dann gilt
    \begin{equation*}
        p_j \geq \varepsilon T, \quad \quad p_j' \geq p_j - \varepsilon^2 T \geq p_j - \varepsilon p_j' \implies p_j \leq (1+\varepsilon) p_j',
    \end{equation*}
    unter der Voraussetzung, dass $\frac{1}{\varepsilon} \in \IN$.
\end{itemize}
Somit sind nicht gerundete Jobgrößen höchstens um den Faktor $(1+\varepsilon)$ größer und somit ist die Last der Maschine $\leq (1+\varepsilon) T$ bezüglich der tatsächlichen Jobgrößen.

Hier der Algorithmus nochmal in Pseudocode:
\begin{algorithm}
    \caption{$B_\varepsilon(T)$}\label{alg:BEpsilon}
    \begin{algorithmic}[1]
        \State{Sei $J' = \set{j \in J \mid p_j > \varepsilon T}$ die Menge der großen Jobs}
        \State{Für $j \in J$ sei $p_j'$ der Wert $p_j$ auf das nächste Vielfache von $\varepsilon^2 T$ abgerundet. Sei $I'$ die Instanz, die nur aus Jobs mit den Größen $p_j'$ mit $j \in J'$ besteht.}
        \State{Konstruiere eine Instanz für Bin-Packing mit Objektgrößen $\set{p_j' \mid j \in J'}$ und Kapazität $T$. Bezeichne $a$ die minimale Anzahl an Bins, die für diese Instanz benötigt werden. Berechne $a$ und die dazugehörige Verteilung $\pi'$ der Objekte mit Hilfe des dynamischen Programms aus Satz~\ref{thm:binPacking}.}
        \If{$a > m$} 
        \State{\Return{$\OPT(I) \geq \OPT(I') > T$}}
        \Else{} 
        \State{Erweitere $\pi'$ auf alle Jobs aus $J$ durch Hinzufügen der Jobs aus $J \setminus J'$ mittels \textsc{LeastLoaded}.}
        \State{\Return{resultierenden Schedule $\pi$}}
        \EndIf{}
    \end{algorithmic}
    
\end{algorithm}

Somit haben wir gezeigt:
\begin{lem}{$B_\varepsilon(T)$}{BEpsilon}
    Es gibt einen Algorithmus, der in Zeit $\cO(m^{\frac{2}{\varepsilon^2}})$ entweder beweist, dass es keinen Schedule mit Makespan $T$ gibt oder einen Schedule mit Makespan $(1+\varepsilon)T$ berechnet.
\end{lem}
Für einen Optimalen Makespan gilt nun
\begin{equation*}
    \begin{split}
        \OPT &\geq \ceil*{\frac{1}{m}\sum_{k=1}^n p_k} \\
        \OPT &\geq p_{\max} \\
        \OPT &\leq \ceil*{\frac{1}{m}\sum_{k=1}^n p_k} + p_{\max}.
    \end{split}
\end{equation*}
Setze initial $L := \max\set*{\ceil*{\frac{1}{m}\sum_{k=1}^n p_k}, p_{\max}}$ und $U := \ceil*{\frac{1}{m}\sum_{k=1}^n p_k} + p_{\max}$. Dabei haben wir folgende Invarianten:
\begin{enumerate}
    \item $L \leq \OPT$
    \item Wir können einen Schedule mit Makespan höchstens $(1+\varepsilon)U$ in Zeit $\cO(m^{\frac{2}{\varepsilon^2}})$ berechnen.
\end{enumerate}
Nun berechne $T := \floor*{\frac{L + U}{2}}$ und prüfe, ob ein Schedule mit Makespan $T$ berechnet werden kann. Falls ja, setze $U := T$, sonst setze $L := T + 1$. Es gilt $U - L \leq \frac{p_{\max}}{2^i}$ nach $i$ Iterationen, somit gilt $U - L \leq \frac{1}{2}$ nach $ \floor*{\log_2(p_{\max})}+1$ Iterationen. Dann muss $L = U$ gelten und wir erhalten als Resultat folgenden Satz:

\begin{thm}{PTAS Scheduling}{PTASScheduling}
    Für Scheduling auf identischen Maschinen existiert ein PTAS mit einer Laufzeit von $\cO(m^{\frac{2}{\varepsilon^2}}\log(p_{\max}))$.
\end{thm}

\subsection{Untere Schranke für Approximierbarkeit}

\begin{defi}{Stark NP-schwer}{stronglyNPhard}
    Sei $\Pi$ ein Optimierungsproblem, in dessen Instanzen ganze Zahlen enthalten sind. $\Pi$ heißt \emph{stark NP-schwer}\index{stark NP-schwer}, wenn es ein Polynom $q$ gibt, sodass $\Pi$ eingeschränkt auf Instanzen $I$, in denen alle Zahlen durch $q(\abs{I})$ beschränkt sind, bereits NP-schwer ist.
\end{defi}

Scheduling auf identischen Maschinen ist tatsächlich stark NP-schwer.

\begin{thm}{Scheduling hat keinen FPTAS}{schedulingNoFPTAS}
    Unter der Annahme $\mathrm{P} \neq \mathrm{NP}$ existiert kein FPTAS für Scheduling auf identischen Maschinen.
\end{thm}
\begin{proof}
    Angenommen, es gäbe einen FPTAS. Weil Scheduling auf identischen Maschinen stark NP-schwer ist, gibt es ein Polynom $q$, sodass es bereits NP-schwer ist, das Problem mit Jobgrößen $p_1 \leq \cdots \leq p_n \in \set{1,\ldots, q(n)}$ zu lösen.

    Berechne mit dem FPTAS eine $(1+\varepsilon)$-Approximation, wobei $\varepsilon = \frac{1}{2nq(n)}$. Die Laufzeit ist polynomiell in $n$ und $\frac{1}{\varepsilon} = 2nq(n)$, also beides polynomiell in $n$. Für den Approximationsfaktor erhalten wir $w_A(I, \varepsilon) \leq (1+\varepsilon) \OPT$.
    Es gilt 
    \begin{equation*}
        \OPT(I) \leq \sum_{i = 1}^n p_i \leq n \cdot q(n)
    \end{equation*}
    und folglich 
    \begin{equation*}
        (1+\varepsilon)\OPT = \OPT + \underset{\leq \frac{1}{2}}{\underbrace{\varepsilon\OPT}}  \leq \OPT(I) + \varepsilon n q(n) = \OPT + \frac{1}{2}.
    \end{equation*}
    Weil alle Jobgrößen ganzzahlig sind, sind auch $w_A(I, \varepsilon)$ und $\OPT(I)$ ganze Zahlen. Da die Differenz höchstens $\frac{1}{2}$ ist, ist $w_A(I, \varepsilon) = \OPT(I)$. Also löst der FPTAS das Problem exakt in polynomieller Zeit. Also muss $\mathrm{P} = \mathrm{NP}$ sein. $\lightning$ zur Annahme
\end{proof}

\section{Grundlagen der Wahrscheinlichkeitsrechnung}

\subsection{Diskrete Wahrscheinlichkeitsräume}

\begin{defi}{Wahrscheinlichkeitsraum}{probabilitySpace}
    Sei $\Omega$ eine endliche oder abzählbare Menge (\emph{Ergebnismenge})\index{Wahrscheinlichkeitsraum!Ergebnismenge $\Omega$}\index{Ergebnismenge $\Omega$}. 
    
    Elemente von $\Omega$ heißen \emph{Elementarereignisse}\index{Elementarereignis}\index{Wahrscheinlichkeitsraum!Elementarereignis}, Teilmengen von $\Omega$ heißen \emph{Ereignisse}\index{Ereignis}\index{Wahrscheinlichkeitsraum!Ereignis}.

    Für $A \subseteq \Omega$ bezeichne $\overline{A} := \Omega \setminus A$ das \emph{Gegenereignis}\index{Ereignis!Gegen-}.

    Ein(-e) \emph{Wahrscheinlichkeit(-smaß)}\index{Wahrscheinlichkeit}\index{Wahrscheinlichkeitsmaß}\index{Wahrscheinlichkeitsraum!Wahrscheinlichkeit}\index{Wahrscheinlichkeitsraum!Wahrscheinlichkeitsmaß} ist eine Funktion $\IP: 2^\Omega \to \edgys{0,1}$ mit $\IP[\Omega] = 1$ und $\IP$ \emph{$\sigma$-additiv}\index{sigma@$\sigma$-additiv}\index{Wahrscheinlichkeitsraum!sigma@$\sigma$-additiv}, d.h. für $A_1, A_2, \ldots \subseteq \Omega$ mit $A_i \cap A_j = \emptyset$ für $i \neq j$ gilt 
    \begin{equation*}
        \IP\left[ \bigcup_{i=1}^\infty A_i \right] = \sum_{i=1}^\infty \IP[A_i].
    \end{equation*}

    Das Paar $(\Omega, \IP)$ heißt (diskreter) \emph{Wahrscheinlichkeitsraum}\index{Wahrscheinlichkeitsraum}.
\end{defi}

Es reicht, $\IP[a]$ für alle $a \in \Omega$ zu definieren, denn 
\begin{equation*}
    \IP[\set{a_1, \ldots, a_n}] = \sum_{i = 1}^n \IP[\set{a_i}].
\end{equation*}

\begin{expl}
    Bei einem Würfelwurf setze $\Omega = \set{1,\ldots, 6}$. Dann ist $\IP[i] = \frac{1}{6}$ für alle $i \in \Omega$. Es ist auch $\IP[\text{\enquote{gerade Zahl}}] = \IP[\set{2,4,6}] = 3 \cdot \frac{1}{6} = \frac{1}{2}$.
\end{expl}

Wir modellieren nun einen Wahrscheinlichkeitsraum für \textsc{RandMaxCut}:

Sei $V = \set{v_1, \ldots, v_n}$. Setze $\Omega = \set{0,1}^n$. Wir berechnen den Schnitt als 
\begin{equation*}
    U = \set{v_i \in V \mid \omega_i = 1}.
\end{equation*}
Für alle $e = \set{v_i, v_j} \in E$, ist die Wahrscheinlichkeit, dass $e$ über den Schnitt läuft, gerade $\frac{1}{2}$, denn es ist 
\begin{equation*}
    \IP[\text{$e$ läuft über den Schnitt $(U, V \setminus U)$}] = \IP[\set{\omega \in \Omega \mid \omega_i \neq \omega_j}] = \frac{2^{n-1}}{2^n} = \frac{1}{2}.
\end{equation*}

\subsection{Unabhängigkeit und bedingte Wahrscheinlichkeiten}
\begin{defi}{Unabhängige Ereignisse}{independentEvents}
    Sei $(\Omega, \IP)$ ein diskreter Wahrscheinlichkeitsraum. Zwei Ereignisse $A, B \subseteq \Omega$ heißen \emph{unabhängig}\index{Ereignis!unabhängige -se}\index{Wahrscheinlichkeitsraum!Ereignis!unabhängige -se}, falls 
    \begin{equation*}
        \IP[A \cap B] = \IP[A] \cdot \IP[B].
    \end{equation*}
    Allgemeiner: Ereignisse $A_1, \ldots, A_k \subseteq \Omega$ heißen \emph{unabhängig}, wenn für alle $I \subseteq \set{1,\ldots,k}$ gilt 
    \begin{equation*}
        \IP\left[ \bigcap_{i \in I} A_i \right] = \prod_{i \in I} \IP[A_i].
    \end{equation*}
\end{defi}

\begin{expl}\leavevmode
    \begin{itemize}
        \item Sei $\Omega = \set{(i, j) \mid i, j \in \set{1,\ldots,6}}$. Dann ist $\IP[(i,j)] = \frac{1}{36}$. 
        
        Seien $A := \text{\enquote{Erster Würfel zeigt eine gerade Zahl}}$ und $B := \text{\enquote{Zweiter Würfel zeigt $\geq 3$}}$. Dann ist $\IP[A] = \frac{1}{2}, \IP[B] = \frac{2}{3}$ und $\IP[A \cap B] = \frac{1}{3} = \IP[A] \cdot \IP[B]$. Also sind $A$ und $B$ unabhängig.
        \item Für eine endlihce Folge von Wahrscheinlichkeitsräumen $(\Omega_1, \IP_1), \ldots, (\Omega_n, \IP_n)$ definieren wir den \emph{Produktraum}\index{Wahrscheinlichkeitsraum!Produktraum} durch $\Omega := \Omega_1 \times \cdots \times \Omega_n$ und $\IP[(x_1, \ldots, x_n)] := \IP_1[x_1] \cdot \ldots \cdot \IP_n[x_n]$.
    \end{itemize}
\end{expl}
In \textsc{RandMaxCut} ist der Wahrscheinlichkeitsraum eigentlich ein Produktraum von $n$ Münzwürfen $(\set{0,1},\IP)$ mit $\IP[0] = \IP[1] = \frac{1}{2}$. Seien $e, e' \in E$ ohne gemeinsame Endpunkte. Dann ist
\begin{equation*}
    \IP[\text{\enquote{$e$ und $e'$ laufen über Schnitt}}] = \frac{1}{4} = \IP[\text{\enquote{$e$ läuft über Schnitt}}] \cdot \IP[\text{\enquote{$e'$ läuft über Schnitt}}].
\end{equation*}

\begin{defi}{Bedingte Wahrscheinlichkeit}{condProb}
    Sei $(\Omega, \IP)$ ein diskreter Wahrscheinlichkeitsraum. Seien $A, B \subseteq \Omega$ mit $\IP[B] > 0$. Dann bezeichne 
    \begin{equation*}
        \IP[A \mid B] := \frac{\IP[A \cap B]}{\IP[B]}
    \end{equation*}
    die \emph{bedingte Wahrscheinlichkeit} von $A$ gegeben $B$.\index{Wahrscheinlichkeit!bedingte -}\index{bedingte Wahrscheinlichkeit}
\end{defi}

\begin{expl}
    Wir betrachten zwei unanbhängige Würfelwürfe. Sei $A := \text{\enquote{Erster Würfel zeigt $6$}}$ und $B := \text{\enquote{Summe beider Würfe ist $10$}}$. Dann sind 
    \begin{equation*}
        \IP[A] = \frac{1}{6}, \quad \IP[B] = \frac{3}{36}, \quad \IP[A \cap B] = \frac{1}{36} \quad \text{und} \quad \IP[A \mid B] = \frac{1}{3}.
    \end{equation*}
\end{expl}

\section{Zufallsvariablen und Erwartungswerte}

\begin{defi}{Zufallsvariable}{randomVar}
    Sei $(\Omega, \IP)$ ein diskreter Wahrscheinlichkeitsraum. eine Abbildung $X: \Omega \to \IR$ heißt (diskrete, reelle) \emph{Zufallsvariable}\index{Zufallsvariable}. Wir schreiben 
    \begin{equation*}
        \IP[X = a] = \IP[\set{\omega \in \Omega \mid X(\omega) = a}] 
    \end{equation*}
    und allgemeiner 
    \begin{equation*}
        \IP[X \in A] = \IP[\set{\omega \in \Omega \mid X(\omega) \in A}].
    \end{equation*}
\end{defi}

\begin{expl}
    Betrachte zwei unabhängige Würfelwürfe. Sei $X: \Omega := \set{1,\ldots,12}$ definiert durch $X(\omega) := \omega_1 + \omega_2$ für alle $\omega = (\omega_1, \omega_2) \in \Omega$.
\end{expl}

\begin{defi}{Unabhängigkeit (Zufallsvariablen)}{independentRandomVar}
    Seien $X, Y: \Omega \to \IR$ Zufallsvariablen. $X$ und $Y$ heißen \emph{unabhängig}\index{Unabhängigkeit}\index{Zufallsvariable!unabhängige -n}\index{Unabhängigkeit!von Zufallsvariablen}, wenn für alle $x,y \in \IR$ gilt
    \begin{equation*}
        \IP[(X = x) \cap (Y = y)] = \IP[X = x] \cdot \IP[Y = y].
    \end{equation*}
    Allgemeiner seien $X_1, \ldots, X_n : \Omega \to \IR$ Zufallsvariablen. Diese heißen \emph{unabhängig}, wenn für alle $I \subseteq \set{1,\ldots,n}, x_i \in \IR$ gilt 
    \begin{equation*}
        \IP\left[ \bigcap_{i \in I} (X_i = x) \right] = \prod_{i \in I} \IP[X_i = x].
    \end{equation*}
\end{defi}

\begin{defi}{Erwartungswert}{expectedValue}
    Sei $X: \Omega \to \IR$ eine diskrete Zufallsvariable. Falls die Reihe 
    \begin{equation*}
        \sum_{\omega\in \Omega} \abs{X(\omega)} \cdot \IP[\omega]
    \end{equation*}
    konvergiert, heißt 
    \begin{equation*}
        \IE[X] := \sum_{\omega \in \Omega} X(\omega) \cdot \IP[\omega]
    \end{equation*}
    \emph{Erwartungswert}\index{Erwartungswert}\index{Zufallsvariable!Erwartungswert} von $X$.
\end{defi}

\newLecture{29.10.2025}

Sei $X$ eine Zufallsvariable, für die der Erwartungswert gemäß~\Cref{def:expectedValue} existiert. Bezeichnen wir mit $R = X(\Omega)$ die abzählbare Teilmenge der reelen Zahlen, die von $X$ angenommen werden können, können wir den Erwartungswert auch schreiben als
\begin{equation*}
    \IE[X] = \sum_{x \in R} x \cdot \IP[X = x].
\end{equation*}
\begin{expl}\leavevmode
    \begin{itemize}
        \item Für die Zufallsvariable $X$, die die Summe zweier unabhängiger Würfelwürfe modelliert, ist
        \begin{equation*}
            \IE[X] = \sum_{x=2}^{12} x \cdot \IP[X = x] = \frac{1}{36}(2 + 2 \cdot 3 + 3 \cdot 4 + 4 \cdot 5 + 5 \cdot 6 + 6 \cdot 7 + 5 \cdot 8 + 4 \cdot 9 + 3 \cdot 10 + 2 \cdot 11 + 12) = 7.
        \end{equation*}
        \item Für die Zufallsvariable $X$, die beschreibt, wie oft in zehn Münzwürfen \enquote{Zahl} geworfen wird, ist es anhand der Definition schon schwieriger, den Erwartungswert zu berechnen. Mit Mühe erhält man 
        \begin{equation*}
            \IE[X] = 0 \cdot \frac{1}{1024} + 1 \cdot \frac{10}{1024} + 2 \cdot \frac{45}{1024} + \cdots + 10 \cdot \frac{1}{1024} = 5.
        \end{equation*}
        \item Wieder betrachten wir den Algorithmus \textsc{RandMaxCut} (Algorithmus~\ref{alg:RandMaxCut}) auf dem obigen Beispiel, in dem die drei Kanten $e_1, e_2$ und $e_3$ ein Dreieck bilden. Sei wieder $X$ die Zufallsvariable, die die Anzahl an Kanten beschreibt, die über den Schnitt laufen. Da $X$ nur die Werte $0$ oder $2$ annehmen kann, gilt 
        \begin{equation*}
            \IE[X] = 0 \cdot \IP[X = 0] + 2 \cdot \IP[X = 2] = 2 \cdot \frac{3}{4} = \frac{3}{2}.
        \end{equation*}
    \end{itemize}
\end{expl}
Im obigen Beispiel für \textsc{RandMaxCut} haben wir die erwartete Anzahl an Kanten, die über den Schnitt laufen, ausgerechnet. Für Graphen mit vielen Knoten und Kanten wäre eine ähnliche Rechnung sehr aufwändig, da wir in diesem Beispiel alle möglichen Realisierungen einzeln betrachtet haben. Um das Ausrechnen von Erwartungswerten zu vereinfachen, lernen wir einige wichtige Rechenregeln kennen.

\begin{thm}{Linearität des Erwartungswerts}{linearityExpectedValue}
    Sei $(\Omega, \IP)$ ein diskreter Wahrscheinlichkeitsraum und seien $X, Y: \Omega \to \IR$ Zufallsvariablen mit existierenden Erwartungswerten. Dann gilt:
    \begin{enumerate}[$(a)$]
        \item Für $c, d \in \IR$ existiert der Erwartungswert von $(cX + d): \Omega \to \IR$ mit $(cX + d)(\omega) := c \cdot X(\omega) + d$ für alle $\omega \in \Omega$. Es gilt $\IE[cX + d] = c \cdot \IE[X] + d$.
        \item Der Erwartungswert der Zufallsvariablen $(X + Y): \Omega \to \IR$ mit $(X + Y)(\omega) := X(\omega) + Y(\omega)$ für alle $\omega \in \Omega$ existiert und es gilt $\IE[X + Y] = \IE[X] + \IE[Y]$.
        \item Sind $X$ und $Y$ unabhängig, so existiert der Erwartungswert der Zufallsvariablen $(XY): \Omega \to \IR$ mit $(XY)(\omega) := X(\omega) \cdot Y(\omega)$ für alle $\omega \in \Omega$ und es gilt $\IE[XY] = \IE[X] \cdot \IE[Y]$.
    \end{enumerate}
\end{thm}

\begin{proof}\leavevmode
    \begin{enumerate}[$(a)$:]
        \item Aus~\Cref{def:expectedValue} folgt 
        \begin{equation*}
            \begin{split}
                \IE[cX + d] &= \sum_{\omega \in \Omega} (cX(\omega) + d) \cdot \IP[\set{\omega}] \\
                &= c \cdot \sum_{\omega \in \Omega} X(\omega) \cdot \IP[\set{\omega}] + d \cdot \sum_{\omega \in \Omega} \IP[\set{\omega}] \\
                &= c \cdot \IE[X] + d.
            \end{split}
        \end{equation*}
        Die Existenz des Erwartungswerts von $cX + d$ folgt aus der Existenz des Erwartungswerts von $X$.
        \item Es gilt, auch mit~\Cref{def:expectedValue}:
        \begin{equation*}
            \begin{split}
                \IE[X + Y] &= \sum_{\omega \in \Omega} (X(\omega) + Y(\omega)) \cdot \IP[\set{\omega}] \\
                &= \sum_{\omega \in \Omega} X(\omega) \cdot \IP[\set{\omega}] + \sum_{\omega \in \Omega} Y(\omega) \cdot \IP[\set{\omega}] \\
                &= \IE[X] + \IE[Y].
            \end{split}
        \end{equation*}
        \item Seien $R = X(\Omega)$ und $S = Y(\Omega)$. Dann gilt die abzählbaren Mengen von Werten, die $X$ und $Y$ annehmen können. Es folgt mit~\Cref{def:expectedValue}:
        \begin{equation*}
            \begin{split}
                \IE[XY] &= \sum_{\omega \in \Omega} X(\omega) \cdot Y(\omega) \cdot \IP[\set{\omega}] \\
                &= \sum_{x \in R} \sum_{y \in S} x \cdot y \cdot \IP[(X = x) \cap (Y = y)] \\
                &= \sum_{x \in R} \sum_{y \in S} x \cdot y \cdot \IP[X = x] \cdot \IP[Y = y] \\
                &= \left( \sum_{x \in R} x \cdot \IP[X = x] \right) \cdot \left( \sum_{y \in S} y \cdot \IP[Y = y] \right) \\
                &= \IE[X] \cdot \IE[Y],
            \end{split}
        \end{equation*}
        wobei wir in der dritten Gleichheit die Unabhängigkeit von $X$ und $Y$ verwendet haben. Die Existenz des Erwartungswerts von $XY$ folgt aus der Existenz der Erwartungswerte von $X$ und $Y$.
    \end{enumerate}
\end{proof}

\begin{thm}{Markov}{markov}
    Sei $(\Omega, \IP)$ ein diskreter Wahrscheinlichkeitsraum und sei $X: \Omega \to \IR_{\geq 0}$ eine Zufallsvariable mit existierendem Erwartungswert. Dann gilt für alle $a > 0$:
    \begin{equation*}
        \IP[X \geq a] \leq \frac{\IE[X]}{a}.
    \end{equation*}
\end{thm}
Wir können die Markov-Ungleichung umformulieren, um eine noch einprägsamere Form zu erhalten. Sei $X: \Omega \to \IR_{\geq 0}$ eine Zufallsvariable. Dann gilt für alle $a \geq 1$
\begin{equation*}
    \IP[X \geq a \cdot \IE[X]] \leq \frac{1}{a}.
\end{equation*}
\begin{proof}[Beweis von Satz~\ref{thm:markov}]
    Es gilt 
    \begin{equation*}
        \IE[X] = \sum_{\omega \in \Omega} X(\omega) \cdot \IP[\set{\omega}] \geq \sum_{\substack{\omega \in \Omega \\ X(\omega) \geq a}} X(\omega) \cdot \IP[\set{\omega}] \geq \sum_{\substack{\omega \in \Omega \\ X(\omega) \geq a}} a \cdot \IP[\set{\omega}] = a \cdot \IP[X \geq a].
    \end{equation*}
    Daraus folgt die Behauptung.
\end{proof}
Mithilfe der Markov-Ungleichung können wir beispielsweise die Wahrscheinlichkeit abschätzen, dass wir in einer Folge von $n$ unabhängigen, fairen Münzwürfen mindestens $(3n/4)$-mal \enquote{Zahl} werfen. Sei $X$ die Zufallsvariable, die die Anzahl an \enquote{Zahl} in den $n$ Münzwürfen beschreibt. Dann ist $\IE[X] = \frac{n}{2}$. Mit der Markov-Ungleichung folgt für $a = \frac{3}{2}$:
\begin{equation*}
    \IP\left[X \geq \frac{3n}{4}\right] = \IP\left[X \geq a \cdot \IE[X]\right] \leq \frac{1}{a} = \frac{2}{3}.
\end{equation*}
Tatsächlich fällt die obige Wahrscheinlichkeit sogar exponentiell schnell mit $n$.

\subsection{Analyse von RandMaxCut}

\begin{defi}{Randomisierter Approximationsalgorithmus}{RandApprox}
    Ein \emph{randomisierter Approximationsalgorithmus} $A$ für ein Minimierungs- oder Maximierungsproblem $\Pi$ erreicht einen \emph{Approximationsfaktor}\index{Approximationsfaktor} von $r \geq 1$ bzw. $r \leq 1$, wenn 
    \begin{equation*}
        \IE[w_A(I)] \leq r \cdot \OPT(I) \quad \text{bzw.} \quad \IE[w_A(I)] \geq r \cdot \OPT(I)
    \end{equation*}
    für alle Instanzen $I \in \cI_\Pi$ gilt. Wir sagen dann, dass $A$ ein \emph{randomisierter $r$-Approximationsalgorithmus}\index{Approximationsalgorithmus!randomisierter $r$-} ist.
    
    Oft hängt der Approximationsfaktor von der Eingabelänge ab. Dann ist $r: \IN \to \edgyround{0,\infty}$ eine Funktion und es muss 
    \begin{equation*}
        \IE[w_A(I)] \leq r(\abs{I}) \cdot \OPT(I) \quad \text{bzw.} \quad \IE[w_A(I)] \geq r(\abs{I}) \cdot \OPT(I)
    \end{equation*}
    für alle Instanzen $I \in \cI_\Pi$ gelten, wobei $\abs{I}$ die Länge der Eingabe $I$ bezeichne.
\end{defi}

\begin{thm}{RandMaxCut}{RandMaxCut}
    \textsc{RandMaxCut} (Algorithmus~\ref{alg:RandMaxCut}) ist ein randomisierter $\frac{1}{2}$-Approximationsalgorithmus für das \textsc{Maximum-Cut}-Problem.
\end{thm}
\begin{proof}
    Sei $G = (V, E)$ ein ungereichteter Graph mit Kantengewichten $w$. Sei $U \subseteq V$ die zufällige Teilmenge von $V$, die \textsc{RandMaxCut} auswählt. Sei $X$ die Zufallsvariable, die das Gesamtgewicht der Kanten, die über den Schnitt laufen, angibt. Außerdem definieren wir für jede Kante $e \in E$ die Zufallsvariable $X_e$ durch
    \begin{equation*}
        X_e := \begin{cases}
            w(e), & \text{falls $e$ über den Schnitt läuft}, \\
            0, & \text{sonst}.
        \end{cases}
    \end{equation*}
    Dann gilt per Definition $X = \sum_{e \in E} X_e$. Mit der Linearität des Erwartungswerts (Satz~\ref{thm:linearityExpectedValue}~$(b)$) folgt
    \begin{equation*}
        \IE[X] = \IE\left[ \sum_{e \in E} X_e \right] = \sum_{e \in E} \IE[X_e].
    \end{equation*}
    Da $X_e$ nur die Werte $0$ und $w(e)$ annehmen kann, gilt
    \begin{equation*}
        \begin{split}
            \IE[X_e] &= 0 \cdot \IP[X_e = 0] + w(e) \cdot \IP[X_e = w(e)] \\
            &= w(e) \cdot \IP[\text{$e$ läuft über den Schnitt}] \\
            &= w(e) \cdot \frac{1}{2}.
        \end{split}
    \end{equation*}
    Für den letzten Schritt haben wir verwendet, dass die Wahrscheinlichkeit, dass eine Kante über den Schnitt läuft, gerade $\frac{1}{2}$ ist. Insgesamt erhalten wir somit
    \begin{equation*}
        \IE[X] = \sum_{e \in E} \IE[X_e] = \sum_{e \in E} w(e) \cdot \frac{1}{2} = \frac{1}{2} \cdot \sum_{e \in E} w(e) = \frac{W}{2}
    \end{equation*}
    für $W = \sum_{e \in E} w(e)$. Außerdem gilt $\OPT \leq W$, da kein Schnitt mehr als das Gesamtgewicht aller Kanten erreichen kann. Somit folgt wie gewünscht 
    \begin{equation*}
        \IE[X] = \frac{W}{2} \geq \frac{\OPT}{2}.
    \end{equation*}
\end{proof}

\subsection{Binomialverteilung und geometrische Verteilung}

Seien $Y_1, \ldots, Y_n$ unabhängige Bernoulli-Zufallsvariablen mit demselben Parameter $p \in \edgys{0,1}$. Die Zufallsvariable $X = Y_1 + \cdots + Y_n$ heißt \emph{binomialverteilt}\index{Binomialverteilung}\index{Zufallsvariable!binomialverteilte -n} mit Parametern $n$ und $p$.

\begin{thm}{Binomialverteilung}{binomialDistribution}
    Sei $X$ eine binomialverteilte Zufallsvariable mit Parametern $n$ und $p$. Dann gilt $\IE[X] = pn$ und für alle $j \in \set{0, \ldots, n}$:
    \begin{equation*}
        \IP[X = j] = \binom{n}{j} p^j (1-p)^{n-j}.
    \end{equation*}    
\end{thm}

\begin{proof}
    Ausrechnen.
\end{proof}

Sei nun $Y_1, Y_2, \ldots$ eine Folge unabhängiger Bernoulli-Zufallsvariablen mit demselben Parameter $p \in \edgys{0,1}$. Wir betrachten nun die Frage, wie lange wir auf den ersten Erfolg warten müssen. Diese wird durch eine \emph{geometrisch verteilte Zufallsvariable mit Parameter $p$}\index{geometrische Verteilung}\index{Zufallsvariable!geometrisch verteilte -n} modelliert. Eine solche Zufallsvariable $X$ nimmt den Wert $n \in \IN$ an, wenn $Y_1 = \cdots = Y_{n-1} = 0$ und $Y_n = 1$. Die Wahrscheinlichkeit hierfür ist $(1-p)^{n-1} p$. Wir definieren für eine geometrische Zufallsvariable $X$ mit Parameter $p$ dementsprechend $\IP[X = n] = (1-p)^{n-1} p$ für alle $n \in \IN$. Um den Erwatungswert einer geometrischen Zufallsvariablen zu bestimmen, leiten wir zunächst eine nützliche alternative Darstellung des Erwartungswerts her, die nur für Zufallsvariablen gilt, die nur Werte aus $\IN_0$ annehmen können.

\begin{lem}{Erwartungswert natürliche Zufallsvariable}{expectedValueNatural}
    Sei $(\Omega, \IP)$ ein diskreter Wahrscheinlichkeitsraum und sei $X: \Omega \to \IN_0$ eine Zufallsvariable. Dann gilt
    \begin{equation*}
        \IE[X] = \sum_{i=1}^\infty \IP[X \geq i].
    \end{equation*}
\end{lem}
\begin{proof}
    Es gilt 
    \begin{equation*}
        \begin{split}
            \sum_{i=1}^\infty \IP[X \geq i] &= \sum_{i=1}^\infty \sum_{j=1}^\infty \IP[X = j] \\
            &= \sum_{j=1}^\infty \sum_{i=1}^{j} \IP[X = j] \\
            &= \sum_{j = 1}^\infty j \cdot \IP[X = j] = \IE[X],
        \end{split}
    \end{equation*}
    wobei wir ausgenutzt haben, dass wir bei absolut konvergenten Reihen die Reihenfolge der Summanden beliebig vertauschen können.
\end{proof}
\begin{thm}{Geometrische Verteilung}{geometricDistribution}
    Sei $X$ eine geometrisch verteilte Zufallsvariable mit Parameter $p$. Dann gilt $\IE[X] = \frac{1}{p}$.
\end{thm}
\begin{proof}
    Für die geometrische Zufallsvariable $X$ und $j \in \IN$ gilt 
    \begin{equation*}
        \IP[X \geq j] = \sum_{i = j}^\infty \IP[X = i] = \sum_{i = j}^\infty (1-p)^{i-1} p = (1-p)^{j-1},
    \end{equation*}
    wobei die letzte Gleichung durch Nachrechnen folgt, oder durch Überlegen, dass $X \geq j$ genau dann gilt, wenn die ersten $(j-1)$ Bernoulli-Versuche Fehlschläge sind. Mit~\Cref{lem:expectedValueNatural} folgt somit
    \begin{equation*}
        \IE[X] = \sum_{i=1}^\infty \IP[X \geq i] = \sum_{i=1}^\infty (1-p)^{i-1} = \frac{1}{1 - (1-p)} = \frac{1}{p},
    \end{equation*}
    wobei wir in der letzten Gleichheit die Formel für die geometrische Reihe verwendet haben.
\end{proof}

Eine weitere wichtige Eigenschaft geometrisch verteilter Zufallsvariablen ist die \emph{Gedächtnislosigkeit}\index{Gedächtnislosigkeit}\index{Zufallsvariable!geometrisch verteilte -n!Gedächtnislosigkeit}\index{geometrische Verteilung!Gedächtnislosigkeit}. Das bedeutet, dass die Wahrscheinlichkeit eines Erfolgs unanbhängig davon ist, wie viele Misserfolge es bis jetzt gegeben hat.

\begin{lem}{Gedächtnislosigkeit geometrischer Verteilung}{memorylessGeometric}
    Sei $X$ eine geometrisch verteilte Zufallsvariable mit Parameter $p$. Dann gilt für alle $n \in \IN$ und $k \in \IN_0$:
    \begin{equation*}
        \IP[X = n + k \mid X > k] = \IP[X = n].
    \end{equation*}
\end{lem}
\begin{proof}
    Es gilt 
    \begin{equation*}
        \begin{split}
            \IP[X = n + k \mid X > k] &= \frac{\IP[(X = n + k) \cap (X > k)]}{\IP[X > k]} = \frac{\IP[X = n + k]}{\IP[X > k]} \\
            &= \frac{(1-p)^{n+k-1} \cdot p}{(1-p)^k} = (1-p)^{n-1} p = \IP[X = n].\qedhere
        \end{split}
    \end{equation*}
\end{proof}

Das \emph{Coupon Collector's Problem}\index{Coupon Collector's Problem} tritt oft in der Analyse randomisierter Algorithmen auf und lässt sich anhant von Sammelbildern beschreiben, wie sie zum Beispiel zu jeder Fußballweltmeisterschaft verkauft werden. Wir gehen davon aus, dass es insgesamt $n$ verschiedene solcher Bilder gibt. Diese kann man nicht gezielt erwerben, sondern man kann nur verschlossene Tüten kaufen, in denen sich jeweils unabhängig ein uniform zufällig ausgewähltes Sammelbild befindet. Wir bezeichnen mit $X$ die Zufallsvariable, die angibt, wie viele Tüten wir kaufen müssen, bis wir jedes Bild mindestens einmal in unserer Sammlung haben.

\begin{thm}{Coupon Collector's Problem}{couponCollector}
    Für die Zufallsvariable $X$ aus dem Coupon Collector's Problem gilt $\IE[X] = n\ln(n) + \Theta(n)$.    
\end{thm}

\begin{proof}
    Wir zerlegen den Prozess in $n$ Phasen. Die erste Phase beginnt mit dem ersten Sammelbild, das wir kaufen. Phase $i$ für $i \in \set{2,\ldots,n}$ beginnt direkt, nachdem wir das $(i-1)$-te verschiedene Sammelbild gekauft haben, und endet, sobald wir das $i$-te verschiedene Sammelbild gekauft haben. Bezeichnen wir mit $X_i$ die Anzahl an Sammelbildern, die wir in Phase $i$ kaufen, dann gilt $X = X_1 + X_2 + \cdots + X_n$. Außerdem ist jedes $X_i$ geometrisch verteilt mit Parameter 
    \begin{equation*}
        p_i = \frac{n-i+1}{n},
    \end{equation*}
    denn die Wahrscheinlichkeit, eines der $n-i+1$ Sammelbilder zu kaufen, die nicht zu den $i-1$ bereits erworbenen Bildern gehören, beträgt in Phase $i$ in jedem Versuch $\frac{n-i+1}{n}$. Mit der Linearität des Erwartungswertes und Satz~\ref{thm:geometricDistribution} folgt somit
    \begin{equation*}
        \IE[X] = \sum_{i=1}^n \IE[X_i] = \sum_{i=1}^n \frac{1}{p_i} = \sum_{i=1}^n \frac{n}{n-i+1} = n \cdot \sum_{i=1}^n \frac{1}{i} = n \cdot ( \ln(n) + \Theta(1) ) = n\ln(n) + \Theta(n).
    \end{equation*}
    Im letzten Schritt haben wir dabei ausgenutzt, dass für die $n$-te harmonische Zahl $H_n$ gerade $\sum_{i=1}^n \frac{1}{i} = \ln(n) + \Theta(1)$ gilt.
\end{proof}

\subsection{Vertex Cover}
Wir betrachten nun einen simplen Algorithmus \textsc{RandVertexCover}\index{RandVertexCover} für das \textsc{Vertex-Cover}-Problem. Für eine Eingabe $G = (V, E)$ startet dieser Algorithmus mit einer leeren Menge $U$ und geht alle Kanten aus $E$ in einer beliebigen Reihenfolge durch. Für jede Kante $e \in E$ überprüft der Algorithmus, ob bereits mindestens ein Endknoten von $e$ in $U$ enthalten ist. Falls ja, so macht der Algorithmus mit der nächsten Kante weiter. Ansonsten wählt er uniform zufällig einen der beiden Endknoten von $e$ aus und fügt diesen zu $U$ hinzu. Anschließend macht er mit der nächsten Kante weiter.

Wir betrachten sein Verhalten auf einem Sterngraphen $G = (V, E)$. Für einen solchen Graphen ist $V = \set{v, u_1, \ldots, u_{n-1}}$ und die Menge $E$ besteht nur aus den Kanten $(v, u_i)$ für alle $i$. Das optimale Vertex Cover besteht nur aus dem Knoten $v$, der alle Kanten abdeckt. Bei jeder Kante fügt \textsc{RandVertexCover} mit einer Wahrscheinlichkeit von $\frac{1}{2}$ den Knoten $v$ zur Menge $U$ hinzu. Sobald dies einmal passiert ist, sind alle Kanten des Graphen abgedeckt und es werden keine weiteren Knoten der Menge $U$ hinzugefügt. Sei nun $X$ eine geometrisch verteilte Zufallsvariable mit Parameter $p = \frac{1}{2}$. Dann ist die Anzahl an Knoten, die \textsc{RandVertexCover} zu $U$ hinzufügt, verteilt wie die Zufallsvariable $\min\set{X, n-1}$, da der Algorithmus maximal $\abs{E} = n-1$ viele Knoten zu $U$ hinzufügen kann. Es gilt 
\begin{equation*}
    \IE[\min\set{X, n-1}] \leq \IE[X] = \frac{1}{p} = 2 = 2 \cdot \OPT
\end{equation*}
und somit handelt es sich bei \textsc{RandVertexCover} bei diesem Beispiel um einen randomisierten $2$-Approximationsalgorithmus.

\begin{thm}{RandVertexCover}{RandVertexCover}
    \textsc{RandVertexCover} ist ein randomisierter $2$-Approximationsalgorithmus für das \textsc{Vertex-Cover}-Problem.
\end{thm}

\begin{proof}
    Aus der Beschreibung des Algorithmus folgt direkt, dass er stets ein Vertex Cover berechnet. Zu zeigen ist also nur, dass dieses im Erwartungswert höchstens doppelt so viele Knoten enthält wie ein minimales Vertex Cover. Sei $G = (V, E)$ eine beliebige Eingabe und sei $Z = \set{z_1, \ldots, z_k} \subseteq V$ ein minimales Vertex Cover von $G$. Wir bezeichnen mit $V_i \subseteq V$ die Menge, die aus $z_i$ und den zu $z_i$ benachbarten Knoten besteht. Die Mengen $V_i$ sind im Allgemeinen nicht paarweise disjunkt. Ferner bezeichne $E_i \subseteq E$ die Menge der zu $z_i$ inzidenten Kanten. Da es sich bei $Z$ um ein Vertex Cover handelt, gilt $E = E_1 \cup \cdots \cup E_k$.

    Es bezeichne $X_i$ die Anzahl an Knoten aus $V_i$, die \textsc{RandVertexCover} bei Betrachtung der Kanten aus $E_i$ zu $U$ hinzufügt. Es seien $e_1, \ldots, e_\ell$ die Kanten aus $E_i$ und wir nehmen an, dass \textsc{RandMaxCut} diese in diese Reihenfolge betrachtet. Zwischendurch betrachtet \textsc{RandMaxCut} im Allgemeinen auch andere Kanten aus $E \setminus E_i$. Wenn der Algorithmus den Knoten $z_i$ bei der Betrachtung einer Kante $e_j \in E_i$ der Menge $U$ hinzufügt, so wird bei allen weiteren Kanten aus $E_i$ kein weiterer Knoten zu $U$ hinzugefügt. Die Situation ist ähnlich zu obigem Sterngraphen, allerdings sind die $X_i$ nicht identisch verteilt wie $\min\set{X, \abs{E_i}}$ für eine geometrisch verteilte Zufallsvariable $X$ mit Parameter $p = \frac{1}{2}$, da zwischen den Betrachtungen der Kanten aus $E_i$ auch Kanten aus $E \setminus E_i$ betrachtet werden können, die den Zustand des Algorithmus verändern. Allerdings gilt dennoch
    \begin{equation*}
        \IE[X_i] \leq \IE[\min\set{X, \abs{E_i}}] \leq \IE[X] = 2.
    \end{equation*}
    Wegen $E_1 \cup \cdots \cup E_k = E$ gilt $\abs{U} = X_1 + \cdots + X_k$ und somit folgt mit der Linearität des Erwartungswerts (Satz~\ref{thm:linearityExpectedValue}~$(b)$):
    \begin{equation*}
        \IE[\abs{U}] = \IE[X_1 + \cdots + X_k] = \sum_{i=1}^k \IE[X_i] \leq \sum_{i=1}^k 2 = 2k = 2 \cdot \OPT.\qedhere
    \end{equation*}
\end{proof}

\section{Max-SAT}\label{sec:maxSATWtheo}

Die Eingabe besteht aus einer aussagenlogischen Formel in konjunktiver Normalform (KNF). Das Ziel ist es, eine Belegung der Variablen zu finden, die möglichst viele Klauseln erfüllt. In der gewichteten Variante besitzt jede Klausel ein Gewicht und das Ziel ist es, eine Belegung zu finden, die das Gesamtgewicht der erfüllten Klauseln maximiert. Aus der NP-Schwere des Entscheidungsproblems SAT folgt direkt, dass auch das \textsc{Max-SAT}-Problem NP-schwer ist.

Wir nennen eine Formel der Form $x$ oder $\neg x$ für eine Variable $x$ ein \emph{Literal}\index{Literal}. Ein Literal der Form $x$ nennen wir ein \emph{positives Literal}\index{Literal!positives -}, ein Literal der Form $\neg x$ ein \emph{negatives Literal}\index{Literal!negatives -}. Wir nennen die Variablen, die in der Eingabe vorkommen, im Folgenden $x_1, \ldots, x_n$. Eine \emph{Klausel}\index{Klausel} $C$ ist eine \emph{Disjunktion}\index{Disjunktion} (Oder-Verknüpfung) von Literalen über diesen Variablen und eine aussagenlogische Formel $\varphi$ ist in \emph{konjunktiver Normalform} (KNF) \index{konjunktive Normalform}, wenn sie eine \emph{Konjunktion}\index{Konjunktion} (Und-Verknüpfung) von Klauseln ist. Oft fassen wir Klauseln auch einfach als Menge von Literalen auf. Eine Formel in KNF ist dann von der Gestalt 
\begin{equation*}
    \varphi = \bigwedge_{i=1}^m \left( \bigvee_{\ell \in C_l} \ell \right),
\end{equation*}
wobei $C_1, \ldots, C_m$ die Klauseln von $\varphi$ sind.

\begin{prob}{Max-SAT}{MaxSAT}\index{Max-SAT}
    \prblmObj{aussagenlogische Formel $\varphi$ in KNF mit Variablen $x_1, \ldots, x_n$ und Klauseln $C_1, \ldots, C_m$; Gewichte $w_1, \ldots, w_m \in \rounds{0,\infty}$}{Belegung der Variablen $x_1, \ldots, x_m$}{Maximiere das Gesamtgewicht der durch die Belegung erfüllten Klauseln}
\end{prob}

Genau wie schon für die Maximum-Cut- und  Vertex Cover-Probleme geben wir für \textsc{Max-SAT} zunächst einen sehr einfachen Approximationsalgorithmus \textsc{RandMaxSAT}\index{RandMaxSAT} an. Dieser wählt einfach eine uniform zufällige Belegung $\alpha$ der Variablen $x_1, \ldots, x_n$. Das bedeutet, er setzt jede Variable unabhängig von den anderen mit einer Wahrscheinlichkeit von $\frac{1}{2}$ auf $0$ bzw. $1$.

\begin{thm}{RandMaxSAT}{randMaxSAT}
    \textsc{RandMaxSAT} ist ein randomisierter $\frac{1}{2}$-Approximationsalgorithmus für \textsc{Max-SAT}.
\end{thm}

\begin{proof}
    Wir definieren zunächst für jede Klausel $C_j$ eine Zufallsvariable
    \begin{equation*}
        Y_j := \begin{cases}
            w_j & \text{falls $\alpha$ Klausel $C_j$ erfüllt,} \\
            0 & \text{sonst.}
        \end{cases}
    \end{equation*}
    Dann können wir das Gesamtgewicht $Y$ der durch $\alpha$ erfüllten Klauseln als die Summe $Y_1 + \cdots + Y_m$ schreiben und es gilt 
    \begin{equation*}
        \IE[Y] = \sum_{j = 1}^m w_j \cdot \IP[\text{$\alpha$ erfüllt $C_j$}].
    \end{equation*}
    Die Klausen $C_j$ wird nur dann nicht durch die zufällige Belegung $\alpha$ erfüllt, wenn alle Variablen, die als positives Literal in $C_j$ vorkommen, auf $0$ gesetzt sind und alle Variablen, die als negatives Literal in $C_j$ vorkommen, auf $1$ gesetzt sind. Es bezeichne $\ell_j = \abs{C_j}$ die Anzahl an Literalen in $C_j$. Da die oben genannte Belegung der Variablen aus $C_j$ die einzige ist, die die Klausel nicht erfüllt, gilt 
    \begin{equation*}
        \IP[\text{$\alpha$ erfüllt $C_j$ nicht}] \leq \left( \frac{1}{2} \right)^{\ell_j} \leq \frac{1}{2}.
    \end{equation*}
    Der erste Schritt ist dabei eine Ungleichung und keine Gleichung, da die Klausel $C_j$ für eine Variable $x$ sowohl das Literal $x$ als auch das Literal $\neg x$ enthalten kann. Da eine solche Klausel durch jede beliebige Belegung erfüllt wird, beträgt die Wahrscheinlichkeit, dass $C_j$ durch $\alpha$ nicht erfüllt wird, in diesem Fall $0$. damit folgt insgesamt 
    \begin{equation*}
        \IE[Y] = \sum_{j = 1}^m w_j \cdot \IP[\text{$\alpha$ erfüllt $C_j$}] \geq \sum_{j = 1}^m w_j \cdot \frac{1}{2} = \frac{1}{2} \sum_{j = 1}^m w_j \geq \frac{1}{2} \cdot \OPT,
    \end{equation*}
    wobei die letzte Schranke daraus folgt, dass das Gesamtgewicht aller Klauseln eine obere Schranke für $\OPT$ ist.
\end{proof}

Damit folgt, dass der Approximationsfaktor von \textsc{RandMaxSAT} besser wird, je länger die Klauseln sind. Eine Klausel mit $\ell$ Literalen ist nur mit Wahrscheinlichkeit von höchstens $(1/2)^\ell$ nicht durch die zufällige Belegung erfüllt. Betrachten wir beispielsweise das $3$-SAT-Problem, bei dem alle Klauseln genau drei Literale enthalten, so ergibt sich ein Approximationsfaktor von $\frac{7}{8}$. Es ist bekannt, dass dies sogar im Wesentlichen der beste erreichbare Approximationsfaktor für $3$-SAT unter der Annahme P$\neq$NP ist.

\textsc{RandMaxCut} und seine Analyse sind vollkommen analog zu dem einfachen Approximationsalgorithmus für das Maximum-Cut Problem. Wir zeigen, wie der Algorithmus \emph{derandomisiert}\index{Approximationsalgorithmus!derandomisierter -}\index{derandomisiert} werden kann. Das bedeutet, wir möchtendie eigentlich randomisierten Entscheidungnen von \textsc{RandMaxSAT} (d.h. die zufällige Belegung $\alpha$) nun deterministisch so treffen, dass der Algorithmus den gleichen Approximationsfaktor erreicht. Danach werden wir sehen, wie wir mit einem besseren randomisierten Ansatz einen besseren Approximationsfaktor erreichen können.

\subsection{Derandomisierung}

\begin{defi}{Bedingte Erwartung}{conditionalExpectation}
    Sei $(\Omega, \IP)$ ein diskreter Wahrscheinlichkeitsraum und $A \subseteq \Omega$ ein Ereignis mit $\IP[A] > 0$. Außerdem sei $X: \Omega \to \IR$ eine diskrete Zufallsvariable mit Werten aus $M \subseteq \IR$. Der \emph{bedingte Erwartungswert von $X$ gegeben $A$}\index{bedingter Erwartungswert}\index{Erwartungswert!bedinger -}\index{Zufallsvariable!Erwartungswert!bedingter -} existiert, falls die Reihe $\sum_{x \in M} \abs{x} \cdot \IP[X = x \mid A]$ konvergiert und ist dann definiert als 
    \begin{equation*}
        \IE[X \mid A] = \sum_{x \in M} x \cdot \IP[X = x \mid A].
    \end{equation*}
\end{defi}
Es gelten folgende Rechenregeln:
\begin{equation}
    \IE[X] = \IP[A] \cdot \IE[X \mid A] + \IP[\overline{A}] \cdot \IE[X \mid \overline{A}] \label{eq:bayesCondExp}
\end{equation}
und (für $B$ unabhängig von $A$)
\begin{equation}
    \IE[X \mid A] = \IP[B] \cdot \IE[X \mid A \land B] + \IP[\overline{B}] \cdot \IE[X \mid A \land \overline{B}]. \label{eq:bayesCondExpIndep}
\end{equation}

In der Derandomisierung von \textsc{RandMaxSAT} wählen wir die Belegung von $x_1, \ldots, x_n$ Schritt für Schritt. In Schritt $i$ bestimmen wir die Belegung von $x_i$ und haben davor schon die Belegung von $x_1, \ldots, x_{i-1}$ fexstgelegt. Wir betrachten zunächst den ersten Schritt, in dem wir die Belegung von $x_1$ festlegen. Mit $Y$ bezeichnen wir wieder die Zufallsvariable, die das Gesamtgewicht der erfüllten Klauseln bei einer uniform zufälligen Belegung $\alpha$ beschreibt. Gemäß~\eqref{eq:bayesCondExp} gilt 
\begin{equation*}
    \IE[Y] = \IP[\alpha_1 = 0] \cdot \IP[Y \mid \alpha_1 = 0] + \IP[\alpha_1 = 1] \cdot \IE[Y \mid \alpha_1 = 1] = \frac{\IE[Y \mid \alpha_1 = 0] + \IE[Y \mid \alpha_1 = 1]}{2}.
\end{equation*}
Aus dieser Gleichung folgt direkt, dass einer der beiden bedingten Erwartungswerte mindestens den Wert $\IE[Y]$ annehmen muss. Sei $b_1 \in \set{0,1}$ mit $\IE[Y \mid \alpha_1 = b_1] \geq \IE[Y]$. Wir setzen $x_1$ nun auf $b_1$. Wir fahren nach dem Schema fort und betrachten Schritt $i$, in dem wir die Belegung von $x_i$ festlegen und gehen davon aus, dass wir bereits Belegungen $b_1, \ldots, b_{i-1}$ der Variablen $x_1, \ldots, x_{i-1}$ gewählt haben, sodass gilt 
\begin{equation*}
    \IE[Y \mid \alpha_1 = b_1, \ldots, \alpha_{i-1} = b_{i-1}] \geq \IE[Y].
\end{equation*}
Mit~\eqref{eq:bayesCondExpIndep} gilt dann 
\begin{equation*}
    \begin{split}
        \IE[Y] &\leq \IE[Y \mid \alpha_1 = b_1, \ldots, \alpha_{i-1} = b_{i-1}] \\
               &= \IP[\alpha_i = 0] \cdot \IE[Y \mid \alpha_1 = b_1, \ldots, \alpha_{i-1} = b_{i-1}, \alpha_i = 0] \\ &\quad+ \IP[\alpha_i = 1] \cdot \IE[Y \mid \alpha_1 = b_1, \ldots, \alpha_{i-1} = b_{i-1}, \alpha_i = 1] \\
               &= \frac{\IE[Y \mid \alpha_1 = b_1, \ldots, \alpha_{i-1} = b_{i-1}, \alpha_i = 0] + \IE[Y \mid \alpha_1 = b_1, \ldots, \alpha_{i-1} = b_{i-1}, \alpha_i = 0]}{2}.
    \end{split}
\end{equation*}
Mit demselben Argument muss einer der beiden bedingten Erwartungswerte mindestens den Wert $\IE[Y]$ haben. Wir setzen $x_i$ nun auf das $b_i$, für das das der Fall ist. Auf diese Weise können wir induktiv fortfahren und deterministisch eine Belegung $b_1, \ldots, b_n$ der Variablen $x_1, \ldots, x_n$ finden, sodass das Gesamtgewicht der erfüllten Klauseln mindestens $\IE[Y]$ beträgt.

Wir müssen nun noch zeigen, dass dieses Verfahren effizient ausgeführt werden kann. Mit Linearität des Erwartungsertes folgt (was auch für bedingte Erwartungen gilt)
\begin{equation*}
    \IE[Y \mid \alpha_1 = b_1, \ldots, \alpha_i = b_i] = \sum_{j = 1}^m w_j \cdot \IP[\text{$\alpha$ erfüllt $C_j$} \mid \alpha_1 = b_1, \ldots, \alpha_i = b_i].
\end{equation*}
Diese bedingten Wahrscheinlichkeiten können wie folgt bestimmt werden: Erfüllt eine der Variablen $x_1, \ldots, x_i$ mit der Belegung $b_1, \ldots, b_i$ bereits Klausel $C_j$, so beträgt die bedingte Wahrscheinlichkeit $1$. Ansonsten sind alle Literale in $C_j$, die aus $x_1, \ldots, x_i$ bestehen, in der Belegung $b_1, \ldots, b_i$ falsch. Diese können aus der Klausel getrichen werden. Bleiben danach noch $\ell$ Literale übrig, so beträgt die bedingte Wahrscheinlichkeit, dass die Klausel durch die zufällige Belegung erfüllt wird, $1 - (1/2)^\ell \geq 1/2$. (Wir nehmen an, dass nicht gleichzeitig $x$ und $\neg x$ in der Klausel vorkommen, da diese sonst mit Wahrscheinlichkeit $1$ erfüllt wäre.)

Wir erhalten insgesamt einen deterministischen $\frac{1}{2}$-Approximationsalgorithmus für \textsc{Max-SAT}. Die Methode der bedingten Erwartungswerte zur Derandomisierung lässt sich auch auf viele andere randomisierte Algorithmen erweitern. Insbesondere können wir mit dieser Methode auch \textsc{RandMaxCut} derandomisieren.

\subsection{Verbesserter Algorithmus}

Wir werden nun sehen, dass der Approximationsfaktor $\frac{1}{2}$ durch einen einfachen Trick verbessert werden kann. Wir setzen nun jede Variable mit Wahrscheinlichkeit $p > \frac{1}{2}$ unabhängig von den anderen Variablen auf $1$. Für eine geeignete Wahl von $p$ führt das zu einem besseren Approximationsfaktor. Zunächst betrachten wir nur Formeln, die keine Klauseln enthalten, die nur aus einem einzigen negativen Literal bestehen. Solche Klauseln nennen wir im Folgenden \emph{negative Einheitsklauseln}\index{negative Einheitsklausel}\index{Klausel!negative Einheits-}.

\begin{lem}{Zufällige Belegung mit Wahrscheinlichkeit $p$}{randomWithProbP}
    Sei $\varphi$ eine Formel in KNF über den Variablen $x_1, \ldots, x_n$ mit den Klauseln $C_1, \ldots, C_m$. Die Formel $\varphi$ enthalte keine negativen Einheitsklauseln. Eine zufällige Belegung $\alpha$ der Variablen $x_1, \ldots, x_n$, die jede Variable unabhängig mit Wahrscheinlichkeit $p \geq \frac{1}{2}$ auf $1$ setzt, erfüllt jede einzelne Klausel mit einer Wahrscheinlichkeit von mindestens $\min\set{p, 1-p^2}$.
\end{lem}

\begin{proof}
    Sei $C$ eine beliebige Klausel. Falls es sich bei $C$ um eine Klausel der Länge $1$ handelt, so besteht sie laut Voraussetzung aus einem positiven Literal. Das bedeutet, die $C$ wird von der zufälligen Belegung mit einer Wahrscheinlichkeit von $p$ erfüllt. Betrachten wir nun den Fall, dass es sich bei $C$ um eine Klausel der Länge $\ell \geq 2$ handelt. Wir bezeichnen mit $a$ und $b$ die Anzahl an positiven bzw. negativen Literalen in $C$, also ist $\ell = a + b$. Enthält $C$ für eine Variable sowohl das entsprechend positive als auch das negative Literal, ist $C$ mit Wahrscheinlichkeit $1$ erfüllt. Ansonsten gibt es genau eine Belegung der in $C$ vorkommenden Variablen, die $C$ nicht erfüllt: Alle positiven Literale sind auf $0$ gesetzt und alle negativen Literale auf $1$. Die Wahrscheinlichkeit, dass diese Belegung gewählt wird, beträgt $p^b (1-p)^a$. Folglich beträgt die Wahrscheinlichkeit, dass $C$ erfüllt ist,
    \begin{equation*}
        1 - p^b(1-p)^a \geq 1 - p^{a + b} = 1 - p^{\ell} \geq 1 - p^2,
    \end{equation*}
    wobei wir $1-p \leq p$ genutzt haben.
\end{proof}

\begin{thm}{Approximationsalgorithmus durch zufällige Belegung mit Wahrscheinlichkeit $p$}{ApproxAlgoRandomWithProbP}
    Für \textsc{Max-SAT} ohne negative Einheitsklauseln erhalten wir einen $\min\set{p, 1-p^2}$-Approximationsalgorithmus, indem wir jede Variable mit Wahrscheinlichkeit $p \geq \frac{1}{2}$ unabhängig auf $1$ setzen.
\end{thm}
\begin{proof}
    Sei $\varphi$ eine Formel ohne negative Einheitsklauseln und $Y$ das Gesamtgewicht der Klauseln durch die zufällige Belegung $\alpha$. Dann gilt 
    \begin{equation*}
        \begin{split}
            \IE[Y] &= \sum_{j = 1}^m w_j \cdot \IP[\text{$\alpha$ erfüllt $C_j$}] \geq \sum_{j = 1}^m w_j \cdot \min\set{p, 1-p^2} \\
            &= \min\set{p, 1-p^2} \cdot \sum_{j = 1}^m w_j \geq \min\set{p, 1-p^2} \cdot \OPT. \qedhere
        \end{split}
    \end{equation*}
\end{proof}

$\min{p, 1-p^2}$ ist maximal, wenn $p = 1-p^2$. In $\edgys{0,1}$ ist das für $p = \frac{\sqrt{5} - 1}{2} \approx 0.618$ der Fall. 

Nun zeigen wir noch, wie wir die Annahme eliminieren können, dass es keine negativen Einheitsklauseln gibt: Sei $\varphi$ eine beliebige KNF-Formel. Wir erhalten eine zu $\varphi$ äquivalente Formel, indem wir für eine Variable $x$ jedes Vorkommen von $x$ durch $\neg x$ und jedes $\neg x$ durch $x$ ersetzen. Das bedeutet, dass oBdA das Gewicht der Einheitsklausel $x$ mindestens so groß ist wie das Gewicht der Einheitsklausel $\neg x$ (mit der impliziten Konvention, dass Klauseln, die nicht in $\varphi$ enthalten sind, Gewicht $0$ haben). Insbesondere bedeutet das, dass nicht der Fall eintreten kann, dass $\varphi$ die Einheitsklausel $\neg x$ mit positiven Gewicht, aber nicht die Einheitsklausel $x$ enthält. Für eine Variable $x_i$ bzeichnen wir im Folgenden mit $v_i$ das Gewicht der negativen Einheitsklausel $\neg x$.

\begin{lem}{$\OPT$ ist höchstens alle Gewichte minus negative Einheitsklauseln}{OPTBoundaryMaxSAT}
    Es gilt $\OPT \leq \sum_{j = 1}^m w_j - \sum_{i = 1}^n v_i$.
\end{lem}
\begin{proof}
    Jede Belegung erfüllt für jede Variable $x_j$ entweder die Einheitsklausel $x_j$ oder die Einheitsklausel $\neg x_j$. Im besten Fall erfüllt eine Belegung alle Klauseln der Länge mindestens zwei und für jede Variable $x_j$ die Einheitsklausel mit größerem Gewicht. Aus dieser Überlegung folgt direkt das Lemma.
\end{proof}

\begin{thm}{randomisierter $\frac{\sqrt{5}-1}{2}$-Approximationsalgorithmus}{randomizedApproxAlgoWithProbPFinal}
    Für \textsc{Max-SAT} erhalten wir einen $\min\set{p, 1-p^2}$-Approximationsalgorithmus, indem wir jede Variable mit Wahrscheinlichkeit $p \geq \frac{1}{2}$ unabhängig auf $1$ setzen. Dabei nehmen wir oBdA an, dass für jede Variable $x$ das Gewicht der Einheitsklausel $x$ mindestens so groß ist wie das Gewicht der Einheitsklausel $\neg x$. Für $p = \frac{\sqrt{5}-1}{2}$ ergibt sich ein Approximationsfaktor von $\frac{\sqrt{5}-1}{2} \approx 0.618$.
\end{thm}

\begin{proof}
    Sei $\varphi$ eine beliebige KNF-Formel. Sei $U$ die Teilmenge der Klauseln von $\varphi$, die alle Klauseln aus $\varphi$ bis auf die negativen Einheitsklauseln enthält. Das Gesamtgewicht der Klauseln in $U$ beträgt $\sum_{j = 1}^m w_j - \sum_{i = 1}^n v_i$.

    Wir setzen nun jede Variable unanbhängig voneinander mit Wahrscheinlichkeit $p$ auf $1$. Sei $C$ eine beliebige Klausel aus $U$. Handelt es sich bei $C$ um eine positive Einheitsklausel, so ist sie mit Wahrscheinlichkeit $p$ erfüllt. Enthält $C$ für eine Variable $x$ die Literale $x$ und $\neg x$, so ist $C$ mit Wahrscheinlichkeit $1$ erfüllt. Ansonsten bezeichne $\ell \geq 2$ die Anzahl an Literalen in $C$. Ferner seien $a$ und $b$ die Anzahlen an positiven bzw. negativen Literalen in $C$. Die Wahrscheinlichkeit, dass $C$ von der zufälligen Belegung der Variablen erfüllt wird, beträgt analog zum Argument aus~\Cref{lem:randomWithProbP} 
    \begin{equation*}
        1 - p^b(1-p)^a \geq 1 - p^{a+b} = 1 - p^\ell \geq 1 - p^2.
    \end{equation*}
    Satz~\ref{thm:randomizedApproxAlgoWithProbPFinal} folgt nun wieder aus der Linearität des Erwartungswertes:
    \begin{equation*}
        \begin{split}
            \IE[Y] &\geq \sum_{C_j \in U} w_j \cdot \IP[\text{$\alpha$ erfüllt $C_j$}] \geq \sum_{C_j \in U} w_j \cdot \min\set{p, 1-p^2} \\
            &= \min\set{p, 1-p^2} \cdot \left( \sum_{j = 1}^m w_j - \sum_{i = 1}^n v_i \right) \\
            &\geq \min\set{p, 1-p^2} \cdot \OPT.
        \end{split}
    \end{equation*}
\end{proof}
Auch der Algorithmus aus Satz~\ref{thm:randomizedApproxAlgoWithProbPFinal} kann mit bedingten Erwartungen derandomisiert werden.

\chapter{Lineare Programmierung und Runden}

\begin{prob}{Lineares Programm (LP)}{linearProgram}\index{Lineares Programm (LP)}
    \prblmObj{$c \in \IR^d, b \in \IR^m, A \in \IR^{m \times d}$}{$x \in \IR^d$ mit $Ax \leq b$}{Minimiere/maximiere $c \cdot x$}
\end{prob}

\begin{expl}[MaxFlow]
    Sei $G = (V, E)$ ein Flussnetzwerk mit Quelle $s \in V$ und Senke $t \in  V$ und Kapazitäten $c: E \to \IN_0$. Setze $x_e$ auf den tatsächlichen Fluss für alle $e \in E$. Dann wollen wir die Funktion 
    \begin{equation*}
        \sum_{e = (s,v)} x_e - \sum_{e = (v,s)}x_e
    \end{equation*}
    maximieren unter folgenden Bedingungen:
    \begin{align*}
        &\forall e \in E : x_e \geq 0 & & \text{(Fluss nicht negativ),} \\
        &\forall e \in E : x_e \leq c(e) & & \text{(Fluss nicht größer als Kapazität),} \\
        &\forall v \in V \setminus \set{s,t}: \sum_{e = (u,v)} x_e - \sum_{e = (v,u)} x_e = 0 & & \text{(Flusserhaltung).}
    \end{align*}
\end{expl}

Lineare Programme können in polynomieller Zeit gelöst werden. Im Folgenden werden \emph{ganzzahlige lineare Programme}\index{Lineares Programm (LP)!ganzzahliges - (ILP)} eine große Rolle spielen -- lineare Programme, bei denen zusätzlich als Nebenbedingung gefordert ist, dass die Werte einiger oder aller Variablen ganzzahlig sein müssen. Das Finden einer solchen Variablenbelegung ist NP-schwer, was z. B. daraus folgt, dass das Rucksackproblem als ILP formuliert werden kann.

\section{Max-SAT}

Mittels LPs können wir einen besseren Approximationsfaktor erreichen.

\subsection{Randomisiertes Runden}

Sei $\varphi$ eine Eingabe für \textsc{Max-SAT} mit Variablen $x_1, \ldots, x_n$, Klauseln $C_1, \ldots, C_m$ und Gewichten $w_1, \ldots, w_m$. Zunächst formulieren wir die Aufgabe, eine Belegung zu finden, für die das Gesamtgewicht der erfüllten Klausen maximal wird, als ILP:
\begin{align*}
    \max \quad & \sum_{j = 1}^m w_j z_j \\ 
    s.t. \quad & \sum_{i \in P_j} y_i + \sum_{i \in N_j} (1-y_i) \geq z_j, & & \forall j \in \set{1,\ldots,m}, \\
    & y_i \in \set{0,1}, & & \forall i \in \set{1,\ldots,n}, \\
    & 0 \leq z_j \leq 1, & & \forall j \in \set{1,\ldots,m},
\end{align*}
wobei $z_j = 1$ genau dann gelten soll, wenn die durch $y_1, \ldots, y_n$ codierte Belegung die Klausel $C_j$ erfüllt. Mit $P_j$ und $N_j$ sind die Indizes der Variablen gemeint, die positiv bzw. negativ in $C_j$ vorkommen.

Wir haben nun das Problem, eine optimale Variablenbelegung zu finden, auf das Lösen eines ILP reduziert. Dies ist auch NP-schwer. Wir werden dieses ILP aber \emph{relaxieren}\index{relaxieren}\index{Lineares Programm (LP)!relaxiertes -}, d.h. die Bedingung, dass nur bestimmte Werte angenommen werden dürfen, aufgeben:

\begin{align*}
    \max \quad & \sum_{j = 1}^m w_j z_j \\ 
    s.t. \quad & \sum_{i \in P_j} y_i + \sum_{i \in N_j} (1-y_i) \geq z_j, & & \forall j \in \set{1,\ldots,m}, \\
    & 0 \leq y_i \leq 1, & & \forall i \in \set{1,\ldots,n}, \\
    & 0 \leq z_j \leq 1, & & \forall j \in \set{1,\ldots,m}.
\end{align*}

Im Folgenden seien $Z^*_{\LP}$ und $Z^*_{\ILP}$ die optimalen Zielfunktionswerte des LP bzw. des ILP. Es gilt dann $Z^*_{\ILP} = \OPT$, wobei $\OPT$ das Gesamtgewicht der Klauseln angibt, die in einer optimalen Belegung der Aussagenvariablen $x_1, \ldots, x_n$ erfüllt sind. Es gilt außerdem offensichtlich $Z^*_{\LP} \geq Z^*_{\ILP}$.

Es bezeichne $(y^*, z^*)$ mit $y^* = (y^*_1, \ldots, y^*_n)$ und $z^* = (z^*_1, \ldots, z^*_n)$ eine optimale Lösung des linearen Programms. Der Algorithmus \textsc{RandomizedRounding}\index{RandomizedRounding} sieht wie folgt aus:

\begin{algorithm}[h]
    \caption{\textsc{RandomizedRounding}}\label{alg:RandomizedRounding}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:MaxSAT}}
        \Output{Wie in~\Cref{prob:MaxSAT}}
        \State{Löse das relaxierte LP}
        \State{Setze jede Aussagenvariable $x_i$ mit Wahrscheinlichkeit $y_i^*$ auf $1$ und sonst auf $0$}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{RandomizedRounding}{randomizedRounding}
    \textsc{RandomizedRounding} ist ein randomisierter $\left(1-\frac{1}{e}\right)$-Approximationsalgorithmus für \textsc{Max-SAT}.
\end{thm}

\begin{proof}
    Wir betrachten wieder für jede Klausel separat die Wahrscheinlichkeit, dass sie durch die zufällige Belegung der Variablen erfüllt wird. Mit obiger Notation erhalten wir für jede Klausel $C_j$
    \begin{equation}
        \IP[\text{$C_j$ nicht erfüllt}] = \prod_{i \in P_j} (1 - y_i^*) \cdot \prod_{i \in N_j} y_i^*.\label{eq:RandomizedRoundingProof}
    \end{equation}
    Dies folgt daraus, dass es nur eine Belegung der in $C_j$ vorkommenden Variablen gibt, die $C_j$ nicht erfüllt: Alle positiv auftretenden Variablen auf $0$, alle negativ auftretenden auf $1$. Die rechte Seite beschreibt genau die Wahrscheinlichkeit, dass diese Belegung gewählt wird. Wir verwenden die AM-GM-Ungleichung $\prod_{i=1}^n a_i \leq \left( \frac{1}{n} \sum_{i = 1}^n a_i \right)^n$ für alle $a_1, \ldots, a_n \in \IR_{\geq 0}$. Angewendet auf~\eqref{eq:RandomizedRoundingProof} ergibt sich mit $\abs{C_j} = \abs{P_j} + \abs{N_j}$
    \begin{equation*}
        \begin{split}
            \prod_{i \in P_j} (1 - y_i^*) \cdot \prod_{i \in N_j} y_i^* &\leq \left( \frac{1}{\abs{C_j}} \left( \sum_{i \in P_j} (1 - y_i^*) + \sum_{i \in y_i^*} \right) \right)^{\abs{C_j}} \\
            &= \left( \frac{1}{\abs{C_j}}\left( \abs{C_j} + \sum_{i \in P_j} (-y_i^*) + \sum_{i \in N_j}(y_i^* - 1) \right) \right)^{\abs{C_j}} \\
            &= \left( 1 - \frac{1}{\abs{C_j}} \left( \sum_{i \in P_j} y_i^* + \sum_{i \in N_j} (1 - y_i^*) \right) \right)^{\abs{C_j}}
        \end{split}
    \end{equation*}
    und mit der Nebenbedingung 
    \begin{equation*}
        \sum_{i \in P_j} y_i^* + \sum_{i \in N_j}(1 - y_i^*) \geq z_j,
    \end{equation*}
    die das LP garantiert, ergibt sich insgesamt 
    \begin{equation*}
        \IP[\text{$C_j$ nicht erfüllt}] \leq \left( 1 - \frac{z_j^*}{\abs{C_j}} \right)^{\abs{C_j}}
    \end{equation*}
    und dementsprechend
    \begin{equation*}
        \IP[\text{$C_j$ erfüllt}] \geq 1 - \left( 1 - \frac{z_j^*}{\abs{C_j}} \right)^{\abs{C_j}}.
    \end{equation*}
    Betrachte die Funktion $f(x) = 1 - \left( 1 - \frac{x}{\abs{C_j}} \right)^{\abs{C_j}}$ genauer. Die zweite Ableitung auf $\edgys{0,1}$ ist für jedes $\abs{C_j} \geq 1$ nicht positiv, folglich ist die Funktion konkav und die Gerade $g$, die durch die Punkte $(0, f(0))$ und $(1, f(1))$ verläuft, liegt an keiner Stelle über der Funktion $f$. Wegen $f(0) = 0$ entspricht $g$ der Funktion
    \begin{equation*}
        g(x) = f(1) \cdot x = \left( 1 - \left( 1 - \frac{1}{\abs{C_j}} \right)^{\abs{C_j}} \right) \cdot x
    \end{equation*}
    und es gilt insgesamt 
    \begin{equation*}
        \IP[\text{$C_j$ erfüllt}] \geq f(z_j^*) \geq g(z_j^*) =  \left( 1 - \left( 1 - \frac{1}{\abs{C_j}} \right)^{\abs{C_j}} \right) \cdot z_j^*.
    \end{equation*}
    Bezeichnen wir wieder mit $Y$ das Gesamtgewicht der erfüllten Klauseln, so ergibt sich mit der Linearität des Erwartungswertes 
    \begin{equation*}
        \begin{split}
            \IE[Y] &= \sum_{j = 1}^m \IP[\text{$C_j$ erfüllt}] \geq \sum_{j = 1}^m w_d \left( 1 - \left( 1 - \frac{1}{\abs{C_j}} \right)^{\abs{C_j}} \right) \cdot z_j^* \\
            &\geq \min_{k \geq 1} \left( 1 - \left( 1 - \frac{1}{k} \right)^k \right) \cdot \sum_{j = 1}^m w_j z_j^* = \min_{k \geq 1} \left( 1 - \left( 1 - \frac{1}{k} \right)^k \right) \cdot Z^*_{\LP} \\
            &\geq \min_{k \geq 1} \left( 1 - \left( 1 - \frac{1}{k} \right)^k \right) \cdot \OPT.
        \end{split}
    \end{equation*}
    Die Funktion $1 - \left( 1 - \frac{1}{k} \right)^k$ ist streng monoton fallend und geht von oben gegen $1 - \frac{1}{e}$. Damit erhalten wir 
    \begin{equation*}
        \IE[Y] \geq \left( 1 - \frac{1}{e} \right) \cdot \OPT. \qedhere
    \end{equation*}
\end{proof}

Es gilt $\left( 1 - \frac{1}{e} \right) \approx 0.632$ und somit erreicht \textsc{RandomizedRounding} einen etwas besseren Approximationsfaktor als die Algorithmen aus Kapitel~\ref{sec:maxSATWtheo}. Kombinieren macht es aber noch besser:

\subsection{Kombination beider Algorithmen}\label{subsec:combinationOfBothAlgos}

Betrachten wir \textsc{RandMaxSAT} und \textsc{RandomizedRounding} genauer, so stellen wir fest, dass der Approximationsfaktor von \textsc{RandMaxSAT} besser wird, je länger die Klauseln wird, und der Approximationsfaktor von \textsc{RandomizedRounding} besser wird, je kürzer die Klauseln sind. Konkret gilt für jede Klausel, dass sie durch \textsc{RandomizedRounding} mit einer Wahrscheinlichkeit von mindestens $\left( 1 - \left( 1 - \frac{1}{\abs{C_j}} \right)^{\abs{C_j}} \right) \cdot z_j^*$ erfüllt wird und durch \textsc{RandMaxSAT} von mindestens $1 - 2^{-\abs{C_j}}$.

\begin{algorithm}[h]
    \caption{\textsc{BetterOfTwo}}\label{alg:betterOfTwo}\index{BetterOfTwo}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:MaxSAT}}
        \Output{Wie in~\Cref{prob:MaxSAT}}
        \State{Führe \textsc{RandomizedRounding} aus}
        \State{Führe \textsc{RandMaxSAT} aus}
        \State{Gib die bessere Belegung zurück}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{BetterOfTwo}{betterOfTwo}
    \textsc{BetterOfTwo} ist ein randomisierter $\frac{3}{4}$-Approximationsalgorithmus für \textsc{Max-SAT}.
\end{thm}

\begin{proof}
    Wir bezeichnen mit $Y_1$ und $Y_2$ das Gesamtgewicht der durch \textsc{RandomizedRounding} und \textsc{RandMaxSAT} erfüllten Klauseln. Da \textsc{BetterOfTwo} beide Algorithmen ausführt und anschließend das bessere Ergebnis wählt, interessieren wir uns für den Erwartungswert von $\max\set{Y_1, Y_2}$. Es gilt
    \begin{equation*}
        \IE[\max\set{Y_1, Y_2}] \geq \IE\left[ \frac{Y_1 + Y_2}{2} \right] = \frac{\IE[Y_1] + \IE[Y_2]}{2}.
    \end{equation*}
    Einsetzen der Infos aus den Beweisen von Sätzen~\ref{thm:randomizedRounding} und~\ref{thm:randMaxSAT} liefert 
    \begin{equation*}
        \begin{split}
            \frac{\IE[Y_1] + \IE[Y_2]}{2} &\geq \frac{1}{2}\sum_{j = 1}^m w_j \left( 1 - \left( 1 - \frac{1}{\abs{C_j}} \right)^{\abs{C_j}} \right) \cdot z_j^* + \frac{1}{2}\sum_{j = 1}^m w_j\left(1 - 2^{-\abs{C_j}}\right) \\
            &\geq \sum_{j = 1}^m w_j z_j^* \left( \frac{1}{2} \left( 1 - \left( 1 - \frac{1}{\abs{C_j}} \right)^{\abs{C_j}} \right) + \frac{1}{2} \left( 1 - 2^{-\abs{C_j}} \right) \right).
        \end{split}
    \end{equation*}
    Wir betrachten nun die Funktion
    \begin{equation*}
        f(x) = \frac{1}{2}\left( 1 - \left( 1 - \frac{1}{x} \right)^x \right) + \frac{1}{2}\left( 1 - 2^{-x} \right).
    \end{equation*}
    Diese Funktion nimmt für alle ganzzahligen Werte mindestens den Wert $\frac{3}{4}$ an:
    \begin{equation*}
        f(1) = \frac{1}{2} + \frac{1}{2} \cdot \frac{1}{2} = \frac{3}{4} \quad \text{und} \quad f(2) = \frac{1}{2}\left( 1 - \frac{1}{4} \right) + \frac{1}{2} \left( 1 - \frac{1}{4} \right) = \frac{3}{4}.
    \end{equation*}
    Für $x \geq 3$ gilt 
    \begin{equation*}
        f(x) \geq \frac{1}{2}\left( 1 - \frac{1}{e} \right) + \frac{1}{2}\left( 1 - 2^{-3} \right) \approx 0.753 \geq \frac{3}{4}.
    \end{equation*}
    Damit gilt insgesamt 
    \begin{equation*}
        \IE[\max\set{Y_1, Y_2}] \geq \frac{3}{4}\sum_{j = 1}^m w_j z_j^* = \frac{3}{4} \cdot Z^*_{\LP} \geq \frac{3}{4} \cdot \OPT. \qedhere
    \end{equation*}
\end{proof}

\subsection{Nichtlineares randomisiertes Runden}
\begin{algorithm}[h]
    \caption{$\textsc{RandRounding}_f$}\index{Rand@$\text{RandRounding}_f$}\label{alg:RandRoundingF}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:MaxSAT}, Funktion $f: \edgys{0,1} \to \edgys{0,1}$}
        \Output{Wie in~\Cref{prob:MaxSAT}}
        \State{Löse das relaxierte LP}
        \State{Setze jede Aussagenvariable $x_i$ mit Wahrscheinlichkeit $f(y_i^*)$ auf $1$ und sonst auf $0$} \LComment{Für $f(x) = x$ ist dies einfach \textsc{RandomizedRounding}.}
    \end{algorithmic}
\end{algorithm}

Wir analysieren den Approximationsfaktor von $\textsc{RandRounding}_f$ für eine beliebige Funktion $f: \edgys{0,1} \to \edgys{0,1}$ mit 
\begin{equation}
    1 - 4^{-x} \leq f(x) \leq 4^{x-1} \label{eq:randRoundingFunction}
\end{equation}
für alle $x \in \edgys{0,1}$. Wir werden sehen, dass damit jede Klausel $C_j$ mit Wahrscheinlichkeit von mindestens $\frac{3}{4}z_j^*$ erfüllt ist.

\begin{thm}{$\text{RandRounding}_f$}{randRoundingF}
    Für jede Funktion $f$ mit Eigenschaft~\eqref{eq:randRoundingFunction} ist $\textsc{RandRounding}_f$ ein randomisiserter $\frac{3}{4}$-Approximationsalgorithmus für \textsc{Max-SAT}.
\end{thm}

\begin{proof}
    Wir zeigen für jede Klausel, dass sie mit Wahrscheinlichkeit von mindestens $\frac{3}{4}z_j^*$ erfüllt ist. Der Satz folgt dann aus dieser Aussage analog zum Beweis von Satz~\ref{thm:randomizedRounding}.

    Sei $C_j$ eine beliebige Klausel. Wieder bezeichnen $P_j$ und $N_j$ die Mengen der Variablen, die positiv bzw. negativ in $C_j$ vorkommen. Die Klausel $C_j$ ist nur dann nicht erfüllt, wenn alle positiv auftretenden Variablen auf $0$ und alle neagtiv auftretenden Variablen auf $1$ gesetzt sind. Daraus folgt 
    \begin{equation*}
        \IP[\text{$C_j$ nicht erfüllt}] = \prod_{i \in P_j} (1 - f(y_i^*)) \cdot \prod_{i \in N_j} f(y_i^*) \leq \prod_{i \in P_j} 4^{-y_i^*} \cdot \prod_{i \in N_j} 4^{y_i^* - 1}
    \end{equation*}
    durch~\eqref{eq:randRoundingFunction}. Es gilt mit der Ungleichung
    \begin{equation*}
        \sum_{i \in P_j} y_i^* + \sum_{i \in N_j} (1 - y_i^*) \geq z_j^*
    \end{equation*}
    aus dem LP gerade 
    \begin{equation*}
        \prod_{i \in P_j} 4^{-y_i^*} \cdot \prod_{i \in N_j} 4^{y_i^* - 1} = 4^{-\left( \sum_{i \in P_j} y_i^* + \sum_{i \in N_j} (1 - y_i^*) \right)} \leq 4^{-z_j^*}.
    \end{equation*}
    Es gilt somit 
    \begin{equation*}
        \IP[\text{$C_j$ erfüllt}] \geq 1 - 4^{-z_j^*}.
    \end{equation*}
    Da die Funktion $g(x) = 1 - 4^{-x}$ in $\edgys{0,1}$ konkav ist, ist die Gerade $h$, die durch die Punkte $(0, g(0))$ und $(1, g(1))$ geht, eine untere Schranke (analog zum Beweis von Satz~\ref{thm:randomizedRounding}). Wegen $g(0) = 0$ gilt $h(x) = g(1) \cdot x$ und somit insgesamt 
    \begin{equation*}
        \IP[\text{$C_j$ erfüllt}] \geq 1 - 4^{-z_j^*} = g(z_j^*) \geq h(z_j^*) = g(1) \cdot z_j^* = \left( 1 - \frac{1}{4} \right)z_j^* = \frac{3}{4} z_j^*.
    \end{equation*}
    Wieder bezeichnen wir mit $Y$ das Gesamtgewicht der erfüllten Klauseln. Insgesamt ergibt sich 
    \begin{equation*}
        \begin{split}
            \IE[Y] &= \sum_{j = 1}^m w_j \cdot \IP[\text{$C_j$ erfüllt}] \\
                   &\geq \sum_{j = 1}^m w_j \cdot \frac{3}{4}z_j^* = \frac{3}{4} \sum_{j = 1}^m w_j z_j^* \\
                   &= \frac{3}{4} \cdot Z^*_{\LP} \geq \frac{3}{4} \cdot \OPT. \qedhere
        \end{split}
    \end{equation*}
\end{proof}

Wir fragen uns nun, ob ein besserer Approximationsfaktor als $\frac{3}{4}$ erreicht werden kann.

\begin{expl}[Instanz für \textsc{Max-SAT}]
    Wir betrachten die Formel 
    \begin{equation*}
        \varphi = (x_1 \vee x_2) \wedge (x_1 \vee \neg x_2) \wedge (\neg x_1 \vee x_2) \wedge (\neg x_1 \vee \neg x_2),
    \end{equation*}
    wobei alle Klauseln das Gewicht $1$ haben. Man sieht leicht, dass jede Belegung der Aussagenvariablen genau drei der Klauseln erfüllt. Somit gilt $\OPT = 3$. In der optimalen Lösung des LP gilt $y_1^* = y_2^* = \frac{1}{2}$ und $z_j = 1$ für alle Klauseln. Folglich ist $Z^*_{\LP} = 4$.
\end{expl}

Es gibt also Eingaben $\varphi$ für \textsc{Max-SAT} für die $\OPT \leq \frac{3}{4}Z^*_{\LP}$ gilt, also können wir keinen besseren Approximationsfaktor als $\frac{3}{4}$ zeigen, wenn wir als obere Schranke nur $Z^*_{\LP}$ verwenden. Dafür haben wir aber folgende Verallgemeinerung:

\begin{defi}{Integrality Gap}{integralityGap}
    Das \emph{Integrality Gap}\index{Integrality Gap}\index{Lineares Programm (LP)!ganzzahliges - (ILP)!Integrality Gap eines -es} ist der Quotient aus dem optimalen Wert des ILPs und dem optimalen Wert des LPs, das durch Relaxierung des ILPs entsteht.

    Haben wir für ein Optimierungsproblem eine Formulierung der Instanzen als ILP gefunden, so bezeichnen wir mit dem \emph{Integrality Gap dieser Formulierung} das schlechteste Integrality Gap einer Instanz des Optimierungsproblems.
\end{defi}
Bei Maximierungsproblemen betrachten wir also das kleinste Integrality Gap über alle Instanzen und bei Minimierungsproblemen das größte.

Allgemein gilt, dass wir für einen Approximationsalgorithmus keinen besseren Approximationsfaktor als das Integrality Gap zeigen können, wenn wir als einzige Schranke für den Wert einer optimalen Lösung die optimale Lösung des relaxierten LPs benutzen.

\section{Facility Location ohne Kapazitäten}
\begin{prob}{Facility Location (ohne Kapazitäten)}{facilityLocNoCapa}\index{Facility Location}
    \prblmObj{Menge $F$ von Standorten (\emph{Facilities}), Menge $D$ von Kunden; Distanzen $d: F \times D \to \IR_{\geq 0}$, Kosten $f: F \to \IR_{\geq 0}$. Wir schreiben $d_{ij}$ für $d(i,j)$ und $f_i$ für $f(i)$.}{Teilmenge $S \subseteq F$ mit $S \neq \emptyset$}{Minimiere $\displaystyle\sum_{i \in S} f_i + \sum_{j \in D} \min_{i \in S} d_{ij}$}
\end{prob}
Es geht darum, eine optimale Menge von Fabrikstandorten zu wählen. Die Kosten, um einen Kunden von einer bestimmten Fabrik zu beliefern, sind durch $d$ gegeben, wobei jeder Kunde von der nächsten Fabrik beliefert wird. Die Gesamtkosten einer Menge von Fabrikstandorten ergeben sich durch die Kosten zum Eröffnen der Standorte (gegeben durch $f$) sowie die Kosten, um alle Kunten zu beliefern. Dieses Problem ist NP-schwer.

\subsection{Reduktion auf Set Cover}

Sei $(D, F, d, f)$ eine Eingabe für Facility Location. Wir konstruieren daraus eine Menge für Set Cover, bei der die Kunden die abzudeckende Grundmenge $S$ bilden. Die Familie der Teilmengen von $S$, die wir zur Abdeckung von $S$ einsetzen können, heiße $T$. Die Familie $T$ enthält dabei jede mögliche Teilmenge $A \subseteq S$ mit $A \neq \emptyset$ genau $\abs{F}$ mal, und zwar für jedes $i \in F$ einmal mit Kosten $f_i + \sum_{j \in A} d_{ij}$. Somit können wir jeder Menge aus $T$ einen Standort $i \in F$ zuordnen.

In einer optimalen Lösung für die konstruierte Instanz von Set Cover können keine zwei Mengen $S_1, S_2 \in T$ ausgewählt sein, die demselben Standort $i \in F$ zugeordnet sind, diese könnten sonst durch die Vereinigung $S_1 \cup S_2$ ersetzt werden, was die Kosten um $f_i$ senken würde. Somit interessieren uns nur Lösungen für die Instanz von Set Cover, die für jedes $i \in F$ höchstens eine Menge aus $T$ auswählen. Ebenso können wir annehmen, dass die ausgewählten Mengen aus $T$ disjunkt sind, da wir ansonsten Elemente aus den Mengen entfernen könnten, ohne die Kosten zu erhöhen.

Zwischen den Teilmengen von $T$, die alle Elemente aus $S$ abdecken, paarweise disjunkt sind und für kein $i \in F$ mehr als eine Menge enthalten, auf der einen Seite, und den Lösungen für die Instanz von Facility Location auf der anderen Seite, existiert eine naheliegende Bijektion. Modulo dieser Bijektion stimmen auch die optimalen Lösungen überein.

Wir wenden nun \textsc{GreedySC} (Algorithmus~\ref{alg:greedySC}) auf die konstruierte Instanz an. Dieser wählt in jedem Schritt eine Menge aus $T$ mit minimalen relativen Kosten aus und fügt diese der Lösung hinzu. Hat er mit den bislang gewählten Mengen aus $T$ bereits eine Teilmenge $D' \subseteq D$ der Kunden abgedeckt, so bedeutet dies, dass er einen Standort $i \in F$ und eine Menge $A \subseteq D$ auswählt, für die 
\begin{equation}
    \frac{f_i + \sum_{j \in A} d_{ij}}{\abs{A \setminus D'}} \label{eq:facilityLocFromSC}
\end{equation}
minimal wird. Gemäß Satz~\ref{thm:greedySC} erreicht dieser Algorithmus einen Approximationsfaktor von $H_{\abs{D}} \in \cO(\log(\abs{D}))$. Allerdings ist nicht klar, dass der Greedy-Algorithmus eine polynomielle Laufzeit besitzt, da $T$ exponentiell groß ist. Dennoch kann in polynomieller Zeit ein Paar $(i,A)$ mit $i \in F$ und $A \subseteq D$ gefunden werden, für das~\eqref{eq:facilityLocFromSC} minimal wird. Dazu beachte, dass für ein Paar $(i, s)$ mit $i \in F$ und $s \in \set{d, \ldots, \abs{D \setminus D'}}$ in polynomieller Zeit eine Menge $A$ mit $\abs{A \setminus D'} = s$ gefunden werden kann, für die~\eqref{eq:facilityLocFromSC} minimal wird. Dazu genügt es, $A$ als die Menge der $s$ nächsten Nachbarn von $i$ aus der Menge $D \setminus D'$ zu wählen. Der Algorutgnzs testet dann alle polynomiell vielen Wahlen von $(i, s)$ mit $i \in F$ und $s \in \set{1,\ldots,\abs{D \setminus D'}}$ und findet so die Teilmenge mit den insgesamt kleinsten relativen Kosten. Es ergibt sich also folgendes Resultat:

\begin{thm}{Facility Location}{facilityLoc}
    Für das Facility-Location-Problem existiert ein $\cO(\log(\abs{D}))$-Approximationsalgorithmus.
\end{thm}

\subsection{Metrisches Facility Location}

Wir nehmen nun an, dass $D \cup F$ in eine Metrik eingebettet ist, folglich also die Dreiecksungleichnug $d_{xy} + d_{yz} \geq d_{xz}$ für alle $x, y, z \in D \cup F$ gilt. Wir schreiben zunächst eine gegebene Eingabe als ILP. Dafür führen wir Variablen $x_{ij} \in \set{0,1}$ und $y_i \in \set{0,1}$ für alle $i \in F$ und $j \in D$ ein. Dabei soll $y_i = 1$ genau dann gelten, wenn Standort $i$ eröffnet wird, und $x_{ij} = 1$ genau dann, wenn Kunde $j$ von Standort $i$ beliefert wird. Wir müssen noch garantieren, dass jeder Kunde von einem \emph{geöffneten} Standort beliefert wird und erhalten folgendes ILP:
\begin{align*}
    \min \quad & \sum_{i \in F} f_i y_i + \sum_{i \in F} \sum_{j \in D} d_{ij} x_{ij} \\ 
    s.t. \quad & \sum_{i \in F} x_{ij} \geq 1, && \forall j \in D, \\
    & x_{ij} \leq y_{i}, && \forall i \in F, j \in D, \\
    & y_i \in \set{0,1}, && \forall i \in F, \\
    & x_{ij} \in \set{0,1}, && \forall i \in F, j \in D.
\end{align*}
Da die Zielfunktion minimiert wird, ist sichergestellt, dass in der optimalen Lösung sogar $\sum_{i \in F} x_{ij} = 1$ gilt. Wir betrachten nun aber wieder die Relaxierung des ILP, indem wir nur noch $x_{ij} \in \edgys{0,1}$ und $y_i \in \edgys{0,1}$ fordern, dann erhalten wir ein LP, welches wir in polynomieller Zeit optimal lösen können. Sei $(x^*, y^*)$ eine optimale Lösung dieses LPs. Wir werden diese Lösung wieder durch eine geeignete Rundung (dieses Mal aber deterministisch) ganzzahlig machen.

Wir bezeichnen mit
\begin{equation*}
    \OPT_{\LP} = \sum_{i \in F} f_i y_i^* + \sum_{i \in F} \sum_{j \in D} d_{ij} x_{ij}^*
\end{equation*}
die Kosten der optimalen Lösung des LPs. Ferner bezeichnen wir mit $F^* = \sum_{i \in F}  f_i y_i^*$ die Kosten für das Eröffnen der Standorte und mit $D^* = \sum_{i \in F} \sum_{j \in D} d_{ij} x_{ij}^*$ die Kosten für das Beliefern der Kunden in der optimalen Lösung des LPs.

Für einen Kunden $j \in D$ seien $D_j^* = \sum_{i \in F} d_{ij} x_{ij}^*$ die Gesamtkosten, um diesen Kunden zu beliefern. Wir definieren die \emph{Nachbarschaft} $N_j$ von Kunden $j$ als 
\begin{equation*}
    N_j = \set*{i \in F \mid d_{ij} \leq 2D_j^* \text{ und } x_{ij}^* > 0}. 
\end{equation*}

\begin{lem}{Kunde von Hälfte der Nachbarschaft beliefert}{customerGetsProductsFromHalfNeigbourhood}
    Für jeden Kunden $j \in D$ gilt
    \begin{equation*}
        \sum_{i \in N_j} x_{ij}^* \geq \frac{1}{2}.
    \end{equation*}
\end{lem}
\begin{proof}
    Angenommen, dies ist nicht der Fall. Dann folgt aus $\sum_{i \in F} x_{ij}^* \geq 1$ als Nebenbedingung direkt $\sum_{i \in F \setminus N_j} x_{ij}^* > \frac{1}{2}$. Dies impliziert den Widerspruch 
    \begin{equation*}
        D_j^* = \sum_{i \in F} d_{ij} x_{ij}^* \geq \sum_{i \in F \setminus N_j} d_{ij} x_{ij}^* > 2D_j^* \sum_{i \in F \setminus N_j} x_{ij}^* > D_j^*. \qedhere
    \end{equation*}
\end{proof}

Wir wandeln die Lösung $(x^*, y^*)$ nun in eine ganzzahlige Lösung um. Dazu sei $j \in D$ ein Kunde mit minimalen Verbindungskosten $D_j^*$, und sei $k = \arg\min_{i \in N_j} f_i$ der günstigste Standort in der Nachbarstaft von Kunden $j$. Wir setzen $y_k'' = 1$ und $y_i'' = 0$ für alle $i \in N_j \setminus \set{k}$. Das heißt, Standort $k$ wird eröffnet und alle anderen Standorte aus der Nachbarschaft von $j$ bleiben geschlossen. Setze außerdem $x_{kj}'' = 1$ und $x_{ij}'' = 0$ für alle $i \neq k$. Das heißt, Kunde $j$ wird in der ganzzahligen Lösung von Standort $k$ beliefert. Ferner ordnen wir alle Kunden $\ell$ mit $N_j \cap N_\ell \neq \emptyset$ ebenfalls Standort $k$ zu, setzen also $x_{k\ell}'' = 1$ und $x_{i\ell}'' = 0$ für alle $i \neq k$.

Solange es noch einen Kunden gibt, der keinem Standort zugewiesen wurde, fahren wir nach diesem Schema fort. Das heißt wir wählen einen Kunden $j \in D$, der unter allen noch nicht zugewiesenen Kunden minimale Verbindungskosten $D_j^*$ besitzt. Für diesen Kunden wählen wir den günstigsten Standort $k$, öffnen diesen und weisen $j$ sowie allen Kunden Standort $k$ zu, die noch nicht zugewiesen sind und einen Standort in der Nachbarschaft haben, der auch in der Nachbarschaft von Kunde $j$ liegt.

Wir zeigen, dass die ganzzahlige Lösung $(x'', y'')$ eine $6$-Approximation berechnet. Da diese effizient gefunden werden kann, ist obiges Verfahren ein $6$-Approximationsalgorithmus für Facility Location.

\begin{thm}{Metrisches Facility Location}{metricFacilityLoc}
    Die Gesamtkosten der ganzzahligen Lösung $(x'', y'')$ sind höchstens $6$-mal so groß wie die Kosten $\OPT_{\LP}$ der optimalen Lösung $(x^*, y^*)$ des LPs.
\end{thm}

\begin{proof}
    Sei $D'$ die Menge aller Kunden, für die ein neuer Standort eröffnet wird. Bei der Betrachtung eines Kunden $j \in D'$ eröffnen wir Standort $k = \arg\min_{i \in N_j} f_i$. Die Kosten dafür betragen
    \begin{equation*}
        f_k \leq 2\sum_{i \in N_j} f_i y_i^*,
    \end{equation*}
    wobei wir ausgenutzt haben, dass $f_k \leq f_i$ für alle $i \in N_j$ gilt und dass nach Lemma~\ref{lem:customerGetsProductsFromHalfNeigbourhood} 
    \begin{equation*}
        \sum_{i \in N_j} y_i^* \geq \sum_{i \in N_j} x_{ij}^* \geq \frac{1}{2}
    \end{equation*}
    gilt. Wir beobachten außerdem, dass die Nachbarschaften der Kunden in $D'$ disjunkt sein müssten, denn wenn $N_j \cap N_{j'} \neq \emptyset$ und es wird für Kunde $j$ ein neuer Standort eröffnet, dann wŕde $j'$ ebenfalls diesem Standort zugeordnet werden. Somit folgt, dass jeder Standort $i \in F$ in höchstens einer Menge $N_j$ mit $j \in D'$ enthalten sein kann. 

    Insgesamt lassen sich die Eröffnungskosten damit beschränken durch 
    \begin{equation*}
        \sum_{j \in D'} \min_{i \in N_j} f_i \leq \sum_{j \in D'} \sum_{i \in N_j} f_i 2y_i^* \leq 2\sum_{i \in F} f_i y_i^* = 2F^*,
    \end{equation*}
    wobei die zweite Ungleichung folgt, da jeder Standort $i \in F$ in höchstens einer Menge $N_j$ mit $j \in D'$ enthalten sein kann.

    Zuletzt müssen wir noch abschätzen, wie hoch die Verbindungskosten sind. Betrachten wir wieder einen Schritt, in dem Kunde $j \in D$ mit minimalen Verbindungskosten $D_j^*$ ausgewählt wird. Dieser wird einem Standort $k$ in seiner Nachbarschaft zugewiesen. Aus der Definition der Nachbarschaft folgt $d_{kj} \leq 2D_j^*$. Alle Kunden, die einen gemeinsamen Standort in der Nachbarschaft von $j$ haben, weden nun ebenfalls dem Standort $k$ zugewiesen. Sei $\ell$ ein solcher Kunde und $i \in N_j \cap N_\ell$. Nun gelten $d_{kj} \leq 2D_j^*$ und $d_{ij} \leq 2D_j^*$, da $k, i \in N_j$. Außerdem gilt $d_{i\ell} \leq 2D_\ell^*$, da $i \in N_\ell$. Nach Dreiecksungleichung gilt damit 
    \begin{equation*}
        d_{k\ell} \leq d_{kj} + d_{ij} + d_{i\ell} \leq 2 D_j^* + 2 D_j^* + 2 D_\ell^* \leq 6 D_\ell^*.
    \end{equation*}
    Im letzten Schritt haben wir ausgenutzt, dass Kunde $j \in D$ gemäß seiner Wahl unter allen noch nicht zugewiesenen Kunden minimale Verbindungskosten $D_j^*$ besitzt und insbesondere also  $D_j^* \leq D_\ell^*$ gilt. Somit betragen die Gesamtkosten der Lösung $(x'', y'')$ höchstens 
    \begin{equation*}
        2F^* + \sum_{j \in D} 6D_j^* = 2F^* + 6D^* \leq 6(F^* + D^*) = 6 \OPT_{\LP}. \qedhere
    \end{equation*}
\end{proof}

Im Beweis fällt auf, dass in der gerundeten Lösung $(x'', y'')$ das Eröffnen der Standorte höchstens doppelt so viel kostet wie in der optimalen Lösung des LPs, während das Beliefern der Kunden bis zu sechsmal so teuer sein kann. Durch eine geschicktere Wahl der Nachbarschaft können diese Beiträge balanciert werden, was insgesamt zu einem besseren Approximationsalgorithmus führt: Sei $\alpha = 1$. Wir definieren die $\alpha$-Nachbarschaft von Kunde $j$ als 
\begin{equation*}
    N_j^{(\alpha)} = \set*{i \in F \mid d_{ij} \leq \alpha D_j^* \text{ und } x_{ij}^* > 0}.
\end{equation*} 
Dann ist $N_j = N_j^{(2)}$. Der Algorithmus läuft nun genauso wie oben, nur mit der $N_j^{(\alpha)}$-Nachbarschaft. Das Lemma würde entsprechend zeigen, dass 
\begin{equation*}
    \sum_{i \in N_j^{(\alpha)}} x_{ij}^* \geq 1 - \frac{1}{\alpha}
\end{equation*}
gilt. Dann kann auch der Beweis von Satz~\ref{thm:metricFacilityLoc} übertragen werden. Damit kann gezeigt werden, dass 
\begin{equation*}
    \sum_{i \in F} f_i y_i' + \sum_{j \in D} 3\alpha D_j^* \leq \frac{\alpha}{\alpha - 1} \sum_{i \in F} f_i y_i^* + 3\alpha D^* \leq \max\set*{\frac{\alpha}{\alpha - 1}, 3\alpha} \cdot \OPT_{\LP},
\end{equation*}
wobei dieser Wert minimal ist für $\alpha = \frac{4}{3}$ und wir erhalten einen $4$-Approximationsalgorithmus.

\section{Scheduling auf allgemeinen Maschinen}

\begin{prob}{Scheduling auf allgemeinen Maschinen}{schedulingGeneralMachines}\index{Scheduling}\index{Scheduling!auf allgemeinen Maschinen}
    \prblmObj{Menge $J = \set{1,\ldots,n}$ von Jobs; Menge $M = \set{1,\ldots,m}$ von Maschinen; Jobgröße $p_{ij} \in \IR_{>0}$ für alle $i \in M$ und $j \in J$}{Abbildung (\emph{Schedule}) $\pi: J \to M$}{Es sei $L_i(\pi) = \displaystyle\sum_{\substack{j \in J \\ \pi(j) = i}} p_{ij}$ die Ausführungszeit von Maschine $i$ in Schedule $\pi$. Minimiere den \emph{Makespan} $C(\pi) = \displaystyle\max_{i \in M} L_i(\pi)$}
\end{prob}
Der einzige Unterschied zum Scheduling auf identischen Maschinen (\Cref{prob:SchedulingIdenticalMachines}) besteht  darin, dass ein Job $j \in J$ nicht eine feste Größe $p_j$, sondern auf jeder Maschine $i \in M$ eine individuelle Größe $p_{ij}$ besitzt. Es ist bekannt, dass für P$\neq$NP für Scheduling auf allgemeinen Maschinen kein Approximationsalgorithmus mit besseren Approximationsfaktor als $\frac{3}{2}$ existiert. Der beste bekannte Approximationsalgorithmus ist ein $2$-Approximationsalgorithmus, welchen wir auch besprechen werden.

Dazu formulieren wir dies wieder als ILP. $x_{ij} \in \set{0,1}$ gibt an, ob Job $j$ Maschine $i$ zugewiesen wird, und $t \in \IR$ beschreibe den Makespan des Schedules:
\begin{align*}
    \min \quad & t \\ 
    s.t. \quad & \sum_{i \in M} x_{ij} = 1, && \forall j \in J, \\ 
    & \sum_{j \in J} p_{ij} x_{ij} \leq t, && \forall i \in M, \\
    & x_{ij} \in \set{0,1}, && \forall i \in M, j \in J.
\end{align*}

Versuchen wir mithilfe dieses ILP einen Approximationsalgorithmus durch Relaxieren zu entwerden, stellen wir fest, dass dies nicht zum Erfolg führen kann, da das Integrality Gap sehr groß ist:

\begin{expl}[Großes Integrality Gap]
    Wir haben einen Job und $m$ Maschinen. Der Job habe überall die Größe $1$.

    Dann besitzt die optimale ganzzahlige Lösung einen Makespan von $1$. In der optimalen Lösung des relaxierten LP wird hingegen jeder Maschine ein $\frac{1}{m}$-Bruchteil zugewisen, also $x_{i1} = \frac{1}{m}$ für alle $i \in M$, sodass die optimale Lösung des relaxierten LPs den Wert $\frac{1}{m}$ besitzt. Das Integrality Gap beträgt damit $m$.
\end{expl}

Unter Verwendung der optimalen Lösung des relaxierten LP als untere Schranke können wir also keinen Approximationsalgorithmus zeigen, der einen besseren Approximationsfaktor als $m$ hat. 

Stattdessen gehen wir davon aus, dass wir als Eingabe bereits einen Makespan $T$ erhalten. Wir testen dann, ob das relaxierte LP eine Lösung mit Wert höchstens $T$ besitzt, wobei wir die obigen Nebenbedingungen sicher stellen. Sei dazu $S_T = \set{(i, j) \in M \times J \mid p_{ij} \leq T}$. Wir betrachten nun das folgende LP $\LP_T$ ohne Zielfunktion:
\begin{align}
    & \sum_{i \in M} x_{ij} = 1, && \forall j \in J, \label{eq:schedulingLPFirst} \\
    & \sum_{j \in J} p_{ij} x_{ij} \leq T, && \forall i \in M, \label{eq:schedulingLPSecond} \\
    & x_{ij} = 0, && \forall (i,j) \notin S_T, \label{eq:schedulingLPThird} \\
    & x_{ij} \geq 0, && \forall (i,j) \in S_T. \label{eq:schedulingLPFourth}
\end{align}

Wir haben bei diesem LP keine Zielfunktion angegeben, da wir nur testen möchten, ob überhaupt eine zulässige Lösung existiert. Dies ist ebenfalls in polynomieller Zeit möglich. Besitzt $\LP_T$ keine Lösung, so offensichtlich auch keine ganzzahlige. Das bedeutet, dass der Makespan $\OPT$ der optimalen Lösung größer als $T$ sein muss. Mit einer binären Suche kann in polynomieller Zeit der Wert 
\begin{equation*}
    T^* = \min\set{T \mid \text{$\LP_T$ besitzt eine Lösung}}
\end{equation*}
bestimmt werden. Für diesen gilt $T^* \leq \OPT$, da $T^* - \varepsilon < \OPT$ für jedes $\varepsilon > 0$ gilt. Wir zeigen nun, wie aus einer zulässigen Lösung für $\LP_{T^*}$ eine ganzzahlige Lösung mit Makespan höchstens $2T^* \leq 2\OPT$ konstruiert werden kann.

Wir müssen uns nun intensiver mit der LP-Struktur beschäftigen und könnnen diese nicht mehr nur als Blackbox behandeln:

In einem LP mit $d$ Variablen entspricht jede Variablenbelegung einem Punkt im $\IR^d$. Jede lineare Nebenbedingung definiert einen Halbraum im $\IR^d$ und der Schnitt all dieser Halbräume entspricht der Menge der zulässigen Lösungen für das LP. Einen solchen Schnitt nennt man \emph{konvexes Polyeder}\index{Polyeder!konvexes -}\index{konvexes Polyeder}, und das zu einem LP gehörende konvexe Polyeder nennen wir auch seinen \emph{Lösungspolyeder}\index{Lösungspolyeder}\index{Lineares Programm (LP)!Lösungspolyeder}\index{Polyeder!Lösungs-}.

Wir betrachten den Rand konvexer Polyeder genauder. Sei $\cP \subseteq \IR^d$ ein beliebiges konvexes Polyeder und sei eine Hyperebene $\cA = \set{x \in \IR^d \mid u \cdot x = w}$ mit beliebigen $u \in \IR^d$ und $w \in \IR$ gegeben. Ist $\cP \cap \cA \neq \emptyset$ und gilt $\cP \subseteq \cH$, wobei $\cH = \set{x \in \IR \mid u \cdot x \leq w}$ den zu $\cA$ gehörenden Halbraum bezeichnet, so sagen wir, dass $\cA$ eine \emph{stützende Hyperebene}\index{stützende Hyperebene}\index{Polyeder!stützende Hyberebene} von $\cP$ ist. Für jede stützende Hyperebene nennen wir die Menge $\cP \cap \cA$ \emph{Fläche von $\cP$}\index{Fläche}\index{Polyeder!Fläche}. Eine nulldimensionale Fläche heißt \emph{Ecke von $\cP$}\index{Ecke}\index{Polyeder!Ecke}, eine eindimensionale Fläche heißt \emph{Kante von $\cP$}\index{Kante}\index{Polyeder!Kante} und eine Fläche der Dimension $d-1$ heißt \emph{Seitenfläche von $\cP$}\index{Seitenfläche}\index{Fläche!Seiten-}\index{Polyeder!Seitenfläche}\index{Polyeder!Fläche!Seiten-}.

Die Ecken des Lösungspolyeders entsprechen den zulässigen Basislösungen. Geometrisch wird eine Ecke durch den Schnitt von $d$ linear unabhängigen Hyperebenen definiert, in einer Ecke sind also mindestens $d$ der Nebenbedingungen mit Gleichheit erfüllt. Eine Lösung $x$ ist genau dann eine zulässige Basislösung, wenn kein $y \in \IR^d \setminus \set{0}$ existiert, sodass $x + y \in \cP$ und $x - y \in \cP$ gilt. Außerdem gibt es zu jedem zulässigen und beschränkten LP eine zulässige Basislösung existiert, die optimal ist, wenn alle Variablen nur nichtnegative Werte annehmen können. Eine solche kann in polynomieller Zeit gefunden werden, wir können also in polynomieller Zeit eine zulässige Basislösung von $\LP_{T^*}$ berechnen.

\begin{lem}{Zulässige Basislösung von $\LP_{T^*}$}{feasibleBasicSolOfLPTStar}
    In einer zulässigen Basislösung von $\LP_{T^*}$ besitzen höchstens $n + m$ Variablen einen Wert echt größer als Null.
\end{lem}
\begin{proof}
    Die Anzahl Variablen in $\LP_{T^*}$ beträgt $nm$. Das bedeutet, in einer zulässigen Basislösung müssen mindestens $nm$ Nebenbedingungen mit Gleichheit erfüllt sein. Es gibt $n$ Bedingungen vom Typ~\eqref{eq:schedulingLPFirst} und $m$ Bedingungen vom Typ~\eqref{eq:schedulingLPSecond}. Selbst wenn alle diese mit Gleichheit erfüllt sind, müssen noch mindestens $nm-(n+m)$ viele Bedingungen vom Typ~\eqref{eq:schedulingLPThird} oder~\eqref{eq:schedulingLPFourth} mit Gleichheit erfüllt sein.
\end{proof}

Sei nun $x$ eine zulässige Basislösung von $\LP_{T^*}$. Zu dieser Lösung definieren wir den \emph{Allokationsgraphen}\index{Allokationsgraph}\index{Lineares Programm (LP)!Allokationsgraph} $G$. Dabei handelt es sich um einen bipartiten Graphen mit Knotenmenge $M \cup J$ und Kantenmenge $E = \set{(i,j) \in M \times J \mid x_{ij} > 0}$. Eine Kante $(i,j)$ in $G$ besagt also, dass Job $j$ in der zulässigen Basislösung $x$ zumindest teilweise Maschine $i$ zugewiesen wird. Wir nennen einen Graphen mit Knotenmenge $V$ einen \emph{Pseudobaum}\index{Pseudobaum}, falls dieser zusammenhängend ist und höchstens $\abs{V}$ Kanten enthält. Da jeder Spannbaum eines solchen Graphen genau $\abs{V} - 1$ Kanten enthält, handelt es sich bei einem Pseudobaum entweder um einen Spannbaum oder um einen Spannbaum mit einer zusätzlichen Kante, die einen Kreis schließt. Das folgende Korollar folgt direkt aus~\Cref{lem:feasibleBasicSolOfLPTStar}:

\begin{cor}\label{cor:pseudotree}
    Falls der Allokationsgraph $G$ zusammenhängend ist, so ist er ein Pseudobaum.
\end{cor}

Wir nennen einen Grapehn einen \emph{Pseudowald}\index{Pseudowald}, wenn jede seiner Zusammenhangskomponenten ein Pseudobaum ist.

\begin{lem}{Allokationsgraph ist Pseudowald}{allocationGraphIsPseudoForest}
    Der Allokationsgraph $G$ ist ein Pseudowald.
\end{lem}

\begin{proof}
    Es sei $G' = (V', E')$ eine beliebige Zusammenhangskomponente von $G$ mit $V' = M' \cup J'$ für $M' \subseteq M$ und $J' \subseteq J$. Wir betrachten nun ein eingeschränktes Schedulingproblem, bei dem nur noch die Jobs aus $J'$ und die Maschinen aus $M'$ vorhanden sind. Wir bezeichnen mit $\LP_{T^*}'$ das zu $\LP_{T^*}$ analoge LP, das sich für die eingeschränkte Instanz ergibt. Außerdem bezeichnen wir mit $x'$ die Einschränkung der Lösung $x$ auf die Variablen, die in $\LP_{T^*}'$ noch auftreten. Dann ist $x'$ eine zulässige Lösung von $\LP_{T^*}'$.

    Wir zeigen nun, dass $x'$ sogar eine zulässige Basislösung von $\LP_{T^*}'$ ist: Angenommen, dies ist nicht der Fall. Dann gibt es $y' \neq 0$, sodass $x'+y'$ und $x'-y'$ zulässige Lösungen von $\LP_{T^*}'$ sind. Wir erweitern $y'$ zu einem $y \in \IR^{nm}$, indem wir $y_{ij} = 0$ für alle $i \notin M'$ oder $j \notin J'$ setzen. Dann sind $x + y$ und $x - y$ zulässige Lösungen von $\LP_{T^*}$ und $x$ ist keine zulässige Basislösung von $\LP_{T^*}$. $\lightning$

    Da $x'$ eine zulässige Basislösung von $\LP_{T^*}'$ ist und $G'$ zusammenhängend ist, ist er laut Korollar~\ref{cor:pseudotree} ein Pseudobaum.
\end{proof}

Mithilfe des Allokationsgraphen runden wir die Lösung $x$ nun in eine ganzzahlinge Lösung $\hat{x}$. Für einen Job $j \in J$, der bereits vollständig einer Maschine $i \in M$ zugewiesen ist (für den also $x_{ij} = 1$ gilt), übernehmen wir die Zuweisung in $\hat{x}$. Solange es noch einen solchen Job gibt, entfernen wir ihn und die entsprechende Kante aus dem Allokationsgraphen. Den Graphen, der danach übrig bleibt, nenenn wir $H$. Die Knotenmenge von $H$ ist $J' \cup M$, wobei $J' \subseteq J$ die Menge der Jobs bezeichnet, die in der Lösung $x$ auf mehr als eine Maschien verteilt werden.

In $H$ berechnen wir ein einseitig perfektes Matching, d. h. eine Kantenmenge $N$, sodass jeder Job aus $J'$ genau eine inzidente Kante in $N$ und jede Maschine aus $M$ höchstens eine inzidente Kante in $N$ besitzt.

\begin{lem}{$H$ hat einseitig perfektes Matching}{semiperfectMatchingExists}
    Der Graph $H$ besitzt ein einseitiges perfektes Matching, welches in polynomieller Zeit gefunden werden kann.
\end{lem}
\begin{proof}
    Zunächst entferne alle isolierten Maschinenknoten aus $H$. Nun sind alle Blätter im Pseudowald $H$ Maschinenknoten, da Jobknoten mit Grad $1$ bereits bei der Erstellung von $H$ entfernt wurden. Solange es in $H$ noch ein Blatt gibt, führe folgende Operation aus: Wähle ein beliebiges Blatt $i \in M$ aus $H$ aus. Dieses besitzt genau einen adjazenten Knoten $j \in J$. Weise Job $j$ Maschine $i$ zu, füge also Kante $(i,j)$ zu $N$ hinzu. Anschließend entferne Knoten $i$ und $j$ und ihre inzidenten Kanten aus $H$. Lösche anschließend alle isolierten Maschinenknoten.

    Werden aus einem Pseudobaum iterativ alle Blätter auf diese Art und Weise entfernt, so bleibt am Ende ein leerer Graph oder ein Kreis übrig. Da es bei $H$ um einen bipartiten Pseudowald handelt, verbleibt somit am Ende eine (möglicherweise leere) Menge an Kreisen gerader Länge. Von jedem dieser Kreise nehmen wir jede zweite Kante zu $N$ hinzul Durch dieses Verfahren stellen wir sicher, dass jeder Knoten aus $J'$ genau eine inzidente Kante und jeder Knoten aus $M$ höchstens eine inzidente Kante erhält.
\end{proof}

Das einseitig perfekte Matching $N$ bildet zusammen mit den bereits in $x$ festgelegten Jobs die ganzzahlige Lösung $\hat{x}$.

\begin{thm}{$2$-Approximationsalgorithmus für Scheduling auf allgemeinen Maschinen}{TwoApproxAlgoForGeneralScheduling}
    Der Makespan von $\hat{x}$ beträgt höchstens $2T^*$.
\end{thm}

\begin{proof}
    Auf jeder Maschine $i \in M$ verursachen die Zuweisungen in $x$ eine Ausführungszeit von höchstens $T^*$. Insbesondere beträgt die Gesamtausführungszeit aller Jobs, die in der Lösung $x$ komplett Maschine $i$ zugewiesen sind, höchstens $T^*$. All diese Jobs erhält Maschine $i$ auch in der Lösung $\hat{x}$. Zusätzlich wird ihr beim Runden der fraktional zugewiesenen Jobs wegen~\Cref{lem:semiperfectMatchingExists} höchstens ein weiterer Job $j$ zugewiesen. Für diesen Job $j$ gilt, sofern er existiert, dass die Kante $(i, j)$ zum Graphen $G$ gehört. Demnach gilt $x_{ij} > 0$ und somit $p_{ij} \leq T^*$. Insgesamt ergibt sich damit für jede Maschine $i \in M$ eine Ausführungszeit von höchstens $2T^*$.
\end{proof}

\chapter{Semidefinite Programmierung und Runden}

Ein \emph{Vektorprogramm}\index{Vektorprogramm (VP)} besitzt $n$ Variablen $z_i \in \IR^n$, jeweils also $n$-dimensionale, reelle Vektoren. Sowohl Zielfunktion als auch Nebenbedingungen sind linear in den Skalarprodukten dieser Vektoren. Ein VP mit $m$ Nebenbedingungen lässt sich also wie folgt schreiben:
\begin{equation}
    \begin{aligned}
        \max/\min \quad & \sum_{i = 1}^n \sum_{j = 1}^n c_{ij} (z_i \cdot z_j) \\
        s.t. \quad & \sum_{i = 1}^n \sum_{j = 1}^n a_{ijk}(z_i \cdot z_j) = b_k, && \forall k \in \set{1,\ldots,m} \\
        & z_i \in \IR^n, && \forall i \in \set{1,\ldots,n}.
    \end{aligned}\label{eq:vectorProgram}
\end{equation}


\section{Semidefinite Programmierung}

Ein bisschen LA-Wiederholung: Für eine Matrix $M \in \IR^{m \times n}$ bezeichne $M^\top \in \IR^{n \times m}$ die transponierte Matrix. $M$ heißt \emph{symmetrisch}\index{Matrix!symmetrische -}, wenn $M = M^\top$ gilt. Alle Vektoren $v \in \IR^n$ fassen wir im Folgenden als Spaltenvektoren auf, d. h. $v^\top v$ entspricht dem Skalarprodukt von $v$ mit sich selbst und $v v^\top$ ist eine $(n \times n)$-Matrix.

\begin{defi}{Positiv semidefinit}{positiveSemidefinite}
    Eine Matrix $M \in \IR^{n \times n}$ heißt \emph{positiv semidefinit}\index{Matrix!positiv semidefinite -}, wenn $x^\top M x \geq 0$ für alle $x \in \IR^n$ gilt. Wir schreiben dann $M \succeq 0$.
\end{defi}

Folgende Aussage kann mit LA gezeigt werden:

\begin{lem}{Eigenschaften positiv semidefiniter Matrizen}{positiveSemidefiniteTraits}
    Für eine symmetrische Matrix $M \in \IR^{n \times n}$ sind folgende Aussagen äquivalent:
    \begin{enumerate}[$(i)$]
        \item $M$ ist positiv semidefinit,
        \item $M$ hat keine negativen Eigenwerte,
        \item Es gilt $M = V^\top V$ für eine Matrix $V \in \IR^{n \times n}$.
    \end{enumerate}
\end{lem}

Wir lernen nun \emph{semidefinite Programme}\index{Semidefinites Programm (SDP)}, eine Art von nichtlinearen Programmen kennen. Die Variablen eines solchen Programms bezeichnen wir mit $x_{ij}$ für $i, j \in \set{1, \ldots, n}$ und wir fordern $x_{ij} = x_{ji}$. Die Zielfunktion und Nebenbedingungen sind linear in diesen Variablen. Der einzige Unterschied zu einem LP ist eine zusätzliche Nebenbedingung, die nur solche Variablenbelegungen zulässt, für die die Matrix $X = (x_{ij})$ positiv semidefinit aus. Es hat also die Form 
\begin{equation}
    \begin{aligned}
        \max/\min \quad & \sum_{i = 1}^n \sum_{j = 1}^n c_{ij} x_{ij} \\ 
        s.t. \quad & \sum_{i=1}^n \sum_{j=1}^n a_{ijk} x_{ij} = b_k, && \forall k \in \set{1,\ldots,m}, \\
        & x_{ij} = x_{ji}, && \forall i,j \in \set{1,\ldots,n}, \\ 
        & X = (x_{ij}) \succeq 0.
    \end{aligned}\label{eq:semidefiniteProgram}
\end{equation}

\begin{lem}{Äquivalenz von Vektor- und semidefiniten Programmen}{vectorSemidefiniteEquiv}
    Das Vektorprogramm~\eqref{eq:vectorProgram} und das semidefinite Programm~\eqref{eq:semidefiniteProgram} sind in dem Sinne äquivalent, das eine Lösung für das eine in polynomieller Zeit in eine Lösung des anderen mit demselben Zielfunktionswert überführt werden kann und umgekehrt. Insbesondere können damit optimale Lösungen ineinander überführt werden.
\end{lem}

\begin{proof}
    Betrachte eine Lösung $z = (z_1, \ldots, z_n)$ für das VP~\eqref{eq:vectorProgram}. Wir konstruieren eine Matrix $Z \in \IR^{n \times n}$, wobei Spalte $i$ durch den Vektor $z_i$ gegeben ist. Es sei $X = (x_{ij}) = Z^\top Z \in \IR^{n \times n}$. Dann gilt $x_{ij} = z_i \cdot z_j = x_{ji}$ und somit ist $X$ symmetrisch. Mit~\Cref{lem:positiveSemidefiniteTraits} folgt dann, dass $X$ positiv semidefinit ist. Die Matrix $X$ ist somit eine zulässige Lösung für das SDP~\eqref{eq:semidefiniteProgram} und besitzt denselben Zielfunktionswert wie die Lösung $z$ im VP.

    Nun betrachten wir eine Lösung $X = (x_{ij})$ für das SDP. Um hieraus eine Lösung für das VP zu konstruieren, berechne eine Cholesky-Zerlegung von $X$, also eine Matrix $Z$ mit $X = Z^\top Z$. Diese kann allgemein nicht in polynomieller Zeit berechnet werden, da die Einträge von $Z$ irrational sein können. Es ist jedoch möglich, die Einträge in polynomieller Zeit mit einem kleinen, additiven Fehler zu approximieren. Da dieser Fehler beliebig klein gemacht werden kann, gehen wir nicht weiter darauf ein und nehmen an, dass die Zerlegung exakt berechnet wurde. Bezeichnen wir mit $z_i$ idie $i$-te Spalte von $Z$, so ist $z = (z_1, \ldots, z_n)$ eine zulässige Lösung für das VP mit demselben Zielfunktionswert wie die Lösung $X$ des SDP, denn es gilt $x_{ij} = z_i \cdot z_j$.
\end{proof}

SDPs können im Allgemeinen nicht in polynomieller Zeit optimal gelöst werden, da die optimale Lösung irrational sein kann. Ein weiteres Problem ist, dass es SDPs gibt, in denen alle Koeffizienten kleine Zahlen sind, der Wert der optimalen Lösung jedoch doppelt exponentiell von der Anzahl der Variablen abhängt. In den für uns relevanten SDPs tritt dieses Problem jedoch nicht auf, und mit der Ellipsoidmethode erhält man folgenden Satz:

\begin{thm}{Fehlerabschätzung optimale Lösung von SDPs}{optimalSolOfSDPsWithError}
    Die optimale Lösung eines semidefiniten Programms kann für jedes $\varepsilon > 0$ mit einem additiven Fehler von höchstens $\varepsilon$ berechnet werden. Die Laufzeit hierfür ist polynomiell in der Eingabegröße und $\log\left( \frac{R}{\varepsilon} \right)$ beschränkt, sofern die Menge der zulässigen Lösung nichtleer und im Hyperwürfel $\edgys{-R,R}^n$ enthalten ist.
\end{thm}

Mit~\Cref{lem:vectorSemidefiniteEquiv} überträgt sich dies auch auf Vektorprogramme.

\section{Maximum-Cut Problem}

Wir kommen zurück zum Maximum-Cut Problem, und zwar auf seine (relaxierte) Darstellung als Vektorprogramm:
\begin{equation}
    \begin{aligned}
        \max \quad & \frac{1}{2}\sum_{i = 1}^{n-1} \sum_{j = i + 1}^n w_{ij} (1 - z_i \cdot z_j) \\
        s.t. \quad & z_i \cdot z_i = 1, && \forall i \in \set{1, \ldots, n}, \\
        & z_i \in \IR^n, && \forall i \in \set{1,\ldots,n}
    \end{aligned}\label{eq:maxCutVP}
\end{equation}

Mithilfe von Satz~\ref{thm:optimalSolOfSDPsWithError} können wir in polynomieller Zeit die optimale Lösung dieser Relaxierung mit einem beliebg kleinen, addtiven Fehler berechnen. Wir ignorieren diesen additiven Fehler und gehen davon aus, dass wir die optimale Lösung $z = (z_1, \ldots, z_n)$ berechnet haben und bezeichnen mit $W^*$ den Wert dieser Lösung. Da~\eqref{eq:maxCutVP} eine Relaxierung des ganzzahligen quadratischen Problems ist, gilt $W^* \geq \OPT$, wobei $\OPT$ den Wert des optimalen Schnittes berechnet.

Das Ziel ist es nun, die optimale Lösung des VPs zu einer ganzzahligen Lösung zu runden, sodass sich die Zielfunktion nur um einen konstanten Faktor verschlechtert. Das Rundungsverfahren, das wir einsetzen werden, ist randomisiert, und es basiert auf der folgenden Beobachtung: Der Beitrag der Kante zwischen den Knoten $v_i$ und $v_j$ zur optimalen Lösung des VPs beträge 
\begin{equation*}
    \frac{w_{ij}}{2}(1 - z_i \cdot z_j) = \frac{w_{ij}}{2}(1 - \norm{z_i} \norm{z_j} \cos(\theta_{ij})) = \frac{w_{ij}}{2}(1 - \cos(\theta_{ij})),
\end{equation*}
wobei $\theta_{ij}$ den Winkel zwischen den Vektoren $z_i$ und $z_j$ bezeichnet. Der Beitrag steigt in dem relevanten Intervall $\edgys{0, \pi}$ für $\theta_{ij}$ also streng monoton von $0$ (für $\theta_{ij} = 0$) auf $w_{ij}$ (für $\theta_{ij} = \pi$).Damit wir durch das Runden nicht allzu viel verlieren, sollte die Wahrscheinlichkeit, dass $v_i$ und $v_j$ in der gerundeten Lösung durch den Schnitt getrennt werden, mit dem Winkel $\theta_{ij}$ ansteigen: Je größer $\theta_{ij}$ ist, desto größer sollte die Wahrscheinlichkeit sein, dass $v_i$ und $v_j$ voneinander getrennt werden.

Das gewünschte Verfahren erhalten wir auf einfache Art und Weise: Die Vektoren $z_i$ liegen aufgrund der Nebenbedingungen $z_i \cdot z_i = 1$ auf der Einheitssphäre $\IS^{n-1} = \set{x \in \IR^n \mid \norm{x} = 1}$. Wir teilen diese Sphäre nun in zwei Hälften, indem wir eine uniform zufällige Hyperebene durch den Mittelpunkt von $\IS^{n-1}$, also den Koordinatenursprung, legen. Diese Hyperebene partitioniert die Menge der Vektoren $z_i$ und damit auch die Menge der Knoten $v_i$ in zwei Teile:

\begin{algorithm}[h]
    \caption{\textsc{SDPMaxCut}}\label{alg:SDPMaxCut}\index{SDPMaxCut}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:MaxCut}}
        \Output{Wie in~\Cref{prob:MaxCut}}
        \State{Löse das Vektorprogramm~\eqref{eq:maxCutVP}}
        \State{Wähle $r \in \IS^{n-1}$ uniform zufällig}
        \State{Setze $U = \set{v_i \mid z_i \cdot r \geq 0}$}
        \State{\Return{$(U, V \setminus U)$}}
    \end{algorithmic}
\end{algorithm}

Wir müssen uns noch überlegen, wie ein uniform zufälliger Vektor $r \in \IS^{n-1}$ effizient ausgewählt werden. Dazu erzeugen wir zunächst $n$ Zahlen $\hat{r}_1, \ldots, \hat{r}_n$ unabhängig zufällig gemäß einer Standardnormalverteilung. Streng genommen ist dies natürlich nicht in endlicher Zeit nötig, da eine standardnormalverteilte Zahl mit Wahrscheinlichkeit $1$ irrational ist. Bei einer Implementierung muss man sich also mit einer Approximation begnügen und die Zahlen $\hat{r}_1, \ldots, \hat{r}_n$ nach endlich vielen Dezimalstellen abschneiden. Standardnormalverteilte Zufallsvariablen, die nach einer endlichen Anzahl an Stellen abgeschnitten sind, können effizient erzeugt werden. Wir ignorieren im Folgenden den Einfluss dieser Approximation, da er beliebig klein gemacht werden kann, und gehen der Einfachheit halber davon aus, dass es sich bei $\hat{r}_1, \ldots, \hat{r}_n$ um nicht-abgeschnittene standardnormalverteilte Zufallsvariablen handelt. Es sei dann $\hat{r} = (\hat{r}_1, \ldots, \hat{r}_n)$ und $r = \frac{\hat{r}}{\norm{\hat{r}}}$.

\begin{lem}{Uniformverteilung auf $\IS^{n-1}$}{unifDistOnUnitSphere}
    Der Vektor $r$ ist uniform zufällig auf der Sphäre $\IS^{n-1}$ verteilt.
\end{lem}

\begin{proof}
    Aus der Definition der Standardnormalverteilung und der Tatsache, dass die Zufallsvariablen $\hat{r}_1, \ldots, \hat{r}_n$ unanbhängig sind, ergibt sich, dass die Dichte des Vektors $\hat{r}$ an der Stelle $(x_1, \ldots, x_n) \in \IR^n$ 
    \begin{equation*}
        \prod_{i = 1}^n \frac{1}{\sqrt{2\pi}}\exp\left( -\frac{x_i^2}{2} \right) = \frac{1}{(2\pi)^{\frac{n}{2}}} \exp\left( -\frac{1}{2} \sum_{i = 1}^n x_i^2 \right)
    \end{equation*}
    beträgt. Diese Dichte hängt nur davon ab, wie weit der Punkt $(x_1, \ldots, x_n)$ vom Koordinatenursprung entfernt ist, nicht aber von seiner genauen Position. Demzufolge ist die Dichte von $\hat{r}$ radialsymmetrisch. Der Vektor $r = \frac{\hat{r}}{\norm{\hat{r}}}$ hat dieselbe Richtung wie der Vektor $\hat{r}$. Aus der Radialsymmetrie folgt somit das Lemma.
\end{proof}


\newLecture{19.11.2025}

Eine wichtige Eigenschaft von Zufallsvektoren ist, dass Projektionen auf Unterräume wieder normalverteilt sind: Seien $e_1, e_2 \in \IR^n$ zwei orthogonale Einheitsvektoren. Dann sind die Projektionen von $\hat{r}$ auf $e_1$ und $e_2$ unabhängige, standardnormalverteilte Zufallsvariablen. Es bezeichne $r'$ die Projektion von $\hat{r}$ auf die von $e_1$ und $e_2$ aufgespannte Ebene. Da die Projektionen von $\hat{r}$ auf $e_1$ und $e_2$ unabhängige standardnormalverteilte Zufallsvariablen sind, folgt mit~\Cref{lem:unifDistOnUnitSphere}, dass der Vektor $\frac{r'}{\norm{r'}}$ uniform zufällig auf der Sphäre $\IS^1$, also dem Rand des Einheitskreises, in der durch $e_1$ und $e_2$ aufgespannten Ebene verteilt ist.

\begin{lem}{Wahrscheinlichkeit, dass Knoten getrent werden}{probOfNodesSeparation}
    Für jedes Paar $v_i$ und $v_j$ von Knoten gilt 
    \begin{equation*}
        \IP[\text{$v_i$ und $v_j$ werden durch den Schnitt $(U, V \setminus U)$ getrennt}] = \frac{\theta_{ij}}{\pi}.
    \end{equation*}
\end{lem}

\begin{proof}
    Für $z_i = z_j$ beträgt die Wahrscheinlichkeit, dass $v_i$ und $v_j$ getrennt werden, gleich null. Die Aussage ist also wahr.

    Sei nun $z_i \neq z_j$. Wir projizieren den Vektor $\hat{r}$ auf die durch $z_i$ und $z_j$ aufgespannte Ebene. Diese Projektion heiße $r'$. Wie oben ist $\frac{r'}{\norm{r'}}$ ein uniform zufälliger Einheitsvektor in dieser Ebene. Um zu bestimmen, wann $v_i$ und $v_j$ getrennt werden, beachte folgendes: Wir zeichnen zu den Vektoren $z_i$ und $z_j$ jeweils eine orthogonale Gerade durch den Koordinatenursprung. Die zu $z_i$ orthogonale Gerade berühre den Einheitskreis in den Punkten $A$ und $C$, die zu $z_j$ orthogonale Gerade in den Punkten $B$ und $D$. Durch diesse Punkte wird der Einheitskreis in vier Sektoren eingeteilt.
    
    Die Knoten $v_i$ und $v_j$ werden genau dann durch den Schnitt getrennt, wenn der Vektor $\frac{r'}{\norm{r'}}$ in den Kreisvektor zwischen $A$ und $B$ oder zwischen $C$ und $D$ fällt. Wir gehen dabei oBdA davon aus, dass von $z_i$ ausgehend der Abstand zu $z_j$ im Uhrzeigersinn mindestens so groß ist wie gegen den Uhrzeigersinn.

    Sei $r'' = \frac{r'}{\norm{r'}}$. Die Wahrscheinlichkeit, dass $r''$ exakt einen der Punkte $A$, $B$, $C$ oder $D$ trifft, ist $0$. Wir können davon ausgehen, dass dieses Ereignis nicht eintritt.
    \begin{itemize}
        \item Fällt $r''$ in den Kreisvektor $AB$, so gilt $r' \cdot z_i < 0$ und $z' \cdot z_j > 0$.
        \item Fällt $r''$ in den Kreisvektor $BC$, so gilt $r' \cdot z_i < 0$ und $z' \cdot z_j < 0$.
        \item Fällt $r''$ in den Kreisvektor $CD$, so gilt $r' \cdot z_i > 0$ und $z' \cdot z_j < 0$.
        \item Fällt $r''$ in den Kreisvektor $DA$, so gilt $r' \cdot z_i > 0$ und $z' \cdot z_j > 0$.
    \end{itemize}
    Die Knoten $v_i$ und $v_j$ werden durch den Schnitt also genau dann getrennt, wenn $r''$ entweder in Sektor $AB$ oder $CD$ fällt. Beide haben einen Mittelpunktswinkel von $\theta_{ij}$. Da $r''$ uniform zufällig auf dem Einheitskreis verteilt ist, beträgt die Wahrscheinlichkeit, dass es in einem der beiden Kreissektoren fällt, somit $\frac{2\theta_{ij}}{2\pi} = \frac{\theta_{ij}}{\pi}$.
\end{proof}

Wir haben bereits $W^* \geq \OPT$ als den optimalen Wert des VPs eingeführt. Wir bezeichnen nun mit $W$ das Gesamtgewicht des Schnittes, den \textsc{SDPMaxCut} berechnet. Sei außerdem 
\begin{equation}
    \alpha = \frac{2}{\pi} \cdot \min_{0 < \theta \leq \pi} \frac{\theta}{1 - \cos(\theta)}. \label{eq:alphaInVP}
\end{equation}

\begin{lem}{Abschätzung Erwartungswert SDPMaxCut}{expectationSDPMaxCut}
    Es gilt $\IE[W] \geq \alpha W^* \geq \alpha \OPT$.
\end{lem}

\begin{proof}
    Mit Linearität des Erwartungswertes folgt aus~\Cref{lem:probOfNodesSeparation}
    \begin{align}
        \IE[W] &= \sum_{i = 1}^{n-1} \sum_{j = i + 1}^n w_{ij} \cdot \IP[\text{$v_i$ und $v_j$ werden durch den Schnitt $(U, V \setminus U)$ getrennt}] \nonumber \\
        &= \sum_{i = 1}^{n-1} \sum_{j = i + 1}^n w_{ij} \cdot  \frac{\theta_{ij}}{\pi}. \label{eq:expectationSDPMaxCut}
    \end{align}
    Aus der Definition von $\alpha$ folgt 
    \begin{equation*}
        \theta_{ij} \geq \alpha \cdot \frac{\pi}{2}(1 - \cos(\theta_{ij})).
    \end{equation*}
    Einsetzen in~\eqref{eq:expectationSDPMaxCut} liefert 
    \begin{equation*}
        \IE[W] \geq \alpha \cdot \sum_{i = 1}^{n-1} \sum_{j = i + 1}^n \frac{w_{ij}}{2} (1 - \cos(\theta_{ij})) = \alpha W^* \geq \alpha \OPT. \qedhere
    \end{equation*}
\end{proof}

\begin{lem}{Abschätzung $\alpha$}{alphaBound}
    Es gilt $\alpha \geq 0.878$.
\end{lem}

\begin{proof}
    Wir verwenden Kurvendiskussion. Die Funktion $\frac{\theta}{1 - \cos(\theta)}$ nimmt ihr Minimum im Intervall $\edgys{0,\pi}$ an der Stelle $\theta \approx 2.3311$ an. Für diesen Wert von $\theta$ gilt 
    \begin{equation*}
        \frac{2}{\pi} \cdot \frac{\theta}{1 - \cos(\theta)} \geq 0.878.\qedhere
    \end{equation*}
\end{proof}

Damit ergibt sich folgendes Resultat:

\begin{thm}{SDPMaxCut}{sdpMaxCut}
    \textsc{SDPMaxCut} ist ein $0.878$-Approximationsalgorithmus für das Maximum-Cut Problem.
\end{thm}


\section{Max-2SAT}

Wir betrachten nun den Spezialfall von \textsc{Max-SAT}, bei dem jede Klausel nur höchstens zwei Literale enthalten darf. Diesen bezeichnen wir mit \textsc{Max-2SAT}.\index{Max-2SAT}\index{Max-SAT!Max-2SAT}

Man kann zwar in polynomieller Zeit testen, ob eine gegebene Formel eine Belegung der Variablen besitzt, die alle Klauseln erfüllt. Es ist aber NP-schwer, eine Belegung zu finden, die für eine gegebene Formel so viele Klauseln wie möglich erfüllt. Sind die Klauseln gewichtet, ist es ebenfalls NP-schwer, eine Belegung der Variablen zu finden, sodass das Gesamtgewicht der erfüllten Klauseln Maximal wird. In Abschnitt~\ref{subsec:combinationOfBothAlgos} für \textsc{Max-SAT} einen $\frac{3}{4}$-Approximationsalgorithmus kennengelernt. Mit SDP können wir für \textsc{Max-2SAT} aber einen besseren Approximationsfaktor zeigen.

Sei im Folgenden $\varphi$ eine Eingabe für \textsc{Max-2SAT} mit Variablen $x_1, \ldots, x_n$, Klauseln $C_1, \ldots, C_m$ und Gewichten $w_1, \ldots, w_m$. Zunächst formulieren wir ein quadratisches Programm. Dazu führen wir Variablen $y_0, y_1, \ldots, y_n$ ein, die nur die Werte $-1$ oder $+1$ annehmen können. Die Idee ist die folgende: Die Aussagenvariable $x_i$ wird genau dann auf $1$ gesetzt, wenn $y_i = y_0$ gilt. Wir definieren den Wert $v(C_j)$ der Klausel $C_j$ als $1$, wenn die Belegung der Aussagenvariablen, die durch $y_0, y_1, \ldots, y_n$ codiert wird, die Klausel erfüllt, und sonst als $0$. Für Klauseln der Länge $1$ gilt 
\begin{equation*}
    v(x_i) = \frac{1 + y_0 y_i}{2} \quad \text{und} \quad v(\neg x_i) = \frac{1 - y_0 y_i}{2}.
\end{equation*}
Betrachte nun eine Klausel der Länge $2$ mit zwei positiven Literalen $x_i$ und $x_j$. Dann:
\begin{equation*}
    \begin{split}
        v(x_i \vee x_j) &= 1 - v(\neg x_i)v(\neg x_j) = 1 - \frac{1 - y_0 y_i}{2} \cdot \frac{1 - y_0 y_j}{2} \\
        &= 1 - \frac{1 - y_0 y_i - y_0 y_j + y_0^2 y_i y_j}{4} \\
        &= \frac{3}{4} + \frac{y_0 y_i + y_0 y_j - y_i y_j}{4} \\
        &= \frac{1 + y0 y_i}{4} + \frac{1 + y_0 y_j}{4} + \frac{1 - y_i y_j}{4},
    \end{split}
\end{equation*}
da $y_0^2 = 1$. Somit können wir den Wert $v(x_i \vee x_j)$ als Linearkombination von Termen der Form $(1 + y_a y_b)$ und $(1 - y_a y_b)$ für $a, b \in \set{0,1\ldots,n}$ darstellen. Das gilt auch, wenn die Klausel der Länge zwei mindestens ein negatives Literal enthält.

Die Zielfunktion, die wir maximieren möchten, ist $\sum_{j = 1}^m w_j v(C_j)$. Da sich die Werte aller Klauseln als Linearkombination von Termen der Form $(1 + y_a y_b)$ und $(1 - y_a y_b)$ für $a, b \in \set{0,1\ldots,n}$ darstellen lassen, können wir geeignete Koeffizienten $a_{ij}$ und $b_{ij}$ finden, sodass 
\begin{equation*}
    \sum_{j = 1}^m w_j v(C_j) = \sum_{i = 0}^{n-1} \sum_{j = i + 1}^n (a_{ij} (1 + y_i y_j) + b_{ij} (1 - y_i y_j))
\end{equation*}
gilt. Wir erhalten damit das quadratische Programm 
\begin{align*}
    \max \quad & \sum_{i = 0}^{n-1} \sum_{j = i + 1}^n (a_{ij} (1 + y_i y_j) + b_{ij} (1 - y_i y_j)) \\
    s.t. \quad & y_i^2 = 1, && \forall i \in \set{0,1,\ldots,n}, \\
    & y_i \in \IZ, && \forall i \in \set{0,1,\ldots,n}.
\end{align*}

Auch hier betrachten wir wieder die Relaxierung als VP:
\begin{equation}
    \begin{aligned}
        \max \quad & \sum_{i = 0}^{n-1} \sum_{j = i + 1}^n (a_{ij} (1 + z_i \cdot z_j) + b_{ij} (1 - z_i \cdot z_j)) \\
        s.t. \quad & z_i \cdot z_i = 1, && \forall i \in \set{0,1,\ldots,n}, \\
        & z_i \in \IR^{n+1}, && \forall i \in \set{0,1,\ldots,n}.
    \end{aligned}\label{eq:max2SATVP}
\end{equation}

Der Approximationsalgorithmus \textsc{SDP2SAT}\index{SDP2SAT}, den wir für \textsc{Max-2SAT} betrachten, arbeitet vollkommen analog zu \textsc{SDPMaxCut}:
\begin{algorithm}[h]
    \caption{\textsc{SDP2SAT}}\label{alg:SDP2SAT}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:MaxSAT}, alle Klauseln haben jedoch Länge $2$}
        \Output{Wie in~\Cref{prob:MaxSAT}}
        \State{Löse das Vektorprogramm~\eqref{eq:max2SATVP}. Die optimale Lösung sei $W^*$}
        \State{Wähle $r \in \IS^{n}$ uniform zufällig}
        \For{$i \in \set{0,\ldots,n}$} 
            \If{$r \cdot z_i \geq 0$} Setze $y_i = +1$
            \Else{} Setze $y_i = -1$
            \EndIf{}
        \EndFor{}
        \State{\Return{$W$ als Gesamtgewicht aller erfüllten Klauseln}}
    \end{algorithmic}
\end{algorithm}

\begin{lem}{Abschätzung Erwartungswert SDP2SAT}{expectationSDP2SAT}
    Es gilt $\IE[W] \geq \alpha W^*$ für den Wert $\alpha$ aus~\eqref{eq:alphaInVP}.
\end{lem}
\begin{proof}
    Mit Lineraität des Erwartungswert gilt 
    \begin{equation*}
        \IE[W] = 2\sum_{i = 0}^{n-1} \sum_{j = i + 1}^n (a_{ij} \cdot \IP[y_i = y_j] + b_{ij} \cdot \IP[y_i \neq y_j]).
    \end{equation*}
    Wir bezeichnen wieder mit $\theta_{ij}$ den Winkel zwischen $z_i$ und $z_j$. Dann folgt analog zur Analyse von \textsc{SDPMaxCut} für alle $i, j \in \set{0,\ldots,n}$
    \begin{equation*}
        \IP[y_i \neq y_j] = \frac{\theta_{ij}}{\pi} \geq \frac{\alpha}{2}(1 - \cos(\theta_{ij})).
    \end{equation*}
    Ebenfalls gilt
    \begin{equation*}
        \IP[y_i = y_j] = 1 - \frac{\theta_{ij}}{\pi} \geq \frac{\alpha}{2}(1 + \cos(\theta_{ij})).
    \end{equation*}
    Nutzen wir genau wie bei der \textsc{SDPMaxCut}-Analyse aus, dass $z_i \cdot z_j = \cos(\theta_{ij})$ gilt, so erhalten wir 
    \begin{equation*}
        \IE[W] \geq \alpha \sum_{i = 0}^{n-1} \sum_{j = i + 1}^n (a_{ij} \cdot (1 + \cos(\theta_{ij})) + b_{ij} \cdot (1 - \cos(\theta_{ij}))) = \alpha W^*. \qedhere
    \end{equation*}
\end{proof}

Zusammengefasst ergibt sich also Folgendes:

\begin{thm}{SDP2SAT}{sdp2SAT}
    \textsc{SDP2SAT} ist ein $0.878$-Approximationsalgorithmus für \textsc{Max-2SAT}.
\end{thm}

\section{Duale Lineare Programme}

Wir betrachten folgendes LP: 
\begin{equation}
    \begin{matrix}
        \min & f(x) = & 4x_1 & + & 7x_2 & + & 2x_3 \\
        s.t. &  \quad \textcolor{red}{(y_1)} \quad & 3x_1 & + & 7x_2 & + & x_3 & \geq & 10 \\
        & \quad \textcolor{red}{(y_2)} \quad & 2x_1 & + & 5x_2 & + & 3x_3 & \geq & 6 \\
        & \quad \textcolor{red}{(y_3)} \quad & 2x_1 & + &  x_2 & - &  x_3 & \geq & 5 \\
        & \quad \textcolor{red}{(y_4)} \quad & 4x_1 & + &  x_2 & + & 3x_3 & \geq & 20 \\
        & \quad \textcolor{red}{(y_5)} \quad & 4x_1 & + & 13x_2& + &  x_3 & \geq & 4\\
        & \quad \textcolor{red}{(y_6)} \quad & x_1, &  & x_2, & & x_3 & \geq & 0
    \end{matrix}\label{eq:primalExpl}
\end{equation}

Wir suchen eine Möglichkeit, die Zielfunktion nach unten abzuschätzen. Beispielsweise gilt 
\begin{equation*}
    \begin{split}
        f(x) &= 4x_1 + 7x_2 + 2x_3 \\
        &\geq 3x_1 + 7x_2 + x_3 \geq 10,
    \end{split}
\end{equation*}
da alle $x_i$ nichtnegativ sind. Somit hat die optimale Lösung keinen kleineren Wert als $10$. Wir erhalten durch Addition der zweiten und dritten Schranke eine bessere Abschätzung:

\begin{equation*}
    \begin{split}
        f(x) &= 4x_1 + 7x_2 + 2x_3 \\
        &\geq 4x_1 + 6x_2 + 2x_3 \\
        &= (2x_1 + 5x_2 + 3x_3) + (2x_1 + 5x_2 - x_3) \\
        &\geq 6 + 5 = 11.
    \end{split}
\end{equation*}

Eine bessere Schranke erhalten wir durch Gewichtung der dritten und vierten Ungleichung jeweils mit $\frac{1}{2}$:

\begin{equation*}
    \begin{split}
        f(x) &= 4x_1 + 7x_2 + 2x_3 \\
        &\geq \frac{1}{2}(2x_1 + x_2 - x_3) + \frac{1}{2}(4x_1 + x_2 + 3x_3) \\
        &\geq \frac{1}{2}(5 + 20) = 12.5
    \end{split}
\end{equation*}

Sei $x^*$ eine optimale Lösung. \emph{Ziel:} Finde eine untere Schranke für $f(x^*)$. Wir setzen nun Variablen $y_i$ ein (eine pro Gleichung), die alle nichtnegativ sein dürfen und jeweils darstellen, wie oft wir eine Zeile mitnehmen (siene~\eqref{eq:primalExpl}). Damit erhalten wir folgendes \emph{Dualsystem}\index{Duales LP}\index{Lineares Programm (LP)!Duales System}:

\begin{equation}
    \begin{matrix}
        \max & & 10y_1 & + & 6y_2 & + & 5y_3 & + & 20y_4 & + & 4y_5 \\
        s.t. & & 3y_1 & + & 2y_2 & + & 2y_3 & + & 4y_4 & + & 4y_5 & \leq & 4 \\
        &  & 7y_1 & + & 5y_2 & + &  y_3 & + &  y_4 & + &13y_5 & \leq & 7 \\
        &  &  y_1 & + & 3y_2 & - &  y_3 & + & 3y_4 & + &  y_5 & \leq & 2 \\
        &  & 3y_1 & + & 2y_2 & + & 2y_3 & + & 4y_4 & + & 4y_5 & \leq & 4 \\
        &  & y_1, &  & y_2, & & y_3, & & y_4, & & y_5  & \geq & 0
    \end{matrix}\label{eq:dualExpl}
\end{equation}

Wir lösen~\eqref{eq:dualExpl} und erhalten $y^* = \left( 0, 0, \frac{2}{5}, \frac{4}{5}, 0 \right)$ und erhalten durch Gewichtung der Gleichungen tatsächlich $f(x) \geq \frac{2}{5} \cdot 5 + \frac{4}{5} \cdot 20 = 18$.

Wir verallgemeinern dies nun auf allgemeine lineare Programme. Wir können LPs schreiben als 
\begin{equation}
    \begin{aligned}
        \min \quad & f(x) = \sum_{j = 1}^n c_j x_j \\ 
        s.t. \quad & \sum_{j = 1}^n a_{ij} x_j \geq b_i, & & \forall i \in \set{1,\ldots,m}, \\
        & x_j \geq 0, && \forall j \in \set{1,\ldots,n}.
    \end{aligned}\label{eq:primalGeneral}
\end{equation}

Wie oben erhalten wir im dualen LP eine Nebenbedingung pro Variable und eine Variable pro Nebenbedingung. Wir erhalten als Duales von~\eqref{eq:primalGeneral}
\begin{equation}
    \begin{aligned}
        \max \quad & g(y) = \sum_{i = 1}^m b_i y_i \\ 
        s.t. \quad & \sum_{i = 1}^1 a_{ij} y_i \geq c_j, & & \forall j \in \set{1,\ldots,n}, \\
        & y_i \geq 0, && \forall i \in \set{1,\ldots,m}.
    \end{aligned}\label{eq:dualGeneral}
\end{equation}

\begin{thm}{Schwache Dualität}{weakDuality}
    Sei $x = (x_1, \ldots, x_n)$ eine Lösung für das primale LP~\eqref{eq:primalGeneral} und $y = (y_1, \ldots, y_n)$ eine Lösung für das duale LP~\eqref{eq:dualGeneral}. Dann gilt $f(x) \geq g(y)$.  
\end{thm}

\begin{proof}
    Es gilt 
    \begin{equation}
        f(x) = \sum_{j = 1}^n c_j x_j \geq \sum_{j = 1}^n \left( \sum_{i = 1}^m a_{ij} y_i \right) x_j = \sum_{i = 1}^m \left( \sum_{j = 1}^m a_{ij} x_j \right) y_i \geq \sum_{i = 1}^m b_i y_i = g(y),
    \end{equation}
    wobei die Ungleichungen folgen, da $y$ eine Lösung für~\eqref{eq:dualGeneral} und $x$ eine Lösung für~\eqref{eq:primalGeneral} ist.
\end{proof}

Für die Analyse von Approximationsalgorithmen reicht uns tatsächlich schon Satz~\ref{thm:weakDuality}. Wir können aber auch noch die folgende stärkere Aussage formulieren:

\begin{thm}{Starke Dualität}{strongDuality}
    Das primale LP~\eqref{eq:primalGeneral} hat genau dann einen beschränkten, optimalen Wert wenn dies für das duale LP~\eqref{eq:dualGeneral} der Fall ist.

    Sei $x^*$ eine optimale Lösung für das primale LP und $y^*$ eine optimale Lösung für das duale LP. Dann gilt $f(x^*) = g(y^*)$.
\end{thm}

\section{Set Cover}
Wir erinnern uns zurück an Set Cover aus Abschnitt~\ref{sec:SetCover}. 

\begin{prob}{Set Cover (Wiederholung)}{SetCoverRev}
    \prblmObj{Menge $S$ mit $n$ Elementen, $n$ Teilmengen $S_1, \ldots, S_m \subseteq S$ mit $\bigcup_{i=1}^m S_i = S$, Kosten $c_1, \ldots, c_m \in \IN$.}{Alle $A \subseteq \set{1, \ldots, m}$ mit $\bigcup_{i \in A} S_i = S$}{Minimiere $\sum_{i \in A} c_i$}\index{Set Cover}
\end{prob}

Formuliert als ILP erhalten wir
\begin{equation*}
    \begin{aligned}
        \min \quad & \sum_{j = 1}^m w_j x_j \\
        s.t. \quad & \sum_{j: i \in S_j} x_j \geq 1, && \forall i \in \set{1,\ldots,n}, \\
                   & x_j \in \set{0,1}, && \forall j \in \set{1,\ldots,m},
    \end{aligned}
\end{equation*}

bzw. relaxiert
\begin{equation}
    \begin{aligned}
        \min \quad & \sum_{j = 1}^m w_j x_j \\
        s.t. \quad & \sum_{j: i \in S_j} x_j \geq 1, && \forall i \in \set{1,\ldots,n}, \\
                   & x_j \geq 0, && \forall j \in \set{1,\ldots,m}.
    \end{aligned}\label{eq:setCoverPrimal}
\end{equation}

Dualisieren ergibt 
\begin{equation}
    \begin{aligned}
        \max \quad & \sum_{i = 1}^n y_i \\
        s.t. \quad & \sum_{i \in S_j} y_i \leq w_j, && \forall j \in \set{1,\ldots,m}, \\
                   & y_i \geq 0, && \forall i \in \set{1,\ldots,n}.
    \end{aligned}\label{eq:setCoverDual}
\end{equation}

\begin{algorithm}
    \caption{\textsc{PrimalDual-SC}}\label{alg:primalDualSC}\index{PrimalDual-SC}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:SetCoverRev}}
        \Output{Wie in~\Cref{prob:SetCoverRev}}
        \State{$y := 0$} \Comment{Gültige Lösung für~\eqref{eq:setCoverDual}}
        \State{$I := \emptyset$}
        \While{$\exists~ i \notin \bigcup_{j \in I} S_j$}
            \State{Erhöhe $y_i$ solange, bis für ein $j$ mit $i \in S_j$ gilt $\sum_{\ell \in S_j} y_\ell = w_\ell$}
            \State{$I := I \cup \set{j}$}
        \EndWhile{}
        \State{\Return{$I$}}
    \end{algorithmic}
\end{algorithm}

\begin{thm}{PrimalDual-SC}{primalDual-SC}
    Sei $f = \max_{i} \abs{\set{j \mid i \in S_j}}$ die maximale Anzahl an Vorkommen eines Elements in den Teilmengen. \textsc{PrimalDual-SC} ist ein $f$-Approximationsalgorithmus für Set Cover.
\end{thm}
\begin{proof}
    Sei $\OPT$ der Wert eines optimalen Set Covers und $Z^*$ der optimale Wert des primalen LP. Dann gilt $Z^* \leq \OPT$.

    Da $y$ eine gültige Lösung im dualen LP ist, gilt 
    \begin{equation*}
        \sum_{i = 1}^n y_i \leq Z^* \leq \OPT.
    \end{equation*}
    Damit genügt es zu zeigen, dass $\sum_{j \in I} w_j \leq f \cdot \sum_{i = 1}^n y_i$. Es gilt
    \begin{equation*}
        \sum_{j \in I} w_j = \sum_{j \in I} \left( \sum_{i \in S_j} y_i \right) = \sum_{i \in S} y_i \cdot \abs{\set{j \in I \mid i \in S_j}} \leq f \cdot \sum_{i \in S} y_i \leq f \cdot \OPT,
    \end{equation*}
    da $j$ nur in $I$ aufgenommen wird, wenn $w_j$ mit Gleichheit erreicht wird.
\end{proof}

Algorithmen, die nach dem Schema von \textsc{PrimalDual-SC} ablaufen, nennt man \emph{primal-duale Algorithmen}\index{primal-dualer Algorithmus}. Diese haben gemeinsam, dass sie stets eine gültige Lösung des dualen LPs mitführen und dabei eine (im Allgemeinen nicht optimale) ganzzahlige Lösung für das primale LP konstruieren. Dabei werden die Lösungen für das primale und duale LP vom Algorithmus explizit konstruiert, ohne auf allgemeine Algorithmen zur Lösung von linearen Programmen zurückzugreifen. Die Dualitätstheorie ist wesentlich für die Analyse der Algorithmen.

\section{Feedback Vertex Set}
\begin{prob}{Feedback Vertex Set}{feedbackVertexSet}\index{Feedback Vertex Set}
    \prblmObj{ungerichteter Graph $G = (V, E)$, Gewichte $w: V \to \IR_{> 0}$}{$S \subseteq V$, sodass $G[V \setminus S]$ kreisfrei ist}{Minimiere $\sum_{v \in S} w(v)$}
\end{prob}

Wir modellieren dies als ILP, wobei $\cC$ die Menge aller Kreise in $G$ sei. Dabei sei $w_v = w(v)$ für alle $v \in V$.
\begin{equation*}
    \begin{aligned}
        \min \quad & \sum_{v \in V} w_v x_v \\
        s.t. \quad & \sum_{v \in C} x_v \geq 1, && \forall C \in \cC, \\
                   & x_v \in \set{0,1}, && \forall v \in V,
    \end{aligned}
\end{equation*}

bzw. relaxiert
\begin{equation}
    \begin{aligned}
        \min \quad & \sum_{v \in V} w_v x_v \\
        s.t. \quad & \sum_{v \in C} x_v \geq 1, && \forall C \in \cC, \\
                   & x_v \geq 0, && \forall v \in V.
    \end{aligned}\label{eq:FVSPrimal}
\end{equation}

Es könnte sein, dass wir exponentiell viele Nebenbedingungen haben. Das ist letztlich aber egal, siehe morgen!

\newLecture{26.11.2025}

Duailsiert erhalten wir 
\begin{equation}
    \begin{aligned}
        \max \quad & \sum_{C \in \cC} y_C \\
        s.t. \quad & \sum_{C: v \in C} y_C \leq w(v), && \forall v \in V, \\
                   & y_C \geq 0, && \forall C \in \cC.
    \end{aligned}\label{eq:FVSDual}
\end{equation}
Sei wieder $\OPT$ der Wert einer optimalen Lösung für die gegebene Instanz von Feedback Vertex Set, $Z^*$ ein optimaler Wert des dualen LP~\eqref{eq:FVSDual}. Sei $y$ eine zulässige Lösung für das duale LP. Dann gilt wieder 
\begin{equation*}
    \OPT \geq Z^* \geq \sum_{C \in \cC} y_C.
\end{equation*}

\begin{algorithm}[h]
    \caption{\textsc{PrimalDual-FVS}}\label{alg:primalDualFVS}\index{PrimalDual-FVS}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:feedbackVertexSet}}
        \Output{Wie in~\Cref{prob:feedbackVertexSet}}
        \State{$y := 0$} \Comment{Lösung von~\eqref{eq:FVSDual}}
        \State{$S := \emptyset$}
        \While{$\exists$ Kreis $C^*$ in $G$}
            \State{Entferne rekursiv alle Knoten mit Grad $1$}
            \State{Erhöhe $y_{C^*}$, bis für ein $v \in C^*$ gilt, dass $\sum_{C: v \in C} y_C = w_v$}
            \State{$S := S \cup \set{v}$}
            \State{Entferne Knoten $v$ und zu $v$ inzidente Kanten aus $G$}
        \EndWhile{}
        \State{\Return{$S$}}
    \end{algorithmic}
\end{algorithm}

Es gilt 
\begin{equation*}
    \sum_{v \in S} w_v = \sum_{v \in S} \left( \sum_{C : v \in C} y_C \right) = \sum_{C \in \cC} y_C \cdot \abs{S \cap C}
\end{equation*}
denn für jeden Knoten, der in $S$ und $C$ liegt, wird $y_C$ einmal aufaddiert.

Angenommen, es ist $\abs{S \cap C} \leq \alpha$ für ein $\alpha$. Dann wäre $\sum_{v \in S} w_v \leq \alpha \sum_{c \in \cC} y_C \leq \alpha \cdot \OPT$. Leider ist das aber a priori nicht klar. Wir modifizieren bzw. konkretisieren, \emph{wie} wir $C^*$ auswählen:

\begin{algorithm}[h]
    \caption{\textsc{PrimalDual-FVS2}}\label{alg:primalDualFVS2}\index{PrimalDual-FVS2}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:feedbackVertexSet}}
        \Output{Wie in~\Cref{prob:feedbackVertexSet}}
        \State{$y := 0$} \Comment{Lösung von~\eqref{eq:FVSDual}}
        \State{$S := \emptyset$}
        \While{$\exists$ Kreis $C^*$ in $G$}
            \State{Entferne rekursiv alle Knoten mit Grad $1$}
            \State{Finde einen Kreis $C^*$ mit höchstens $2 \cdot \ceil*{\log_2(n)}$ Knoten mit Grad $\geq 3$.}
            \State{Erhöhe $y_{C^*}$, bis für ein $v \in C^*$ gilt, dass $\sum_{C: v \in C} y_C = w_v$}
            \State{$S := S \cup \set{v}$}
            \State{Entferne Knoten $v$ und zu $v$ inzidente Kanten aus $G$}
        \EndWhile{}
        \State{\Return{$S$}}
    \end{algorithmic}
\end{algorithm}

Hier bleibt die Frage, warum dieser Kreis existieren muss.

\begin{lem}{Graph hat Kreis mit $\leq2\ceil*{\log_2(n)}$ Knoten mit Grad $\geq 3$}{graphHasCycleWithFewBigNodes}
    In jedem Graphen mit $n$ Knoten, der keine Knoten mit Grad $1$, aber mindestens eine Kante hat, existiert ein Kreis mit höchstens $2\cdot \ceil*{\log_2(n)}$ Knoten mit Grad $\geq 3$.
\end{lem}
\begin{proof}
    Zunächste entfernen wir aus $G$ alle Knoten mit Grad $2$, inden wir nach und nach jeden Knoten mit Grad $2$ durch eine direkte Kante zwischen seinen beiden Nachbarknoten ersetzen. Es bezeichne $G'$ den resultirenden Graphen. Jeder Kreis in $G'$ entspricht einem Kreis in $G$.

    Wir machen Breitensuche in $G'$ beginnend bei einem beliebigen Knoten. Solange kein Knoten in der Breitensuche mehrfach besucht wird, gibt es auf Level $i \geq 1$ mindetsens $3 \cdot 2^{i-1}$ Knoten, insgesamt also 
    \begin{equation*}
        1 + \sum_{i = 1}^k 3 \cdot 2^{i-1} = 1 + 3 \cdot (2^k - 1) = 3 \cdot 2^k - 2.
    \end{equation*}
    Für $K = \ceil*{\log_2\left( \frac{n+2}{3} \right)}$ gilt $3 \cdot 2^K - 2 \geq n$. Damit ergibt sich ein Kreis der Länge $2 K + 1$, da die BFS spätestens auf Tiefe $K$ den ersten Kreis liefert. In $G$ entspricht das einem Kreis mit höchstens $2K + 1$ Knoten mit Grad $\geq 3$. Es gilt
    \begin{equation*}
        2 \cdot \ceil*{\log_2\left( \frac{n+2}{3} \right)} + 1 \leq 2 \cdot \ceil*{\log_2(n)}. \qedhere
    \end{equation*}
\end{proof}

\emph{Ziel:} Beschränke $\abs{S \cap C}$ für alle Kreise $C$ mit $y_C > 0$.

Knoten mit Grad $\geq 3$ teilen $C$ in maximal $2\ceil*{\log_2(n)}$ Segmente ein, wobei jedes Segment nur Grad-$2$-Knoten enthält. Folglich kann $S$ höchstens alle Grad-$\geq 3$-Knoten von $C$ und pro Segment einen weiteren Knoten enthalten. Damit folgt $\abs{S \cap C} \leq 4\ceil*{\log_2(n)}$ und wir erhalten
\begin{equation*}
    \sum_{v \in S} w_v = \sum_{v \in S}\left( \sum_{C: v \in C} y_C \right) = \sum_{C \in \cC} y_C \cdot \abs{S \cap C} \leq 4 \cdot \ceil*{\log_2(n)} \cdot \sum_{C \in \cC} y_C \leq 4 \cdot \ceil*{\log_2(n)} \cdot \OPT.
\end{equation*}
Wir erhalten zusammengefasst folgenden Satz:
\begin{thm}{PrimalDual-FVS2}{primalDualFVS2}
    \textsc{PrimalDual-FVS2} ist ein $4\ceil*{\log_2(n)}$-Approximationsalgorithmus für Feedback Vertex Set.
\end{thm}

\section{Das Kürzeste-Wege-Problem}\label{sec:shortestPath}

\begin{prob}{Kürzeste-Wege-Problem}{shortestPath}\index{Kürzeste-Wege-Problem}
    \prblmObj{Ungerichteter Graph $G = (V, E)$, Kantengewichte $w: E \to \IR_{> 0}$, Startknoten $s$, Zielknoten $t$}{$s$-$t$-Pfad in $G$}{Minimiere $\sum_{e \in P} w(e)$}
\end{prob}

Wir formulieren dies wieder als ILP (für $w_e = w(e)$ und $\cS = \set{S \subseteq V \mid s \in S, t \notin S}$ als Menge der $s$-$t$-Schnitte, mit $\delta(S) = \set{\set{x,y} \in E \mid \abs{\set{x,y} \cap S} = 1}$ für $S \in \cS$)
\begin{equation}
    \begin{aligned}
        \min \quad & \sum_{e \in E} w_e x_e \\
        s.t. \quad & \sum_{e \in \delta(S)} x_e \geq 1, && \forall S \in \cS \\
        & x_e \in \set{0,1}, && \forall e \in E,
    \end{aligned}\label{eq:shortestPathPrimalUnrelaxed}
\end{equation}

bzw. relaxiert 
\begin{equation}
    \begin{aligned}
        \min \quad & \sum_{e \in E} w_e x_e \\
        s.t. \quad & \sum_{e \in \delta(S)} x_e \geq 1, && \forall S \in \cS \\
        & x_e \geq 0, && \forall e \in E.
    \end{aligned}\label{eq:shortestPathPrimal}
\end{equation}

\newLecture{02.12.2025}

Sei $P$ ein $s$-$t$-Weg. Setze $x_e = \1_{\set{e \in P}}(e)$ für alle $e \in E$. Dann ist $x$ eine gültige Lösung von~\eqref{eq:shortestPathPrimalUnrelaxed}. 

Sei andersrum $x$ eine zulässige Lösung von~\eqref{eq:shortestPathPrimalUnrelaxed}. Dann codiert $x$ eine Knotenauswahl, die einen $s$-$t$-Weg enthält.

Dualisieren von~\eqref{eq:shortestPathPrimal} ergibt 
\begin{equation}
    \begin{aligned}
        \max \quad & \sum_{S \in \cS} y_S \\
        s.t. \quad & \sum_{S: e \in \delta(S)} y_S \geq w_e, && \forall e \in E \\
        & y_S \geq 0, && \forall S \in \cS.
    \end{aligned}\label{eq:shortestPathDual}
\end{equation}

\begin{algorithm}[h]
    \caption{\textsc{PrimalDual-ShortestPath}}\label{alg:primalDualShortestPath}\index{PrimalDual-ShortestPath}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:shortestPath}}
        \Output{Wie in~\Cref{prob:shortestPath}}
        \State{$y := 0$} \Comment{Lösung von~\eqref{eq:shortestPathDual}}
        \State{$F := \emptyset$}
        \While{$\not\exists$ $s$-$t$-Weg in $(V, F)$}
            \State{$S := \set{v \in V \mid \text{$\exists$ $s$-$v$-Weg in $(V, F)$}}$}
            \State{Erhöhe $y_S$ bis für ein $e \in \delta(S)$ gilt $\sum_{S': e \in \delta(S)} y_{S'} = w_e$}
            \State{$F := F \cup \set{e}$}
        \EndWhile{}
        \State{Sei $P$ ein beliebiger $s$-$t$-Weg in $(V, F)$}
        \State{\Return{$P$}}
    \end{algorithmic}    
\end{algorithm}

\begin{lem}{PrimalDual-ShortestPath wählt Baum aus}{primalDualSPSelectsTree}
    Zu jedem Zeitpunkt gilt, dass $F$ einen Baum auf den von $s$ in $(V, F)$ erreichbaren Knoten bildet.
\end{lem}

\begin{proof}
    Zu Beginn ist $F = \emptyset$ und die Aussage ist offensichtlich wahr.

    Nun werde $e = \set{x,y}$ der Menge $F$ hinzugefügt. Es ist also $e \in \delta(S)$ für die Menge $S$, der von $s$ aus erreichbaren Knoten in $(V, F)$. Sei $x \in S$ und $y \notin S$. Da $y$ von $s$ aus über $F$ nicht erreichbar ist, kann die Hinzunahme von $e$ nicht zu einem Kreis führen (da der Graph ungerichtet ist). Damit ist die Invariante gezeigt.
\end{proof}

\begin{thm}{PrimalDual-ShortestPath}{primalDualSP}
    \textsc{PrimalDual-ShortestPath} ist ein $1$-Approximationsalgorithmus für das Kürzeste-Wege-Problem, also wird stets ein kürzester Weg ausgegeben.
\end{thm}

\begin{proof}
    Es gilt 
    \begin{equation*}
        \sum_{e \in P} w_e = \sum_{e \in P} \sum_{S: e \in \delta(S)} y_S = \sum_{S \in \cS} y_S \cdot \abs{P \cap \delta(S)}. 
    \end{equation*}
    Es bleibt also zu zeigen, dass $\abs{P \cap \delta(S)} \leq 1$ für alle $S$ mit $y_S > 0$, durch schwache Dualität (Satz~\ref{thm:weakDuality}) folgt damit nämlich
    \begin{equation*}
        \sum_{e \in P} w_e \leq \sum_{e \in P} y_S \leq \OPT.
    \end{equation*}
    Angenommen, es gibt einen Schnitt $S \in \cS$ mit $y_S > 0$ und $\abs{P \cap \delta(S)} \geq 2$. Dann gilt sogar $\abs{P \cap \delta(S)} \geq 3$, da der Schnitt immer ungerade oft gekreuzt wird. Sei $w$ der erste Knoten, den wir beim Wiedereintritt in $S$ erreichen. Da zu jedem Zeitpunkt die Kanten $F$ einen Baum bildet, und $w \in S$, gibt es einen Weg von $s$ zu $w$ innerhalb von $S$. Damit gibt es also einen Kreis. Da nach~\ref{lem:primalDualSPSelectsTree} $(V, F)$ aber kreisfrei ist, folgt der Widerspruch. 
\end{proof}

FunFact: \textsc{PrimalDual-ShortestPath} ist literally \textsc{Dijkstra}.

\section{Steinerwaldproblem}

\begin{prob}{Steinerwaldproblem}{steinerForest}\index{Steinerwaldproblem}
    \prblmObj{Ungerichteter Graph $G = (V, E)$ mit Kantengewichten $w: E \to \IR_{> 0}$, Knotenpaare $(s_i, t_i)$ für $i \in \set{1,\ldots,k}$ des Graphen}{Kantenauswahl $F \subseteq E$, sodass für alle $i$ ein $s_i$-$t_i$-Weg in $F$ existiert}{Minimiere $\sum_{e \in F} w(e)$}
\end{prob}

Wir modellieren das Steinerwaldproblem nun wieder mit Linearen Programmen. Es geht sehr analog zu Abschnitt~\ref{sec:shortestPath}. Wir setzen $\cS_i := \set{S \subseteq V \mid \abs{S \cap \set{s_i, t_i}} = 1}$ und $\cS = \cS_1 \cup \dots \cup \cS_k$. Dann können wir die LPs übernehmen und erhalten

\begin{equation*}
    \begin{aligned}
        \min \quad & \sum_{e \in E} w_e x_e \\
        s.t. \quad & \sum_{e \in \delta(S)} x_e \geq 1, && \forall S \in \cS \\
        & x_e \in \set{0,1}, && \forall e \in E,
    \end{aligned}
\end{equation*}

bzw. relaxiert 
\begin{equation}
    \begin{aligned}
        \min \quad & \sum_{e \in E} w_e x_e \\
        s.t. \quad & \sum_{e \in \delta(S)} x_e \geq 1, && \forall S \in \cS \\
        & x_e \geq 0, && \forall e \in E.
    \end{aligned}\label{eq:steinerForestPrimal}
\end{equation}

Dualisieren ergibt wieder

\begin{equation}
    \begin{aligned}
        \max \quad & \sum_{S \in \cS} y_S \\
        s.t. \quad & \sum_{S: e \in \delta(S)} y_S \geq w_e, && \forall e \in E \\
        & y_S \geq 0, && \forall S \in \cS.
    \end{aligned}\label{eq:steinerForestDual}
\end{equation}

Wir werden versuchen, einen sehr analogen Algorithmus zu \textsc{PrimalDual-ShortestPath} aufzustellen:

\begin{algorithm}[h]
    \caption{\textsc{PrimalDual-SteinerForest1}}\label{alg:primalDualSteinerForest1}\index{PrimalDual-SteinerForest1}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:steinerForest}}
        \Output{Wie in~\Cref{prob:steinerForest}}
        \State{$y := 0$} \Comment{Lösung von~\eqref{eq:steinerForestDual}}
        \State{$F := \emptyset$}
        \While{$\exists~i:~\not\exists~s_i$-$t_i$-Weg in $(V, F)$}
            \State{Sei $S$ eine Zusammenhangskomponente von $(V, F)$ mit $\abs{S \cap \set{s_i, t_i}} = 1$}
            \State{Erhöhe $y_S$ bis für ein $e \in \delta(S)$ gilt $\sum_{S': e \in \delta(S)} y_{S'} = w_e$}
            \State{$F := F \cup \set{e}$}
        \EndWhile{}
        \State{\Return{$F$}}
    \end{algorithmic}    
\end{algorithm}

Es gilt wieder 
\begin{equation*}
    \sum_{e \in F} w_e = \sum_{e \in F} \sum_{S: e \in \delta(S)} y_S = \sum_{S \in \cS} y_S \cdot \abs{F \cap \delta(S)}.
\end{equation*}

Betrachte folgendes Beispiel: Wir haben eine $(k+1)$-Clique, wobei $s_1 = \dots = s_k = v$ und die $t_i$ alle verschieden sind und alle Kanten die Länge $1$ haben. Sei $S = \set{v}$ der erste Schnitt. Nun wird $y_S$ erhöht, bis $y_S = 1$ ist. Dann sind plötzlich alle Knoten erreichbar, und wir fügen eine dieser Kanten zu $F$ hinzu. Nun wird wieder eine Zusammenhangskomponente ausgesucht, und eine weitere duale Variable erhöht. Es kann aber tatsächlich für keine weitere Menge $S' \in \cS$ die Variable $y_{S'}$ noch um mehr als $0$ erhöht werden. Damit ist für $S = \set{v}$ am Ende $\abs{F \cap \delta(S)} = k$. Durch diese Analyse bekommen wir also keinen besseren Approximationsfaktor als durch die triviale Betrachtung, einfach alle $k$ kürzesten Wege zu wählen.

Auch ein konkreteres Wählen der Zusammenhangskomponente $S$ funktioniert nicht trivial sinnvoll. Wir brauchen einen anderen Ansatz, und erhöhen die dualen Variablen aller Zusammenhangskomponenten parallel gleichzeitig:

\newLecture{09.12.2025}


\begin{algorithm}[h]
    \caption{\textsc{PrimalDual-SteinerForest2}}\label{alg:primalDualSteinerForest2}\index{PrimalDual-SteinerForest2}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:steinerForest}}
        \Output{Wie in~\Cref{prob:steinerForest}}
        \State{$y := 0$} \Comment{Lösung von~\eqref{eq:steinerForestDual}}
        \State{$F := \emptyset$}
        \State{$\ell := 0$} \Comment{Schleifenindex}
        \While{$\exists~ i: \not\exists s_i$-$t_i$-Weg in $(V, F)$}
            \State{$\ell := \ell + 1$}
            \State{$C := \set{\text{Zusammenhangskomponente $S$ von $(V, F)$} \mid \exists~ i : \abs{S \cap \set{s_i, t_i}} = 1}$}
            \State{Erhöhe $y_S$ gleichmäßig für alle $S \in C$, bis für ein $S$ aus $C$ und eine Kante $e_\ell \in \delta(S)$ gilt $\sum_{S' : e_\ell \in \delta(S')} y_{S'} = w_{e_\ell}$.}
            \State{$F := F \cup \set{e_\ell}$}
        \EndWhile{}
        \State{$F' := F$}
        \For{$k = \ell,\ell-1,\ldots,1$}
            \If{$F' \setminus \set{e_k}$ gültige Lösung} 
            \State{$F' := F' \setminus \set{e_k}$}
            \EndIf{}
        \EndFor{}
        \State{\Return{$F'$}}
    \end{algorithmic}
\end{algorithm}

\begin{lem}{PrimalDual-SteinerForest2: Iterationen}{primalDualSteinerForest2}
    In jeder Iteration von \textsc{PrimalDual-SteinerForest2} gilt
    \begin{equation*}
        \sum_{S \in C} \abs{F' \cap \delta(S)} \leq 2 \abs{C}.
    \end{equation*}
\end{lem}

\begin{thm}{PrimalDual-SteinerForest2}{primalDualSteinerForest2}
    \textsc{PrimalDual-SteinerForest2} ist ein $2$-Approximationsalgorithmus.
\end{thm}

\begin{proof}
    Es gilt 
    \begin{equation*}
        \sum_{e \in F'} w_e = \sum_{e \in F'} \sum_{S': e \in \delta(S)} y_S = \sum_{S \in \cS} y_S \cdot \abs{F' \cap \delta(S)}.
    \end{equation*}
    Wir wollen also zeigen, dass 
    \begin{equation*}
        \sum_{S \in \cS} y_S \abs{F' \cap \delta(S)} \leq 2 \sum_{S \in \cS} y_S %\label{eq:primalDualSteinerForest2}
    \end{equation*}
    gilt. Per Induktion über $\ell$: Für den Induktionsanfang gilt $y_S = 0$ für alle $S \in \cS$, es ist also nichts zu zeigen. Für den Induktionsschritt betrachte die Iteration, in der jedes $y_S$ für $S \in C$ um $\varepsilon$ erhöht wird. Dann wird $\sum_{S \in \cS} y_S \cdot \abs{F' \cap \delta(S)}$ um $\varepsilon \cdot \sum_{S \in C} \abs{F' \cap \delta(S)}$ erhöht. Die rechte Seite $2\sum_{S \in \cS} y_S$ wird um $2\varepsilon\abs{C}$ erhöht. Mit~\Cref{lem:primalDualSteinerForest2} wird die linke Seite höchstens so stark erhöht wie die rechte Seite.
\end{proof}

\begin{proof}[Beweis von Lemma~$\ref{lem:primalDualSteinerForest2}$]
    Sei $F_\ell = \set{e_1, \dots, e_{\ell - 1}}$ und $H = F' \setminus F_\ell$. Dann ist $F_\ell \cup H = F_\ell \cup F'$ eine Lösung. Betrachte Iteraiton $\ell$ und lege einen Graphen $G'$ an. Dabei setzen wir die Knoten $V'$ als die Zusammenhangskomponenten von $(V, F_\ell)$, und die Kanten entsprechen den Kanten aus $H$.

    Jede Kante $e \in H$ verläuft zwischen verschiedenen Zusammenhangskomponenten von $(V, F_\ell)$. $G'$ ist außerdem kein Multigraph, also zwischen zwei Knoten aus $G'$ gibt maximal eine Kante.

    Für $v \in V'$ sei $d(v)$ der Grad von $v$. Wir sagen, $v \in V'$ ist \emph{aktiv}, wenn für die entsprechende Zusammenhangskomponente $S$ gilt, dass $S \in C$. Wir sagen, $v \in V'$ ist \emph{passiv}, wenn $v$ nicht aktiv ist, aber $d(v) > 0$ gilt. Die Menge $A \subseteq V'$ beschreibt die aktiven Knoten, $P \subseteq V'$ beshreibt die passiven Knoten.

    Wir wollen zeigen:
    \begin{equation}
        \sum_{S \in C} \abs{F' \cap \delta(S)} \leq 2 \abs{C} \label{eq:primalDualSteinerForest2-lem1},
    \end{equation}
    dabei ist $\abs{F' \cap \delta(S)} = \delta(v_S)$ für $v_S$ den entsprechenden Knoten in $G'$, die rechte Seite ist das gleiche wie $2 \cdot \abs{A}$, somit ist~\eqref{eq:primalDualSteinerForest2-lem1} äquivalent zu
    \begin{equation}
        \sum_{v \in A} d(v) \leq 2 \abs{A}.\label{eq:primalDualSteinerForest2-lem2}
    \end{equation}
    Es gilt 
    \begin{equation*}
        \sum_{v \in A} d(v) = \sum_{v \in A \cup P} d(v) - \sum_{v \in P} d(v).
    \end{equation*}
    Wäre $d(v) \geq 2$ für alle $v \in P$, dann gilt 
    \begin{equation*}
        \sum_{v \in A} d(v) = \sum_{v \in A \cup P} d(v) - \sum_{v \in P} d(v) \leq 2 \cdot (\abs{A} + \abs{P}) - 2 \abs{P} = 2 \abs{A},
    \end{equation*}
    wobei $\sum_{v \in A \cup P} d(v) \leq 2 \cdot (\abs{A} + \abs{P})$, weil $G'$ kreisfrei (und kein Multigraph) ist und somit $\abs{E(G')} \leq \abs{V(G')}-1 \leq \abs{A} + \abs{P} - 1 \leq \abs{A} + \abs{P}$.

    Angenommen, für $v \in P$ ist $d(v) = 1$. Der Fall $d(v) = 0$ kann nach Definition passiver Knoten nicht auftreten. Sei $e$ die zu $v$ inzidente Kante. $e$ gehört zu $H$ und damit zu $F'$. Damit wird $e$ gebraucht, um Zusammenhang zwischen $s_i$ und $t_i$ für ein $i$ herzustellen. Dann ist $e$ aktiv. $\lightning$ 
\end{proof}


\part{Online-Algorithmen}
\setcounter{chapter}{10}
\chapter{Einleitung}

In den vergangenen Problemen kannten wir von Anfang an die komplette Eingabe. Nun stellen wir uns vor, dass im Laufe eines Algorithmus weitere Eingaben bekannt werden bzw. dazukommen, beispielsweise auf dem Aktienmarkt. Ein Algorithmus dafür heißt \emph{Online-Algorithmus}\index{Online-Algorithmus}. Mehr dazu nächste Woche.

\newLecture{16.12.2025}

Das klassische Einführungsproblem ist das fiktive \emph{Ski Rental}-Problem: Entweder zahlen wir $50$ pro Tag, oder $800$ als Flat Rate. Wir wissen aber vorher nicht, wie gut das Wetter ist und ob man an diesem Tag Ski fahren möchte. Angenommen, wir gehen an $x$ Tagen Ski fahren.

\begin{description}
    \item[1. Strategie] Wir kaufen die Flat Rate am ersten Tag. Dann ist $\OPT = \min\set{800, 50x}$ und $\frac{800}{\OPT} \leq 16$.
    \item[2. Strategie] Wir kaufen die Flat Rate nie. Dann ist $\frac{50 x}{\min\set{800, 50x}} \overset{x \to \infty}{\longrightarrow} \infty$.
    \item[3. Strategie] Wir kaufen die Flat Rate am 17. Tag. Dann bezahlen wir $50x$ für $x \leq 16$ und $1600$ für $x \geq 16$ und wir erhalten $\frac{\text{Kosten}}{\OPT} \leq 2$.
\end{description}

\begin{defi}{Kompetitiver Faktor}{competitiveFactor}\index{Kompetitiver Faktor}
    Ein Online-Algorithmus $A$ für ein Minimierungsproblem \emph{erreicht einen kompetitiven Faktor $r \geq 1$} (oder \emph{ist $r$-kompetitiv}), wenn
    \begin{equation*}
        \exists \tau \in \IR ~ \forall \sigma \in \cI_\Pi: \quad w_A(\sigma) \leq r \cdot \OPT(\sigma) + \tau.
    \end{equation*}
    Gilt das sogar für $\tau = 0$, so heißt $A$ \emph{strikt $r$-kompetitiv}.\index{Kompetitiver Faktor!strikt -}
\end{defi}

Ein strikt $r$-kompetitiver, polynomieller Online-Algorithmus ist tatsächlich auch ein $r$-Approxi\-mastionsalgorithmus. Andersrum gilt das im Allgemeinen nicht.

\chapter{Paging}

Wir betrachten die Frage \emph{Cache vs. RAM}. Der Cache habe eine Größe $k$. Es sei $\sigma = (\sigma_1, \sigma_2, \ldots)$ mit $\sigma_i \in \IN$. Dieses entspricht einer aufgerufenen Seite. Wir können also nur $k$ Seiten gleichzeitig in unserem Cache aktiv haben. Die Kosten sind nun die Anzahl der Zugriffe, bei denen eine Seite in den Cache geladen werden muss. Außerdem sei der Cache am Anfang leer.\index{Paging}

\begin{prob}{Paging}{paging}
    \prblmObj{Seiten $\set{1,\ldots,m}$, Cache-Größe $k$, Eingabe $\sigma = (\sigma_1, \ldots, \sigma_n) \in \set{1,\ldots,m}^n$}{Verteilung der aktuell im Cache befindlichen Seiten}{Minimiere Anzahl der Seitenladungen}
\end{prob}

\section{Deterministische Algorithmen}
Wir haben folgende Algorithmen-Ideen:
\begin{description}
    \item[LRU] (Least Recently Used)\index{Paging!LRU}
    \item[FIFO] (First In, First Out)\index{Paging!FIFO}
    \item[FWF] (Flush When Full)\index{Paging!FWF}
    \item[RANDOM] \emph{nicht deterministisch}\index{Paging!RANDOM}
    \item[LFU] (Least Frequently Used)\index{Paging!LFU}
    \item[LIFO] (Last In, First Out)\index{Paging!LIFO}
    \item[LFD] (Longest Forward Distance) \emph{keine Online-Strategie}\index{Paging!LFD}
\end{description}

\subsection{Markierungsalgorithmen}

Wir betrachten eine \emph{Phaseneinteilung}\index{Phaseneinteilung}: Sei $k = 3$ und $\sigma = (1,2,4,2,1,3,5,2,3,5,1,2,3,4)$. Wir unterteilen dies in Phasen, in denen nur auf $k$ verschiedene Seiten zugegriffen wird (mit maximaler Länge). 

\begin{expl}
    Für obiges Beispiel erhalten wir die folgende Phaseneinteilung:
    \begin{equation*}
        \sigma = (\overset{P1}{\overbrace{1,2,4,2,1}},\overset{P2}{\overbrace{3,5,2,3,5}},\overset{P3}{\overbrace{1,2,3}},\overset{P4}{\overbrace{4}}).
    \end{equation*}
\end{expl}

\begin{defi}{Markierungsalgorithmus}{markAlgo}\index{Markierungsalgorithmus}
    Ein Algorithmus ist ein \emph{Markierungsalgorithmus}, wenn wir nie eine Seite verdrängen, auf die in der aktuellen Phase bereits zugegriffen wurde.
\end{defi}

\begin{thm}{Beispiele: Markierungsalgorithmen auf Paging}{explsMarkAlgosPaging}
    LRU und FWF sind Markierungsalgorithmen.
\end{thm}
\begin{proof}
    \begin{description}
        \item[LRU:] Sei $\sigma$ eine Eingabe, auf der LRU eine markierte Seite $x$ verdränkt, zum Zeitpunkt $t$ in Phase $i$. Dann wird Seite $x$ (das vor Seite $\sigma_t$, zum Zeitpunkt $t' < t$ vorkommt) entfernt. Damit müssen zwischen $\sigma_{t'}$ und $\sigma_t$ $k-1$ Seiten $\neq x$ vorkommen. Damit gibt es aber $k+1$ verschiedene Seiten in Phase $i$. $\lightning$
        \item[FWF:] Wenn die erste Phase vorbei ist, wird der gesamte Cache geleert. Somit besteht der Cache gerade aus den Seiten, die in der aktuellen Phase bereits vorgekommen sind.
    \end{description}
\end{proof}

\begin{thm}{Markierungsalgorithmen auf Paging}{markAlgosPaging}
    Jeder Markierungsalgorithmus ist strikt $k$-kompetitiv.
\end{thm}
\begin{proof}
    Sei $A$ ein Markierungsalgorithmus und $\sigma$ eine Eingabe. Sei $\ell$ die Anzahl der Phasen in $\sigma$ und sei $\ell \geq 2$. Es gilt $w_A(\sigma) \leq k \cdot \ell$, da in einer Phase höchstens $k$ Seiten neu in den Cache geladen werden (sonst hätte die Phase mehr als $k$ Seiten!).

    Wir teilen eine Eingabe in \emph{Teilsequenzen} auf, wobei Teilsequenz $i$ (mit $i \in \set{1,\ldots,\ell - 2}$) mit dem zweiten Zugriff der $(i+1)$-ten Phase beginnt und mit dem ersten Zugriff in Phase $i+2$.

    \emph{Behauptung:} Jeder Algorithmus muss in jeder Teilsequenz mindestens einen Seitenfehler produzieren: Mindestens eine der Seiten aus der Teilsequenz ist zu Beginn der Teilsequenz nicht im Cache.

    Somit erhalten wir $\OPT(\sigma) \geq k + (\ell - 2) = k + \ell - 2 \geq \ell$, da wir in der ersten Phase $k$ Seiten in den leeren Cache laden.

    Somit erhalten wir $w_A(\sigma) \leq k \cdot \ell \leq \OPT(\sigma) \cdot k$.
\end{proof}

\begin{corollary}
    LRU und FWF sind strikt $k$-kompetitiv.
\end{corollary}

\subsection{Untere Schranken}

\begin{thm}{LFU, LIFO}{lfuAndLIFO}
    LFU und LIFO sind nicht kompetitiv.
\end{thm}
Das heißt, wir finden keinen konstanten Kompetitiven Faktor.
\begin{proof}\leavevmode
    \begin{description}
        \item[LIFO:] Sei $\sigma = (1,2,\ldots,k,(k+1,k)^\ell)$. Dann ist $w_{\text{LIFO}}(\sigma) = 2\ell + (k-1)$ und $\OPT(\sigma) = k + 1$.
        \item[LFU:] Sei $\sigma = (1^{\ell+1},2^{\ell+1},\ldots,(k-1)^{\ell+1},k,(k+1,k)^\ell)$. Dann ist ebenfalls $w_{\text{LFU}}(\sigma) = 2\ell + (k-1)$ und $\OPT(\sigma) = k + 1$.
    \end{description}
\end{proof}

\begin{thm}{Deterministische Online-Algorithmen für Paging}{deterministicPagingAlgo}
    Für kein $r < k$ gibt es einen \emph{deterministischen} $r$-kompetitiven Online-Algorithmus für Paging.
\end{thm}

\begin{proof}
    Sei $A$ ein beliebiger Online-Algorithmus. Wir konstruieren $\sigma$ wie folgt:
    
    $\sigma^{(\ell)} = (1,\ldots,k,\sigma_{k+1}, \ldots, \sigma_\ell)$, und $\sigma_i$ mit $i > k$ entspricht der \emph{eindeutigen} Seite aus $\set{1,\ldots,k+1}$, die $A$ nicht im Cache hält. Dann ist $w_A(\sigma) = \ell$, da in jedem Schritt ein Seitenfehler passiert.
    
    \newLecture{17.12.2025}

    Andererseits ist $\OPT \leq w_{\text{LFD}}(\sigma^{\ell}) \leq k + \ceil*{\frac{\ell}{k}}$
    Nun gibt es für alle $r < k$ und für alle $\tau$ ein $\ell$, sodass $w_A(\sigma^{(\ell)}) > \tau + r \cdot \OPT(\sigma^{(\ell)})$ -- wähle $\ell > \tau + r \cdot \left( k + \ceil*{\frac{\ell}{k}} \right)$ und wir erhalten somit den Widerspruch. $\lightning$
\end{proof}

\section{Randomisierte Algorithmen}
Wir betrachten folgende Algorithmen:
\begin{description}
    \item[RANDOM] Verdränge bei Seitenfehler bei vollem Cache eine uniform zufällig gewählte Seite\index{Paging!RANDOM}
    \item[MARK] Verdränge bei Seitenfehler bei vollem Cache uniform zufällig eine nicht-markierte Seite\index{Paging!MARK}
\end{description}

\begin{defi}{Kompetitiver Faktor für randomisierte Algorithmen}{competitiveFactorRandom}\index{Kompetitiver Faktor}\index{Kompetitiver Faktor!randomisierter -}
    Ein randomisierter Online-Algorithmus $A$ für ein Minimierungsproblem \emph{erreicht einen kompetitiven Faktor $r \geq 1$} (oder \emph{ist $r$-kompetitiv}), wenn
    \begin{equation*}
        \exists \tau \in \IR ~ \forall \sigma \in \cI_\Pi:\quad \IE[w_A(\sigma)] \leq r \cdot \OPT(\sigma) + \tau.
    \end{equation*}
    Gilt das sogar für $\tau = 0$, so heißt $A$ \emph{strikt $r$-kompetitiv}.\index{Kompetitiver Faktor!strikt -}
\end{defi}

\subsection{Potentialfunktionen}

Sei $A$ ein Online-Algorithmus und $\sigma = (\sigma_1, \ldots, \sigma_n)$ eine Eingabe. Wir wollen für $A$ und $\sigma$ Potentiale $\Phi_0, \ldots, \Phi_n \in \IR$ finden, die intuitiv einen \enquote{Kontostand} darstellen sollen -- ist $\Phi$ positiv, haben wir gegenüber $\OPT$ mehr Kosten gebraucht, für $\Phi$ negativ analog.

\begin{defi}{Amortisierte Kosten}{amortisedCost}\index{Amortisierte Kosten}
    Seien $A_i$ und $\OPT_i$ die Kosten von $A$ bzw. $\OPT$ bei Ereignis $\sigma_i$. Wir sagen,
    \begin{equation*}
        a_i := A_i + \Phi_i - \Phi_{i-1}
    \end{equation*}
    sind die \emph{amortisierten Kosten} von $A$ für $\sigma_i$.
\end{defi}

\begin{thm}{Potentialfunktion}{potentialFunction}
    Sei $A$ ein Online-Algorithmus und $r \geq 1$. Gibt es eine Konstante $b \geq 0$ und eine Potentialfunktion $\Phi$ mit den folgenden Eigenschaften, so ist $A$ $r$-kompetitiv:
    \begin{enumerate}
        \item Für alle $i \geq 1$ gilt $\IE[a_i] \leq r \cdot \OPT_i$.
        \item $\IE[\Phi_0] \leq b$.
        \item Für alle $i \geq 1$ gilt $\IE[\Phi_i] \geq -b$.
    \end{enumerate}
\end{thm}

\begin{proof}
    Es gilt 
    \begin{equation*}
        \begin{split}
            \IE[w_A(\sigma)] &= \IE\left[ \sum_{i = 1}^n A_i \right] = \IE\left[ \sum_{i = 1}^n (a_i - \Phi_i + \Phi_{i-1}) \right]  \\
                             &= \sum_{i = 1}^n \IE[a_i] - \IE[\Phi_i] + \IE[\Phi_{i-1}] = \left(\sum_{i = 1}^n \IE[a_i]\right) + \IE[\Phi_0] - \IE[\Phi_n] \\
                             &\leq \sum_{i = 1}^n r \cdot \OPT_i + b + b = r \cdot \sum_{i = 1}^n \OPT_i + 2b = r \cdot \OPT + 2b.
        \end{split}
    \end{equation*}
    mit Linearität des Erwartungswerts und den gegebenen Bedingungen.
\end{proof}

\subsection{Analyse von RANDOM}\index{Paging!RANDOM}

Wur definieren das Potential basierend auf den Cacheinhalten von RANDOM und $\OPT$. Sei $z$ die Anzahl an gemeinsamen Seiten. Wir setzen dabei $\Phi(z) = k(k-z)$.
\begin{thm}{RANDOM ist $k$-kompetitiv}{RANDOMisKcompetitive}
    RANDOM ist $k$-kompetitiv.
\end{thm}
\begin{proof}
    Es ist $\Phi_0 = \Phi(0) = k^2$, da beide Caches leer sind und entsprechend keine gemeinsamen Seiten haben. Ebenso gilt $\Phi_i \geq 0$ für alle $i$, dementsprechend sind die hinteren beiden Eigenschaften erfüllt.

    Zur ersten Eigenschaft: Zu zeigen ist, dass 
    \begin{equation}
        \IE[a_i] \leq r \cdot \OPT_i \iff \IE[A_i + \Phi_i - \Phi_{i-1}] \leq r \cdot \OPT \iff \IE[\Phi_i - \Phi_{i-1}] \leq r \cdot \OPT_i - \IE[A_i],\label{eq:analysisRANDOM1}
    \end{equation}
    wobei $A = \text{RANDOM}$.

    Sei das Verhalten von RANDOM auf $\sigma_1, \ldots, \sigma_{i-1}$ bereits aufgedeckt, und sei $z_{i-1}$ die Anzahl an gemeinsamen Seiten nach $\sigma_1, \ldots, \sigma_{i-1}$, und $P$ mit $\abs{P} = z_{i-1}$ die gemeinsamen Seiten, und sei $\sigma_i = p$.
    \begin{description}
        \item[Fall 1: $p$ ist im Cache von RANDOM.] Dann ist $A_i = 0$ und $z_i - z_{i-1} \in \set{0,1}$. Dementsprechend ist $\Phi_i - \Phi_{i-1} \in \set{0,-k}$. Damit ist die linke Seite von~\eqref{eq:analysisRANDOM1} nichtpositiv und die rechte Seite nichtnegativ.
        \item[Fall 2: $p$ ist nicht im Cache von RANDOM, aber im Cache von OPT.] Dann ist $A_i = 1$ und $z_i - z_{i-1} \in \set{0,1}$. Entsprechend ist $\Phi_i - \Phi_{i-1} \in \set{0, -k}$. Die Wahrscheinlichkeit, eine  Seite zu verdrängen, die nicht zu $P$ gehört, beträgt $\frac{k - z_{i-1}}{k}$. Damit gilt 
        \begin{equation}
            \IE[\Phi_i - \Phi_{i-1}] = \frac{k - z_{i-1}}{k} \cdot (-k) = z_{i-1} - k \leq -1,\label{eq:analysisRANDOM2}
        \end{equation}
        wobei $z_{i-1} \leq k-1$, da $p$ nur im Cache von OPT enthalten ist. Mit $k \cdot \OPT - A_i = -1$ ist~\eqref{eq:analysisRANDOM1} damit bewiesen.
        \item[Fall 3: $p$ ist weder im Cache von RANDOM noch im Cache von OPT.] Dann nimmt die rechte Seite von~\eqref{eq:analysisRANDOM1} den Wert $k-1$ an.
        \begin{description}
            \item[Fall 3.1: OPT verdrängt eine Seite, die nicht zu $P$ gehört.] Dann ist $\Phi_i - \Phi_{i-1} \in \set{0, -k}$, je nachdem, ob RANDOM eine Seite aus $P$ oder eine andere Seite verdrängt. Somit ist die linke Seite von~\eqref{eq:analysisRANDOM1} kleiner als $k-1$.
            \item[Fall 3.2: OPT verdrängt eine Seite aus $P$.] Verdrängt RANDOM dieselbe Seite, so ändert sich das Potential nicht, ebenso wenn RANDOM eine andere Seite, die nicht zu $P$ gehört, verdrängt. Nur der Fall, dass RANDOM eine andere Seite aus $P$ als OPT verdrängt, erhöht sich das Potential um $k$ -- eintreten tut dies mit Wahrscheinlichkeit $\frac{z_{i-1} - 1}{k}$, daraus folgt 
            \begin{equation}
                \IE[\Phi_i - \Phi_{i-1}] = \frac{z_{i-1} - 1}{k} \cdot k \leq k - 1. \label{eq:analysisRANDOM3}
            \end{equation}
        \end{description} 
    \end{description}
    Damit ist~\eqref{eq:analysisRANDOM1} gezeigt für jedes $P$.
\end{proof}

\subsection{Untere Schranke für randomisierte Online-Algorithmen}

\begin{thm}{Randomisierte Online-Algorithmen für Paging}{randomizedOnlineAlgoPaging}
    Es gibt keinen randomisierten Online-Algorithmus für Paging, der besser als $H_k$-kompetitiv ist.
\end{thm}
\begin{proof}
    Sei $A$ ein beliebiger, randomisierter Online-Algorithmus. Konstruiere eine Eingabe $\sigma$ mit $\sigma_i \in \set{1, \ldots, k+1}$ und $\sigma = (1, \ldots, k, \ldots)$.

    Für $\sigma$ gibt es eine Wahrscheinlichkeitsverteilung:
    \begin{equation*}
        \forall i \in \set{1,\ldots,k+1} \text{ definiere } p_i := p_i^\sigma := \IP[\text{Seite $i$ nach $\sigma$ nicht im Cache}].
    \end{equation*}
    Dann gilt $\sum_{i = 1}^{k + 1} p_i^\sigma = 1$.

    \emph{Erste Idee:} Wähle $\sigma_i = j$ für das $j$, für das $p_j^{\sigma_1, \ldots, \sigma_{i-1}}$ maximal ist. Das Problem: Für RANDOM gilt stets $p^{\sigma_1, \ldots, \sigma_i}_{\sigma_i} = 0$ und für alle $j \neq \sigma_i$ ist $p^{\sigma_1, \ldots, \sigma_i}_j = \frac{1}{k}$. Erwartete Kosten von RANDOM sind pro Schritt $\frac{1}{k}$. Somit erwartet RANDOM pro Schritt Kosten von $\frac{1}{k}$. Je nach Tiebreaking verursacht auch OPT in jedem $k$-ten Schritt einen Seitenfehler.

    \emph{Konstruktion einer Phase $\sigma'$:} Wir unterteilen nun $\sigma$ wieder in Phasen $\set{\sigma'}$, diese bestehen jeweils aus mehreren Teilphasen $\sigma_1', \ldots, \sigma_k'$. Eine Seite heißt \emph{markiert}, wenn sie in der aktuellen Phase schon vorgekommen ist oder wenn es die letzte Seite aus der vorherigen Phase ist.

    Eine Teilphase $\sigma_i'$ besteht aus einer (ggf. leeren) Sequenz von Zugriffen auf markierte Seiten, gefolgt von einer unmarkierten Seite. Dann ist $\OPT(\sigma) \leq (k + 1) + (m - 1)$, wobei $m$ die Gesamtzahl der Phasen ist.

    \emph{Konstruktion von Teilphase $\sigma_j'$:} Zu Beginn von $\sigma_j'$ gibt es $u = k - j + 1$ unmarkierte Seiten. Wir setzen $\gamma := \sum_{i \in M} p_i$, wobei $M$ die menge der markierten Seiten ist.
    \begin{description}
        \item[Fall 1: $\gamma = 0$.] Dann gibt es $a \notin M$ mit $p_a \geq \frac{1}{u}$. Dann ist $\sigma_j' = (a)$.
        \item[Fall 2: $\gamma > 0$.] Sei $a \in M$ mit $p_a > 0$ beliebig und sei $\varepsilon := p_a$. Dann beginne $\sigma_j'$ mit Zugriff auf $a$. Wir betrachten folgenden Algorithmus:
        \begin{algorithmic}[1]
            \State{$E := p_a$}
            \While{$E < \frac{1}{u}$ \textbf{and} $\gamma > u \varepsilon$}
                \State{Hänge Zugriff $b \in M$ mit $b := \arg\max_{i \in M} p_i$ an $\sigma_j'$ an.}
                \State{$E := E + p_b$}
                \State{Aktualisiere $p_1, \ldots, p_{k + 1}$ und $\gamma = \sum_{i \in M} p_i$.}
            \EndWhile{}
            \State{Hänge Zugriff auf $b' \notin M$ mit $b' := \arg\max_{i \notin M} p_i$ an $\sigma_j'$ an.}
        \end{algorithmic}
        Terminiert die Schleife wegen $E \geq \frac{1}{u}$, so betragen die erwarteten Kosten von $a$ auf $\sigma_j'$ mindestens $\frac{1}{u}$. Betrachte also den Fall, dass die Schleife wegen $\gamma \leq u \varepsilon$ terminiert. Die erwarteten Kosten von $A$ auf $\sigma_j'$ betragen mindestens
        \begin{equation*}
            p_a + p_{b'} = \varepsilon + \frac{1 - \gamma}{u} \geq \varepsilon + \frac{1 - u\varepsilon}{u} = \frac{1}{u}.
        \end{equation*}
    \end{description} 
    Damit betragen die erwarteten Kosten von Algorithmus $A$ auf der Teilphase $\sigma_j'$ mindestens $\frac{1}{u_j}$ mit $u_j := k - j + 1$. Aufsummiert ergibt sich
    \begin{equation*}
        \IE[\text{Kosten von $A$ auf Phase $i \geq 2$}] \geq \sum_{j = 1}^k \frac{1}{u_j} = \sum_{j = 1}^k \frac{1}{k - j + 1} = H_k.
    \end{equation*}
    Als Gesamtkosten erhalten wir
    \begin{equation*}
        \IE[w_A(\sigma)] \geq k + 1 + (m-1) H_k, \quad \quad \OPT = k + m
    \end{equation*}
    und für $m$ groß gilt $\frac{\IE[w_A(\sigma)]}{\OPT} \to H_k$.
\end{proof}

\chapter{Das $k$-Server-Problem}

Wir betrachten einen metrischen Raum $\cM := (M, d)$ für einen endlichen Raum $M$ und eine Metrik $d$.
\begin{prob}{$k$-Server}{kServer}\index{k-@$k$-Server-Problem}
    \prblmObj{Metrischer Raum $\cM = (M, d)$, Folge $\sigma = (\sigma_1, \ldots, \sigma_n)$ von Anfragen $\sigma_i \in M$, $k$ Server mit Startstandorten in $M$}{Folge von Serverstandorten, sodass bei jeder Anfrage ein Server bei diesem Standort steht}{Minimiere Gesamtkosten der Serververschiebungen}
\end{prob}

Das Paging-Problem ist ein Spezialfall des $k$-Server-Problems, wo $d(x,y) = 1$ für alle $x \neq y$ ist.

\begin{algorithm}[h]
    \caption{\textsc{Greedy-KServ}}\label{alg:greedy-kserv}
    \begin{algorithmic}[1]
        \Input{Wie in~\Cref{prob:kServer}}
        \Output{Wie in~\Cref{prob:kServer}}
        \For{$\sigma_i \in \sigma$}
            \If{Kein Server befindet sich bei $\sigma_i$}
                \State{Verschiebe den nächsten Server zu $\sigma_i$}
            \EndIf{}
        \EndFor{}
    \end{algorithmic}
\end{algorithm}

Dieser Algorithmus hat das Problem, dass eine Eingabe $\sigma = (c, a, b, a, b, a, b, \ldots)$ beliebig hohe Kosten erzeugen kann (für $k = 2$ und $d(a,b) < \min\set{d(a,c), d(b,c)}$ und $a$, $b$ und $c$ auf einer Linie).
